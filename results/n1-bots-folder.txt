def sell_regulatory_fees(price: float, qty: int) -> float:
    if qty <= 0 or price <= 0:
        return 0.0
    notional = float(price) * int(qty)
    sec = notional * SEC_RATE
    finra = min(8.30, FINRA_TAF_PER_SHARE * qty)
    cat = CAT_PER_SHARE * qty
    return float(sec + finra + cat)

def load_and_prep_data() -> Dict[str, pd.DataFrame]:
    dfs: Dict[str, pd.DataFrame] = {}
    required = UNIVERSE + [BENCHMARK]

    for sym in required:
        p = DATA_DIR / f"{sym}_daily.csv"
        if not p.exists():
            raise FileNotFoundError(f"Missing required data file: {p}")

        df = pd.read_csv(p, parse_dates=["timestamp"])
        df = df.set_index("timestamp").sort_index()
        df = df[~df.index.duplicated(keep="last")].copy()

        # Moving Averages
        close = df["close"]
        df["ema20"] = close.ewm(span=20, adjust=False).mean()
        df["sma5"] = close.rolling(5, min_periods=5).mean()
        df["sma50"] = close.rolling(50, min_periods=50).mean()
        df["sma200"] = close.rolling(200, min_periods=200).mean()

        # Donchian Channels (shifted to prevent lookahead)
        df["donch_high_30"] = df["high"].rolling(30, min_periods=30).max().shift(1)
        df["donch_low_30"] = df["low"].rolling(30, min_periods=30).min().shift(1)

        # Average True Range (ATR 14)
        prev_close = close.shift(1)
        tr = np.maximum(
            df["high"] - df["low"],
            np.maximum((df["high"] - prev_close).abs(), (df["low"] - prev_close).abs())
        )
        df["atr14"] = tr.rolling(14, min_periods=14).mean()
        df["atr_ratio"] = df["atr14"] / close.replace(0, np.nan)
        atr_baseline = df["atr14"].rolling(30, min_periods=30).mean()
        df["atr_expansion"] = df["atr14"] / atr_baseline.replace(0, np.nan)

        # RSI(2)
        delta = close.diff()
        gain = delta.clip(lower=0).rolling(2, min_periods=2).mean()
        loss = (-delta.clip(upper=0)).rolling(2, min_periods=2).mean()
        rs = gain / loss.replace(0, np.nan)
        df["rsi2"] = 100.0 - (100.0 / (1.0 + rs))
        df.loc[(loss == 0) & (gain > 0), "rsi2"] = 100.0
        df.loc[(loss == 0) & (gain == 0), "rsi2"] = 50.0

        # Bollinger Bands
        df["bb_mid"] = close.rolling(20, min_periods=20).mean()
        df["bb_std"] = close.rolling(20, min_periods=20).std()
        df["bb_upper"] = df["bb_mid"] + 2.0 * df["bb_std"]
        df["bb_lower"] = df["bb_mid"] - 2.0 * df["bb_std"]

        # Volume / Momentum
        df["vol_sma20"] = df["volume"].rolling(20, min_periods=20).mean()
        df["vol_surge"] = df["volume"] / df["vol_sma20"].replace(0, np.nan)
        df["roc5"] = close.pct_change(5)

        dfs[sym] = df

    return dfs

def close_trade(
    expert: str,
    pos: Position,
    exit_date: pd.Timestamp,
    exit_price: float,
    exit_reason: str,
) -> Tuple[Trade, float]:
    qty = pos.qty
    if pos.side == "BUY":
        gross = (exit_price - pos.entry_price) * qty
        exit_fees = sell_regulatory_fees(exit_price, qty)
        pnl_pct = (exit_price / pos.entry_price) - 1.0
    else:
        gross = (pos.entry_price - exit_price) * qty
        exit_fees = 0.0
        pnl_pct = (pos.entry_price - exit_price) / pos.entry_price

    total_fees = pos.entry_fees + exit_fees
    net_pnl = gross - total_fees
    init_risk_dollars = pos.initial_risk_per_share * qty
    r_mult = net_pnl / init_risk_dollars if init_risk_dollars > 0 else 0.0

    trade = Trade(
        symbol=pos.symbol,
        expert=expert,
        side=pos.side,
        entry_date=str(pos.entry_date.date()),
        entry_price=pos.entry_price,
        exit_date=str(exit_date.date()),
        exit_price=exit_price,
        qty=qty,
        initial_stop=pos.initial_stop,
        initial_risk_per_share=pos.initial_risk_per_share,
        initial_risk_dollars=init_risk_dollars,
        gross_pnl=gross,
        fees=total_fees,
        pnl=net_pnl,
        pnl_pct=pnl_pct,
        r_mult=r_mult,
        exit_reason=exit_reason,
    )
    nav_change = gross - exit_fees
    return trade, nav_change


def check_open_gap(pos: Position, row: pd.Series) -> Optional[Tuple[float, str]]:
    open_px = float(row["open"])
    if pos.side == "BUY":
        if open_px <= pos.current_stop:
            return open_px * (1.0 - SLIPPAGE_PCT), "STOP_GAP_LOSS"
        if pos.tp is not None and open_px >= pos.tp:
            return max(pos.tp, open_px) * (1.0 - SLIPPAGE_PCT), "TARGET_GAP"
    else:
        if open_px >= pos.current_stop:
            return open_px * (1.0 + SLIPPAGE_PCT), "STOP_GAP_LOSS"
        if pos.tp is not None and open_px <= pos.tp:
            return min(pos.tp, open_px) * (1.0 + SLIPPAGE_PCT), "TARGET_GAP"
    return None


def check_intraday_ohlc(pos: Position, row: pd.Series) -> Optional[Tuple[float, str]]:
    high = float(row["high"])
    low = float(row["low"])
    stop = pos.current_stop
    tp = pos.tp

    if pos.side == "BUY":
        stop_hit = low <= stop
        tp_hit = tp is not None and high >= tp
        if stop_hit:
            return stop * (1.0 - SLIPPAGE_PCT), "STOP_LOSS"
        if tp_hit:
            return tp * (1.0 - SLIPPAGE_PCT), "TAKE_PROFIT"
    else:
        stop_hit = high >= stop
        tp_hit = tp is not None and low <= tp
        if stop_hit:
            return stop * (1.0 + SLIPPAGE_PCT), "STOP_LOSS"
        if tp_hit:
            return tp * (1.0 + SLIPPAGE_PCT), "TAKE_PROFIT"
    return None


def calculate_mtm_nav(realized_nav: float, open_positions: Dict[str, Position], dfs: Dict[str, pd.DataFrame], date: pd.Timestamp, price_col: str = "close") -> float:
    nav = realized_nav
    for sym, pos in open_positions.items():
        if date in dfs[sym].index:
            px = float(dfs[sym].loc[date, price_col])
            unrealized = (px - pos.entry_price) * pos.qty if pos.side == "BUY" else (pos.entry_price - px) * pos.qty
            nav += unrealized
    return nav

def backtest_expert(name: str, fn: Callable, dfs: Dict[str, pd.DataFrame]) -> Tuple[dict, List[Trade], pd.DataFrame]:
    spy_df = dfs[BENCHMARK]
    common_dates = sorted(set.intersection(*[set(dfs[s].index) for s in UNIVERSE + [BENCHMARK]]))

    if len(common_dates) < 250:
        raise RuntimeError(f"Insufficient history: {len(common_dates)} common bars.")

    realized_nav = INITIAL_NAV
    open_positions: Dict[str, Position] = {}
    trades: List[Trade] = []
    equity_rows: List[dict] = []

    for t_idx in range(199, len(common_dates) - 1):
        curr_date = common_dates[t_idx]
        next_date = common_dates[t_idx + 1]

        # 1. Signals at bar t close
        exit_intents: Dict[str, str] = {}
        for sym, pos in list(open_positions.items()):
            df = dfs[sym]
            idx = df.index.get_loc(curr_date)
            res = fn(df, idx, pos, spy_df) if name == "8. Sector Top-Down" else fn(df, idx, pos)
            if res.get("signal") == "EXIT":
                exit_intents[sym] = res.get("reason", "SIGNAL_EXIT")

        entry_intents: List[Tuple[str, dict]] = []
        for sym in UNIVERSE:
            if sym in open_positions:
                continue
            df = dfs[sym]
            idx = df.index.get_loc(curr_date)
            res = fn(df, idx, None, spy_df) if name == "8. Sector Top-Down" else fn(df, idx, None)
            if res.get("signal") in ("BUY", "SELL"):
                entry_intents.append((sym, res))

        # 2. t+1 Open: Gap exits & signal exits
        for sym in list(open_positions.keys()):
            pos = open_positions[sym]
            row = dfs[sym].loc[next_date]
            gap_event = check_open_gap(pos, row)
            if gap_event:
                px, reason = gap_event
                t, nav_chg = close_trade(name, pos, next_date, px, reason)
                realized_nav += nav_chg

def calc_rsi(series, period=2):
    delta = series.diff()
    gain = delta.clip(lower=0)
    loss = -delta.clip(upper=0)
    avg_gain = gain.ewm(com=period - 1, adjust=False).mean()
    avg_loss = loss.ewm(com=period - 1, adjust=False).mean()
    rs = avg_gain / np.where(avg_loss == 0, 1e-5, avg_loss)
    return 100.0 - (100.0 / (1.0 + rs))

def with_retries(fn, *args, **kwargs):
    last_err = None
    for attempt in range(1, ExecConfig.MAX_RETRIES + 1):
        try:
            return fn(*args, **kwargs)
        except Exception as e:
            last_err = e
            log.warning(f"{fn.__name__} failed (attempt {attempt}/{ExecConfig.MAX_RETRIES}): {e}")
            if attempt < ExecConfig.MAX_RETRIES:
                time.sleep(ExecConfig.RETRY_BACKOFF_SECONDS * attempt)
    log.error(f"{fn.__name__} failed after {ExecConfig.MAX_RETRIES} attempts: {last_err}")
    raise last_err

def add_atr(df, period=14):
    df = df.copy()
    df['shift_close'] = df['close'].shift(1)
    high_low = df['high'] - df['low']
    high_close = np.abs(df['high'] - df['shift_close'])
    low_close = np.abs(df['low'] - df['shift_close'])
    ranges = pd.concat([high_low, high_close, low_close], axis=1)
    true_range = np.max(ranges, axis=1)
    df['atr'] = true_range.rolling(period).mean().bfill()
    return df

        if current_close > sma20 and current_close >= prev_close:

        elif current_close < sma20 and current_close <= prev_close:

    except Exception as e:

    return None

def calculate_position_size(instrument, side, entry_price, stop_price, balance):
    risk_amount = balance * (ExecConfig.RISK_PER_TRADE_PCT / 100.0)
    stop_distance = abs(entry_price - stop_price)
    if stop_distance <= 0:
        return 0

    if instrument.startswith("USD"):
        risk_per_unit = stop_distance
    elif instrument.endswith("JPY"):
        risk_per_unit = stop_distance / entry_price
    else:
        risk_per_unit = stop_distance

    if risk_per_unit <= 0:
        return 0

    raw_units = risk_amount / risk_per_unit
    max_units_allowed = int(ExecConfig.MAX_NOTIONAL / entry_price) if entry_price > 0 else 100000
    units = int(round(min(raw_units, max_units_allowed), -1))
    units = max(units, 100)

    return units if side == "buy" else -units

def compute_stop_target(df, side):
    atr_now = df.iloc[-1]["atr"]
    price = df.iloc[-1]["close"]
    
    if side == "buy":
        stop = price - atr_now * BTConfig.STOP_ATR_MULT
        target = price + atr_now * BTConfig.TARGET_ATR_MULT
    else:
        stop = price + atr_now * BTConfig.STOP_ATR_MULT
        target = price - atr_now * BTConfig.TARGET_ATR_MULT
    return stop, target

        if units == 0:
            log.warning(f"{instrument}: computed 0 units, skipping order.")
            last_seen[instrument] = latest_time

        try:

            n_open += 1
        except Exception as e:
            log.error(f"Order placement failed for {instrument}: {e}")

        last_seen[instrument] = latest_time

    return last_seen

        sleep_for = max(0, ExecConfig.POLL_SECONDS - elapsed)

            data = [{
                'time': b['t'],
                'open': float(b['o']),
                'high': float(b['h']),
                'low': float(b['l']),
                'close': float(b['c']),
                'volume': float(b['v'])
            } for b in bars]
            return pd.DataFrame(data)
    except Exception as e:
        print(f"Alpaca Equity fetch error {symbol}: {e}")
    return None

def simulate_strategy(df, asset_type="FOREX"):
    if df is None or len(df) < 55:
        return None

    df = df.copy()
    df['ema21'] = df['close'].ewm(span=21, adjust=False).mean()
    df['bb_mid'] = df['close'].rolling(20).mean()
    df['bb_std'] = df['close'].rolling(20).std()
    df['bb_upper'] = df['bb_mid'] + (2.0 * df['bb_std'])
    df['bb_lower'] = df['bb_mid'] - (2.0 * df['bb_std'])

    tr = pd.concat([
        df['high'] - df['low'],
        (df['high'] - df['close'].shift(1)).abs(),
        (df['low'] - df['close'].shift(1)).abs()
    ], axis=1).max(axis=1)
    df['atr'] = tr.rolling(14).mean()

    delta = df['close'].diff()
    gain = (delta.where(delta > 0, 0)).rolling(14).mean()
    loss = (-delta.where(delta < 0, 0)).rolling(14).mean()
    rs = gain / (loss + 1e-9)
    df['rsi'] = 100 - (100 / (1 + rs))

    df['donchian_high'] = df['high'].shift(1).rolling(20 if asset_type == "EQUITY" else 55).max()
    df['donchian_low'] = df['low'].shift(1).rolling(20 if asset_type == "EQUITY" else 55).min()

    results = {
        "FIXED": {"pnl": 0.0, "trades": 0, "wins": 0, "losses": 0},
        "VIRTUAL_RATCHET": {"pnl": 0.0, "trades": 0, "wins": 0, "losses": 0}
    }

    for mode in ["FIXED", "VIRTUAL_RATCHET"]:
        pos = None
        for i in range(55, len(df)):
            row = df.iloc[i]
            prev = df.iloc[i-1]

            buy_sig = False
            sell_sig = False

            if asset_type == "FOREX":
                if (prev['low'] <= prev['bb_lower']) and (row['close'] > row['bb_lower']) and (row['rsi'] < 40):
                    buy_sig = True
                elif (prev['high'] >= prev['bb_upper']) and (row['close'] < row['bb_upper']) and (row['rsi'] > 60):
                    sell_sig = True
            else:
                if row['close'] > prev['donchian_high'] and row['close'] > row['ema21']:
                    buy_sig = True
                elif row['close'] < prev['donchian_low'] and row['close'] < row['ema21']:
                    sell_sig = True

            if pos is not None:
                entry_px = pos['entry']
                direction = pos['direction']
                init_risk = pos['risk_dist']
                units = pos['units']
                cur_close = row['close']
                bb_mid = row['bb_mid']
                ema21 = row['ema21']
                atr = row['atr']

                mfe = (cur_close - entry_px) if direction == 'LONG' else (entry_px - cur_close)
                r_mult = mfe / (init_risk + 1e-9)

                trade_closed = False
                exit_pnl = 0.0

                if mode == "FIXED":
                    if direction == 'LONG':
                        if row['low'] <= pos['sl']:
                            trade_closed = True
                            exit_pnl = -BASE_RISK
                        elif row['high'] >= pos['tp']:
                            trade_closed = True
                            exit_pnl = BASE_RISK * 1.5
                    else:
                        if row['high'] >= pos['sl']:
                            trade_closed = True
                            exit_pnl = -BASE_RISK
                        elif row['low'] <= pos['tp']:
                            trade_closed = True
                            exit_pnl = BASE_RISK * 1.5

                elif mode == "VIRTUAL_RATCHET":
                    if direction == 'LONG':
                        if cur_close <= pos['virtual_sl']:
                            trade_closed = True
                            exit_pnl = (pos['virtual_sl'] - entry_px) * units
                        elif asset_type == "FOREX" and cur_close >= bb_mid and mfe > 0:
                            trade_closed = True
                            exit_pnl = (cur_close - entry_px) * units
                        else:
                            if r_mult >= 2.0:
                                pos['virtual_sl'] = max(pos['virtual_sl'], ema21 - (0.5 * atr))
                            elif r_mult >= 1.2:
                                pos['virtual_sl'] = max(pos['virtual_sl'], entry_px + (0.2 * atr))
                    else:
                        if cur_close >= pos['virtual_sl']:
                            trade_closed = True
                            exit_pnl = (entry_px - pos['virtual_sl']) * units
                        elif asset_type == "FOREX" and cur_close <= bb_mid and mfe > 0:
                            trade_closed = True
                            exit_pnl = (entry_px - cur_close) * units
                        else:
                            if r_mult >= 2.0:
                                pos['virtual_sl'] = min(pos['virtual_sl'], ema21 + (0.5 * atr))
                            elif r_mult >= 1.2:
                                pos['virtual_sl'] = min(pos['virtual_sl'], entry_px - (0.2 * atr))

                if trade_closed:
                    results[mode]['pnl'] += exit_pnl
                    results[mode]['trades'] += 1
                    if exit_pnl > 0:
                        results[mode]['wins'] += 1
                    else:
                        results[mode]['losses'] += 1
                    pos = None

            if pos is None and (buy_sig or sell_sig):
                direction = 'LONG' if buy_sig else 'SHORT'
                entry_px = row['close']
                risk_dist = max(1.5 * row['atr'], entry_px * 0.005)
                units = BASE_RISK / risk_dist

                pos = {
                    "direction": direction,
                    "entry": entry_px,
                    "risk_dist": risk_dist,
                    "units": units,
                    "sl": entry_px - risk_dist if direction == 'LONG' else entry_px + risk_dist,
                    "tp": entry_px + (1.5 * risk_dist) if direction == 'LONG' else entry_px - (1.5 * risk_dist),
                    "virtual_sl": entry_px - risk_dist if direction == 'LONG' else entry_px + risk_dist
                }

    return results

def load_resampled_daily(sym: str) -> pd.DataFrame:
    p_large = LARGE_DIR / f"{sym}_1MIN_SIP_2017_2026.parquet"
    p_mid = MID_DIR / f"{sym}_1MIN_SIP.parquet"
    p = p_large if p_large.exists() else p_mid
    if not p.exists():
        return pd.DataFrame()

    df = pd.read_parquet(p)
    df["time"] = pd.to_datetime(df["time"], utc=True)
    df["time_et"] = df["time"].dt.tz_convert("America/New_York")

    daily = df.groupby("date").agg({
        "open": "first", "high": "max", "low": "min", "close": "last", "volume": "sum"
    }).reset_index().sort_values("date").reset_index(drop=True)

    return daily

def run_oos():
    print("=" * 80)
    print("STAN WEINSTEIN STAGE 2 — OUT-OF-SAMPLE TEST (2025-2026)")
    print("=" * 80)

    all_trades = []
    slip_pct = 0.0015
    fee_pct = 0.0002

    for sym in SYMBOLS:
        df = load_resampled_daily(sym)
        if len(df) < 160:
            continue

        df["sma150"] = df["close"].rolling(150).mean()
        df["sma150_slope"] = df["sma150"] - df["sma150"].shift(10)
        df["sma20"] = df["close"].rolling(20).mean()
        df["vol_sma50"] = df["volume"].rolling(50).mean()
        df["high_20d"] = df["high"].rolling(20).max().shift(1)
        df["low_20d"] = df["low"].rolling(20).min().shift(1)

        c = df["close"].values
        h = df["high"].values
        l = df["low"].values
        tr = np.maximum(h - l, np.maximum(np.abs(h - np.roll(c, 1)), np.abs(l - np.roll(c, 1))))
        tr[0] = h[0] - l[0]
        df["atr"] = pd.Series(tr).rolling(14).mean()

        in_pos = False
        entry_px = 0.0
        stop_px = 0.0
        target_px = 0.0
        risk = 0.0
        entry_idx = 0

        for i in range(150, len(df) - 1):
            row = df.iloc[i]
            next_row = df.iloc[i + 1]

            if not in_pos:
                # Stage 2 criteria
                is_stage2 = (row["close"] > row["sma150"]) and (row["sma150_slope"] > 0)
                vol_burst = row["volume"] > (row["vol_sma50"] * 1.3)
                breakout = row["close"] > row["high_20d"]

                if is_stage2 and vol_burst and breakout:
                    # Check if entry is within the OOS window
                    entry_date = next_row["date"]
                    entry_px = next_row["open"] * (1.0 + slip_pct)
                    risk = max(row["atr"] * 1.8, entry_px - row["low_20d"])
                    stop_px = entry_px - risk
                    target_px = entry_px + 3.0 * risk
                    in_pos = True
                    entry_idx = i + 1

            else:
                days_held = (i + 1) - entry_idx
                trade_recorded = False

                if next_row["low"] <= stop_px:
                    exit_px = stop_px * (1.0 - slip_pct)
                    pnl = (exit_px - entry_px) - (fee_pct * entry_px)
                    trade_recorded = True
                    hit = "STOP"
                elif next_row["high"] >= target_px:
                    exit_px = target_px * (1.0 - slip_pct)
                    pnl = (exit_px - entry_px) - (fee_pct * entry_px)
                    trade_recorded = True
                    hit = "TARGET"
                elif row["close"] < row["sma20"] and (row["close"] > entry_px):
                    exit_px = next_row["open"] * (1.0 - slip_pct)
                    pnl = (exit_px - entry_px) - (fee_pct * entry_px)
                    trade_recorded = True
                    hit = "TRAIL_20SMA"
                elif row["high"] >= entry_px + 1.5 * risk:
                    stop_px = max(stop_px, entry_px)

                if trade_recorded:
                    # Only log trades that initiated in OOS period (2025-2026)
                    if df.iloc[entry_idx]["date"] >= OOS_START_DATE:
                        all_trades.append({
                            "symbol": sym, "pnl_r": pnl / risk, "win": 1 if pnl > 0 else 0,
                            "hit": hit, "days": days_held
                        })
                    in_pos = False

    if not all_trades:
        print("No OOS trades met criteria.")
        return

    res = pd.DataFrame(all_trades)
    tot_trades = len(res)
    wr = res["win"].mean() * 100.0
    tot_r = res["pnl_r"].sum()
    exp_r = res["pnl_r"].mean()

    losses = abs(res[res["pnl_r"] < 0]["pnl_r"].sum())
    pf = round(wins / losses if losses > 0 else 0, 2)

    peak = np.maximum.accumulate(cum_r)
    dd = peak - cum_r
    max_dd = round(np.max(dd), 1) if len(dd) > 0 else 0.0

def signal_velez_sector_topdown(df: pd.DataFrame, idx: int, pos: Optional[Position], spy_df: pd.DataFrame) -> dict:
    """Hybrid A: Velez + Sector Relative Strength Filter"""
    row = df.iloc[idx]
    date = df.index[idx]

    if date not in spy_df.index or idx < 20:
        return {"signal": "HOLD"}

    if pd.isna(row["sma200"]) or pd.isna(row["ema20"]) or pd.isna(row["atr14"]):
        return {"signal": "HOLD"}

    stock_base = float(df["close"].iloc[idx - 20])
    spy_hist = spy_df.loc[:date]
    if len(spy_hist) < 21:
        return {"signal": "HOLD"}

    spy_now = float(spy_hist["close"].iloc[-1])
    spy_base = float(spy_hist["close"].iloc[-21])
    if stock_base <= 0 or spy_base <= 0:
        return {"signal": "HOLD"}

    stock_perf_20 = (float(row["close"]) / stock_base) - 1.0
    spy_perf_20 = (spy_now / spy_base) - 1.0

    if pos is None:
        if (
            row["close"] > row["sma200"]
            and row["close"] > row["ema20"]
            and abs(row["low"] - row["ema20"]) / row["ema20"] < 0.015
            and row["close"] > row["open"]
            and row["atr_ratio"] <= 0.055
            and stock_perf_20 > (spy_perf_20 + 0.03)
        ):
            return {
                "signal": "BUY",
                "stop": float(row["low"]) - 0.5 * float(row["atr14"]),
                "target_r": 3.0,
            }
    return {"signal": "HOLD"}

def run_hybrid_backtest(name: str, fn: Callable, dfs: Dict[str, pd.DataFrame]) -> Tuple[dict, List, pd.DataFrame]:
    spy_df = dfs[BENCHMARK]
    common_dates = sorted(set.intersection(*[set(dfs[s].index) for s in UNIVERSE + [BENCHMARK]]))
    needs_spy = len(inspect.signature(fn).parameters) == 4

    realized_nav = INITIAL_NAV
    open_positions: Dict[str, Position] = {}
    trades: List = []
    equity_rows: List[dict] = []

    def call_signal(df, idx, pos):
        return fn(df, idx, pos, spy_df) if needs_spy else fn(df, idx, pos)

    for t_idx in range(199, len(common_dates) - 1):
        curr_date = common_dates[t_idx]
        next_date = common_dates[t_idx + 1]

        # 1. Evaluate at bar t close
        exit_intents: Dict[str, str] = {}
        for sym, pos in list(open_positions.items()):
            df = dfs[sym]
            idx = df.index.get_loc(curr_date)
            res = call_signal(df, idx, pos)
            if res.get("signal") == "EXIT":
                exit_intents[sym] = res.get("reason", "SIGNAL_EXIT")

        entry_intents: List[Tuple[str, dict]] = []
        for sym in UNIVERSE:
            if sym in open_positions:
                continue
            df = dfs[sym]
            idx = df.index.get_loc(curr_date)
            res = call_signal(df, idx, None)
            if res.get("signal") in ("BUY", "SELL"):
                entry_intents.append((sym, res))

        # 2. t+1 Open: Gap exits & signal exits
        for sym in list(open_positions.keys()):
            pos = open_positions[sym]
            row = dfs[sym].loc[next_date]
            gap_event = check_open_gap(pos, row)
            if gap_event:
                px, reason = gap_event
                t, nav_chg = close_trade(name, pos, next_date, px, reason)
                realized_nav += nav_chg
                trades.append(t)
                del open_positions[sym]

        for sym, reason in exit_intents.items():
            if sym in open_positions:
                pos = open_positions[sym]
                row = dfs[sym].loc[next_date]
                raw_open = float(row["open"])
                px = raw_open * (1.0 - SLIPPAGE_PCT) if pos.side == "BUY" else raw_open * (1.0 + SLIPPAGE_PCT)
                t, nav_chg = close_trade(name, pos, next_date, px, reason)
                realized_nav += nav_chg
                trades.append(t)
                del open_positions[sym]

        # 3. t+1 Open: Execute new entries
        for sym, res in entry_intents:
            if len(open_positions) >= MAX_POSITIONS_PER_EXPERT or sym in open_positions:
                continue

            row = dfs[sym].loc[next_date]
            side = res["signal"]
            raw_open = float(row["open"])
            entry_fill = raw_open * (1.0 + SLIPPAGE_PCT) if side == "BUY" else raw_open * (1.0 - SLIPPAGE_PCT)
            initial_stop = float(res["stop"])

            if (side == "BUY" and initial_stop >= entry_fill) or (side == "SELL" and initial_stop <= entry_fill):
                continue

            risk_per_share = abs(entry_fill - initial_stop)
            if not np.isfinite(risk_per_share) or risk_per_share <= 0:
                continue

            nav_open = calculate_mtm_nav(realized_nav, open_positions, dfs, next_date, price_col="open")
            risk_dollars = nav_open * RISK_PER_TRADE_PCT
            qty = int(risk_dollars / risk_per_share)
            if qty <= 0:
                continue

            entry_fees = sell_regulatory_fees(entry_fill, qty) if side == "SELL" else 0.0
            realized_nav -= entry_fees

            target_r = res.get("target_r")
            tp = None
            if target_r is not None:
                tp = entry_fill + (float(target_r) * risk_per_share) if side == "BUY" else entry_fill - (float(target_r) * risk_per_share)

            open_positions[sym] = Position(
                symbol=sym,
                side=side,
                entry_date=next_date,
                entry_price=entry_fill,
                initial_stop=initial_stop,
                current_stop=initial_stop,
                initial_risk_per_share=risk_per_share,
                qty=qty,
                tp=tp,
                entry_fees=entry_fees,
            )

        # 4. t+1 Intraday: Conservative stop-first
        for sym in list(open_positions.keys()):
            pos = open_positions[sym]
            row = dfs[sym].loc[next_date]
            event = check_intraday_ohlc(pos, row)
            if event:
                px, reason = event
                t, nav_chg = close_trade(name, pos, next_date, px, reason)
                realized_nav += nav_chg
                trades.append(t)
                del open_positions[sym]

        # 5. MTM Close
        mtm = calculate_mtm_nav(realized_nav, open_positions, dfs, next_date, price_col="close")
        equity_rows.append({"date": next_date, "realized_nav": realized_nav, "mtm_nav": mtm, "positions": len(open_positions)})

    # Final Close Liquidation
    final_date = common_dates[-1]
    for sym in list(open_positions.keys()):
        pos = open_positions[sym]
        raw_close = float(dfs[sym].loc[final_date, "close"])
        px = raw_close * (1.0 - SLIPPAGE_PCT) if pos.side == "BUY" else raw_close * (1.0 + SLIPPAGE_PCT)
        t, nav_chg = close_trade(name, pos, final_date, px, "END_OF_TEST")
        realized_nav += nav_chg
        trades.append(t)

    equity_df = pd.DataFrame(equity_rows).drop_duplicates(subset=["date"], keep="last").reset_index(drop=True)

    pnls = np.array([t.pnl for t in trades], dtype=float)
    rvals = np.array([t.r_mult for t in trades], dtype=float)

    wins = pnls[pnls > 0].sum()
    losses = -pnls[pnls < 0].sum()
    pf = (wins / losses) if losses > 0 else (99.0 if wins > 0 else 0.0)
    win_rate = float((pnls > 0).mean() * 100.0) if len(pnls) else 0.0
    exp_r = float(np.mean(rvals)) if len(rvals) else 0.0
    med_r = float(np.median(rvals)) if len(rvals) else 0.0

    eq = equity_df["mtm_nav"].to_numpy()
    if len(eq):
        peaks = np.maximum.accumulate(eq)
        drawdowns = (peaks - eq) / np.where(peaks != 0, peaks, np.nan)
        max_dd = float(np.nanmax(drawdowns) * 100.0)
    else:
        max_dd = 0.0

    total_net = realized_nav - INITIAL_NAV
    total_ret = (realized_nav / INITIAL_NAV - 1.0) * 100.0
    passes_gate = bool(len(trades) >= 15 and pf >= 1.40 and exp_r >= 0.15 and max_dd <= 8.0)

    metrics = {
        "trades": len(trades),
        "win_rate": win_rate,
        "profit_factor": pf,
        "expectancy_r": exp_r,
        "median_r": med_r,
        "net_pnl": total_net,
        "return_pct": total_ret,
        "max_drawdown": max_dd,
        "final_nav": realized_nav,
        "pass_phase1": passes_gate,
    }

    return metrics, trades, equity_df

    equity_curve = [equity]

    for dt in all_dates:

            pos = open_positions[sym]
            df = data[sym]

                continue

            bar = df.loc[dt]
            pos['days_held'] += 1

            if bar['Low'] <= pos['stop_loss']:
                exit_price = pos['stop_loss']
                pnl = pos['units'] * (exit_price - pos['entry_price'])
                equity += pnl
                equity_curve.append(equity)
                completed_trades.append({
                    "symbol": sym, "type": pos['type'], "pnl": pnl,
                    "r_mult": (exit_price - pos['entry_price']) / pos['risk_per_unit'],
                    "days": pos['days_held'], "exit": "STOP_LOSS"
                })
                del open_positions[sym]
                continue

            if not pos['scaled'] and bar['High'] >= pos['take_profit']:
                scaled_units = pos['units'] * 0.5
                pnl_half = scaled_units * (pos['take_profit'] - pos['entry_price'])
                equity += pnl_half
                pos['units'] -= scaled_units
                pos['scaled'] = True
                pos['stop_loss'] = pos['entry_price']

            if pos['scaled'] and bar['Close'] < bar['sma20']:
                exit_price = bar['Close']
                pnl_rem = pos['units'] * (exit_price - pos['entry_price'])
                equity += pnl_rem
                equity_curve.append(equity)
                completed_trades.append({
                    "symbol": sym, "type": pos['type'], "pnl": pnl_rem,
                    "r_mult": (exit_price - pos['entry_price']) / pos['risk_per_unit'],
                    "days": pos['days_held'], "exit": "TRAILED_20SMA"
                })
                del open_positions[sym]

        if len(open_positions) < MAX_CONCURRENT:
            for sym in SYMBOLS:
                if sym in open_positions or sym not in data:
                    continue
                if len(open_positions) >= MAX_CONCURRENT:
                    break

                df = data[sym]
                if dt not in df.index:
                    continue

                pos_idx = df.index.get_loc(dt)
                if pos_idx < 1 or pos_idx >= len(df) - 1:
                    continue

                curr, prev = df.iloc[pos_idx], df.iloc[pos_idx - 1]
                setup_type, stop_price = evaluate_signal(curr, prev)

                    next_open = df.iloc[pos_idx + 1]['Open']
                    risk_per_unit = next_open - stop_price
                    if risk_per_unit <= 0:
                        continue

                    units = int(risk_dollars / risk_price) if 'risk_price' in locals() else int(risk_dollars / risk_per_unit)
