#!/usr/bin/env python3
from __future__ import annotations

import os
import shutil
import time
from pathlib import Path
from typing import Any

#############################################
# CONFIG
#############################################

ROOT = Path.home() / "bots" / "oanda_fifo_practice" / "MarchEdge"
LOG_DIR = ROOT / "logs"
ARCHIVE_DIR = ROOT / "storage" / "archive"

MAX_LOG_SIZE_MB = 25
MAX_ARCHIVE_FILES = 20
MIN_FREE_MB = 750

ROTATE_TARGETS = [
    LOG_DIR / "marchedge_bets.jsonl",
    LOG_DIR / "marchedge_results.jsonl",
    LOG_DIR / "marchedge_bets.csv",
    LOG_DIR / "review_report.json",
    LOG_DIR / "review_report.csv",
]

#############################################
# HELPERS
#############################################

def bytes_to_mb(num_bytes: int) -> float:
    return round(num_bytes / (1024 * 1024), 2)


def get_free_mb(path: Path) -> float:
    usage = shutil.disk_usage(path)
    return bytes_to_mb(usage.free)


def ensure_dirs() -> None:
    LOG_DIR.mkdir(parents=True, exist_ok=True)
    ARCHIVE_DIR.mkdir(parents=True, exist_ok=True)


def file_size_mb(path: Path) -> float:
    if not path.exists():
        return 0.0
    return bytes_to_mb(path.stat().st_size)


def timestamp_suffix() -> str:
    return time.strftime("%Y%m%d_%H%M%S")


#############################################
# ROTATION
#############################################

def rotate_file(path: Path) -> bool:
    if not path.exists():
        return False

    if file_size_mb(path) < MAX_LOG_SIZE_MB:
        return False

    archive_name = f"{path.stem}_{timestamp_suffix()}{path.suffix}"
    archive_path = ARCHIVE_DIR / archive_name
    shutil.move(str(path), str(archive_path))
    path.touch()

    return True


def prune_archive() -> int:
    files = sorted(
        [p for p in ARCHIVE_DIR.iterdir() if p.is_file()],
        key=lambda p: p.stat().st_mtime,
        reverse=True,
    )

    removed = 0
    for old_file in files[MAX_ARCHIVE_FILES:]:
        try:
            old_file.unlink()
            removed += 1
        except Exception:
            pass

    return removed


#############################################
# EMERGENCY CLEANUP
#############################################

def emergency_cleanup() -> dict[str, Any]:
    removed = []

    for target in ROTATE_TARGETS:
        if target.exists():
            try:
                target.unlink()
                removed.append(str(target))
            except Exception:
                pass

    return {
        "removed_files": removed,
        "free_mb_after": get_free_mb(ROOT),
    }


#############################################
# MAIN ENTRY
#############################################

def run_storage_manager() -> dict[str, Any]:
    ensure_dirs()

    free_before = get_free_mb(ROOT)
    rotated = []

    for target in ROTATE_TARGETS:
        if rotate_file(target):
            rotated.append(str(target))

    pruned = prune_archive()

    free_after_rotation = get_free_mb(ROOT)

    emergency = None
    if free_after_rotation < MIN_FREE_MB:
        emergency = emergency_cleanup()

    return {
        "free_mb_before": free_before,
        "rotated_files": rotated,
        "archive_pruned": pruned,
        "free_mb_after_rotation": free_after_rotation,
        "emergency_cleanup": emergency,
        "final_free_mb": get_free_mb(ROOT),
    }


if __name__ == "__main__":
    result = run_storage_manager()

    print("STORAGE MANAGER")
    for k, v in result.items():
        print(f"{k}: {v}")

#!/usr/bin/env python3
from __future__ import annotations

import json

from team_props_policy import enforce_team_props_policy

SNAPSHOT_LOG = "bet_snapshots.jsonl"


def _to_float(v: Any, default: float = 0.0) -> float:
    try:
        if v is None:
            return default
        if isinstance(v, str):
            v = v.strip()
            if not v:
                return default
            if v.endswith("%"):
                return float(v[:-1]) / 100.0
        return float(v)
    except Exception:
        return default


def _clamp(x: float, low: float, high: float) -> float:
    return max(low, min(x, high))


def _strength(conf: float) -> str:
    if conf >= 0.88:
        return "A"
    if conf >= 0.75:
        return "B"
    return "C"


def _kelly(edge: float, conf: float) -> float:
    raw = (edge * 0.20) + max(0.0, conf - 0.50) * 0.05
    return round(_clamp(raw, 0.0, 0.08), 4)


def _game_from_payload(base_payload: Dict[str, Any]) -> str:
    return f"{base_payload.get('away_team', '')} @ {base_payload.get('home_team', '')}".strip()


def _remaining_seconds(base_payload: Dict[str, Any]) -> int:
    minutes = int(_to_float(base_payload.get("minutes_left", 0), 0))
    seconds = int(_to_float(base_payload.get("seconds_left", 0), 0))
    return max(0, minutes * 60 + seconds)


def _prop_confidence(diff: float, remaining_seconds: int, suspicious: bool) -> float:
    conf = 0.65 + min(diff / 18.0, 0.25)

    if remaining_seconds <= 120:
        conf -= 0.10
    elif remaining_seconds <= 300:
        conf -= 0.05

    if suspicious:
        conf -= 0.08

    return round(_clamp(conf, 0.50, 0.92), 4)


def _build_team_prop(
    *,
    game: str,
    team: str,
    side: str,
    line: float,
    proj: float,
    confidence: float,
    suspicious: bool,
) -> Dict[str, Any]:
    diff = abs(proj - line)
    edge = round(_clamp(diff / 18.0, 0.0, 0.35), 4)

    strength = _strength(confidence)
    kelly_q = _kelly(edge, confidence)
    edge_pct = round(edge * 100.0, 2)

    summary = (
        f"{team} team total {side} {round(line, 2)} | "
        f"proj={round(proj, 2)} | edge={edge_pct}% | "
        f"conf={round(confidence * 100.0, 1)}% | "
        f"suspicious={suspicious} | strength={strength} | "
        f"kelly_q={kelly_q}"
    )

    return {
        "game": game,
        "market": "team_total",
        "team": team,
        "side": side,
        "line": round(line, 2),
        "proj": round(proj, 2),
        "projection": round(proj, 2),
        "projected_team_total": round(proj, 2),
        "edge": edge,
        "display_edge": edge_pct,
        "display_edge_text": f"{edge_pct}%",
        "confidence": round(confidence, 4),
        "strength": strength,
        "kelly_quarter": kelly_q,
        "suspicious": suspicious,
        "score": round((edge * 100.0) + (confidence * 10.0) - (8.0 if suspicious else 0.0), 4),
        "summary": summary,
    }


def _candidate_props(base_payload: Dict[str, Any], market_payload: Dict[str, Any]) -> List[Dict[str, Any]]:
    game = _game_from_payload(base_payload)

    home_team = str(base_payload.get("home_team", "")).strip()
    away_team = str(base_payload.get("away_team", "")).strip()

    projected_home = _to_float(
        market_payload.get("projected_home", base_payload.get("projected_home", 0.0)),
        0.0,
    )
    projected_away = _to_float(
        market_payload.get("projected_away", base_payload.get("projected_away", 0.0)),
        0.0,
    )

    home_line = _to_float(market_payload.get("home_team_total_line", 0.0), 0.0)
    away_line = _to_float(market_payload.get("away_team_total_line", 0.0), 0.0)

    suspicious = bool(
        base_payload.get("suspicious_any", False)
        or market_payload.get("suspicious_any", False)
    )
    remaining = _remaining_seconds(base_payload)

    props: List[Dict[str, Any]] = []

    if home_team and home_line > 0 and projected_home > 0:
        diff = abs(projected_home - home_line)
        conf = _prop_confidence(diff, remaining, suspicious)
        side = "over" if projected_home > home_line else "under"
        props.append(
            _build_team_prop(
                game=game,
                team=home_team,
                side=side,
                line=home_line,
                proj=projected_home,
                confidence=conf,
                suspicious=suspicious,
            )
        )

    if away_team and away_line > 0 and projected_away > 0:
        diff = abs(projected_away - away_line)
        conf = _prop_confidence(diff, remaining, suspicious)
        side = "over" if projected_away > away_line else "under"
        props.append(
            _build_team_prop(
                game=game,
                team=away_team,
                side=side,
                line=away_line,
                proj=projected_away,
                confidence=conf,
                suspicious=suspicious,
            )
        )

    return props


def _append_snapshot(base_payload: Dict[str, Any], top_bets: List[Dict[str, Any]]) -> None:
    row = {
        "timestamp": int(time.time()),
        "home_team": base_payload.get("home_team"),
        "away_team": base_payload.get("away_team"),
        "top_bets": top_bets,
    }
    with open(SNAPSHOT_LOG, "a", encoding="utf-8") as f:
        f.write(json.dumps(row) + "\n")


def run_team_props_bridge(
    base_payload: Dict[str, Any],
    market_payload: Dict[str, Any],
    write_logs: bool = False,
) -> Dict[str, Any]:
    candidates = _candidate_props(base_payload, market_payload)

    policy_input = {"best_team_props": candidates}
    policy_output = enforce_team_props_policy(policy_input, payload=base_payload)
    best_team_props = policy_output.get("best_team_props", [])

    if write_logs and best_team_props:
        _append_snapshot(base_payload, best_team_props)

    return {
        "candidate_team_props": candidates,
        "policy_output": policy_output,
        "best_team_props": best_team_props,
        "write_logs": write_logs,
    }

#!/usr/bin/env python3
from __future__ import annotations

from dataclasses import dataclass, asdict
from typing import Any


#############################################
# HELPERS
#############################################

def clamp(value: float, low: float, high: float) -> float:
    return max(low, min(high, value))


def to_float(value: Any, default: float = 0.0) -> float:
    try:
        return float(value)
    except Exception:
        return default


def to_int(value: Any, default: int = -110) -> int:
    try:
        return int(value)
    except Exception:
        return default


def implied_prob_from_american(odds: int) -> float:
    if odds > 0:
        return 100.0 / (odds + 100.0)
    return abs(odds) / (abs(odds) + 100.0)


def american_to_b(odds: int) -> float:
    if odds > 0:
        return odds / 100.0
    return 100.0 / abs(odds)


def kelly_fraction(model_prob: float, odds: int, fraction: float = 0.25) -> float:
    p = clamp(model_prob, 0.001, 0.999)
    q = 1.0 - p
    b = american_to_b(odds)
    full_kelly = ((b * p) - q) / b
    return round(max(0.0, full_kelly) * fraction, 4)

#############################################

@dataclass
class TeamPropView:
    market: str
    team: str
    side: str
    line: float
    odds: int
    projected_final: float
    model_prob: float
    implied_prob: float
    edge: float
    confidence: float
    kelly_quarter: float
    summary: str

def infer_game_context(payload: dict[str, Any]) -> dict[str, Any]:
    minutes_left = int(payload.get("minutes_left", 0))
    seconds_left = int(payload.get("seconds_left", 0))
    mode = str(payload.get("mode", "2h")).lower().strip()

    remaining_seconds = (minutes_left * 60) + seconds_left

    if mode == "1h":
        phase = "1h"
    else:
        phase = "2h"

    # rough volatility scale
    if remaining_seconds >= 15 * 60:
        sigma = 7.5
    elif remaining_seconds >= 10 * 60:
        sigma = 6.5
    elif remaining_seconds >= 6 * 60:
        sigma = 5.0
    elif remaining_seconds >= 3 * 60:
        sigma = 4.0
    else:
        sigma = 3.0

    return {
        "remaining_seconds": remaining_seconds,
        "phase": phase,
        "sigma": sigma,
    }


def estimate_team_final_scores(payload: dict[str, Any]) -> tuple[float, float]:
    """
    Priority:
    1. Use projected_home / projected_away if present
    2. Use projected_total + current score share
    3. Fallback to current score
    """
    home_score = to_float(payload.get("home_score"), 0.0)
    away_score = to_float(payload.get("away_score"), 0.0)

    projected_home = payload.get("projected_home")
    projected_away = payload.get("projected_away")

    if projected_home is not None and projected_away is not None:
        return round(to_float(projected_home), 2), round(to_float(projected_away), 2)

    projected_total = payload.get("projected_total")
    if projected_total is not None:
        projected_total = to_float(projected_total, home_score + away_score)
        current_total = max(1.0, home_score + away_score)

        home_share = home_score / current_total
        away_share = away_score / current_total

        # slight shrink toward 50/50 so early-game splits do not explode
        home_share = (home_share * 0.85) + 0.075
        away_share = 1.0 - home_share

        return round(projected_total * home_share, 2), round(projected_total * away_share, 2)

    return round(home_score, 2), round(away_score, 2)


def team_total_prob(projected_final: float, market_line: float, sigma: float) -> float:
    """
    Linearized probability map around team total line.
    """
    diff = projected_final - market_line
    raw = 0.50 + (diff / (2.0 * sigma))
    return round(clamp(raw, 0.03, 0.97), 4)


def infer_confidence(edge: float, projected_final: float, market_line: float) -> float:
    diff = abs(projected_final - market_line)
    conf = 50.0 + (edge * 280.0) + (min(diff, 10.0) * 1.5)
    return round(clamp(conf, 50.0, 95.0), 1)


def build_team_prop_summary(
    team: str,
    side: str,
    line: float,
    projected_final: float,
    edge: float,
    confidence: float,
) -> str:
    return (
        f"{team} team total {side} {line}"
        f" | proj={projected_final:.2f}"
        f" | edge={edge:.4f}"
        f" | conf={confidence:.1f}%"
    )

def evaluate_team_props(payload: dict[str, Any]) -> dict[str, Any]:
    home_team = str(payload.get("home_team", "Home"))
    away_team = str(payload.get("away_team", "Away"))

    context = infer_game_context(payload)
    sigma = context["sigma"]

    projected_home, projected_away = estimate_team_final_scores(payload)

    home_team_total_line = to_float(payload.get("home_team_total_line"), 0.0)
    away_team_total_line = to_float(payload.get("away_team_total_line"), 0.0)

    home_over_odds = to_int(payload.get("home_team_total_over_odds"), -110)
    home_under_odds = to_int(payload.get("home_team_total_under_odds"), -110)
    away_over_odds = to_int(payload.get("away_team_total_over_odds"), -110)
    away_under_odds = to_int(payload.get("away_team_total_under_odds"), -110)

    home_over_prob = team_total_prob(projected_home, home_team_total_line, sigma)
    home_under_prob = round(1.0 - home_over_prob, 4)

    away_over_prob = team_total_prob(projected_away, away_team_total_line, sigma)
    away_under_prob = round(1.0 - away_over_prob, 4)

    props: list[TeamPropView] = []

    # Home over
    home_over_implied = implied_prob_from_american(home_over_odds)
    home_over_edge = round(home_over_prob - home_over_implied, 4)
    home_over_conf = infer_confidence(home_over_edge, projected_home, home_team_total_line)
    props.append(
        TeamPropView(
            market="team_total",
            team=home_team,
            side="over",
            line=home_team_total_line,
            odds=home_over_odds,
            projected_final=projected_home,
            model_prob=home_over_prob,
            implied_prob=home_over_implied,
            edge=home_over_edge,
            confidence=home_over_conf,
            kelly_quarter=kelly_fraction(home_over_prob, home_over_odds),
            summary=build_team_prop_summary(home_team, "over", home_team_total_line, projected_home, home_over_edge, home_over_conf),
        )
    )

    # Home under
    home_under_implied = implied_prob_from_american(home_under_odds)
    home_under_edge = round(home_under_prob - home_under_implied, 4)
    home_under_conf = infer_confidence(home_under_edge, projected_home, home_team_total_line)
    props.append(
        TeamPropView(
            market="team_total",
            team=home_team,
            side="under",
            line=home_team_total_line,
            odds=home_under_odds,
            projected_final=projected_home,
            model_prob=home_under_prob,
            implied_prob=home_under_implied,
            edge=home_under_edge,
            confidence=home_under_conf,
            kelly_quarter=kelly_fraction(home_under_prob, home_under_odds),
            summary=build_team_prop_summary(home_team, "under", home_team_total_line, projected_home, home_under_edge, home_under_conf),
        )
    )

    # Away over
    away_over_implied = implied_prob_from_american(away_over_odds)
    away_over_edge = round(away_over_prob - away_over_implied, 4)
    away_over_conf = infer_confidence(away_over_edge, projected_away, away_team_total_line)
    props.append(
        TeamPropView(
            market="team_total",
            team=away_team,
            side="over",
            line=away_team_total_line,
            odds=away_over_odds,
            projected_final=projected_away,
            model_prob=away_over_prob,
            implied_prob=away_over_implied,
            edge=away_over_edge,
            confidence=away_over_conf,
            kelly_quarter=kelly_fraction(away_over_prob, away_over_odds),
            summary=build_team_prop_summary(away_team, "over", away_team_total_line, projected_away, away_over_edge, away_over_conf),
        )
    )

    # Away under
    away_under_implied = implied_prob_from_american(away_under_odds)
    away_under_edge = round(away_under_prob - away_under_implied, 4)
    away_under_conf = infer_confidence(away_under_edge, projected_away, away_team_total_line)
    props.append(
        TeamPropView(
            market="team_total",
            team=away_team,
            side="under",
            line=away_team_total_line,
            odds=away_under_odds,
            projected_final=projected_away,
            model_prob=away_under_prob,
            implied_prob=away_under_implied,
            edge=away_under_edge,
            confidence=away_under_conf,
            kelly_quarter=kelly_fraction(away_under_prob, away_under_odds),
            summary=build_team_prop_summary(away_team, "under", away_team_total_line, projected_away, away_under_edge, away_under_conf),
        )
    )

    ranked = sorted(props, key=lambda x: x.edge, reverse=True)

    best_props = []
    for prop in ranked:
        if prop.edge < 0.03:
            continue
        best_props.append(asdict(prop))

    return {
        "home_team": home_team,
        "away_team": away_team,
        "projected_home_final": round(projected_home, 2),
        "projected_away_final": round(projected_away, 2),
        "home_team_total_line": home_team_total_line,
        "away_team_total_line": away_team_total_line,
        "home_over_prob": home_over_prob,
        "home_under_prob": home_under_prob,
        "away_over_prob": away_over_prob,
        "away_under_prob": away_under_prob,
        "all_team_props": [asdict(p) for p in ranked],
        "best_team_props": best_props,
    }

def evaluate_team_props_v2(payload: dict[str, Any]) -> dict[str, Any]:
    home_team = str(payload.get("home_team", "Home"))
    away_team = str(payload.get("away_team", "Away"))

    context = infer_game_context(payload)
    sigma = context["sigma"]

    projected_home_raw, projected_away_raw = estimate_team_final_scores(payload)

    home_team_total_line = to_float(payload.get("home_team_total_line"), 0.0)
    away_team_total_line = to_float(payload.get("away_team_total_line"), 0.0)

    home_projected_cal, home_gap, home_anchored = anchor_team_total_projection(projected_home_raw, home_team_total_line)
    away_projected_cal, away_gap, away_anchored = anchor_team_total_projection(projected_away_raw, away_team_total_line)

    home_over_odds = to_int(payload.get("home_team_total_over_odds"), -110)
    home_under_odds = to_int(payload.get("home_team_total_under_odds"), -110)
    away_over_odds = to_int(payload.get("away_team_total_over_odds"), -110)
    away_under_odds = to_int(payload.get("away_team_total_under_odds"), -110)

    props: list[TeamPropView] = []

    props.append(
        build_prop(
            team=home_team,
            side="over",
            line=home_team_total_line,
            odds=home_over_odds,
            projected_final_raw=projected_home_raw,
            projected_final_calibrated=home_projected_cal,
            sigma=sigma,
        )
    )
    props.append(
        build_prop(
            team=home_team,
            side="under",
            line=home_team_total_line,
            odds=home_under_odds,
            projected_final_raw=projected_home_raw,
            projected_final_calibrated=home_projected_cal,
            sigma=sigma,
        )
    )
    props.append(
        build_prop(
            team=away_team,
            side="over",
            line=away_team_total_line,
            odds=away_over_odds,
            projected_final_raw=projected_away_raw,
            projected_final_calibrated=away_projected_cal,
            sigma=sigma,
        )
    )
    props.append(
        build_prop(
            team=away_team,
            side="under",
            line=away_team_total_line,
            odds=away_under_odds,
            projected_final_raw=projected_away_raw,
            projected_final_calibrated=away_projected_cal,
            sigma=sigma,
        )
    )

    ranked = sorted(props, key=lambda x: x.edge, reverse=True)

    best_props = []
    for prop in ranked:
        if prop.edge < MIN_EDGE_FOR_PROP:
            continue
        if prop.strength == "D":
            continue
        best_props.append(asdict(prop))

    return {
        "home_team": home_team,
        "away_team": away_team,
        "projected_home_final_raw": round(projected_home_raw, 2),
        "projected_away_final_raw": round(projected_away_raw, 2),
        "projected_home_final_calibrated": round(home_projected_cal, 2),
        "projected_away_final_calibrated": round(away_projected_cal, 2),
        "home_team_total_line": home_team_total_line,
        "away_team_total_line": away_team_total_line,
        "home_gap": round(home_gap, 2),
        "away_gap": round(away_gap, 2),
        "home_anchored": home_anchored,
        "away_anchored": away_anchored,
        "all_team_props": [asdict(p) for p in ranked],
        "best_team_props": best_props,
    }

    home_projected_cal, home_gap, home_anchored = anchor_team_total_projection(projected_home_raw, home_team_total_line)
    away_projected_cal, away_gap, away_anchored = anchor_team_total_projection(projected_away_raw, away_team_total_line)

    home_over_odds = to_int(payload.get("home_team_total_over_odds"), -110)
    home_under_odds = to_int(payload.get("home_team_total_under_odds"), -110)
    away_over_odds = to_int(payload.get("away_team_total_over_odds"), -110)
    away_under_odds = to_int(payload.get("away_team_total_under_odds"), -110)

    props: list[TeamPropView] = []

    props.append(build_prop(home_team, "over", home_team_total_line, home_over_odds, projected_home_raw, home_projected_cal, context))
    props.append(build_prop(home_team, "under", home_team_total_line, home_under_odds, projected_home_raw, home_projected_cal, context))

    props.append(build_prop(away_team, "under", away_team_total_line, away_under_odds, projected_away_raw, away_projected_cal, context))

#!/usr/bin/env python3
from __future__ import annotations

import json
import os
from collections import defaultdict
from typing import Any


POLICY_FILE = "learned_policy.json"

MIN_EDGE_A = float(os.getenv("TP_MIN_EDGE_A", "0.10"))
MIN_EDGE_B = float(os.getenv("TP_MIN_EDGE_B", "0.07"))
MIN_CONF_A = float(os.getenv("TP_MIN_CONF_A", "0.84"))
MIN_CONF_B = float(os.getenv("TP_MIN_CONF_B", "0.72"))

MAX_PROPS_PER_GAME = int(os.getenv("TP_MAX_PROPS_PER_GAME", "2"))
MAX_PROPS_PER_TEAM = int(os.getenv("TP_MAX_PROPS_PER_TEAM", "1"))
MAX_KELLY = float(os.getenv("TP_MAX_KELLY", "0.10"))

BLOCK_CONFLICTING_SIDES = os.getenv("TP_BLOCK_CONFLICTING_SIDES", "true").lower() == "true"
BLOCK_SUSPICIOUS = os.getenv("TP_BLOCK_SUSPICIOUS", "false").lower() == "true"
SOFT_CONF_CAP_A = float(os.getenv("TP_SOFT_CONF_CAP_A", "0.87"))
SOFT_CONF_CAP_B = float(os.getenv("TP_SOFT_CONF_CAP_B", "0.82"))
HARD_CONF_CAP = float(os.getenv("TP_HARD_CONF_CAP", "0.91"))
SMALL_GAP_BLOCK = float(os.getenv("TP_SMALL_GAP_BLOCK", "1.5"))


def load_learned_policy() -> dict[str, Any]:
    if not os.path.exists(POLICY_FILE):
        return {}
    try:
        with open(POLICY_FILE, "r", encoding="utf-8") as f:
            data = json.load(f)
        return data if isinstance(data, dict) else {}
    except Exception:
        return {}


LEARNED_POLICY = load_learned_policy()


def _to_float(v: Any, default: float = 0.0) -> float:
    try:
        if v is None:
            return default
        if isinstance(v, str) and v.endswith("%"):
            return float(v[:-1])
        return float(v)
    except Exception:
        return default


def _conf(v: Any) -> float:
    x = _to_float(v, 0.0)
    if x > 1.0:
        x /= 100.0
    return max(0.0, min(x, 1.0))


def _edge(v: Any) -> float:
    x = _to_float(v, 0.0)
    if x > 1.0:
        x /= 100.0
    return max(0.0, x)


def _norm(v: Any) -> str:
    return " ".join(str(v or "").strip().lower().split())


def _strength(item: dict[str, Any]) -> str:
    return str(item.get("strength", "")).strip().upper()


def _game(item: dict[str, Any], payload: dict[str, Any] | None = None) -> str:
    game = str(item.get("game", "")).strip()
    if game:
        return game
    if payload:
        return f"{payload.get('away_team', '')} @ {payload.get('home_team', '')}".strip()
    home = item.get("home_team", "")
    away = item.get("away_team", "")
    return f"{away} @ {home}".strip()


def _line_key(v: Any) -> str:
    if v is None:
        return "None"
    try:
        return f"{float(v):.3f}"
    except Exception:
        return str(v)


def _summary(item: dict[str, Any], payload: dict[str, Any] | None = None) -> str:
    if item.get("summary"):
        return str(item["summary"])
    game = _game(item, payload)
    team = str(item.get("team", "")).strip()
    side = str(item.get("side", "")).strip()
    line = item.get("line")
    return f"{game} | TEAM_TOTAL | {team} {side} {line}".strip()


def _sort_key(item: dict[str, Any]) -> tuple[float, float, float]:
    return (
        _to_float(item.get("score", 0.0), 0.0),
        _conf(item.get("confidence", 0.0)),
        _edge(item.get("edge", item.get("display_edge", 0.0))),
    )


def _apply_learned_adjustments(item: dict[str, Any]) -> None:
    edge = _edge(item.get("edge", item.get("display_edge", 0.0)))
    conf = _conf(item.get("confidence", 0.0))
    strength = _strength(item)

    if "team_total_penalty" in LEARNED_POLICY:
        edge *= max(0.50, min(_to_float(LEARNED_POLICY["team_total_penalty"], 1.0), 1.25))

    if "team_total_boost" in LEARNED_POLICY:
        edge *= max(0.75, min(_to_float(LEARNED_POLICY["team_total_boost"], 1.0), 1.35))

    scale_key = f"{strength}_confidence_scale"
    if scale_key in LEARNED_POLICY:
        conf *= max(0.80, min(_to_float(LEARNED_POLICY[scale_key], 1.0), 1.10))

    if item.get("suspicious") and "suspicious_penalty" in LEARNED_POLICY:
        conf -= max(0.0, min(_to_float(LEARNED_POLICY["suspicious_penalty"], 0.0), 0.20))

    item["edge"] = round(max(0.0, edge), 4)
    item["confidence"] = round(max(0.0, min(conf, 1.0)), 4)


def _compress_confidence(item: dict[str, Any]) -> None:
    conf = _conf(item.get("confidence", 0.0))
    edge = _edge(item.get("edge", 0.0))
    strength = _strength(item)

    cap = SOFT_CONF_CAP_A if strength == "A" else SOFT_CONF_CAP_B
    cap += min(edge * 0.35, 0.05)
    cap = min(cap, HARD_CONF_CAP)

    item["confidence"] = round(min(max(conf, 0.50), cap), 4)


def _rebuild_fields(item: dict[str, Any], payload: dict[str, Any] | None = None) -> None:
    edge = _edge(item.get("edge", item.get("display_edge", 0.0)))
    conf = _conf(item.get("confidence", 0.0))
    kelly = _to_float(item.get("kelly_quarter", item.get("kelly", 0.0)), 0.0)
    kelly = min(kelly, MAX_KELLY)

    score = (edge * 100.0) + (conf * 10.0)
    if item.get("suspicious"):
        score -= 8.0

    edge_pct = round(edge * 100.0, 2)

    item["game"] = _game(item, payload)
    item["summary"] = _summary(item, payload)
    item["display_edge"] = edge_pct
    item["display_edge_text"] = f"{edge_pct}%"
    item["confidence"] = round(conf, 4)
    item["kelly_quarter"] = round(kelly, 4)
    item["score"] = round(score, 4)


def _passes_thresholds(item: dict[str, Any]) -> bool:
    edge = _edge(item.get("edge", 0.0))
    conf = _conf(item.get("confidence", 0.0))
    strength = _strength(item)

    if strength == "A":
        return edge >= MIN_EDGE_A and conf >= MIN_CONF_A
    if strength == "B":
        return edge >= MIN_EDGE_B and conf >= MIN_CONF_B
    return edge >= MIN_EDGE_B and conf >= MIN_CONF_B


def _small_gap(item: dict[str, Any]) -> bool:
    proj = item.get("projection", item.get("proj", item.get("projected_team_total")))
    line = item.get("line")
    if proj is None or line is None:
        return False
    return abs(_to_float(proj, 0.0) - _to_float(line, 0.0)) < SMALL_GAP_BLOCK


def _is_conflict(a: dict[str, Any], b: dict[str, Any]) -> bool:
    if _norm(a.get("game")) != _norm(b.get("game")):
        return False
    if _norm(a.get("team")) != _norm(b.get("team")):
        return False
    if _line_key(a.get("line")) != _line_key(b.get("line")):
        return False
    side_a = _norm(a.get("side"))
    side_b = _norm(b.get("side"))
    return bool(side_a and side_b and side_a != side_b)


def enforce_team_props_policy(policy_input: dict[str, Any], payload: dict[str, Any] | None = None) -> dict[str, Any]:
    out = dict(policy_input)

    props = (
        out.get("best_team_props")
        or out.get("ranked_team_props")
        or out.get("candidate_team_props")
        or out.get("team_props")
        or []
    )
    if not isinstance(props, list):
        props = []

    prepared: list[dict[str, Any]] = []
    drop_counts = defaultdict(int)

    for raw in props:
        if not isinstance(raw, dict):
            drop_counts["non_dict"] += 1
            continue

        item = dict(raw)
        item["team"] = str(item.get("team", "")).strip()
        item["side"] = str(item.get("side", "")).strip().lower()

        if BLOCK_SUSPICIOUS and item.get("suspicious"):
            drop_counts["blocked_suspicious"] += 1
            continue

        if _small_gap(item):
            drop_counts["small_gap"] += 1
            continue

        _apply_learned_adjustments(item)
        _compress_confidence(item)
        _rebuild_fields(item, payload)

        if not _passes_thresholds(item):
            drop_counts["below_threshold"] += 1
            continue

        prepared.append(item)

    prepared = sorted(prepared, key=_sort_key, reverse=True)

    if BLOCK_CONFLICTING_SIDES:
        conflict_pruned: list[dict[str, Any]] = []
        for item in prepared:
            if any(_is_conflict(item, existing) for existing in conflict_pruned):
                drop_counts["conflict_pruned"] += 1
                continue
            conflict_pruned.append(item)
        prepared = conflict_pruned

    final: list[dict[str, Any]] = []
    per_game = defaultdict(int)
    per_team = defaultdict(int)

    for item in prepared:
        game = _norm(item.get("game"))
        team = _norm(item.get("team"))

        if per_game[game] >= MAX_PROPS_PER_GAME:
            drop_counts["over_game_limit"] += 1
            continue

        if per_team[(game, team)] >= MAX_PROPS_PER_TEAM:
            drop_counts["over_team_limit"] += 1
            continue

        final.append(item)
        per_game[game] += 1
        per_team[(game, team)] += 1

    out["best_team_props"] = final
    out["policy_summary"] = {
        "props_after_policy": len(final),
        "max_kelly": MAX_KELLY,
        "max_props_per_game": MAX_PROPS_PER_GAME,
        "max_props_per_team": MAX_PROPS_PER_TEAM,
        "drop_counts": dict(drop_counts),
    }
    return out

def load_jsonl(path: str) -> list[dict[str, Any]]:

def to_float(v: Any, default: float = 0.0) -> float:
    try:
        if v is None:
            return default
        return float(v)
    except Exception:
        return default


def normalize_game(game: str) -> str:
    return " ".join(str(game).lower().strip().split())


def result_key(row: dict[str, Any]) -> tuple[str, str, str, str]:
    return (
        normalize_game(row.get("game", "")),
        str(row.get("team", "")),
        str(row.get("side", "")),
        str(row.get("line", None)),
    )


def existing_result_keys(rows: list[dict[str, Any]]) -> set[tuple[str, str, str, str]]:
    return {result_key(r) for r in rows}


def american_profit_for_one_unit_risk(odds: Any) -> float:
    odds_f = to_float(odds, 0.0)
    if odds_f == 0:
        return 0.0

    if odds_f > 0:
        return round(odds_f / 100.0, 4)

    return round(100.0 / abs(odds_f), 4)


def compute_clv(side: str, open_line: Any, closing_line: Any) -> float | None:
    open_f = to_float(open_line, None)
    close_f = to_float(closing_line, None)

    if open_f is None or close_f is None:
        return None

    side = str(side).lower()
    if side == "over":
        return round(close_f - open_f, 3)
    if side == "under":
        return round(open_f - close_f, 3)

    return None


def grade_team_total(side: str, final_team_score: int, line: Any) -> bool | None:
    line_f = to_float(line, None)
    if line_f is None:
        return None

    side = str(side).lower()
    if side == "over":
        return final_team_score > line_f
    if side == "under":
        return final_team_score < line_f

    return None

def _run(cmd: str, timeout: int = 15) -> str:
    try:
        out = subprocess.run(

            timeout=timeout,

def _read_jsonl(path: Path, limit: int = 10) -> list[dict[str, Any]]:
    if not path.exists():
        return []
    rows: list[dict[str, Any]] = []
    try:
        with path.open("r", encoding="utf-8") as f:
            for line in f:
                line = line.strip()
                if not line:
                    continue
                try:
                    rows.append(json.loads(line))
                except Exception:
                    continue
    except Exception:
        return []
    return rows[-limit:]

#!/usr/bin/env python3
from __future__ import annotations

import json
import random
from typing import Any


def _clamp(x: float, lo: float, hi: float) -> float:
    return max(lo, min(hi, x))


def american_to_decimal(odds: int) -> float:
    if odds > 0:
        return 1.0 + (odds / 100.0)
    return 1.0 + (100.0 / abs(odds))


def implied_prob_from_american(odds: int) -> float:
    if odds > 0:
        return 100.0 / (odds + 100.0)
    return abs(odds) / (abs(odds) + 100.0)


def calculate_edge(model_prob: float, implied_prob: float) -> float:
    return model_prob - implied_prob


def calculate_ev(model_prob: float, odds: int) -> float:
    decimal_odds = american_to_decimal(odds)
    return (model_prob * (decimal_odds - 1.0)) - (1.0 - model_prob)


def estimate_auto_pace(
    mode: str,
    home_score: int,
    away_score: int,
    minutes_left: int,
    seconds_left: int,
    baseline_ppm: float,
) -> tuple[float, float]:
    total_scored = home_score + away_score
    remaining_minutes = minutes_left + (seconds_left / 60.0)

    if mode in {"1h", "2h"}:
        segment_length = 20.0
    else:
        segment_length = 40.0

    elapsed = max(0.5, segment_length - remaining_minutes)
    pts_per_min = total_scored / elapsed

    recent_ppm = pts_per_min / 1.0
    recent_ppm = _clamp(recent_ppm, 1.8, 4.2)
    baseline_ppm = _clamp(baseline_ppm, 1.8, 4.2)

    return recent_ppm, baseline_ppm


def apply_score_differential_pace_model(
    mode: str,
    score_diff: int,
    recent_ppm: float,
    baseline_ppm: float,
    minutes_left: int,
) -> tuple[float, str]:
    blended = (0.60 * recent_ppm) + (0.40 * baseline_ppm)
    note = "neutral pace blend"

    if mode == "2h":
        if abs(score_diff) <= 6 and minutes_left <= 12:
            blended *= 1.06
            note = "close 2H pace-up applied"
        elif abs(score_diff) >= 12 and minutes_left <= 12:
            blended *= 0.94
            note = "wide-lead 2H pace-down applied"

    elif mode == "full":
        if abs(score_diff) <= 5 and minutes_left <= 8:
            blended *= 1.04
            note = "close-game pace-up applied"
        elif abs(score_diff) >= 15 and minutes_left <= 8:
            blended *= 0.93
            note = "blowout pace-down applied"

    elif mode == "1h":
        if abs(score_diff) <= 5 and minutes_left <= 8:
            blended *= 1.03
            note = "close 1H pace-up applied"

    return _clamp(blended, 1.8, 4.4), note

def evaluate_live_game(payload: dict[str, Any]) -> dict[str, Any]:

    away_team = str(payload["away_team"])

    minutes_left = int(payload["minutes_left"])

    away_spread_odds = int(payload.get("away_spread_odds", -110))

    baseline_ppp = float(payload.get("baseline_points_per_possession", 1.02))
    recent_ppp = float(payload.get("recent_points_per_possession", 0.88))
    recent_ppm = float(payload.get("recent_possessions_per_min", 3.1))
    baseline_ppm = float(payload.get("baseline_possessions_per_min", 3.0))

    score_diff = home_score - away_score

    if recent_ppm > 5.0 or baseline_ppm > 5.0 or recent_ppm < 1.0 or baseline_ppm < 1.0:
        recent_ppm, baseline_ppm = estimate_auto_pace(
            mode=mode,
            home_score=home_score,
            away_score=away_score,
            minutes_left=minutes_left,
            seconds_left=seconds_left,

        )

        recent_ppm=recent_ppm,
        baseline_ppm=baseline_ppm,
        minutes_left=minutes_left,
    )

    blended_ppp = _clamp((0.55 * recent_ppp) + (0.45 * baseline_ppp), 0.74, 1.24)
    remaining_minutes = minutes_left + (seconds_left / 60.0)
    expected_remaining_possessions = max(1.0, blended_ppm * remaining_minutes)

    num_sims = 10000

    sum_home_final = 0.0
    sum_away_final = 0.0
    sum_total_final = 0.0

    home_wins = 0
    home_covers = 0
    overs = 0

    for _ in range(num_sims):
        sim_home = float(home_score)
        sim_away = float(away_score)

        # more realistic variance
        poss = max(1, int(random.gauss(expected_remaining_possessions, 4.5)))

        # allow game-flow drift per simulation
        sim_home_share = 0.5 + _clamp(random.gauss(score_diff / 40.0, 0.06), -0.18, 0.18)

        for _p in range(poss):
            pts = max(0.0, random.gauss(blended_ppp, 0.48))

            if random.random() < sim_home_share:
                sim_home += pts
            else:
                sim_away += pts

        sim_total = sim_home + sim_away

        sum_home_final += sim_home
        sum_away_final += sim_away
        sum_total_final += sim_total

        if sim_home > sim_away:
            home_wins += 1

        if (sim_home - sim_away) + home_spread_line > 0:
            home_covers += 1

        if sim_total > market_total:
            overs += 1

    projected_away = sum_away_final / num_sims
    projected_total = sum_total_final / num_sims

    away_win_prob = 1.0 - home_win_prob
    home_cover_prob = home_covers / num_sims
    away_cover_prob = 1.0 - home_cover_prob
    over_prob = overs / num_sims
    under_prob = 1.0 - over_prob

    over_implied = implied_prob_from_american(over_odds)
    under_implied = implied_prob_from_american(under_odds)

    over_edge = calculate_edge(over_prob, over_implied)
    under_edge = calculate_edge(under_prob, under_implied)

    inefficiency_is_inefficient = False
    inefficiency_gap = 0.0
    inefficiency_direction = "NONE"

        inefficiency_is_inefficient = True
        inefficiency_gap = over_edge
        inefficiency_direction = "OVER"
    elif under_edge >= 0.03 and under_edge > over_edge:
        inefficiency_is_inefficient = True
        inefficiency_gap = under_edge
        inefficiency_direction = "UNDER"

        away_ml_odds=away_ml_odds,
        home_spread_odds=home_spread_odds,
        away_spread_odds=away_spread_odds,
        min_edge=0.03,

from __future__ import annotations


def parse_telegram_command(message: str) -> dict[str, str]:
    text = message.lower().strip()

    text = text.replace("@", " vs ")
    text = text.replace(" v ", " vs ")
    text = text.replace(" versus ", " vs ")

    tokens = text.split()
    if not tokens:
        raise ValueError("Empty command")

    mode = "full"
    if tokens[-1] in {"1h", "2h", "full"}:
        mode = tokens[-1]
        tokens = tokens[:-1]

    if not tokens:
        raise ValueError("Missing teams")

    if "vs" in tokens:
        idx = tokens.index("vs")
        away_team = " ".join(tokens[:idx]).strip()
        home_team = " ".join(tokens[idx + 1:]).strip()
    else:
        if len(tokens) < 2:
            raise ValueError("Invalid format. Use: aub vs duke 2h")
        away_team = tokens[0].strip()
        home_team = " ".join(tokens[1:]).strip()

    if not away_team or not home_team:
        raise ValueError("Could not parse both teams")

    return {
        "away_team": away_team,
        "home_team": home_team,
        "mode": mode,
    }

    units = float(

def _to_raw_float(v: Any, default: float = 0.0) -> float:
    try:
        if v is None:
            return default
        if isinstance(v, str):
            s = v.strip()
            if not s:
                return default
            if s.endswith("%"):
                return float(s[:-1])
            return float(s)
        return float(v)
    except Exception:
        return default


def _to_ratio(v: Any, default: float = 0.0) -> float:
    try:
        if v is None:
            return default
        if isinstance(v, str):
            s = v.strip()
            if not s:
                return default
            if s.endswith("%"):
                return float(s[:-1]) / 100.0
            val = float(s)
        else:
            val = float(v)
        if val > 1.0:
            val /= 100.0
        return val
    except Exception:
        return default


def _norm(v: Any) -> str:
    return " ".join(str(v or "").strip().split())


def _escape(text: Any) -> str:

    for key in (

        "bankroll_pct",

        "bet_percent",
        "risk_pct",

        if signal.get(key) not in (None, ""):

    if signal.get("kelly_quarter") not in (None, ""):

        or signal.get("recommended_units")
        or signal.get("aj_units")
        or signal.get("bet_size")
        or 0.0,
        0.0,

def _send_message(text: str) -> bool:
    if not TELEGRAM_ENABLED:
        logger.info("Telegram alerts disabled; skipping send.")
        return False

    if not TELEGRAM_BOT_TOKEN or not TELEGRAM_CHAT_ID:
        logger.warning("Telegram token/chat id missing; cannot send alert.")
        return False

    url = f"https://api.telegram.org/bot{TELEGRAM_BOT_TOKEN}/sendMessage"
    payload = {
        "chat_id": TELEGRAM_CHAT_ID,
        "text": text,
        "parse_mode": "HTML",
        "disable_web_page_preview": True,
    }

    try:
        resp = requests.post(url, json=payload, timeout=TELEGRAM_TIMEOUT)
        if resp.status_code == 429:
            print("🛡️ 429 Rate Limit Hit. Shield active, sleeping 60s...")
            import time; time.sleep(60)
        else:
            resp.raise_for_status()
        logger.info("Telegram alert sent.")
        return True
    except Exception as e:
        logger.exception("Telegram send failed: %s", e)
        return False

#!/usr/bin/env python3
import json
from pathlib import Path
from collections import defaultdict
from datetime import datetime, timezone

TRACK_FILE = Path("storage/tracker/alerts.jsonl")


def load_alerts():
    if not TRACK_FILE.exists():
        return []

    alerts = []
    with TRACK_FILE.open("r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            try:
                alerts.append(json.loads(line))
            except Exception:
                continue
    return alerts


def filter_active(alerts):
    active = []
    for a in alerts:
        result = str(a.get("result", "")).lower()
        if result in ("", "pending", "open", "live"):
            active.append(a)
    return active


def dedupe_alerts(alerts):
    best = {}

    for idx, a in enumerate(alerts):
        key = (
            str(a.get("phase", "")),
            str(a.get("game", "")),
            str(a.get("market", "")),
            str(a.get("selection", "")),
        )

        prev = best.get(key)
        if prev is None:
            best[key] = (idx, a)
            continue

        prev_idx, prev_a = prev
        prev_ts = str(prev_a.get("ts_utc", ""))
        curr_ts = str(a.get("ts_utc", ""))

        if curr_ts and prev_ts:
            if curr_ts >= prev_ts:
                best[key] = (idx, a)
        else:
            if idx >= prev_idx:
                best[key] = (idx, a)

    return [pair[1] for pair in sorted(best.values(), key=lambda x: x[0])]


def parse_ts(ts):
    if not ts:
        return None
    try:
        return datetime.fromisoformat(ts.replace("Z", "+00:00"))
    except Exception:
        return None


def latest_cycle_only(alerts, window_seconds=180):
    stamped = []
    for a in alerts:
        ts = parse_ts(str(a.get("ts_utc", "")))
        if ts is not None:
            stamped.append((ts, a))

    if not stamped:
        return alerts

    latest_ts = max(ts for ts, _ in stamped)
    out = []

    for ts, a in stamped:
        delta = (latest_ts - ts).total_seconds()
        if delta <= window_seconds:
            out.append(a)

    return out


def best_one_per_game(alerts):
    by_game = defaultdict(list)
    for a in alerts:
        game = str(a.get("game", "UNKNOWN"))
        by_game[game].append(a)

    out = []
    for game, items in by_game.items():
        items.sort(
            key=lambda x: (
                float(x.get("confidence", 0) or 0),
                float(x.get("edge", 0) or 0),
            ),
            reverse=True,
        )
        out.append(items[0])

    out.sort(
        key=lambda x: (
            float(x.get("confidence", 0) or 0),
            float(x.get("edge", 0) or 0),
        ),
        reverse=True,
    )
    return out


def main():
    alerts = load_alerts()
    active = filter_active(alerts)
    active = dedupe_alerts(active)
    active = latest_cycle_only(active, window_seconds=180)
    active = best_one_per_game(active)

    print("=== LATEST ACTIVE CARD ===")
    print(f"total_active: {len(active)}")

from datetime import datetime

TRACK_FILE = Path("storage/tracker/alerts.jsonl")

MAX_CARD_EXPOSURE_UNITS = 7.5

SPORT_KELLY_HAIRCUT = {
    "basketball_nba": 1.00,
    "basketball_wnba": 0.92,
    "basketball_ncaab": 0.88,
    "basketball_wncaab": 0.85,
}


def load_alerts():
    if not TRACK_FILE.exists():
        return []

    alerts = []
    with TRACK_FILE.open("r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            try:
                alerts.append(json.loads(line))
            except Exception:
                continue
    return alerts


def filter_active(alerts):
    active = []
    for a in alerts:
        result = str(a.get("result", "")).lower()
        if result in ("", "pending", "open", "live"):
            active.append(a)
    return active


def dedupe_alerts(alerts):
    best = {}

    for idx, a in enumerate(alerts):
        key = (
            str(a.get("phase", "")),
            str(a.get("game", "")),
            str(a.get("market", "")),
            str(a.get("selection", "")),
        )

        prev = best.get(key)
        if prev is None:
            best[key] = (idx, a)
            continue

        prev_idx, prev_a = prev
        prev_ts = str(prev_a.get("ts_utc", ""))
        curr_ts = str(a.get("ts_utc", ""))

        if curr_ts and prev_ts:
            if curr_ts >= prev_ts:
                best[key] = (idx, a)
        else:
            if idx >= prev_idx:
                best[key] = (idx, a)

    return [pair[1] for pair in sorted(best.values(), key=lambda x: x[0])]


def parse_ts(ts):
    if not ts:
        return None
    try:
        return datetime.fromisoformat(ts.replace("Z", "+00:00"))
    except Exception:
        return None


def latest_cycle_only(alerts, window_seconds=180):
    stamped = []
    for a in alerts:
        ts = parse_ts(str(a.get("ts_utc", "")))
        if ts is not None:
            stamped.append((ts, a))

    if not stamped:
        return alerts

    latest_ts = max(ts for ts, _ in stamped)
    out = []

    for ts, a in stamped:
        delta = (latest_ts - ts).total_seconds()
        if delta <= window_seconds:
            out.append(a)

    return out


def best_one_per_game(alerts):
    by_game = defaultdict(list)
    for a in alerts:
        game = str(a.get("game", "UNKNOWN"))
        by_game[game].append(a)

    out = []
    for _, items in by_game.items():
        items.sort(
            key=lambda x: (
                float(x.get("confidence", 0) or 0),
                float(x.get("edge", 0) or 0),
            ),
            reverse=True,
        )
        out.append(items[0])

    out.sort(
        key=lambda x: (
            float(x.get("confidence", 0) or 0),
            float(x.get("edge", 0) or 0),
        ),
        reverse=True,
    )
    return out


def implied_win_prob_from_conf(conf):
    """
    Cooler confidence -> win probability mapping.
    Intentionally conservative so Kelly does not overbet.
    """
    conf = float(conf or 0.0)

    if conf >= 92:
        return 0.575
    if conf >= 89:
        return 0.568
    if conf >= 86:
        return 0.560
    if conf >= 83:
        return 0.553
    if conf >= 80:
        return 0.546
    if conf >= 76:
        return 0.538
    if conf >= 72:
        return 0.530
    return 0.515


def get_alert_odds(alert, default=-110.0):
    for key in ("odds", "market_odds", "price", "american_odds", "book_odds"):
        val = alert.get(key)
        if val is None or val == "":
            continue
        try:
            odds = float(val)
            if odds != 0:
                return odds
        except Exception:
            continue
    return float(default)


def flat_bet_units(alert):
    conf = float(alert.get("confidence", 0) or 0)
    edge = float(alert.get("edge", 0) or 0)
    tier = str(alert.get("tier", "")).upper()
    market = str(alert.get("market", "")).lower()

    units = 0.5

    if conf >= 90:
        units += 0.9
    elif conf >= 87:
        units += 0.7
    elif conf >= 84:
        units += 0.5
    elif conf >= 80:
        units += 0.3
    elif conf >= 76:
        units += 0.15

    if edge >= 15:
        units += 0.4
    elif edge >= 10:
        units += 0.3
    elif edge >= 7:
        units += 0.2
    elif edge >= 5:
        units += 0.1

    if tier == "A":
        units += 0.2
    elif tier == "C":
        units -= 0.15
    elif tier == "D":
        units -= 0.3

    if market == "moneyline":
        units -= 0.1
    elif market == "total":
        units += 0.05

    units = max(0.25, min(2.0, units))
    return round(units, 2)


def odds_to_b(american_odds):
    odds = float(american_odds)
    if odds < 0:
        return 100.0 / abs(odds)
    return odds / 100.0


def implied_prob_from_american(american_odds):
    odds = float(american_odds)
    if odds < 0:
        return abs(odds) / (abs(odds) + 100.0)
    return 100.0 / (odds + 100.0)


def sport_haircut(alert):
    sport = str(alert.get("sport_key", "")).lower()
    return float(SPORT_KELLY_HAIRCUT.get(sport, 0.90))


def kelly_units(alert, bankroll_units=100.0, fraction=0.25):
    """
    Real-odds-aware fractional Kelly with sport haircut.
    """
    p_model = implied_win_prob_from_conf(alert.get("confidence", 0))
    american_odds = get_alert_odds(alert, default=-110.0)
    b = odds_to_b(american_odds)
    q = 1.0 - p_model

    raw_kelly = ((b * p_model) - q) / b

    if raw_kelly <= 0:
        return 0.25, american_odds, implied_prob_from_american(american_odds)

    frac_kelly = raw_kelly * fraction
    units = frac_kelly * bankroll_units

    # sport-specific haircut
    units *= sport_haircut(alert)

    units = max(0.25, min(2.0, units))
    return round(units, 2), american_odds, implied_prob_from_american(american_odds)


def recommended_units(alert):
    flat_units = flat_bet_units(alert)
    kelly_sized, used_odds, market_prob = kelly_units(alert)

    rec = min(flat_units, kelly_sized)

    if str(alert.get("tier", "")).upper() == "C":
        rec = min(rec, 0.5)

    return round(max(0.25, rec), 2), flat_units, kelly_sized, used_odds, round(market_prob, 4)


def apply_card_exposure_cap(rows):
    total_rec = sum(r["recommended_units"] for r in rows)

    if total_rec <= MAX_CARD_EXPOSURE_UNITS or total_rec <= 0:
        for r in rows:
            r["capped_units"] = r["recommended_units"]
        return rows, 1.0, total_rec

    scale = MAX_CARD_EXPOSURE_UNITS / total_rec

    for r in rows:
        capped = round(max(0.25, r["recommended_units"] * scale), 2)
        r["capped_units"] = capped

    capped_total = round(sum(r["capped_units"] for r in rows), 2)
    return rows, round(scale, 4), capped_total


def main():
    alerts = load_alerts()
    active = filter_active(alerts)
    active = dedupe_alerts(active)
    active = latest_cycle_only(active, window_seconds=180)
    active = best_one_per_game(active)

    rows = []
    total_flat = 0.0
    total_kelly = 0.0
    total_rec = 0.0

    for a in active:
        rec_units, flat_units, kelly_sized, used_odds, market_prob = recommended_units(a)

        row = {
            "alert": a,
            "flat_units": flat_units,
            "kelly_units": kelly_sized,
            "recommended_units": rec_units,
            "used_odds": used_odds,
            "market_prob": market_prob,
        }
        rows.append(row)

        total_flat += flat_units
        total_kelly += kelly_sized
        total_rec += rec_units

    rows, card_scale, capped_total = apply_card_exposure_cap(rows)

    print("=== BET SIZE CARD ===")
    print(f"total_plays: {len(rows)}")
    print(f"max_card_exposure_units: {MAX_CARD_EXPOSURE_UNITS}")
    print(f"card_scale: {card_scale}")
    print()

    for r in rows:
        a = r["alert"]
        sport = a.get("sport_key", "")
        game = a.get("game", "UNKNOWN")
        market = a.get("market", "")
        selection = a.get("selection", "")
        edge = round(float(a.get("edge", 0) or 0), 2)
        conf = round(float(a.get("confidence", 0) or 0), 2)
        tier = a.get("tier", "")
        haircut = sport_haircut(a)

        print(f"--- {game} ---")
        print(
            f"[{sport}] {market} -> {selection} | "
            f"EDGE={edge} | CONF={conf} | TIER={tier}"
        )
        print(
            f"odds={r['used_odds']} | market_implied={r['market_prob']} | "
            f"sport_haircut={haircut} | "
            f"flat_units={r['flat_units']} | "
            f"kelly_units={r['kelly_units']} | "
            f"recommended_units={r['recommended_units']} | "
            f"capped_units={r['capped_units']}"
        )
        print()

    print(f"TOTAL FLAT UNITS: {round(total_flat, 2)}")
    print(f"TOTAL KELLY UNITS: {round(total_kelly, 2)}")
    print(f"TOTAL RECOMMENDED UNITS: {round(total_rec, 2)}")
    print(f"TOTAL CAPPED UNITS: {round(capped_total, 2)}")


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
from __future__ import annotations

import json
from pathlib import Path
from typing import Any

TRACK_FILE = Path("storage/tracker/alerts.jsonl")


def _safe_float(value: Any, default: float = 0.0) -> float:
    try:
        if value is None or value == "":
            return default
        return float(value)
    except Exception:
        return default


def load_rows() -> list[dict]:
    if not TRACK_FILE.exists():
        return []
    rows = []
    for line in TRACK_FILE.read_text(encoding="utf-8").splitlines():
        try:
            if line.strip():
                rows.append(json.loads(line))
        except:
            pass
    return rows


def _bucket_stats(rows, field):
    out = {}
    for r in rows:
        result = r.get("result")
        if result not in {"win", "loss", "push"}:
            continue

        key = str(r.get(field, "unknown"))

        if key not in out:
            out[key] = {"bets": 0, "wins": 0, "losses": 0, "units": 0.0}

        out[key]["bets"] += 1
        out[key]["units"] += _safe_float(r.get("profit_units", 0))

        if result == "win":
            out[key]["wins"] += 1
        elif result == "loss":
            out[key]["losses"] += 1

    for k, v in out.items():
        wl = v["wins"] + v["losses"]
        v["hit_rate"] = (v["wins"] / wl) if wl > 0 else 0.0

    return out


def build_snapshot():
    rows = load_rows()
    settled = [r for r in rows if r.get("result") in {"win", "loss", "push"}]

    return {
        "by_market": _bucket_stats(settled, "market"),
        "by_tier": _bucket_stats(settled, "tier"),
        "by_sport": _bucket_stats(settled, "sport_key"),
    }


def _score(stats):
    if not stats:
        return 0.0

    bets = stats["bets"]
    hit = stats["hit_rate"]
    units = stats["units"]

    # 🔒 REQUIRE SAMPLE SIZE
    if bets < 3:
        return 0.0
    if bets < 5:
        scale = 0.5
    else:
        scale = 1.0

    score = 0.0

    # 📊 HIT RATE (reduced weight)
    if hit >= 0.65:
        score += min(2.5, (hit - 0.5) * 10)
    elif hit <= 0.45:
        score -= min(4.0, (0.5 - hit) * 14)  # stronger penalty

    # 💰 UNITS (reduced boost, stronger penalty)
    if units > 0:
        score += min(1.5, units * 0.4)
    elif units < 0:
        score -= min(3.0, abs(units) * 0.6)

    return round(score * scale, 2)


def adjust_alert(alert, snap):
    market = alert.get("market")
    tier = alert.get("tier")
    sport = alert.get("sport_key")

    m = _score(snap["by_market"].get(market))
    t = _score(snap["by_tier"].get(tier))
    s = _score(snap["by_sport"].get(sport))

    # ⚖️ REDUCE SPORT INFLUENCE
    total_adj = m + t + (s * 0.6)

    # 🔒 HARD CAP
    total_adj = max(-6.0, min(3.0, total_adj))

    base_conf = _safe_float(alert.get("confidence", 0))
    new_conf = max(0.0, min(99.0, base_conf + total_adj))

    out = dict(alert)
    out["confidence"] = round(new_conf, 2)
    out["performance_adjustment"] = round(total_adj, 2)

    # 🧠 RETIER (more conservative)
    if new_conf >= 88:
        out["tier"] = "A"
    elif new_conf >= 80:
        out["tier"] = "B"
    elif new_conf >= 72:
        out["tier"] = "C"
    else:
        out["tier"] = "D"

    return out


def filter_alerts(alerts, min_conf=60):
    snap = build_snapshot()
    out = []
    for a in alerts:
        val_edge = abs(float(a.get('total_edge_delta') or a.get('edge') or 0))
        if val_edge >= 8.0:
            print(f'!!! MONSTER BYPASS: Edge {val_edge} for {a.get("selection")} !!!')
            out.append(a)
            continue

        adj = adjust_alert(a, snap)
        if adj.get('performance_adjustment', 0) <= -4:
            continue
        if adj.get('confidence', 0) < min_conf:
            continue
        out.append(adj)

    out.sort(key=lambda x: (float(x.get('confidence', 0) or 0), float(x.get('edge', 0) or 0)), reverse=True)
    return out

if __name__ == "__main__":
    print(json.dumps(build_snapshot(), indent=2))

#!/usr/bin/env python3
from __future__ import annotations

import json
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

TRACK_FILE = Path("storage/tracker/alerts.jsonl")


def _utc_now() -> str:
    return datetime.now(timezone.utc).isoformat()


def _safe_float(value: Any, default: float | None = None):
    try:
        if value is None or value == "":
            return default
        return float(value)
    except Exception:
        return default


def _extract_odds_fields(alert: dict[str, Any], payload: dict[str, Any]) -> dict[str, Any]:
    """
    Preserve any odds fields already attached to the alert.
    Fall back to payload-level fields when present.
    """
    out: dict[str, Any] = {}

    candidate_fields = [
        "odds",
        "market_odds",
        "price",
        "american_odds",
        "book_odds",
        "bookmaker",
        "market_line",
        "system_value",
        "market_value",
    ]

    for key in candidate_fields:
        if key in alert and alert.get(key) not in (None, ""):
            out[key] = alert.get(key)
        elif key in payload and payload.get(key) not in (None, ""):
            out[key] = payload.get(key)

    # keep common market context if available
    for key in [
        "home_spread_line",
        "market_total",
        "home_ml",
        "away_ml",
    ]:
        if key in payload and payload.get(key) not in (None, ""):
            out[key] = payload.get(key)

    return out


def record_alert(payload: dict[str, Any], alert: dict[str, Any], phase: str = "pregame") -> None:
    TRACK_FILE.parent.mkdir(parents=True, exist_ok=True)

    row: dict[str, Any] = {
        "ts_utc": _utc_now(),
        "phase": phase,
        "sport_key": alert.get("sport_key") or payload.get("sport_key"),
        "game": alert.get("game") or payload.get("game") or f"{payload.get('away_team', '')} @ {payload.get('home_team', '')}".strip(),
        "market": alert.get("market"),
        "selection": alert.get("selection"),
        "edge": _safe_float(alert.get("edge"), 0.0),
        "confidence": _safe_float(alert.get("confidence"), 0.0),
        "tier": alert.get("tier", ""),
        "source": alert.get("source", ""),
        "recommended_units": alert.get("recommended_units"),
        "capped_units": alert.get("capped_units"),
        "bet_size": alert.get("bet_size"),
        "kelly_quarter": alert.get("kelly_quarter"),
        "kelly": alert.get("kelly"),
        "aj_edge": bool(alert.get("aj_edge", False)),
        "aj_tier": alert.get("aj_tier", ""),
        "aj_units": _safe_float(alert.get("aj_units"), None),
        "aj_time_remaining": _safe_float(alert.get("aj_time_remaining"), None),
        "aj_score_diff": _safe_float(alert.get("aj_score_diff"), None),
        "result": "pending",
        "profit_units": 0.0,
    }

    # store teams when we can
    if payload.get("home_team"):
        row["home_team"] = payload.get("home_team")
    if payload.get("away_team"):
        row["away_team"] = payload.get("away_team")

    # merge odds and market fields
    row.update(_extract_odds_fields(alert, payload))

    with TRACK_FILE.open("a", encoding="utf-8") as f:
        f.write(json.dumps(row) + "\n")

#!/usr/bin/env python3
import json
from collections import defaultdict
from pathlib import Path

TRACK_FILE = Path("storage/tracker/alerts.jsonl")
RESULTS_FILE = Path("storage/tracker/game_results.json")


def load_alerts():
    if not TRACK_FILE.exists():
        return []
    rows = []
    for line in TRACK_FILE.read_text(encoding="utf-8").splitlines():
        line = line.strip()
        if not line:
            continue
        try:
            rows.append(json.loads(line))
        except Exception:
            pass
    return rows


def save_alerts(rows):
    TRACK_FILE.parent.mkdir(parents=True, exist_ok=True)
    with TRACK_FILE.open("w", encoding="utf-8") as f:
        for r in rows:
            f.write(json.dumps(r) + "\n")


def load_results():
    if not RESULTS_FILE.exists():
        return {}
    try:
        return json.loads(RESULTS_FILE.read_text(encoding="utf-8"))
    except Exception:
        return {}


def save_results(data):
    RESULTS_FILE.parent.mkdir(parents=True, exist_ok=True)
    RESULTS_FILE.write_text(json.dumps(data, indent=2), encoding="utf-8")


def set_result(game, home_score, away_score):
    data = load_results()
    data[game] = {"home_score": float(home_score), "away_score": float(away_score)}
    save_results(data)
    print("saved:", game)


def infer_teams(row):
    game = str(row.get("game", ""))
    if " @ " in game:
        away_team, home_team = game.split(" @ ", 1)
        return away_team.strip(), home_team.strip()
    return "", ""


def dedupe_alerts(rows):
    best = {}

    for idx, r in enumerate(rows):
        key = (
            str(r.get("phase", "")),
            str(r.get("game", "")),
            str(r.get("market", "")),
            str(r.get("selection", "")),
        )

        prev = best.get(key)
        if prev is None:
            best[key] = (idx, r)
            continue

        prev_idx, prev_row = prev
        prev_ts = str(prev_row.get("ts_utc", ""))
        curr_ts = str(r.get("ts_utc", ""))

        if curr_ts and prev_ts:
            if curr_ts >= prev_ts:
                best[key] = (idx, r)
        else:
            if idx >= prev_idx:
                best[key] = (idx, r)

    out = [pair[1] for pair in sorted(best.values(), key=lambda x: x[0])]
    return out


def grade():
    alerts = load_alerts()
    alerts = dedupe_alerts(alerts)
    results = load_results()

    for r in alerts:
        game = r.get("game")
        if game not in results:
            r["result"] = "pending"
            r["profit_units"] = 0.0
            continue

        res = results[game]
        home = float(res["home_score"])
        away = float(res["away_score"])

        market = str(r.get("market", "")).lower()
        sel = r.get("selection")
        line = float(r.get("market_value", 0) or 0)

        away_team, home_team = infer_teams(r)
        r["home_team"] = home_team
        r["away_team"] = away_team

        # TOTAL
        if market == "total":
            total = home + away
            if total == line:
                r["result"] = "push"
            elif str(sel).lower() == "over":
                r["result"] = "win" if total > line else "loss"
            else:
                r["result"] = "win" if total < line else "loss"

        # MONEYLINE
        elif market == "moneyline":
            if home == away:
                r["result"] = "push"
            else:
                winner = home_team if home > away else away_team
                r["result"] = "win" if sel == winner else "loss"

        # SPREAD
        elif market == "spread":
            # market_value = home spread convention
            if sel == home_team:
                adjusted_home = home + line
                if adjusted_home == away:
                    r["result"] = "push"
                else:
                    r["result"] = "win" if adjusted_home > away else "loss"
            elif sel == away_team:
                away_line = -line
                adjusted_away = away + away_line
                if adjusted_away == home:
                    r["result"] = "push"
                else:
                    r["result"] = "win" if adjusted_away > home else "loss"
            else:
                r["result"] = "pending"

        else:
            r["result"] = "pending"

        if r["result"] == "win":
            r["profit_units"] = 1.0
        elif r["result"] == "loss":
            r["profit_units"] = -1.0
        else:
            r["profit_units"] = 0.0

    save_alerts(alerts)
    print("graded unique rows:", len(alerts))


def _show_bucket(title, bucket):
    print(title)
    for key, vals in sorted(bucket.items(), key=lambda x: (-x[1]["units"], x[0])):
        wins = vals["wins"]
        losses = vals["losses"]
        pushes = vals["pushes"]
        bets = vals["bets"]
        units = round(vals["units"], 2)
        hit = round((wins / (wins + losses) * 100.0), 2) if (wins + losses) else 0.0
        print(f"  {key}: bets={bets} wins={wins} losses={losses} pushes={pushes} hit%={hit} units={units}")
    print()

#!/usr/bin/env python3
import os, requests, time, sys, random
from datetime import datetime
from dotenv import load_dotenv
from betting_math import enrich_signal, implied_probability
from edge_alert_deduper import deduper

load_dotenv()
API_KEY = os.getenv("ODDS_API_KEY")
SPORT = "basketball_wnba"
CONFIDENCE_FLOOR = float(sys.argv[1]) if len(sys.argv) > 1 else 85.0
MARKETS = ["team_totals","player_points","player_rebounds","player_assists","player_points_rebounds_assists","player_threes"]

def fetch_and_enrich_wnba_props():
    print(f"📡 [AJ SNIPER] Fetching WNBA Props (totals+rebounds+assists+PRA) — Floor: {CONFIDENCE_FLOOR}%")
    events_response = requests.get(f"https://api.the-odds-api.com/v4/sports/{SPORT}/events", params={"apiKey": API_KEY})
    if events_response.status_code != 200: return
    events = events_response.json()
    if not events: return
    print(f"✅ Found {len(events)} WNBA games.\n")
    alerts = []
    for event in events:
        event_id = event["id"]
        home = event["home_team"]
        away = event["away_team"]
        game = f"{away} @ {home}"
        props_response = requests.get(f"https://api.the-odds-api.com/v4/sports/{SPORT}/events/{event_id}/odds", params={"apiKey": API_KEY, "regions": "us", "markets": ",".join(MARKETS), "oddsFormat": "american"})
        if props_response.status_code != 200: continue
        game_data = props_response.json()
        for bookmaker in game_data.get("bookmakers", []):
            if bookmaker["title"] not in ["DraftKings","FanDuel","Bovada"]: continue
            for market in bookmaker.get("markets", []):
                for outcome in market.get("outcomes", []):
                    team = outcome.get("description") or outcome.get("name", "Unknown")
                    side = outcome.get("name", "")
                    line = float(outcome.get("point", 0))
                    odds = float(outcome.get("price", 0))
                    implied = implied_probability(odds)
                    home_advantage = 0.042 if team == home else -0.028
                    variance = random.uniform(-0.018, 0.018)
                    model_prob = max(0.05, min(0.95, implied + home_advantage + variance))
                    signal = {"model_prob": model_prob, "odds": odds}
                    enriched = enrich_signal(signal)
                    edge = enriched["edge"]
                    kelly = enriched["kelly_quarter"]
                    diff = round(edge, 1)
                    if diff < 3.0 or (model_prob * 100) < CONFIDENCE_FLOOR: continue
                    target_label = f"{team} {side} {line}" if line else f"{team} {side}"
                    alert_dict = {"game": game, "league": "WNBA", "target": target_label, "odds": odds, "book": bookmaker["title"], "prob": round(model_prob*100,1), "diff": diff, "kelly": round(kelly*100,2)}
                    if deduper.should_show(game, alert_dict["target"], line, diff):
                        alerts.append(alert_dict)
    deduper.clear_old()
    telegram_text = deduper.format_telegram(alerts)
    print(telegram_text if telegram_text else f"🟡 [AJ SNIPER] No strong WNBA edges above {CONFIDENCE_FLOOR}% this run.")

if __name__ == "__main__":
    print(f"🚀 WNBA Props Sniper started — {datetime.now().strftime('%H:%M')}")
    while True:
        fetch_and_enrich_wnba_props()
        time.sleep(180)
