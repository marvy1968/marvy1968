        flips = 0

import math

class MicroBeastRiskManager:
    def __init__(self, nav):
        self.nav = float(nav)  # Current balance (e.g., 87718.54)
        self.base_risk = 0.015  # 1.5% Risk per trade
        self.jpy_ceiling_guard = 159.00 # High intervention risk above 159
        
    def calculate_units(self, symbol, stop_loss_pips, current_price, bias_multiplier=1.0):
        """
        Calculates unit size based on a 1.5% risk of total NAV.
        """
        # 1. Calculate Dollar Risk (The amount we are willing to lose)
        risk_amount = self.nav * self.base_risk * bias_multiplier

        # 2. Apply "JPY Intervention Guard" (April 2026 Special)
        # If we are near the 160.00 BoJ ceiling, we cut risk by 75% for safety
        if "JPY" in symbol and current_price > self.jpy_ceiling_guard:
            print(f"⚠️  JPY Intervention Guard Active for {symbol}. Cutting risk to 0.375% for safety.")
            risk_amount *= 0.25 

        # 3. Calculate Position Size in Units
        # Formula for standard pairs (EUR/USD, GBP/CAD): Units = Risk / (StopLoss * PipValue)
        # Pip value for 1 unit is usually 0.0001
        try:
            if "JPY" in symbol:
                # JPY pairs have a different decimal structure (0.01)
                unit_size = risk_amount / (stop_loss_pips * 0.01 * (1 / current_price))
            else:
                # Standard 5-digit pairs
                unit_size = risk_amount / (stop_loss_pips * 0.0001)
            
            final_units = math.floor(unit_size)
            print(f"📊 Sizing Audit | Risk: ${risk_amount:.2f} | Units: {final_units} | Stop: {stop_loss_pips} pips")
            return final_units
        
        except ZeroDivisionError:
            return 0

import os

class StrategyEngine:
    def __init__(self, db_bias_dict=None):
        self.symbol_bias = db_bias_dict or {}
        self.trigger_mult = float(os.getenv("TRIGGER_MULT", "0.8"))
        self.min_heat_score = int(os.getenv("MIN_HEAT_SCORE", "60"))

    def enrich_data(self, df):
        if df is None or len(df) < 200:
            return None
            
        df['ema8'] = df['c'].ewm(span=8, adjust=False).mean()
        df['ema20'] = df['c'].ewm(span=20, adjust=False).mean()
        df['ema50'] = df['c'].ewm(span=50, adjust=False).mean()
        df['ema200'] = df['c'].ewm(span=200, adjust=False).mean()
        
        df['tr0'] = abs(df['h'] - df['l'])
        df['tr1'] = abs(df['h'] - df['c'].shift())
        df['tr2'] = abs(df['l'] - df['c'].shift())
        df['tr'] = df[['tr0', 'tr1', 'tr2']].max(axis=1)
        df['atr'] = df['tr'].rolling(window=14).mean()
        
        return df

    def calculate_heat_score(self, pair, side, price, ema200, current_candle_size, atr):
        score = 0
        if (side == "BUY" and price > ema200) or (side == "SELL" and price < ema200):
            score += 30
            
        if atr > 0:
            vol_ratio = current_candle_size / atr
            score += min(vol_ratio * 20, 40)
            
        bias = self.symbol_bias.get(pair, 1.0)
        score += (bias * 15) 
        
        return round(score, 2)

    def generate_signal(self, pair, raw_df):
        df = self.enrich_data(raw_df)
        if df is None:
            return None

        price = df['c'].iloc[-1]
        ema8 = df['ema8'].iloc[-1]
        ema20 = df['ema20'].iloc[-1]
        ema50 = df['ema50'].iloc[-1]
        ema200 = df['ema200'].iloc[-1]
        atr = df['atr'].iloc[-1]
        
        prev_close = df['c'].iloc[-2]
        current_candle_size = abs(df['h'].iloc[-1] - df['l'].iloc[-1])
        
        if pd.isna(atr) or atr <= 0 or price <= 0:
            return None
            
        dist20 = (price - ema20) / ema20
        trigger = (atr / price) * self.trigger_mult
        
        bullish = price > prev_close
        bearish = price < prev_close
        
        side = None
        signal_type = None
        
        if price > ema50 and dist20 < -trigger and bullish:
            side = "BUY"
            signal_type = "DEEP_PULLBACK"
        elif price < ema50 and dist20 > trigger and bearish:
            side = "SELL"
            signal_type = "DEEP_PULLBACK"
        elif price > ema8 and price > ema20 and bullish:
            side = "BUY"
            signal_type = "MOMENTUM_BREAKOUT"
        elif price < ema8 and price < ema20 and bearish:
            side = "SELL"
            signal_type = "MOMENTUM_BREAKOUT"

        if not side:
            return None

        heat = self.calculate_heat_score(pair, side, price, ema200, current_candle_size, atr)
        
        if heat < self.min_heat_score:
            logging.debug(f"[{pair}] Signal Ignored. Heat {heat} < {self.min_heat_score}")
            return None

        return {
            "pair": pair,
            "side": side,
            "type": signal_type,
            "heat": heat,
            "price": price,
            "atr": atr
        }

TARGET_ATR = 2.2

def fetch_candles(instrument, days=DAYS_BACK):

    r = InstrumentsCandles(instrument=instrument, params=params)

    data = r.response['candles']
    df = pd.DataFrame([{
        'time': pd.to_datetime(c['time']),
        'open': float(c['mid']['o']),
        'high': float(c['mid']['h']),
        'low': float(c['mid']['l']),
        'close': float(c['mid']['c']),
    } for c in data])
    return df.set_index('time')

def rsi(series, period=14):
    delta = series.diff()
    gain = delta.where(delta > 0, 0).rolling(window=period).mean()
    loss = -delta.where(delta < 0, 0).rolling(window=period).mean()
    rs = gain / loss
    return 100 - (100 / (1 + rs))

def bollinger_bands(series, period=20, std=2):
    sma = series.rolling(window=period).mean()
    stddev = series.rolling(window=period).std()
    return sma - stddev * std, sma + stddev * std

def adx(high, low, close, period=14):
    tr = pd.concat([high - low, abs(high - close.shift()), abs(low - close.shift())], axis=1).max(axis=1)
    plus_dm = high - high.shift()
    minus_dm = low.shift() - low
    plus_dm = plus_dm.where((plus_dm > minus_dm) & (plus_dm > 0), 0)
    minus_dm = minus_dm.where((minus_dm > plus_dm) & (minus_dm > 0), 0)
    atr = tr.rolling(period).mean()
    plus_di = 100 * (plus_dm.rolling(period).mean() / atr)
    minus_di = 100 * (minus_dm.rolling(period).mean() / atr)
    dx = (abs(plus_di - minus_di) / (plus_di + minus_di + 1e-10)) * 100
    return dx.rolling(period).mean()

def atr(high, low, close, period=14):
    tr = pd.concat([high - low, abs(high - close.shift()), abs(low - close.shift())], axis=1).max(axis=1)
    return tr.rolling(period).mean()

def run_backtest(df, pair):
    df = df.copy()
    df['rsi'] = rsi(df['close'])
    df['bb_lower'], df['bb_upper'] = bollinger_bands(df['close'])
    df['adx'] = adx(df['high'], df['low'], df['close'])
    df['atr'] = atr(df['high'], df['low'], df['close'])

    df['long_signal'] = (df['rsi'] < RSI_LONG) & (df['close'] <= df['bb_lower'] * BB_MULT_LONG) & (df['adx'] < ADX_MAX)
    df['short_signal'] = (df['rsi'] > RSI_SHORT) & (df['close'] >= df['bb_upper'] * BB_MULT_SHORT) & (df['adx'] < ADX_MAX)

    trades = []
    position = None

    for i in range(80, len(df)):
        row = df.iloc[i]
        prev = df.iloc[i-1]

        if position is None:
            if row['long_signal']:
                entry = row['close']
                stop = entry - row['atr'] * STOP_ATR
                target = entry + row['atr'] * TARGET_ATR
                position = {'direction': 'LONG', 'entry': entry, 'stop': stop, 'target': target, 'entry_time': row.name}
            elif row['short_signal']:
                entry = row['close']
                stop = entry + row['atr'] * STOP_ATR
                target = entry - row['atr'] * TARGET_ATR
                position = {'direction': 'SHORT', 'entry': entry, 'stop': stop, 'target': target, 'entry_time': row.name}

        elif position is not None:
            if position['direction'] == 'LONG':
                if row['low'] <= position['stop']:
                    pnl = (position['stop'] - position['entry']) / position['entry']
                    trades.append({**position, 'exit_time': row.name, 'exit_price': position['stop'], 'pnl': pnl, 'result': 'LOSS'})
                    position = None

def get_historical_data(instrument, count=3000, granularity="M15"):
    """Fetch historical data from OANDA (safe method)"""
    params = {
        "count": count,
        "granularity": granularity
    }
    
    r = InstrumentsCandles(instrument=instrument, params=params)
    client.request(r)

            'time': pd.to_datetime(candle['time']),
            'open': float(candle['mid']['o']),
            'high': float(candle['mid']['h']),
            'low': float(candle['mid']['l']),
            'close': float(candle['mid']['c']),
        })
    return pd.DataFrame(data).set_index('time')

def calculate_indicators(df):
    df = df.copy()
    
    # RSI
    delta = df['close'].diff()
    gain = delta.where(delta > 0, 0).rolling(14).mean()
    loss = -delta.where(delta < 0, 0).rolling(14).mean()
    rs = gain / loss
    df['rsi'] = 100 - (100 / (1 + rs))
    
    # Bollinger Bands
    sma = df['close'].rolling(20).mean()
    std = df['close'].rolling(20).std()
    df['bb_lower'] = sma - std * 2
    
    # ATR
    tr = pd.concat([
        df['high'] - df['low'],
        abs(df['high'] - df['close'].shift()),
        abs(df['low'] - df['close'].shift())
    ], axis=1).max(axis=1)
    df['atr'] = tr.rolling(14).mean()
    
    # ADX
    plus_dm = df['high'] - df['high'].shift()
    minus_dm = df['low'].shift() - df['low']
    plus_dm = plus_dm.where((plus_dm > minus_dm) & (plus_dm > 0), 0)
    minus_dm = minus_dm.where((minus_dm > plus_dm) & (minus_dm > 0), 0)
    atr = tr.rolling(14).mean()
    plus_di = 100 * (plus_dm.rolling(14).mean() / atr)
    minus_di = 100 * (minus_dm.rolling(14).mean() / atr)
    dx = (abs(plus_di - minus_di) / (plus_di + minus_di)) * 100
    df['adx'] = dx.rolling(14).mean()
    
    return df

def run_backtest(df, rsi_entry=35, bb_mult=1.008, adx_max=26, 
                 stop_atr=1.6, target_atr=2.2, trail_atr=1.2, 
                 initial_balance=10000):
    
    df = calculate_indicators(df).dropna()
    
    balance = initial_balance
    equity_curve = [balance]
    trades = []
    in_position = False
    entry_price = stop_price = target_price = 0
    
    for i in range(1, len(df)):
        row = df.iloc[i]
        
        # Manage open position
        if in_position:
            if row['low'] <= stop_price:
                pnl = stop_price - entry_price
                balance += pnl
                trades.append({'type': 'loss', 'pnl': pnl})
                in_position = False
            elif row['high'] >= target_price:
                pnl = target_price - entry_price
                balance += pnl
                trades.append({'type': 'win', 'pnl': pnl})
                in_position = False
            else:
                new_trail = row['close'] - (row['atr'] * trail_atr)
                if new_trail > stop_price:
                    stop_price = new_trail
        
        # New entry signal
        if not in_position:
            if (row['rsi'] < rsi_entry and 
                row['close'] <= row['bb_lower'] * bb_mult and 
                row['adx'] < adx_max):
                
                entry_price = row['close']
                stop_price = entry_price - (row['atr'] * stop_atr)
                target_price = entry_price + (row['atr'] * target_atr)
                in_position = True
        
        equity_curve.append(balance)
    
    wins = [t for t in trades if t['type'] == 'win']
    losses = [t for t in trades if t['type'] == 'loss']
    
    win_rate = len(wins) / len(trades) * 100 if trades else 0
    profit_factor = sum(t['pnl'] for t in wins) / abs(sum(t['pnl'] for t in losses)) if losses else float('inf')
    total_pnl = balance - initial_balance
    
    return {
        'final_balance': round(balance, 2),
        'total_pnl': round(total_pnl, 2),
        'total_trades': len(trades),
        'win_rate': round(win_rate, 1),
        'profit_factor': round(profit_factor, 2),
    }

from __future__ import annotations
from dataclasses import dataclass
from typing import List, Optional, Tuple
from .types import EngineContext, EngineSignal, clamp
from .experts import (
    TrendBreakoutExpert,
    MeanReversionExpert,
    SqueezeExpert,
    ReversalExpert,
    RiskExecutionExpert,
)

@dataclass
class BrainDecision:
    # "long" / "short" / None
    direction: Optional[str]
    confidence: float
    block: bool
    reason: str
    signals: List[EngineSignal]

class ExpertBrain:
    """
    Runs 5 experts IN PARALLEL, then:
      1) Applies hard blocks from risk/execution expert
      2) Chooses the best directional signal by score
      3) Adds regime-aware weighting
    """
    def __init__(self):
        self.e_trend = TrendBreakoutExpert()
        self.e_mr = MeanReversionExpert()
        self.e_sq = SqueezeExpert()
        self.e_rev = ReversalExpert()
        self.e_risk = RiskExecutionExpert()

        self.experts = [self.e_trend, self.e_mr, self.e_sq, self.e_rev]
        self.risk_expert = self.e_risk

    def decide(self, c: EngineContext) -> BrainDecision:
        risk_sig = self.risk_expert.evaluate(c)
        sigs: List[EngineSignal] = [risk_sig]

        # hard block takes precedence
        if risk_sig.block:
            return BrainDecision(
                direction=None,
                confidence=0.0,
                block=True,
                reason=f"BLOCKED: {risk_sig.reason}",
                signals=sigs,
            )

        # evaluate strategy experts in parallel
        strat_sigs = [e.evaluate(c) for e in self.experts]
        sigs.extend(strat_sigs)

        # score each directional signal
        best: Optional[Tuple[float, EngineSignal]] = None
        for s in strat_sigs:
            if not s.direction:
                continue

            # base score = confidence
            score = float(s.confidence)

            # regime weighting (institutional style allocation)
            if c.regime == "trend":
                if s.name in ("trend_breakout", "squeeze_expansion"):
                    score *= 1.12
                if s.name in ("mean_reversion",):
                    score *= 0.85
            else:  # range
                if s.name in ("mean_reversion", "reversal_exhaustion"):
                    score *= 1.10
                if s.name in ("trend_breakout",):
                    score *= 0.80

            # bias alignment bonus
            if c.bias == s.direction:
                score *= 1.08

            if best is None or score > best[0]:
                best = (score, s)

        if best is None:
            return BrainDecision(
                direction=None,
                confidence=0.0,
                block=False,
                reason="no expert directional edge",
                signals=sigs,
            )

        score, s = best
        return BrainDecision(
            direction=s.direction,
            confidence=clamp(score, 0.0, 1.0),
            block=False,
            reason=f"{s.name}: {s.reason}",
            signals=sigs,
        )

from __future__ import annotations
from dataclasses import dataclass
from typing import Optional, List
from .types import EngineContext, EngineSignal, clamp

# ---------------------------------------------------------
# Expert 1: Trend Breakout (Donchian / HH-LL continuation)
# ---------------------------------------------------------
@dataclass
class TrendBreakoutExpert:
    name: str = "trend_breakout"

    def evaluate(self, c: EngineContext) -> EngineSignal:
        if c.donH is None or c.donL is None:
            return EngineSignal(self.name, None, 0.0, reason="missing donchian")

        # Only favor in trend regime
        if c.regime != "trend":
            return EngineSignal(self.name, None, 0.15, reason="not trend regime")

        # breakout logic
        long_ok = c.price >= (c.donH + c.buf)
        short_ok = c.price <= (c.donL - c.buf)

        if long_ok and c.bias in ("neutral", "long"):
            conf = 0.55 + 0.01 * float(c.adx or 0)
            return EngineSignal(self.name, "long", clamp(conf, 0.0, 0.95), reason="donchian breakout long")
        if short_ok and c.bias in ("neutral", "short"):
            conf = 0.55 + 0.01 * float(c.adx or 0)
            return EngineSignal(self.name, "short", clamp(conf, 0.0, 0.95), reason="donchian breakout short")

        return EngineSignal(self.name, None, 0.20, reason="no breakout")

# ---------------------------------------------------------
# Expert 2: Range Mean Reversion (fade edges in balance)
# ---------------------------------------------------------
@dataclass
class MeanReversionExpert:
    name: str = "mean_reversion"

    def evaluate(self, c: EngineContext) -> EngineSignal:
        if c.donH is None or c.donL is None:
            return EngineSignal(self.name, None, 0.0, reason="missing donchian")

        if c.regime != "range":
            return EngineSignal(self.name, None, 0.15, reason="not range regime")

        mid = (c.donH + c.donL) / 2.0
        span = max(1e-9, (c.donH - c.donL))

        # how close to edges
        near_low = (c.price - c.donL) <= (0.18 * span)
        near_high = (c.donH - c.price) <= (0.18 * span)

        if near_low and c.bias in ("neutral", "long"):
            # more confident if ADX low (more rangy)
            adx = float(c.adx or 0.0)
            conf = 0.55 + clamp((22.0 - adx) / 50.0, 0.0, 0.25)
            return EngineSignal(self.name, "long", clamp(conf, 0.0, 0.85), reason="fade lower range edge")

        if near_high and c.bias in ("neutral", "short"):
            adx = float(c.adx or 0.0)
            conf = 0.55 + clamp((22.0 - adx) / 50.0, 0.0, 0.25)
            return EngineSignal(self.name, "short", clamp(conf, 0.0, 0.85), reason="fade upper range edge")

        return EngineSignal(self.name, None, 0.25, reason="no range edge")

# ---------------------------------------------------------
# Expert 3: Volatility Squeeze / Expansion (compression -> break)
# ---------------------------------------------------------
@dataclass
class SqueezeExpert:
    name: str = "squeeze_expansion"
    atr_floor: float = 1e-9

    def evaluate(self, c: EngineContext) -> EngineSignal:
        if c.atr is None or c.atr <= self.atr_floor:
            return EngineSignal(self.name, None, 0.0, block=True, reason="bad atr")

        # crude squeeze: ATR small vs price
        atr_pct = float(c.atr) / max(1e-9, float(c.price))
        squeeze = atr_pct < 0.00035  # tune for EURUSD

        if not squeeze:
            return EngineSignal(self.name, None, 0.20, reason="no squeeze")

        # if squeezing, prefer breakout direction if price is near a boundary
        if c.donH is not None and c.price >= (c.donH - 0.25 * (c.donH - (c.donL or c.donH))):
            return EngineSignal(self.name, "long", 0.60, reason="squeeze near high -> expansion long")
        if c.donL is not None and c.price <= (c.donL + 0.25 * ((c.donH or c.donL) - c.donL)):
            return EngineSignal(self.name, "short", 0.60, reason="squeeze near low -> expansion short")

        return EngineSignal(self.name, None, 0.40, reason="squeeze but no directional edge")

# ---------------------------------------------------------
# Expert 4: Reversal / Exhaustion (failed breakout / stop run)
# ---------------------------------------------------------
@dataclass
class ReversalExpert:
    name: str = "reversal_exhaustion"
    wick_factor: float = 2.2  # how strong reversal needs to be

    def evaluate(self, c: EngineContext) -> EngineSignal:
        # needs bid/ask (or last extremes) to infer "wick"/rejection
        if c.bid is None or c.ask is None or c.atr is None or c.atr <= 0:
            return EngineSignal(self.name, None, 0.0, reason="missing bid/ask/atr")

        # rejection proxy: spread-adjusted move vs ATR (simple)
        # If price is above donH but snaps back under it -> short reversal
        if c.donH is not None and c.price < c.donH and c.ask > c.donH:
            # stronger if ADX is low-ish (reversal more likely in balance)
            adx = float(c.adx or 0.0)
            conf = 0.55 + clamp((28.0 - adx) / 80.0, 0.0, 0.20)
            return EngineSignal(self.name, "short", clamp(conf, 0.0, 0.80), reason="failed break above high -> reverse short")

        # If price is below donL but snaps back over it -> long reversal
        if c.donL is not None and c.price > c.donL and c.bid < c.donL:
            adx = float(c.adx or 0.0)
            conf = 0.55 + clamp((28.0 - adx) / 80.0, 0.0, 0.20)
            return EngineSignal(self.name, "long", clamp(conf, 0.0, 0.80), reason="failed break below low -> reverse long")

        return EngineSignal(self.name, None, 0.20, reason="no reversal pattern")

class RiskExecutionExpert:
    name: str = "risk_execution"
    max_spread_pips: float = 2.0
    stale_ticks_block: int = 3

    # internal state (per-bot instance)
    _last_tick_time: Optional[str] = None
    _stale_run: int = 0

    def evaluate(self, c: EngineContext) -> EngineSignal:

        else:
            self._stale_run = 0

        if self._stale_run >= self.stale_ticks_block:

        # spread gate
        if c.spr_pips is not None and float(c.spr_pips) >= self.max_spread_pips:
            return EngineSignal(self.name, None, 1.0, block=True, reason=f"SPREAD HIGH {c.spr_pips:.2f}p")

        return EngineSignal(self.name, None, 0.10, block=False, reason="ok")

Manages Swing Trend, Daytrade/Scalp, and Crypto Range bots

import os
import sys
import time
import subprocess
import signal
from pathlib import Path

BOTS = {
    "swing": "alpaca_swing_trend_bot.py",
    "daytrade": "alpaca_daytrade_scalp_bot.py",
    "crypto": "alpaca_crypto_range_bot.py"
}

LOG_DIR = Path("logs")
LOG_DIR.mkdir(exist_ok=True)

processes = {}

def load_env():
    env_file = Path("../RangeBox_Master/start_env.sh")
    if env_file.exists():
        os.system(f"source {env_file} && env > /tmp/env.txt")
        with open("/tmp/env.txt") as f:
            for line in f:
                if "=" in line:
                    key, value = line.strip().split("=", 1)
                    os.environ[key] = value

def start_bot(name, script):
    if name in processes and processes[name].poll() is None:
        print(f"{name} is already running.")
        return

    log_file = LOG_DIR / f"{name}.log"
    with open(log_file, "a") as log:
        proc = subprocess.Popen(
            [sys.executable, script],
            stdout=log,
            stderr=log,
            preexec_fn=os.setsid
        )
        processes[name] = proc
        print(f"✅ Started {name} (PID: {proc.pid})")

def stop_bot(name):
    if name in processes:
        proc = processes[name]
        if proc.poll() is None:
            os.killpg(os.getpgid(proc.pid), signal.SIGTERM)
            print(f"🛑 Stopped {name}")
        del processes[name]

def stop_all():
    for name in list(processes.keys()):
        stop_bot(name)
    print("All bots stopped.")

def status():
    print("\n=== Bot Status ===")
    for name, script in BOTS.items():
        if name in processes and processes[name].poll() is None:
            print(f"  {name}: RUNNING (PID {processes[name].pid})")
        else:
            print(f"  {name}: STOPPED")

def main():
    load_env()

    print("=== Trading Bot Master Controller ===")
    print("Commands: start <bot>, stop <bot>, status, stopall, exit")

    while True:
        try:
            cmd = input("\n> ").strip().lower().split()
            if not cmd:
                continue

            if cmd[0] == "start" and len(cmd) > 1:
                bot = cmd[1]
                if bot in BOTS:
                    start_bot(bot, BOTS[bot])
                elif bot == "all":
                    for b, s in BOTS.items():
                        start_bot(b, s)
                else:
                    print("Unknown bot. Use: swing, daytrade, crypto, all")

            elif cmd[0] == "stop" and len(cmd) > 1:
                bot = cmd[1]
                if bot in BOTS:
                    stop_bot(bot)
                elif bot == "all":
                    stop_all()
                else:
                    print("Unknown bot.")

            elif cmd[0] == "status":
                status()

            elif cmd[0] == "stopall":
                stop_all()

            elif cmd[0] in ["exit", "quit"]:
                stop_all()
                break

            else:
                print("Commands: start <bot|all>, stop <bot|all>, status, stopall, exit")

        except KeyboardInterrupt:
            stop_all()
            break

if __name__ == "__main__":
    main()

import numpy as np

class MasterSimulator:
    def __init__(self, initial_equity=100000, simulations=3000, days=252):
        self.initial_equity = initial_equity
        self.simulations = simulations
        self.days = days

    def run(self, kelly=0.6, vol_rank=70, use_wyckoff=True, use_lps=True):
        results = []
        max_dds = []

        for _ in range(self.simulations):
            equity = self.initial_equity
            peak = equity
            drawdowns = []

            for day in range(self.days):
                is_trending = np.random.random() < 0.6
                vol = 0.018 * (1.4 if is_trending else 0.8)

                # Volume Rank Filter
                if np.random.random() * 100 < vol_rank:
                    continue

                # Wyckoff / LPS bias (higher win rate in accumulation)
                wyckoff_bonus = 0.08 if (use_wyckoff and np.random.random() < 0.35) else 0
                lps_bonus = 0.12 if (use_lps and np.random.random() < 0.25) else 0

                win_rate = 0.58 + wyckoff_bonus + lps_bonus
                risk = min(kelly, 0.05)

                if np.random.random() < win_rate:
                    ret = np.random.normal(2.5 * vol, vol)
                else:
                    ret = np.random.normal(-1.0 * vol, vol)

                equity *= (1 + ret * risk)
                peak = max(peak, equity)
                dd = (peak - equity) / peak
                drawdowns.append(dd)

            results.append(equity)
            max_dds.append(max(drawdowns) if drawdowns else 0)

        results = np.array(results)
        max_dds = np.array(max_dds)

        print("\n" + "="*75)
        print("MASTER SIMULATION - ALL EXPERTS + FEATURES")
        print("="*75)

def calculate_indicators(df):
    delta = df['close'].diff()
    gain = delta.where(delta > 0, 0).rolling(14).mean()
    loss = -delta.where(delta < 0, 0).rolling(14).mean()
    df['rsi'] = 100 - (100 / (1 + gain / loss))

    sma = df['close'].rolling(20).mean()
    std = df['close'].rolling(20).std()
    df['bb_lower'] = sma - std * 2
    df['bb_upper'] = sma + std * 2

    tr = pd.concat([df['high'] - df['low'],
                    abs(df['high'] - df['close'].shift()),
                    abs(df['low'] - df['close'].shift())], axis=1).max(axis=1)
    df['atr'] = tr.rolling(14).mean()

    plus_dm = df['high'] - df['high'].shift()
    minus_dm = df['low'].shift() - df['low']
    plus_dm = plus_dm.where((plus_dm > minus_dm) & (plus_dm > 0), 0)
    minus_dm = minus_dm.where((minus_dm > plus_dm) & (minus_dm > 0), 0)
    atr = tr.rolling(14).mean()
    plus_di = 100 * (plus_dm.rolling(14).mean() / atr)
    minus_di = 100 * (minus_dm.rolling(14).mean() / atr)
    dx = (abs(plus_di - minus_di) / (plus_di + minus_di + 1e-10)) * 100
    df['adx'] = dx.rolling(14).mean()

    return df

def run_backtest(df, params):
    df = df.copy()
    df['long_signal'] = (df['rsi'] < params['rsi_long']) & \
                        (df['close'] <= df['bb_lower'] * params['bb_mult_long']) & \
                        (df['adx'] < params['adx_max'])

    df['short_signal'] = (df['rsi'] > params['rsi_short']) & \
                         (df['close'] >= df['bb_upper'] * params['bb_mult_short']) & \
                         (df['adx'] < params['adx_max'])

    trades = []
    position = None

    for i in range(80, len(df)):
        row = df.iloc[i]

        if position is None:
            if row['long_signal']:
                entry = row['close']
                stop = entry - row['atr'] * params['stop_atr']
                target = entry + row['atr'] * params['target_atr']
                position = {'dir': 'LONG', 'entry': entry, 'stop': stop, 'target': target}
            elif row['short_signal']:
                entry = row['close']
                stop = entry + row['atr'] * params['stop_atr']
                target = entry - row['atr'] * params['target_atr']
                position = {'dir': 'SHORT', 'entry': entry, 'stop': stop, 'target': target}

        elif position is not None:
            if position['dir'] == 'LONG':
                if row['low'] <= position['stop']:
                    pnl = (position['stop'] - position['entry']) / position['entry']
                    trades.append({'pnl': pnl, 'result': 'LOSS'})
                    position = None
                elif row['high'] >= position['target']:
                    pnl = (position['target'] - position['entry']) / position['entry']
                    trades.append({'pnl': pnl, 'result': 'WIN'})
                    position = None
            else:
                if row['high'] >= position['stop']:
                    pnl = (position['entry'] - position['stop']) / position['entry']
                    trades.append({'pnl': pnl, 'result': 'LOSS'})
                    position = None
                elif row['low'] <= position['target']:
                    pnl = (position['entry'] - position['target']) / position['entry']
                    trades.append({'pnl': pnl, 'result': 'WIN'})
                    position = None

    return pd.DataFrame(trades)

    pf = abs(df_trades[df_trades['result'] == 'WIN']['pnl'].sum() /

def calculate_indicators(df):
    delta = df['close'].diff()
    gain = delta.where(delta > 0, 0).rolling(14).mean()
    loss = -delta.where(delta < 0, 0).rolling(14).mean()
    df['rsi'] = 100 - (100 / (1 + gain / loss))

    sma = df['close'].rolling(20).mean()
    std = df['close'].rolling(20).std()
    df['bb_lower'] = sma - std * 2
    df['bb_upper'] = sma + std * 2

    tr = pd.concat([
        df['high'] - df['low'],
        abs(df['high'] - df['close'].shift()),
        abs(df['low'] - df['close'].shift())
    ], axis=1).max(axis=1)
    df['atr'] = tr.rolling(14).mean()

    plus_dm = df['high'] - df['high'].shift()
    minus_dm = df['low'].shift() - df['low']
    plus_dm = plus_dm.where((plus_dm > minus_dm) & (plus_dm > 0), 0)

#!/usr/bin/env python3
"""
Automated Performance Reporter for Trading Bots
"""

import pandas as pd
import json
from pathlib import Path
from datetime import datetime, timedelta
import glob

class PerformanceReporter:
    def __init__(self, journal_path="trade_journal.csv", logs_dir="logs"):
        self.journal_path = Path(journal_path)
        self.logs_dir = Path(logs_dir)

    def load_trades(self):
        if not self.journal_path.exists():
            # Fallback: try to parse from logs
            return self.parse_from_logs()
        
        try:
            df = pd.read_csv(self.journal_path)
            df['timestamp'] = pd.to_datetime(df['timestamp'])
            return df
        except:
            return pd.DataFrame()

    def parse_from_logs(self):
        # Simple fallback parser
        trades = []
        for log_file in glob.glob(str(self.logs_dir / "*.log")):
            with open(log_file) as f:
                for line in f:
                    if "CLOSE" in line and "PnL" in line:
                        try:
                            parts = line.split("|")
                            pnl_str = [p for p in parts if "PnL" in p][0]
                            pnl = float(pnl_str.split("$")[1].split()[0])
                            trades.append({"pnl": pnl, "timestamp": datetime.now()})
                        except:
                            pass
        return pd.DataFrame(trades)

    def generate_report(self, days=30):
        df = self.load_trades()
        if df.empty:
            return "No trade data found."

        # Filter recent trades
        cutoff = datetime.now() - timedelta(days=days)
        if 'timestamp' in df.columns:
            df = df[df['timestamp'] >= cutoff]

        if df.empty:
            return f"No trades in the last {days} days."

        total_pnl = df['pnl'].sum()
        wins = df[df['pnl'] > 0]
        win_rate = len(wins) / len(df) * 100
        avg_win = wins['pnl'].mean() if not wins.empty else 0
        avg_loss = df[df['pnl'] <= 0]['pnl'].mean() if len(df[df['pnl'] <= 0]) > 0 else 0

        # Simple max drawdown approximation
        equity_curve = df['pnl'].cumsum()
        peak = equity_curve.cummax()
        drawdown = (peak - equity_curve) / peak.max() * 100
        max_dd = drawdown.max()

        report = f"""
╔══════════════════════════════════════════════════════════════╗
║           TRADING PERFORMANCE REPORT - LAST {days} DAYS       ║
╚══════════════════════════════════════════════════════════════╝

Total Trades     : {len(df)}
Win Rate         : {win_rate:.1f}%
Total P&L        : ${total_pnl:,.2f}
Average Win      : ${avg_win:,.2f}
Average Loss     : ${avg_loss:,.2f}
Max Drawdown     : {max_dd:.2f}%

Profit Factor    : {abs(avg_win / avg_loss) if avg_loss != 0 else 'N/A':.2f}

Report generated: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}
"""
        return report

if __name__ == "__main__":
    reporter = PerformanceReporter()
    print(reporter.generate_report(days=7))   # Last 7 days
    print(reporter.generate_report(days=30))  # Last 30 days

#!/usr/bin/env python3
"""
Ross Cameron Momentum Backtester - Realistic
"""

import pandas as pd
import numpy as np

def ross_cameron_backtest(df, initial_capital=100000):
    capital = initial_capital
    position = 0
    entry_price = 0
    highest_since_entry = 0
    trades = []
    equity_curve = [initial_capital]

    for i in range(30, len(df)):
        price = df['close'].iloc[i]
        volume = df['volume'].iloc[i]
        avg_vol = df['volume'].rolling(20).mean().iloc[i]
        high_5 = df['high'].rolling(5).max().iloc[i-1]

        # Entry
        if position == 0 and price > high_5 * 1.01 and volume > avg_vol * 2.2:
            position = capital * 0.97 / price
            entry_price = price
            highest_since_entry = price
            trades.append({"type": "entry", "price": price})

        # Trailing stop + target
        elif position > 0:
            highest_since_entry = max(highest_since_entry, price)
            trail_stop = highest_since_entry * 0.96   # 4% trailing

            pnl_pct = (price - entry_price) / entry_price
            if price >= entry_price * 1.05 or price <= trail_stop:
                capital = position * price
                position = 0
                trades.append({"type": "exit", "price": price, "pnl": pnl_pct})

        equity_curve.append(capital if position == 0 else position * price)

    final_capital = capital if position == 0 else position * df['close'].iloc[-1]
    win_trades = len([t for t in trades if t.get("pnl", 0) > 0])
    total_closed = len([t for t in trades if "pnl" in t])
    win_rate = (win_trades / total_closed * 100) if total_closed > 0 else 0.0

    print("\n" + "="*75)
    print("ROSS CAMERON MOMENTUM BACKTEST - REALISTIC")
    print("="*75)
    print(f"Initial Capital : ${initial_capital:,.0f}")
    print(f"Final Capital   : ${final_capital:,.0f}")
    print(f"Total Return    : {(final_capital/initial_capital - 1)*100:.1f}%")
    print(f"Total Trades    : {total_closed}")
    print(f"Win Rate        : {win_rate:.1f}%")
    print(f"Max Equity      : ${max(equity_curve):,.0f}")
    print("="*75)

#!/usr/bin/env python3
"""
Ross Cameron Momentum Parameter Optimizer
"""

import pandas as pd
import numpy as np
from itertools import product

def backtest_with_params(df, breakout_mult=1.01, vol_mult=2.2, profit_target=0.05, trail_pct=0.04, initial_capital=100000):
    capital = initial_capital
    position = 0
    entry_price = 0
    highest_since_entry = 0
    trades = 0
    wins = 0

    for i in range(30, len(df)):
        price = df['close'].iloc[i]
        volume = df['volume'].iloc[i]
        avg_vol = df['volume'].rolling(20).mean().iloc[i]
        high_5 = df['high'].rolling(5).max().iloc[i-1]

        if position == 0 and price > high_5 * breakout_mult and volume > avg_vol * vol_mult:
            position = capital * 0.97 / price
            entry_price = price
            highest_since_entry = price

        elif position > 0:
            highest_since_entry = max(highest_since_entry, price)
            trail_stop = highest_since_entry * (1 - trail_pct)
            pnl_pct = (price - entry_price) / entry_price

            if price >= entry_price * (1 + profit_target) or price <= trail_stop:
                capital = position * price
                position = 0
                trades += 1
                if pnl_pct > 0:
                    wins += 1

    final_capital = capital
    win_rate = (wins / trades * 100) if trades > 0 else 0
    return final_capital, trades, win_rate

if __name__ == "__main__":
    # Generate realistic data
    np.random.seed(42)
    n = 1200
    dates = pd.date_range('2024-01-01', periods=n)
    returns = np.random.normal(0.0005, 0.016, n)
    returns[90:180] *= 4.0
    returns[420:510] *= 3.5
    returns[780:860] *= 3.8
    returns[1050:1130] *= 3.2
    close = 100 * np.exp(np.cumsum(returns))
    high = close * (1 + np.random.uniform(0.01, 0.035, n))
    volume = np.random.randint(100000, 1200000, n).astype(float)
    volume[90:180] *= 4.5
    volume[420:510] *= 4.0
    volume[780:860] *= 4.2
    volume[1050:1130] *= 3.8

    df = pd.DataFrame({'close': close, 'high': high, 'volume': volume}, index=dates)

    print("Starting Parameter Optimization...\n")

    best_return = 0
    best_params = None

    # Grid search
    for breakout_mult, vol_mult, profit_target, trail_pct in product(
        [1.005, 1.008, 1.01, 1.015],
        [1.8, 2.0, 2.2, 2.5],
        [0.04, 0.05, 0.06],
        [0.035, 0.04, 0.045]
    ):
        final, trades, win_rate = backtest_with_params(df, breakout_mult, vol_mult, profit_target, trail_pct)
        ret = (final / 100000 - 1) * 100

        if ret > best_return:
            best_return = ret
            best_params = (breakout_mult, vol_mult, profit_target, trail_pct)

        if trades > 5:
            print(f"Breakout: {breakout_mult:.3f} | Vol: {vol_mult:.1f} | Target: {profit_target:.2f} | Trail: {trail_pct:.3f} → Return: {ret:.1f}% | Trades: {trades} | Win: {win_rate:.1f}%")

    print("\n" + "="*70)
    print("BEST PARAMETERS FOUND")
    print("="*70)
    print(f"Breakout Mult : {best_params[0]}")
    print(f"Volume Mult   : {best_params[1]}")
    print(f"Profit Target : {best_params[2]}")
    print(f"Trail %       : {best_params[3]}")
    print(f"Best Return   : {best_return:.1f}%")
    print("="*70)

import os
import numpy as np
import pandas as pd
from dotenv import load_dotenv
import v20

load_dotenv("/home/moonrider1968/bots/.env")

OANDA_API_KEY = os.getenv("OANDA_API_KEY")
OANDA_ACCOUNT_ID = os.getenv("OANDA_ACCOUNT_ID")
OANDA_HOST = "api-fxpractice.oanda.com"  # Use practice/paper environment

def fetch_oanda_candles(instrument="EUR_USD", granularity="M15", count=250):
    ctx = v20.Context(OANDA_HOST, 443, ssl=True, token=OANDA_API_KEY)
    response = ctx.instrument.candles(
        instrument,
        granularity=granularity,
        count=count,
        price="M"
    )
    
    if response.status != 200:
        raise RuntimeError(f"OANDA API Error: {response.body}")
        
    records = []
    for c in response.body.get('candles', []):
        if not c.complete:
            continue
        records.append({
            "time": c.time,
            "open": float(c.mid.o),
            "high": float(c.mid.h),
            "low": float(c.mid.l),
            "close": float(c.mid.c),
            "volume": int(c.volume)
        })
    df = pd.DataFrame(records)
    df['time'] = pd.to_datetime(df['time'])
    return df

def compute_indicators(df):
    close = df['close']
    
    # 1. RSI (14)
    delta = close.diff()
    gain = delta.where(delta > 0, 0.0)
    loss = -delta.where(delta < 0, 0.0)
    avg_gain = gain.rolling(14).mean()
    avg_loss = loss.rolling(14).mean()
    rs = avg_gain / avg_loss.replace(0, np.nan)
    df['rsi'] = 100 - (100 / (1 + rs))

    # 2. Bollinger Bands (20, 2)
    df['bb_mid'] = close.rolling(20).mean()
    bb_std = close.rolling(20).std()
    df['bb_lower'] = df['bb_mid'] - (2 * bb_std)
    df['bb_upper'] = df['bb_mid'] + (2 * bb_std)

    # 3. Stochastic Oscillator (14, 3, 3)
    low14 = df['low'].rolling(14).min()
    high14 = df['high'].rolling(14).max()
    df['stoch_k'] = 100 * ((close - low14) / (high14 - low14))
    df['stoch_d'] = df['stoch_k'].rolling(3).mean()

    # 4. Z-Score (Distance from 20-period SMA in standard deviations)
    df['zscore'] = (close - df['bb_mid']) / bb_std

    # 5. Commodity Channel Index (CCI 20)
    tp = (df['high'] + df['low'] + close) / 3
    tp_sma = tp.rolling(20).mean()
    mad = (tp - tp_sma).abs().rolling(20).mean()
    df['cci'] = (tp - tp_sma) / (0.015 * mad)

    return df

def evaluate_consensus(df):
    df = compute_indicators(df)
    
    # 5 BUY Signals
    b1 = df['rsi'] < 30
    b2 = df['close'] <= df['bb_lower']
    b3 = (df['stoch_k'] < 20) & (df['stoch_k'] > df['stoch_d'])
    b4 = df['zscore'] < -2.0
    b5 = df['cci'] < -100
    
    df['buy_votes'] = b1.astype(int) + b2.astype(int) + b3.astype(int) + b4.astype(int) + b5.astype(int)
    
    # 3 SELL / EXIT Signals
    s1 = df['rsi'] > 55
    s2 = df['close'] >= df['bb_mid']
    s3 = df['stoch_k'] > 70
    
    df['sell_votes'] = s1.astype(int) + s2.astype(int) + s3.astype(int)
    
    return df

if __name__ == "__main__":
    instrument = "EUR_USD"
    print(f"[*] Fetching candles for {instrument}...")
    df = fetch_oanda_candles(instrument=instrument, granularity="M15", count=250)
    df = evaluate_consensus(df)
    
    latest = df.iloc[-1]
    print("=" * 60)
    print(f"Latest Bar:   {latest['time']}")
    print(f"Close Price:  {latest['close']:.5f}")
    print(f"RSI(14):      {latest['rsi']:.1f}")
    print(f"Z-Score:      {latest['zscore']:.2f}")
    print(f"CCI(20):      {latest['cci']:.1f}")
    print("-" * 60)
    print(f"BUY Consensus:  {int(latest['buy_votes'])} / 5 {'[BUY TRIGGER]' if latest['buy_votes'] >= 4 else '[NO ENTRY]'}")
    print(f"SELL Consensus: {int(latest['sell_votes'])} / 3 {'[EXIT TRIGGER]' if latest['sell_votes'] >= 2 else '[HOLD]'}")
    print("=" * 60)
