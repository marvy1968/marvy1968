def transport_with(statuses):
    calls = []

    def t(url, params):
        calls.append(params["apiKey"])
        code = statuses.get(params["apiKey"], 200)
        rem = {"PAIDKEY123": 19997, "FREEKEY456": 492}[params["apiKey"]]
        return code, {"x-requests-remaining": str(rem), "x-requests-last": "3" if code == 200 else "0"}, \
            ([] if code == 200 else {"message": "nope"})
    return t, calls

def add(db, strategy="value", league="nfl", market="h2h", selection="Chicago Bears", won=True, p=0.5,
        price=100, stake=100.0, tag=None, sent=False, details=None, player=None, n=1):
    for _ in range(n):
        aid = db.insert_alert(dict(dedupe_key=f"k{db.conn.execute('SELECT COUNT(*) FROM alerts').fetchone()[0]}",
                                   event_id="e", league=league, strategy=strategy, market=market,
                                   selection=selection, price=price, book="bovada", fair_p=p,
                                   trigger="watchlist" if tag else "x", message="m",
                                   details_json=json.dumps(details or {})))
        if sent:
            db.mark_sent(aid)
        bid = db.insert_paper_bet(dict(alert_id=aid, event_id="e", league=league, strategy=strategy, market=market,
                                       selection=selection, side="home", price=price, book="bovada", stake=stake,
                                       to_win=stake * price / 100.0, fair_p=p, tag=tag, sport=league, player=player))
        db.settle_bet(bid, "won" if won else "lost", stake * price / 100.0 if won else -stake, "1-0")

class FakeOdds:
    def __init__(self, home, away, with_bovada=True, market="pitcher_strikeouts"):
        self.requests, self.home, self.away, self.bov, self.market = [], home, away, with_bovada, market
        self.remaining = 90000

    def __call__(self, url, params):
        self.requests.append((url, dict(params)))
        if url.endswith("/events"):
            return 200, {"x-requests-last": "0", "x-requests-remaining": str(self.remaining)}, [
                {"id": "ev1", "home_team": self.home, "away_team": self.away, "commence_time": START}]
        if "/events/" in url:
            n = len(params["markets"].split(","))
            self.remaining -= n
            books = [_book(b, -110, -110, self.market) for b in ("draftkings", "fanduel", "betmgm")]
            if self.bov:
                books.append(_book("bovada", 125, -160, self.market))
            return 200, {"x-requests-last": str(n), "x-requests-remaining": str(self.remaining)}, \
                {"id": "ev1", "home_team": self.home, "away_team": self.away, "commence_time": START, "bookmakers": books}
        raise AssertionError(f"unexpected call {url}")       # never a featured / live /odds call

    def __init__(self, cfg, db, pcfg=None, answer=None, fail=False):
        super().__init__(cfg, db, pcfg)
        self.answer, self.fail = answer, fail

    def odds(self, sport_key, markets, strategy="value"):
        self.budget.check(1)
        if self.fail:
            raise ProviderError("down")
        self.calls += 1
        self.budget.record("odds", sport_key, 200, 1, "", strategy)
        return copy.deepcopy(self.answer)

class R_:
    def __init__(self, code, body=None):
        self.status_code, self._b = code, body or {}

    def json(self):
        return self._b


def live_notifier(cfg, db=None, responses=None):
    n = Notifier(cfg, dry_run=False)
    n._token, n._chat, n._static, n._rfile = "TKN", "111", [], None
    n.sleeps, n.posts = [], []
    seq = list(responses or [])

    def post(data):
        n.posts.append(data)
        r = seq.pop(0) if seq else R_(200)
        if isinstance(r, Exception):
            raise r
        return r
    n._post = post
    n._sleep = lambda s: n.sleeps.append(s)
    if db is not None:
        n.attach_outbox(db, 900)
    return n

def test_daily_live_cap_is_a_hard_stop_across_providers(cfg):
    from ajbet.providers.hub import OddsHub
    from ajbet.db import today_local
    db = DB()
    calls = []

    class P:
        def __init__(self, name, role, prio, exc=None):
            self.name, self.role, self.priority, self.available, self.calls, self.exc = name, role, prio, True, 0, exc

        def odds(self, *a, **k):
            calls.append(self.name)
            if self.exc:
                raise self.exc
            return []
    hub = OddsHub([P("theoddsapi", "primary", 10, BudgetExceeded("live daily allocation: used 599 + 3 > 600")),
                   P("oddsapiio", "fallback", 20)], db, allocations={"live": {"daily": 600}})
    try:
        hub.live_odds("americanfootball_ncaaf", strategy="live")
        raise AssertionError("expected BudgetExceeded")
    except BudgetExceeded:
        pass
    assert calls == ["theoddsapi"]                      # the allocation refusal never falls through to odds-api.io
    db.log_usage("odds", "x", 200, 600, 1000, 1, "", "live", "paid")
    calls.clear()
    try:
        hub.live_odds("basketball_wnba", strategy="live")
        raise AssertionError("expected BudgetExceeded")
    except BudgetExceeded as e:
        assert "hard stop" in str(e)
    assert calls == []                                  # cap reached -> no provider is even tried
    assert db.credits_used_on(today_local(), "live") == 600

def odds_event(fav_ml, dog_ml, fav_spread, books=1):
    bks = []
    for i in range(books):
        bks.append({"key": f"b{i}", "title": f"Book{i}", "markets": [
            {"key": "h2h", "outcomes": [{"name": "Home Team", "price": fav_ml}, {"name": "Away Team", "price": dog_ml}]},
            {"key": "spreads", "outcomes": [{"name": "Home Team", "price": -110, "point": fav_spread},
                                            {"name": "Away Team", "price": -110, "point": -fav_spread}]}]})
    return {"id": "oa1", "home_team": "Home Team", "away_team": "Away Team",
            "commence_time": "2026-10-01T23:00:00Z", "bookmakers": bks}

class FakeResp:
    def __init__(self, body, code=200):
        self.body, self.status_code = body, code

    def json(self):
        return self.body


class FakeHTTP:
    def __init__(self, updates):
        self.updates, self.sent = updates, []

    def post(self, url, data=None, timeout=None):
        method = url.rsplit("/", 1)[-1]
        if method == "getUpdates":
            ups, self.updates = self.updates, []
            return FakeResp({"ok": True, "result": ups})
        self.sent.append((method, data))
        return FakeResp({"ok": True, "result": {}})

def _season(team, season, covers, games, start_day=1):
    out = []
    for i in range(games):
        hs = 30 if i < covers else 10                 # spread 3.0 -> covers iff margin > 3
        out.append(g(season, f"{season}-09-{start_day + i:02d}", team, f"O{i}", hs, 20, 3.0, 45.0))
    return out


def test_qualifies_only_when_discovery_and_out_of_sample_agree():
    games = _season("KC", 2022, 7, 10) + _season("KC", 2023, 7, 10) + _season("KC", 2024, 6, 10) \
        + _season("KC", 2025, 6, 9)
    ev = T.evaluate("nfl", T.team_rows(games, "nfl"), 2025, {})
    q = {x["id"]: x for x in ev["teams"]["KC"]["qualified"]}
    assert "ats_cover_all" in q and "ats_cover_home" in q
    x = q["ats_cover_all"]
    assert x["disc"] == [20, 30] and x["test"] == [6, 9] and x["disc_seasons"] == [2022, 2023, 2024]
    assert abs(x["p"] - (26 + 10) / (39 + 20)) < 1e-3                   # shrunk toward 50%
    # same discovery, but the latest season flips -> not confirmed out-of-sample
    games2 = games[:30] + _season("KC", 2025, 3, 9)

    assert not any(x["id"] == "ats_cover_all" for x in ev2["teams"]["KC"]["qualified"])
    # one discovery season under 50% -> fails "holds every season"
    games3 = _season("KC", 2022, 9, 10) + _season("KC", 2023, 9, 10) + _season("KC", 2024, 4, 10) \
        + _season("KC", 2025, 6, 9)
    ev3 = T.evaluate("nfl", T.team_rows(games3, "nfl"), 2025, {})
    assert not any(x["id"] == "ats_cover_all" for x in ev3["teams"]["KC"]["qualified"])

    ev4 = T.evaluate("nfl", T.team_rows(games4, "nfl"), 2025, {})

def test_srs_recovers_ratings():
    res = []
    truth = {"a": 7, "b": 0, "c": -7}
    for _ in range(5):
        for h in truth:
            for a in truth:
                if h != a:
                    m = truth[h] - truth[a] + 2
                    res.append(Result(h, a, 20 + max(m, 0), 20 + max(-m, 0)))
    s = SRS(res, hfa=2.0, ridge=0.01)
    assert s.r["a"] - s.r["c"] == pytest.approx(14, abs=0.2)
    assert s.margin("a", "b") == pytest.approx(9, abs=0.2)
    assert SRS.win_prob(0, 13.5) == 0.5

"""Forecast-weather ledger for outdoor NFL games."""
from datetime import datetime, timezone

from ajbet.db import DB
from ajbet.espn import Game, Lines, Team
from ajbet.wx_ledger import WeatherLedger

KO = "2026-10-04T17:00Z"
KO_TS = datetime(2026, 10, 4, 17, 0, tzinfo=timezone.utc).timestamp()


def game(state="pre", hs=0, as_=0, eid="9"):
    return Game(event_id=eid, league="nfl", kind="us", name="X", start=KO, state=state, completed=state == "post",
                period=0, clock_s=0, display_clock="", detail="", home=Team(id="1", name="Chicago Bears", abbr="CHI", score=hs),
                away=Team(id="2", name="Green Bay Packers", abbr="GB", score=as_),
                pregame=Lines(total=44.5, total_under_price=-110, provider="ESPN BET"))


class Ctx:
    def __init__(self, w):
        self.w, self.calls = w, 0

    def nfl_weather(self, g):
        self.calls += 1
        return self.w


def test_dayof_prekick_then_settle():
    db, t = DB(), [KO_TS - 6 * 3600]
    ctx = Ctx({"roof": "outdoor", "wind": 14.0, "precip": 0.0, "temp": 50.0})
    L = WeatherLedger(db, ctx, clock=lambda: t[0])
    assert L.record({"nfl": [game()]})["captured"] == 1           # day-of only
    assert L.record({"nfl": [game()]})["captured"] == 0 and ctx.calls == 1
    t[0] = KO_TS - 3600
    assert L.record({"nfl": [game()]})["captured"] == 1           # prekick
    t[0] = KO_TS + 4 * 3600
    assert L.record({"nfl": [game("post", 17, 20)]})["settled"] == 2
    rows = db.conn.execute("SELECT slot, wind_mph, wind10_flag, total_line, final_total, under_result FROM nfl_weather ORDER BY slot").fetchall()
    assert [tuple(r) for r in rows] == [("dayof", 14.0, 1, 44.5, 37.0, "win"), ("prekick", 14.0, 1, 44.5, 37.0, "win")]
    assert L.record({"nfl": [game("post", 17, 20)]})["settled"] == 0


def test_skips_domes_and_far_games():
    db = DB()
    L = WeatherLedger(db, Ctx({"roof": "dome"}), clock=lambda: KO_TS - 3600)
    assert L.record({"nfl": [game()]})["captured"] == 0
