import numpy as np
import pandas as pd

class PowerCandleBacktester:
    def __init__(self, initial_balance=10000.0, risk_pct=0.01):
        self.initial_balance = initial_balance
        self.risk_pct = risk_pct

    def generate_session_data(self):
        np.random.seed(42)
        periods = 1200
        price = 100.0
        prices, highs, lows, opens, volumes = [], [], [], [], []

        for i in range(periods):
            drift = 0.16 if 150 < i < 450 else -0.22 if 550 < i < 750 else 0.42 if 850 < i < 1100 else -0.01
            price += np.random.normal(0, 0.4) + drift
            prices.append(price)
            opens.append(price + np.random.uniform(-0.15, 0.15))
            highs.append(max(price, opens[-1]) + np.random.uniform(0.01, 0.3))
            lows.append(min(price, opens[-1]) - np.random.uniform(0.01, 0.3))
            
            # Simulate high-volume clusters near breakout points
            vol_mult = 3.0 if (148 < i < 160 or 848 < i < 865) else 1.0
            volumes.append(np.random.randint(100, 1000) * vol_mult)

        return pd.DataFrame({"open": opens, "high": highs, "low": lows, "close": prices, "volume": volumes})

    def calculate_adx(self, df: pd.DataFrame, period: int = 14):
        high, low, close = df["high"], df["low"], df["close"]
        up_move = high - high.shift(1)
        down_move = low.shift(1) - low

        plus_dm = np.where((up_move > down_move) & (up_move > 0), up_move, 0.0)
        minus_dm = np.where((down_move > up_move) & (down_move > 0), down_move, 0.0)

        tr = pd.concat([high - low, (high - close.shift(1)).abs(), (low - close.shift(1)).abs()], axis=1).max(axis=1)
        atr = tr.ewm(alpha=1/period, adjust=False).mean()

        plus_di = 100 * (pd.Series(plus_dm, index=df.index).ewm(alpha=1/period, adjust=False).mean() / (atr + 1e-9))
        minus_di = 100 * (pd.Series(minus_dm, index=df.index).ewm(alpha=1/period, adjust=False).mean() / (atr + 1e-9))

        dx = 100 * ((plus_di - minus_di).abs() / ((plus_di + minus_di) + 1e-9))
        adx = dx.ewm(alpha=1/period, adjust=False).mean()

        return adx, plus_di, minus_di

    def run_backtest(self, df: pd.DataFrame, use_confluence: bool = False):
        df = df.copy()
        
        # Technical Indicators
        df["sma20"] = df["close"].rolling(20).mean()
        df["sma200"] = df["close"].rolling(200).mean()
        df["vol_sma20"] = df["volume"].rolling(20).mean()
        
        # 20 SMA Slope (5-bar lookback)
        df["sma20_slope"] = ((df["sma20"] - df["sma20"].shift(5)) / (df["sma20"].shift(5) + 1e-9)) * 100

        # ADX & +DI/-DI
        df["adx"], df["plus_di"], df["minus_di"] = self.calculate_adx(df, period=14)

        # Baseline Trigger: Close > 20 SMA & Volume > 2.0x Average
        baseline_trigger = (df["close"] > df["sma20"]) & (df["volume"] > 2.0 * df["vol_sma20"])

        if use_confluence:
            # 4-Indicator Confluence Filter:
            # 1. Price > 20 SMA > 200 SMA (Macro Stage 2)
            # 2. 20 SMA Slope > +0.10% (Micro trajectory)
            # 3. Volume Surge > 2.0x
            # 4. ADX >= 25.0 and +DI > -DI (Directional Power)
            confluence_filter = (
                (df["close"] > df["sma20"]) & 
                (df["sma20"] > df["sma200"]) & 
                (df["sma20_slope"] > 0.10) & 
                (df["adx"] >= 25.0) & 
                (df["plus_di"] > df["minus_di"])
            )
            df["entry_signal"] = baseline_trigger & confluence_filter
        else:
            df["entry_signal"] = baseline_trigger

        balance = self.initial_balance
        equity_curve = [balance]
        trades = []
        in_position = False
        entry_price = 0.0
        stop_loss = 0.0
        position_size = 0

        for i in range(200, len(df)):
            curr = df.iloc[i]

                in_position = True
                entry_price = curr["close"]
                stop_loss = curr["low"]
                risk_per_share = max(0.05, entry_price - stop_loss)
                risk_amount = balance * self.risk_pct
                position_size = int(risk_amount / risk_per_share)
                continue

            if in_position:
                target_price = entry_price + (2.0 * (entry_price - stop_loss))

                if curr["low"] <= stop_loss:
                    pnl = position_size * (stop_loss - entry_price)
                    balance += pnl

                    in_position = False
                elif curr["high"] >= target_price:
                    pnl = position_size * (target_price - entry_price)
                    balance += pnl
                    trades.append(pnl)
                    in_position = False

            equity_curve.append(balance)

        return self.compute_metrics(trades, equity_curve)

    def compute_metrics(self, trades, equity_curve):
        if not trades:
            return {
                "final_balance": self.initial_balance,
                "net_pnl": 0.0,
                "total_trades": 0,
                "win_rate_pct": 0.0,
                "profit_factor": 0.0,
                "max_drawdown_pct": 0.0,
                "sharpe_ratio": 0.0
            }

        trades_arr = np.array(trades)
        wins = trades_arr[trades_arr > 0]
        losses = trades_arr[trades_arr < 0]

        total_trades = len(trades)
        win_rate = (len(wins) / total_trades) * 100 if total_trades > 0 else 0.0
        gross_profit = np.sum(wins) if len(wins) > 0 else 0.0
        gross_loss = abs(np.sum(losses)) if len(losses) > 0 else 0.0
        profit_factor = (gross_profit / gross_loss) if gross_loss > 0 else gross_profit

        eq = np.array(equity_curve)
        peak = np.maximum.accumulate(eq)
        drawdown = (eq - peak) / peak
        max_drawdown = float(np.min(drawdown)) * 100

        returns = np.diff(eq) / eq[:-1]
        sharpe = float((np.mean(returns) / (np.std(returns) + 1e-9)) * np.sqrt(252)) if len(returns) > 1 else 0.0

        return {
            "final_balance": round(float(eq[-1]), 2),
            "net_pnl": round(float(eq[-1] - self.initial_balance), 2),
            "total_trades": total_trades,
            "win_rate_pct": round(win_rate, 2),
            "profit_factor": round(profit_factor, 2),
            "max_drawdown_pct": round(max_drawdown, 2),
            "sharpe_ratio": round(sharpe, 2)
        }

def calculate_atr(bars_slice, period=14):
    if len(bars_slice) < period + 1:
        return bars_slice[-1]["h"] - bars_slice[-1]["l"]
    tr_list = []
    for j in range(1, len(bars_slice)):
        h = bars_slice[j]["h"]
        l = bars_slice[j]["l"]
        cp = bars_slice[j-1]["c"]
        tr = max(h - l, abs(h - cp), abs(l - cp))
        tr_list.append(tr)
    return np.mean(tr_list[-period:])

def calculate_adx(highs_slice, lows_slice, closes_slice, period=14):
    if len(closes_slice) < (period * 2):
        return 0.0
    tr_list, plus_dm_list, minus_dm_list = [], [], []
    for i in range(1, len(closes_slice)):
        h, l = highs_slice[i], lows_slice[i]
        prev_c, prev_h, prev_l = closes_slice[i-1], highs_slice[i-1], lows_slice[i-1]
        tr = max(h - l, abs(h - prev_c), abs(l - prev_c))
        tr_list.append(tr)
        up_move = h - prev_h
        down_move = prev_l - l
        plus_dm = up_move if (up_move > down_move and up_move > 0) else 0.0
        minus_dm = down_move if (down_move > up_move and down_move > 0) else 0.0
        plus_dm_list.append(plus_dm)
        minus_dm_list.append(minus_dm)
    if len(tr_list) < period:
        return 0.0
    atr_s = sum(tr_list[:period])
    p_di_s = sum(plus_dm_list[:period])
    m_di_s = sum(minus_dm_list[:period])
    dx_list = []
    for i in range(period, len(tr_list)):
        atr_s = atr_s - (atr_s / period) + tr_list[i]
        p_di_s = p_di_s - (p_di_s / period) + plus_dm_list[i]
        m_di_s = m_di_s - (m_di_s / period) + minus_dm_list[i]
        p_di = 100 * (p_di_s / atr_s) if atr_s != 0 else 0
        m_di = 100 * (m_di_s / atr_s) if atr_s != 0 else 0
        di_sum = p_di + m_di
        dx = 100 * (abs(p_di - m_di) / di_sum) if di_sum != 0 else 0
        dx_list.append(dx)
    return sum(dx_list[-period:]) / period if len(dx_list) >= period else 0.0

def is_valid_tod(timestamp_str):
    # Timestamp format: 2026-08-13T14:35:00Z (UTC)
    # Market Open 9:30 AM EST = 13:30 UTC (Standard/Daylight converted)
    # Extract Hour and Minute directly
    try:
        dt = datetime.fromisoformat(timestamp_str.replace("Z", "+00:00"))
        # Convert UTC to US Eastern (UTC-4 in Daylight Saving Time)
        hour_est = (dt.hour - 4) % 24
        minute = dt.minute
        time_decimal = hour_est + (minute / 60.0)

        # Window 1: 10:00 AM - 11:30 AM EST
        # Window 2: 1:30 PM - 3:30 PM EST (13.5 - 15.5)
        in_morning_window = (10.0 <= time_decimal <= 11.5)
        in_afternoon_window = (13.5 <= time_decimal <= 15.5)
        return in_morning_window or in_afternoon_window
    except Exception:
        return True

def calculate_rsi(closes, period=2):
    if len(closes) < period + 1:
        return 50.0
    deltas = np.diff(closes)
    gains = np.where(deltas > 0, deltas, 0.0)
    losses = np.where(deltas < 0, -deltas, 0.0)
    avg_gain = np.mean(gains[-period:])
    avg_loss = np.mean(losses[-period:])
    if avg_loss == 0:
        return 100.0
    rs = avg_gain / avg_loss
    return 100.0 - (100.0 / (1.0 + rs))

def calculate_atr(bars_slice, period=14):
    if len(bars_slice) < period + 1:
        return bars_slice[-1]["h"] - bars_slice[-1]["l"]

def calculate_adx(highs_slice, lows_slice, closes_slice, period=14):
    if len(closes_slice) < (period * 2):
        return 0.0

    tr_list, plus_dm_list, minus_dm_list = [], [], []

    for i in range(1, len(closes_slice)):
        h, l = highs_slice[i], lows_slice[i]
        prev_c, prev_h, prev_l = closes_slice[i-1], highs_slice[i-1], lows_slice[i-1]

        tr = max(h - l, abs(h - prev_c), abs(l - prev_c))
        tr_list.append(tr)

        up_move = h - prev_h
        down_move = prev_l - l

        plus_dm = up_move if (up_move > down_move and up_move > 0) else 0.0
        minus_dm = down_move if (down_move > up_move and down_move > 0) else 0.0

        plus_dm_list.append(plus_dm)
        minus_dm_list.append(minus_dm)

    if len(tr_list) < period:
        return 0.0

    atr_smooth = sum(tr_list[:period])
    plus_di_smooth = sum(plus_dm_list[:period])
    minus_di_smooth = sum(minus_dm_list[:period])

    dx_list = []

    for i in range(period, len(tr_list)):
        atr_smooth = atr_smooth - (atr_smooth / period) + tr_list[i]
        plus_di_smooth = plus_di_smooth - (plus_di_smooth / period) + plus_dm_list[i]
        minus_di_smooth = minus_di_smooth - (minus_di_smooth / period) + minus_dm_list[i]

        plus_di = 100 * (plus_di_smooth / atr_smooth) if atr_smooth != 0 else 0
        minus_di = 100 * (minus_di_smooth / atr_smooth) if atr_smooth != 0 else 0

        di_sum = plus_di + minus_di
        dx = 100 * (abs(plus_di - minus_di) / di_sum) if di_sum != 0 else 0
        dx_list.append(dx)

    if len(dx_list) < period:
        return 0.0

    return sum(dx_list[-period:]) / period

    df['sma20'] = df['close'].rolling(20).mean()
    df['sma200'] = df['close'].rolling(200).mean()
    tr = pd.concat([df['high'] - df['low'], (df['high'] - df['close'].shift(1)).abs(), (df['low'] - df['close'].shift(1)).abs()], axis=1).max(axis=1)
    df['atr'] = tr.rolling(14).mean()

    df['sma20_slope'] = ((df['sma20'] - df['sma20'].shift(5)) / (df['sma20'].shift(5) + 1e-9)) * 100

    z1 = df['sma20'] - (0.50 * df['atr'])

    entry = 0.0
    stop = 0.0
    shares = 0
    up_bars = 0

    for i in range(200, len(df)):
        curr = df.iloc[i]
        bar_time = df.index[i]
        
        # Check market hour bounds (Avoid overnight holding)
        is_eod = hasattr(bar_time, 'hour') and bar_time.hour >= 15 and bar_time.minute >= 45

        # 1. Manage Active Position
        if in_pos:
            # EOD Mandatory Exit or Hard Stop Hit
            if is_eod or curr['low'] <= stop:
                exit_price = stop if curr['low'] <= stop else curr['close']
                pnl = shares * (exit_price - entry)
                balance += pnl
                trades.append(pnl)
                in_pos = False
                equity_curve.append(balance)

            is_green = curr['close'] > curr['open']

def calculate_atr(bars_slice, period=14):
    if len(bars_slice) < period + 1:
        return bars_slice[-1]["h"] - bars_slice[-1]["l"]
    tr_list = []
    for j in range(1, len(bars_slice)):
        h = bars_slice[j]["h"]
        l = bars_slice[j]["l"]
        cp = bars_slice[j-1]["c"]
        tr = max(h - l, abs(h - cp), abs(l - cp))
        tr_list.append(tr)
    return np.mean(tr_list[-period:])

def calculate_adx(highs_slice, lows_slice, closes_slice, period=14):
    if len(closes_slice) < (period * 2):
        return 0.0

    tr_list = []
    plus_dm_list = []
    minus_dm_list = []

    for i in range(1, len(closes_slice)):
        h = highs_slice[i]
        l = lows_slice[i]
        prev_c = closes_slice[i-1]
        prev_h = highs_slice[i-1]
        prev_l = lows_slice[i-1]

        tr = max(h - l, abs(h - prev_c), abs(l - prev_c))
        tr_list.append(tr)

        up_move = h - prev_h
        down_move = prev_l - l

        plus_dm = up_move if (up_move > down_move and up_move > 0) else 0.0
        minus_dm = down_move if (down_move > up_move and down_move > 0) else 0.0

        plus_dm_list.append(plus_dm)
        minus_dm_list.append(minus_dm)

    if len(tr_list) < period:
        return 0.0

    atr_smooth = sum(tr_list[:period])
    plus_di_smooth = sum(plus_dm_list[:period])
    minus_di_smooth = sum(minus_dm_list[:period])

    dx_list = []

    for i in range(period, len(tr_list)):
        atr_smooth = atr_smooth - (atr_smooth / period) + tr_list[i]
        plus_di_smooth = plus_di_smooth - (plus_di_smooth / period) + plus_dm_list[i]
        minus_di_smooth = minus_di_smooth - (minus_di_smooth / period) + minus_dm_list[i]

        plus_di = 100 * (plus_di_smooth / atr_smooth) if atr_smooth != 0 else 0
        minus_di = 100 * (minus_di_smooth / atr_smooth) if atr_smooth != 0 else 0

        di_sum = plus_di + minus_di
        dx = 100 * (abs(plus_di - minus_di) / di_sum) if di_sum != 0 else 0
        dx_list.append(dx)

    if len(dx_list) < period:
        return 0.0

    return sum(dx_list[-period:]) / period

def compute_indicators(df):
    df['ema20'] = df['close'].ewm(span=20, adjust=False).mean()
    df['ema50'] = df['close'].ewm(span=50, adjust=False).mean()
    df['sma150'] = df['close'].rolling(150).mean()
    
    # BB
    df['bb_mid'] = df['close'].rolling(20).mean()
    df['bb_std'] = df['close'].rolling(20).std()
    df['bb_upper'] = df['bb_mid'] + (2.0 * df['bb_std'])
    df['bb_lower'] = df['bb_mid'] - (2.0 * df['bb_std'])
    
    # ATR & ADX
    tr1 = df['high'] - df['low']
    tr2 = (df['high'] - df['close'].shift(1)).abs()
    tr3 = (df['low'] - df['close'].shift(1)).abs()
    df['atr'] = pd.concat([tr1, tr2, tr3], axis=1).max(axis=1).rolling(14).mean()
    
    up = df['high'] - df['high'].shift(1)
    down = df['low'].shift(1) - df['low']
    plus_dm = np.where((up > down) & (up > 0), up, 0.0)
    minus_dm = np.where((down > up) & (down > 0), down, 0.0)
    plus_di = 100.0 * (pd.Series(plus_dm).rolling(14).mean() / (df['atr'] + 1e-9))
    minus_di = 100.0 * (pd.Series(minus_dm).rolling(14).mean() / (df['atr'] + 1e-9))
    df['adx'] = (100.0 * (abs(plus_di - minus_di) / (plus_di + minus_di + 1e-9))).rolling(14).mean()
    
    # Swing High/Low for Fibonacci calculation
    df['swing_high_20'] = df['high'].rolling(20).max()
    df['swing_low_20'] = df['low'].rolling(20).min()
    return df.dropna().reset_index(drop=True)

def calculate_indicators(df_15m):
    df = df_15m.copy()
    
    # Relative Volume (RVOL)
    vol_sma = df['volume'].rolling(20).mean()
    df['rvol'] = df['volume'] / vol_sma.replace(0, 1e-9)
    
    # Fast / Slow Moving Averages
    df['ema_fast'] = df['close'].ewm(span=9, adjust=False).mean()
    df['ema_slow'] = df['close'].ewm(span=21, adjust=False).mean()
    
    # Session VWAP
    df['date'] = pd.to_datetime(df.index).date
    typical_price = (df['high'] + df['low'] + df['close']) / 3
    tp_vol = typical_price * df['volume']
    df['cum_tp_vol'] = tp_vol.groupby(df['date']).cumsum()
    df['cum_vol'] = df['volume'].groupby(df['date']).cumsum()
    df['vwap'] = df['cum_tp_vol'] / df['cum_vol']
    
    # ATR & ADX
    tr0 = abs(df['high'] - df['low'])
    tr1 = abs(df['high'] - df['close'].shift(1))
    tr2 = abs(df['low'] - df['close'].shift(1))
    tr = pd.concat([tr0, tr1, tr2], axis=1).max(axis=1)
    df['atr'] = pd.Series(tr).ewm(alpha=1/14, adjust=False).mean()
    
    up_move = df['high'] - df['high'].shift(1)
    down_move = df['low'].shift(1) - df['low']
    plus_dm = np.where((up_move > down_move) & (up_move > 0), up_move, 0)
    minus_dm = np.where((down_move > up_move) & (down_move > 0), down_move, 0)
    
    atr_smooth = df['atr'].replace(0, 1e-9)
    plus_di = 100 * (pd.Series(plus_dm, index=df.index).ewm(alpha=1/14, adjust=False).mean() / atr_smooth)
    minus_di = 100 * (pd.Series(minus_dm, index=df.index).ewm(alpha=1/14, adjust=False).mean() / atr_smooth)
    
    denom = (plus_di + minus_di).replace(0, 1e-9)
    dx = 100 * (abs(plus_di - minus_di) / denom)
    df['adx'] = dx.ewm(alpha=1/14, adjust=False).mean().fillna(20.0)
    
    return df

class ATRTrailingStopManager:

    def update_stop_level(self, side: str, entry_price: float, current_price: float, current_stop: float, atr_value: float) -> float:
        """
        Calculates the new trailing stop price if activation threshold is reached.
        """
        if side.upper() == 'BUY' or side.upper() == 'LONG':
            unrealized_r = (current_price - entry_price) / max(0.0001, (entry_price - current_stop))
            if unrealized_r >= self.activation_r_multiple:
                proposed_stop = current_price - (atr_value * self.atr_multiplier)
                # Ratchet upwards only
                return max(current_stop, proposed_stop)
        elif side.upper() == 'SELL' or side.upper() == 'SHORT':
            unrealized_r = (entry_price - current_price) / max(0.0001, (current_stop - entry_price))
            if unrealized_r >= self.activation_r_multiple:
                proposed_stop = current_price + (atr_value * self.atr_multiplier)
                # Ratchet downwards only
                return min(current_stop, proposed_stop)
        return current_stop

def compute_indicators(df_stock: pd.DataFrame) -> Optional[dict]:
    if df_stock is None or len(df_stock) < 40: return None
    try:
        c, o, h, l = df_stock["close"], df_stock["open"], df_stock["high"], df_stock["low"]
        sma20 = c.rolling(20).mean()
        tr = pd.concat([h - l, (h - c.shift(1)).abs(), (l - c.shift(1)).abs()], axis=1).max(axis=1)
        atr14 = tr.rolling(14).mean()
        return {
            "close": float(c.iloc[-1]),
            "sma20": float(sma20.iloc[-1]),
            "atr": float(atr14.iloc[-1]),
            "is_green_bar": float(c.iloc[-1]) > float(o.iloc[-1])

def calculate_adx(df: pd.DataFrame, period: int = 14) -> Tuple[pd.Series, pd.Series, pd.Series]:
    """Calculates ADX, +DI, and -DI using Wilder's Smoothing method."""
    high, low, close = df["high"], df["low"], df["close"]
    
    up_move = high - high.shift(1)
    down_move = low.shift(1) - low

    plus_dm = np.where((up_move > down_move) & (up_move > 0), up_move, 0.0)
    minus_dm = np.where((down_move > up_move) & (down_move > 0), down_move, 0.0)

    tr = pd.concat([high - low, (high - close.shift(1)).abs(), (low - close.shift(1)).abs()], axis=1).max(axis=1)

    atr = tr.ewm(alpha=1/period, adjust=False).mean()
    plus_di = 100 * (pd.Series(plus_dm, index=df.index).ewm(alpha=1/period, adjust=False).mean() / (atr + 1e-9))
    minus_di = 100 * (pd.Series(minus_dm, index=df.index).ewm(alpha=1/period, adjust=False).mean() / (atr + 1e-9))

    dx = 100 * ((plus_di - minus_di).abs() / ((plus_di + minus_di) + 1e-9))
    adx = dx.ewm(alpha=1/period, adjust=False).mean()

    return adx, plus_di, minus_di

def compute_technical_indicators(df_stock: pd.DataFrame, df_spy: pd.DataFrame) -> Optional[dict]:
    if df_stock is None or len(df_stock) < 200: return None
    try:
        c = df_stock["close"]
        o = df_stock["open"]
        h = df_stock["high"]
        l = df_stock["low"]

        # 1. Moving Averages
        sma20 = c.rolling(20).mean()
        sma200 = c.rolling(200).mean()

        # 2. 20 SMA Slope (5-bar lookback)
        sma20_curr = sma20.iloc[-1]
        sma20_prev = sma20.iloc[-6]
        sma20_slope_pct = ((sma20_curr - sma20_prev) / (sma20_prev + 1e-9)) * 100

        # 3. Mansfield Relative Strength vs SPY
        common_idx = c.index.intersection(df_spy.index)
        if len(common_idx) < 50: return None
        raw_rs = (c.loc[common_idx] / df_spy["close"].loc[common_idx]) * 100
        mansfield_rs = ((raw_rs / (raw_rs.rolling(50).mean() + 1e-9)) - 1) * 100
        curr_mrs = mansfield_rs.iloc[-1]

        # 4. ADX Calculation
        adx_series, plus_di_series, minus_di_series = calculate_adx(df_stock, period=14)
        curr_adx = float(adx_series.iloc[-1])
        curr_plus_di = float(plus_di_series.iloc[-1])
        curr_minus_di = float(minus_di_series.iloc[-1])

        # ATR Calculation
        tr = pd.concat([h - l, (h - c.shift(1)).abs(), (l - c.shift(1)).abs()], axis=1).max(axis=1)
        atr14 = tr.rolling(14).mean().iloc[-1]

        curr_close = float(c.iloc[-1])
        curr_open = float(o.iloc[-1])
        curr_sma200 = float(sma200.iloc[-1])

        # --- 4-INDICATOR CONFLUENCE CHECK ---
        macro_ok = (curr_close > sma20_curr) and (sma20_curr > curr_sma200) # Price > 20 MA > 200 MA
        slope_ok = sma20_slope_pct > 0.10                                   # 20 MA curving up

def calculate_atr(bars_slice, period=14):
    if len(bars_slice) < period + 1:
        return bars_slice[-1]["h"] - bars_slice[-1]["l"]
    tr_list = []
    for j in range(1, len(bars_slice)):
        h = bars_slice[j]["h"]
        l = bars_slice[j]["l"]
        cp = bars_slice[j-1]["c"]
        tr = max(h - l, abs(h - cp), abs(l - cp))
        tr_list.append(tr)
    return np.mean(tr_list[-period:]) if tr_list else 1.0

def calculate_rsi(closes, period=2):
    if len(closes) < period + 1:
        return 50.0
    deltas = np.diff(closes)
    gains = np.where(deltas > 0, deltas, 0.0)
    losses = np.where(deltas < 0, -deltas, 0.0)
    avg_gain = np.mean(gains[-period:])
    avg_loss = np.mean(losses[-period:])
    if avg_loss == 0:
        return 100.0
    rs = avg_gain / avg_loss
    return 100.0 - (100.0 / (1.0 + rs))
