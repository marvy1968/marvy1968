def outlier_dependency(tr):
    pnl = np.sort(tr["pnl"].to_numpy())[::-1]
    out = {"all trades": pnl.sum()}
    for pct in (1, 5):
        k = max(1, int(len(pnl) * pct / 100))
        out[f"without best {pct}% ({k} trades)"] = pnl[k:].sum()
    return pd.Series(out).round(2)


def halves(tr, equity):
    tr = tr.sort_values("closed")
    days = pd.unique(tr["date"].astype(str))
    cut = days[len(days) // 2] if len(days) > 1 else days[0]
    first, second = tr[tr["date"].astype(str) < cut], tr[tr["date"].astype(str) >= cut]
    rows = {}
    for name, part in (("first half", first), ("second half", second)):
        if len(part):
            rows[name] = run_summary(part, equity)
    exp = pd.DataFrame({"first half": first.groupby("expert")["pnl"].sum(),
                        "second half": second.groupby("expert")["pnl"].sum()}).fillna(0).round(2)
    return pd.DataFrame(rows).T, exp


def pct_table(x, name):
    q = np.percentile(x, [5, 25, 50, 75, 95])
    return pd.Series(q.round(3), index=["p5", "p25", "median", "p75", "p95"], name=name)

def verdict(summary, by_exp, boot, base_bps):
    """Heuristic checklist - rules of thumb, not guarantees."""
    print("\n================ VERDICT (rules of thumb) ================")
    checks = []
    if summary is not None:
        be = breakeven_bps(summary.index.tolist(), summary["net_pnl"].tolist())
        checks.append((be >= 5, f"Overall break-even slippage {be:.1f} bps (want >= 5 bps cushion)"))
        pf = summary.loc[base_bps, "profit_factor"] if base_bps in summary.index else np.nan
        checks.append((pf >= 1.2, f"Profit factor at {base_bps:g} bps = {pf:.2f} (want >= 1.2)"))
        weak = [e for e, v in by_exp["breakeven_bps"].items() if v < base_bps]
        checks.append((not weak, f"Experts losing money at {base_bps:g} bps: {weak or 'none'}"))
    checks.append((boot["p_loss"] <= 10, f"P(losing period) = {boot['p_loss']:.1f}% (want <= 10%)"))
    od = boot["outliers"]
    checks.append((od.iloc[-1] > 0, f"Still profitable without best 5% of trades: ${od.iloc[-1]:,.2f}"))
    for ok, msg in checks:
        print(f"  [{'PASS' if ok else 'FAIL'}] {msg}")
    print("  An expert failing these should be switched off or re-tuned (config.EXPERT_OVERRIDES)")

def minute_bars(p0, days, rng, vol=0.0008, drift=0.0, end_hhmm=(16, 0)):
    idx = []
    for d in days:
        o = pd.Timestamp(d, tz=NYZ) + pd.Timedelta(hours=9, minutes=30)
        n = (end_hhmm[0] * 60 + end_hhmm[1]) - 570
        idx += list(pd.date_range(o, periods=n, freq="min"))
    idx = pd.DatetimeIndex(idx).tz_convert("UTC")
    r = rng.normal(drift, vol, len(idx))
    c = p0 * np.exp(np.cumsum(r))
    o = np.r_[p0, c[:-1]]
    h = np.maximum(o, c) * (1 + abs(rng.normal(0, vol / 2, len(c))))
    l = np.minimum(o, c) * (1 - abs(rng.normal(0, vol / 2, len(c))))
    v = rng.integers(1000, 20000, len(c)).astype(float)
    return pd.DataFrame(dict(open=o, high=h, low=l, close=c, volume=v), index=idx)


def daily_bars(p0, n, drift, rng, end="2026-08-28"):
    idx = pd.date_range(end=end, periods=n, freq="B", tz=NYZ).tz_convert("UTC")
    c = p0 * np.exp(np.cumsum(rng.normal(drift, 0.01, n)))
    return pd.DataFrame(dict(open=c, high=c * 1.01, low=c * 0.99, close=c, volume=1e6), index=idx)


def world(seed=0, days=None, per_sector=5, vol=0.0008, mutate=None, end_hhmm=None, prep=None):
    """A full synthetic market: 11 sectors x N stocks + ETFs.  mutate(sym, df) -> df lets a test corrupt data."""
    rng = np.random.default_rng(seed)
    days = days if days is not None else pd.bdate_range("2026-08-17", "2026-08-28")
    arrays, frames, sym_sector = {}, {}, {}
    for k, (sec, spec) in enumerate(config.SECTORS.items()):
        drift = [0.003, -0.003, 0.0][k % 3]
        frames[spec["etf"]] = daily_frame(daily_bars(100, 300, drift, rng, end=str(days[-1].date())))
        syms = [spec["etf"]] + spec["candidates"][:per_sector]
        for s in syms:
            df = minute_bars(rng.uniform(30, 400), days, rng, vol=vol, drift=0.0,

            arrays[s] = to_arrays((prep or prepare_intraday)(df))
            if s != spec["etf"]:
                sym_sector[s] = sec
    return arrays, frames, sym_sector, days

@test
def lookahead_indicators():
    """Every indicator value at bar t must be identical whether or not future bars exist."""
    rng = np.random.default_rng(1)
    df = minute_bars(100, pd.bdate_range("2026-08-24", "2026-08-26"), rng)
    full = prepare_intraday(df)
    cols = [c for c in full.columns if c not in ("session",)]
    bad = set()
    for k in rng.integers(60, len(df), 40):
        part = prepare_intraday(df.iloc[:k])
        t = part.index[-1]
        a, b = part.loc[t, cols].astype(float), full.loc[t, cols].astype(float)
        diff = ~(np.isclose(a, b, equal_nan=True, rtol=1e-9, atol=1e-9))
        bad |= set(np.array(cols)[np.asarray(diff)])
    assert not bad, f"look-ahead in columns: {sorted(bad)}"


@test
def no_edge_on_pure_noise():
    """On random-walk data with ZERO costs, the system must not show a significant edge.
    A strong positive result here would mean look-ahead or a fill bug."""
    rs = []
    for seed in range(3):
        arrays, frames, ss, days = world(seed=100 + seed)
        _, tr = run(arrays, frames, ss, days, slip=0.0)
        rs.append(tr["r"].to_numpy())
    r = np.concatenate(rs)
    t_stat = r.mean() / (r.std(ddof=1) / np.sqrt(len(r)))
    print(f"      noise: {len(r)} trades, mean R {r.mean():+.3f}, t={t_stat:+.1f}")
    assert t_stat < 3, f"suspicious edge on noise (t={t_stat:.1f})"

def flash_crash_losses_bounded_and_halt():
    """A -15% one-bar crash in every stock at 11:00.  Losses must equal the gap (not be hidden)
    and the daily-loss kill switch must use mark-to-market equity."""
    def crash(sym, df):

        hit = (ny.hour == 11) & (ny.minute == 0)
        after = (ny.hour > 11) | ((ny.hour == 11) & (ny.minute >= 0))
        same_day = ny.normalize() == ny.normalize()[hit][-1] if hit.any() else hit
        df = df.copy()
        mask = after & same_day

class ApiErr(Exception):
    def __init__(self, code, msg="err"):
        super().__init__(f"{code} {msg}")
        self.status_code = code


class FakeTC:
    """Simulates one position + its bracket legs, with scriptable failures.
    Enforces Alpaca's rule: exit orders can't exceed qty_available (no over-allocation)."""

    def __init__(self, side=1, qty=10, replace=None, oco=None, stop=None, close=None,
                 stuck_cancel=False, fill_on_cancel=False, submit_transient=0):
        self.side, self.qty = side, qty
        self.script = {"replace": list(replace or []), "oco": list(oco or []),
                       "stop": list(stop or []), "close": list(close or [])}
        self.stuck_cancel, self.fill_on_cancel = stuck_cancel, fill_on_cancel
        self.submit_transient = submit_transient
        self.log, self.seen_coids = [], []
        self.orders = {
            "tp": types.SimpleNamespace(id="tp", status="new", stop_price=None, limit_price=102.0, legs=[], qty=qty),
            "sl": types.SimpleNamespace(id="sl", status="held", stop_price=99.0, limit_price=None, legs=[], qty=qty)}

    def _next(self, kind):
        seq = self.script[kind]
        r = seq.pop(0) if seq else None
        if isinstance(r, Exception):
            raise r
        return r

    def _reserved(self):
        return sum(o.qty for o in self.orders.values() if o.status in ("new", "held", "accepted", "pending_cancel"))

    # --- API surface used by Broker ---
    def get_orders(self, filter=None):
        return [o for o in self.orders.values() if o.status in ("new", "held", "accepted", "pending_cancel")]

    def get_open_position(self, sym):
        if self.qty == 0:
            raise ApiErr(404, "position does not exist")
        return types.SimpleNamespace(qty=str(self.qty), side="long" if self.side > 0 else "short",
                                     qty_available=str(max(self.qty - self._reserved(), 0)), avg_entry_price="100")

    def replace_order_by_id(self, oid, req):
        self.log.append(("replace", oid))
        self._next("replace")
        self.orders[oid].stop_price = req.stop_price

    def cancel_order_by_id(self, oid):
        self.log.append(("cancel", oid))
        if self.fill_on_cancel:
            self.qty = 0
            for o in self.orders.values():
                o.status = "filled"
            return
        self.orders[oid].status = "pending_cancel" if self.stuck_cancel else "canceled"

    def submit_order(self, order_data):
        kind = type(order_data).__name__
        oc = str(getattr(order_data, "order_class", "")).lower()
        tag = "oco" if "oco" in oc else "stop" if kind == "StopOrderRequest" else "entry"
        self.seen_coids.append(getattr(order_data, "client_order_id", None))
        if tag == "entry" and self.submit_transient:
            self.submit_transient -= 1
            raise ApiErr(503, "unavailable")
        self.log.append(("submit", tag, float(order_data.qty), str(order_data.side)))
        if tag != "entry" and float(order_data.qty) > self.qty - self._reserved():
            raise ApiErr(403, "insufficient qty available")   # Alpaca's over-allocation rejection
        if tag in ("oco", "stop"):
            self._next(tag)
        oid = f"n{len(self.log)}"
        self.orders[oid] = types.SimpleNamespace(id=oid, status="new", stop_price=1.0, limit_price=None,
                                                 legs=[], qty=float(order_data.qty))
        return types.SimpleNamespace(id=oid)

    def close_position(self, sym):
        self.log.append(("close", sym))
        self._next("close")
        self.qty = 0


def ladder(tc, last=101.5, new_stop=100.05, side=1):
    import broker as brk
    b = object.__new__(brk.Broker)
    b.tc, b._assets = tc, {}
    real_time = brk.time
    brk.time = types.SimpleNamespace(time=real_time.time, sleep=lambda s: None)
    try:
        pos = Position("AAPL", "EMA", "Technology", side, tc.qty, 100.0, 99.0 if side > 0 else 101.0,
                       102.0 if side > 0 else 98.0, 1.0, "2026-08-24T14:00:00+00:00")
        return b.move_stop("AAPL", pos, new_stop, last), brk
    finally:
        brk.time = real_time

# ---------------------------------------------------------------- swing study (daily, mid/small caps)
def _daily_panel(n=40, days=1100, drift=0.0, vol=0.02, seed=0):
    rng = np.random.default_rng(seed)
    idx = pd.bdate_range("2016-01-04", periods=days)
    out = {}
    for k in range(n):
        c = 50 * np.exp(np.cumsum(rng.normal(drift, vol, days)))
        o = c * np.exp(rng.normal(0, vol / 3, days))
        h = np.maximum(o, c) * (1 + abs(rng.normal(0, vol / 2, days)))
        l = np.minimum(o, c) * (1 - abs(rng.normal(0, vol / 2, days)))
        out[f"S{k}"] = pd.DataFrame(dict(open=o, high=h, low=l, close=c,
                                         volume=rng.integers(200_000, 600_000, days).astype(float)), index=idx)
    return out

# ---------------------------------------------------------------- Weinstein Stage 2 module
def _stage2_stock(breakout=True, weeks=62, seed=0):
    """50 weeks rising 20->40, 10-week flat base near 40, then a breakout week (3x volume) to ~44."""
    rng = np.random.default_rng(seed)
    days = pd.bdate_range("2019-01-07", periods=weeks * 5)
    wk = np.arange(len(days)) // 5
    lvl = np.where(wk < 50, 20 + 20 * wk / 50, 40.0)
    last = wk == wk.max()
    lvl = np.where(last, 44.0 if breakout else 39.8, lvl)
    c = lvl * (1 + rng.normal(0, 0.002, len(days)))
    d = pd.DataFrame(dict(open=c, high=c * 1.005, low=c * 0.995, close=c, volume=1e6), index=days)
    if breakout:
        d.loc[last, "volume"] = 3e6
    return d


def _flat_spy(index):
    return pd.Series(100.0, index=index)


@test
def weinstein_weekly_bars_and_breakout():
    import weinstein as ws
    d = _stage2_stock()
    w = ws.weekly(d)
    assert len(w) == 62 and np.isclose(w["volume"].iloc[0], 5e6)
    f = ws.stage_frame(w, ws.weekly(pd.DataFrame(dict(open=100.0, high=100.0, low=100.0, close=100.0, volume=0.0),
                                                 index=d.index))["close"])
    assert bool(f["breakout"].iloc[-1]), f.iloc[-1]
    assert not f["breakout"].iloc[-6:-1].any()                        # inside the base: no signal
    quiet = _stage2_stock()
    quiet["volume"] = 1e6                                             # same price move, no volume surge
    fq = ws.stage_frame(ws.weekly(quiet), ws.weekly(pd.DataFrame(dict(open=100.0, high=100.0, low=100.0,
                                                                      close=100.0, volume=0.0), index=d.index))["close"])
    assert not bool(fq["breakout"].iloc[-1]), "breakout without 2x volume must not signal"

    def __init__(self, positions=None, orders=None, is_open=True):
        self.pos = dict(positions or {})
        self.orders = list(orders or [])
        self.submitted, self.cancelled, self.is_open = [], [], is_open

    def get_clock(self):
        return types.SimpleNamespace(is_open=self.is_open)

    def get_account(self):
        return types.SimpleNamespace(equity="100000")

    def get_all_positions(self):
        return [types.SimpleNamespace(symbol=s, qty=str(q)) for s, q in self.pos.items()]

    def get_orders(self, filter=None):
        return list(self.orders)

    def cancel_order_by_id(self, oid):
        self.cancelled.append(oid)
        self.orders = [o for o in self.orders if o.id != oid]

    def submit_order(self, order_data):
        kind = type(order_data).__name__
        self.submitted.append((kind, order_data.symbol, float(order_data.qty), str(order_data.side),
                               getattr(order_data, "stop_price", None), order_data.client_order_id))
        oid = f"o{len(self.submitted)}"
        if kind == "MarketOrderRequest" and "BUY" in str(order_data.side).upper():
            self.pos[order_data.symbol] = self.pos.get(order_data.symbol, 0) + float(order_data.qty)
        self.orders.append(types.SimpleNamespace(id=oid, symbol=order_data.symbol, client_order_id=order_data.client_order_id,
                                                 stop_price=getattr(order_data, "stop_price", None)))
        return types.SimpleNamespace(id=oid)

    def get_order_by_id(self, oid):
        q = next(x[2] for x in self.submitted if f"o{self.submitted.index(x) + 1}" == oid)
        return types.SimpleNamespace(status="filled", filled_qty=str(q))

@test
def zones_engine_invariants_and_noise():

    tr, eq = zones.run(P, spy, "2017-06-01", "2021-01-29", cost_mult=0.0)
    assert len(tr) > 30, len(tr)
    assert (tr["entry"] >= tr["trigger"] - 1e-9).all(), "filled below the stop-entry trigger"
    assert (tr["trigger"] > tr["stop0"]).all()
    assert tr.loc[tr.reason == "breakeven_stop", "tp1_hit"].all(), "breakeven stop without TP1"

    ev = sorted([(d, 1) for d in pd.to_datetime(tr.entry_date)] + [(d, -1) for d in pd.to_datetime(tr.exit_date)],
                key=lambda x: (x[0], x[1]))
    cur = peak = 0
    for _, x in ev:
        cur += x
        peak = max(peak, cur)
    assert peak <= zones.MAX_POSITIONS
    t = tr.ret_pct.mean() / (tr.ret_pct.std() / np.sqrt(len(tr)))
    print(f"      zones on noise: {len(tr)} trades, avg {tr.ret_pct.mean():+.2f}%, TP1 hit "
          f"{tr.tp1_hit.mean() * 100:.0f}%, t={t:+.2f}")

class _ZFake:
    """Fake Alpaca modelling bracket / OTO parents with legs (GTC), standalone stops, positions."""

    def __init__(self, is_open=True):
        self.pos, self.top, self.submitted, self.cancelled, self.is_open, self.n = {}, {}, [], [], is_open, 0

    def _id(self):
        self.n += 1
        return f"z{self.n}"

    def get_clock(self):
        return types.SimpleNamespace(is_open=self.is_open)

    def get_account(self):
        return types.SimpleNamespace(equity="100000")

    def get_all_positions(self):
        return [types.SimpleNamespace(symbol=s, qty=str(q)) for s, q in self.pos.items() if q > 0]

    def get_orders(self, filter=None):
        return [o for o in self.top.values() if o.status in ("new", "held", "accepted")]

    def get_order_by_id(self, oid, filter=None):
        return self.top[oid]

    def _find(self, oid):
        for o in self.top.values():
            if o.id == oid:
                return o
            for l in o.legs:
                if l.id == oid:
                    return l

    def cancel_order_by_id(self, oid):
        self.cancelled.append(oid)
        self._find(oid).status = "canceled"

    def submit_order(self, order_data):
        od, kind = order_data, type(order_data).__name__
        oc = str(getattr(od, "order_class", "")).lower()
        side = "buy" if "buy" in str(od.side).lower() else "sell"
        self.submitted.append(dict(kind=kind, cls=oc, sym=od.symbol, qty=float(od.qty), side=side,
                                   tif=str(od.time_in_force).lower(), cid=od.client_order_id,
                                   stop=getattr(od, "stop_price", None) or getattr(getattr(od, "stop_loss", None), "stop_price", None),
                                   tp=getattr(getattr(od, "take_profit", None), "limit_price", None)))
        mk = lambda **k: types.SimpleNamespace(id=self._id(), symbol=od.symbol, client_order_id=k.pop("cid", None),
                                               legs=[], created_at=pd.Timestamp.now(tz="UTC"), **k)
        if kind == "MarketOrderRequest" and side == "buy":
            self.pos[od.symbol] = self.pos.get(od.symbol, 0) + float(od.qty)
            par = mk(cid=od.client_order_id, status="filled", stop_price=None, limit_price=None, qty=od.qty)
            if "bracket" in oc:
                par.legs.append(mk(status="new", limit_price=od.take_profit.limit_price, stop_price=None, qty=od.qty))
            if "bracket" in oc or "oto" in oc:
                par.legs.append(mk(status="held", stop_price=od.stop_loss.stop_price, limit_price=None, qty=od.qty))
        elif kind == "StopOrderRequest":
            par = mk(cid=od.client_order_id, status="new", stop_price=od.stop_price, limit_price=None, qty=od.qty)
        else:
            self.pos[od.symbol] = self.pos.get(od.symbol, 0) - float(od.qty)
            par = mk(cid=od.client_order_id, status="filled", stop_price=None, limit_price=None, qty=od.qty)
        self.top[par.id] = par
        return par

    def fill_tp(self, sym):
        for o in self.top.values():
            tp = [l for l in o.legs if l.limit_price and l.status == "new"]
            if o.symbol == sym and tp:
                tp[0].status = "filled"
                for l in o.legs:
                    if l.stop_price:
                        l.status = "canceled"
                self.pos[sym] -= float(tp[0].qty)
                return

def _pbs_stock(pullback=3):
    days = pd.bdate_range("2021-01-04", periods=80)
    c = list(np.linspace(50, 70, 80 - pullback))
    for k in range(pullback):

    c = np.array(c)
    return pd.DataFrame(dict(open=c, high=c + 0.3, low=c - 0.3, close=c, volume=1e6), index=days)


def _path_panel(n, days, seed, steps=26, vol=0.02):
    """TRUE random walk: each day's O/H/L/C come from a simulated intraday path, so a buy stop has exactly
    zero expected edge. (The simpler generator makes highs pure one-day spikes, which unfairly punishes
    any stop-entry strategy.)"""
    rng = np.random.default_rng(seed)
    idx = pd.bdate_range("2016-01-04", periods=days)
    out = {}
    for k in range(n):
        x = np.log(50) + np.cumsum(rng.normal(0, vol / np.sqrt(steps), days * steps)).reshape(days, steps)
        px = np.exp(x)
        out[f"S{k}"] = pd.DataFrame(dict(open=px[:, 0], high=px.max(1), low=px.min(1), close=px[:, -1],
                                         volume=rng.integers(200_000, 600_000, days).astype(float)), index=idx)
    return out

                             volume=rng.integers(5e5, 2e6, n) * 1.0), index=days)

"""Daily swing study: 5-indicator overbought/oversold engine on mid & small caps.

    python swing.py --start 2016-01-01 --end 2021-12-31          # discovery backtest
    python swing.py --scan                                        # today's setups (no trading)

SETUP ("promising"): long-term uptrend (close > SMA200 and SMA50 > SMA200) AND an oversold pullback:
at least OBOS_MIN_VOTES of these 5 indicators oversold on the day's close:
    RSI(2) < 10   Stochastic %K(14,3) < 20   Bollinger %B(20,2) < 0   CCI(20) < -100   MFI(14) < 20
ENTRY next day's open. EXITS (fixed up front): RSI(2) > 60 or close > SMA5 -> sell next open;
stop 3 x ATR(14) below entry (intraday, gap-aware); 10-trading-day time stop. Long only.
UNIVERSE (point-in-time, monthly): US stocks ranked by 20-day dollar volume, EXCLUDING the top 500
(~large caps), price >= $5, 20-day dollar volume >= $5M. Costs scale with liquidity.
"""
import argparse
import hashlib

# ---------------- preregistered parameters ----------------
OBOS_MIN_VOTES = 4
RSI2_LO, RSI2_HI = 10, 90
STOCH_LO, STOCH_HI = 20, 80
PCTB_LO, PCTB_HI = 0.0, 1.0
CCI_LO, CCI_HI = -100, 100
MFI_LO, MFI_HI = 20, 80
EXIT_RSI2 = 60
STOP_ATR = 3.0
MAX_HOLD_DAYS = 10
MAX_POSITIONS = 15
POSITION_PCT = 0.05
LARGE_CAP_RANK = 500
MIN_ADV = 5e6
MIN_PRICE = 5.0
DISCOVERY_END = date(2021, 12, 31)       # 2022 onward is held out
COST_TIERS = [(50e6, 5.0), (20e6, 10.0), (0.0, 20.0)]   # (min 20-day $ volume, bps per side)


# ---------------- indicators (all use only the current and earlier daily bars) ----------------
def obos_frame(df):
    c, h, l, v = (df[k].astype(float) for k in ("close", "high", "low", "volume"))
    out = pd.DataFrame(index=df.index)
    out["rsi2"] = rsi(c, 2)
    ll, hh = l.rolling(14).min(), h.rolling(14).max()
    out["stoch"] = (100 * (c - ll) / (hh - ll).replace(0, np.nan)).rolling(3).mean()
    m20, sd20 = c.rolling(20).mean(), c.rolling(20).std(ddof=0)
    out["pctb"] = (c - (m20 - 2 * sd20)) / (4 * sd20).replace(0, np.nan)
    tp = (h + l + c) / 3
    mad = np.full(len(tp), np.nan)
    if len(tp) >= 20:
        w = np.lib.stride_tricks.sliding_window_view(tp.to_numpy(), 20)
        mad[19:] = np.abs(w - w.mean(axis=1, keepdims=True)).mean(axis=1)
    out["cci"] = (tp - tp.rolling(20).mean()) / (0.015 * pd.Series(mad, index=df.index)).replace(0, np.nan)
    flow = tp * v
    up = flow.where(tp > tp.shift(1), 0.0).rolling(14).sum()
    dn = flow.where(tp < tp.shift(1), 0.0).rolling(14).sum()
    mfi = 100 - 100 / (1 + up / dn.replace(0, np.nan))
    out["mfi"] = mfi.where(dn != 0, 100.0).where(up.notna())
    out["os_votes"] = ((out.rsi2 < RSI2_LO).astype(int) + (out.stoch < STOCH_LO) + (out.pctb < PCTB_LO)
                       + (out.cci < CCI_LO) + (out.mfi < MFI_LO))
    out["ob_votes"] = ((out.rsi2 > RSI2_HI).astype(int) + (out.stoch > STOCH_HI) + (out.pctb > PCTB_HI)
                       + (out.cci > CCI_HI) + (out.mfi > MFI_HI))
    out["obos_score"] = out["ob_votes"] - out["os_votes"]          # -5 very oversold ... +5 very overbought
    s50, s200 = c.rolling(50).mean(), c.rolling(200).mean()
    out["uptrend"] = (c > s200) & (s50 > s200)
    out["sma5"] = c.rolling(5).mean()
    out["atr14"] = atr(df.astype(float), 14)
    out["adv20"] = (c * v).rolling(20).mean()
    return out


def _wide(panel, col, frames=None):
    src = frames if frames is not None else panel
    return pd.DataFrame({s: d[col] for s, d in src.items()}).sort_index()


def membership(adv, close, large_rank=LARGE_CAP_RANK, min_adv=MIN_ADV, min_price=MIN_PRICE):
    """Point-in-time monthly universe: uses the PREVIOUS day's 20-day $ volume and close."""
    a, c = adv.shift(1), close.shift(1)
    rank = a.rank(axis=1, ascending=False)
    ok = (rank > large_rank) & (a >= min_adv) & (c >= min_price)
    month = ok.index.to_period("M")
    first = ~pd.Series(month).duplicated().to_numpy()
    snap = ok.astype(float)
    snap.loc[~first, :] = np.nan
    return snap.ffill().fillna(0).astype(bool)


def slip_bps(adv_value, cost_mult=1.0):
    for floor, bps in COST_TIERS:
        if adv_value >= floor:
            return bps * cost_mult
    return COST_TIERS[-1][1] * cost_mult


# ---------------- daily portfolio backtest ----------------
def run(panel, start, end, cost_mult=1.0, equity0=100_000.0, large_rank=LARGE_CAP_RANK, min_votes=OBOS_MIN_VOTES,
        allow=None):
    """panel: {symbol: daily OHLCV DataFrame indexed by date}. Returns (trades DataFrame, equity Series)."""
    feats = {s: obos_frame(d) for s, d in panel.items() if len(d) >= 30}
    syms = sorted(feats)
    O, H, L, C = (_wide(panel, k).reindex(columns=syms) for k in ("open", "high", "low", "close"))
    F = {k: _wide(None, k, feats).reindex(columns=syms) for k in
         ("os_votes", "rsi2", "sma5", "atr14", "adv20", "uptrend", "stoch", "pctb", "cci", "mfi")}
    member = membership(F["adv20"], C, large_rank)
    dates = C.index
    di0 = int(np.searchsorted(dates, pd.Timestamp(start)))
    di1 = int(np.searchsorted(dates, pd.Timestamp(end), side="right"))
    o, h, l, c = O.to_numpy(), H.to_numpy(), L.to_numpy(), C.to_numpy()
    votes, r2, s5, at, adv = (F[k].to_numpy() for k in ("os_votes", "rsi2", "sma5", "atr14", "adv20"))
    up, mem = F["uptrend"].fillna(False).to_numpy(bool), member.reindex(columns=syms).fillna(False).to_numpy(bool)
    alw = None if allow is None else allow.reindex(index=dates, columns=syms).fillna(True).to_numpy(bool)

    cash, pos, pend_in, pend_out = equity0, {}, [], {}
    trades, curve, last_px = [], [], {}

    def close_pos(j, px, di, why):
        nonlocal cash
        p = pos.pop(j)
        cash += p["qty"] * px
        trades.append(dict(symbol=syms[j], signal_date=dates[p["sig_di"]].date(), entry_date=dates[p["di"]].date(),
                           exit_date=dates[di].date(), entry=round(p["px"], 4), exit=round(px, 4),
                           ret_pct=round((px / p["px"] - 1) * 100, 3), days=di - p["di"], reason=why,
                           votes=int(p["votes"]), cost_bps=p["bps"],
                           pnl=round(p["qty"] * (px - p["px"]), 2)))

    for di in range(max(di0, 1), di1):
        # 1) exits decided yesterday -> today's open
        for j in list(pend_out):
            if j not in pos:
                pend_out.pop(j)
            elif np.isfinite(o[di, j]):
                close_pos(j, o[di, j] * (1 - pos[j]["bps"] / 1e4), di, pend_out.pop(j))
        # 2) entries decided yesterday -> today's open
        eq_prev = curve[-1][1] if curve else equity0
        for j, sig_di in pend_in:
            if j in pos or not np.isfinite(o[di, j]) or not np.isfinite(at[sig_di, j]):
                continue
            bps = slip_bps(adv[sig_di, j], cost_mult)
            px = o[di, j] * (1 + bps / 1e4)
            qty = int(POSITION_PCT * eq_prev // px)
            if qty < 1 or qty * px > cash:
                continue
            cash -= qty * px
            pos[j] = dict(px=px, qty=qty, di=di, sig_di=sig_di, stop=px - STOP_ATR * at[sig_di, j],
                          bps=bps, votes=votes[sig_di, j])
        pend_in = []
        # 3) intraday stops (gap-aware)
        for j in list(pos):
            if np.isfinite(l[di, j]) and l[di, j] <= pos[j]["stop"]:
                px = min(o[di, j], pos[j]["stop"]) if di > pos[j]["di"] else pos[j]["stop"]
                close_pos(j, px * (1 - pos[j]["bps"] / 1e4), di, "stop")
        # 4) end-of-day exit signals
        for j, p in pos.items():
            if not np.isfinite(c[di, j]):
                continue
            last_px[j] = c[di, j]
            if di - p["di"] + 1 >= MAX_HOLD_DAYS:
                pend_out[j] = "time"
            elif r2[di, j] > EXIT_RSI2 or c[di, j] > s5[di, j]:
                pend_out[j] = "engine_neutral"
        # 5) new setups at today's close -> enter tomorrow
        slots = MAX_POSITIONS - len(pos) + len(pend_out)
        if slots > 0:
            ok = mem[di] & up[di] & (np.nan_to_num(votes[di]) >= min_votes) & np.isfinite(c[di])
            if alw is not None:
                ok &= alw[di]                      # optional external filter (e.g. news engine)
            cand = [j for j in np.flatnonzero(ok) if j not in pos]
            cand.sort(key=lambda j: (-votes[di, j], r2[di, j]))
            pend_in = [(j, di) for j in cand[:slots]]
        # 6) mark to market
        eq = cash + sum(p["qty"] * last_px.get(j, p["px"]) for j, p in pos.items())
        curve.append((dates[di], eq))
    for j in list(pos):
        close_pos(j, last_px.get(j, pos[j]["px"]) * (1 - pos[j]["bps"] / 1e4), di1 - 1, "period_end")
    eqs = pd.Series(dict(curve), dtype=float)
    return pd.DataFrame(trades), eqs


# ---------------- reporting ----------------
def perf(eq):
    eq = eq.dropna()
    if len(eq) < 2:
        return dict(cagr_pct=np.nan, max_dd_pct=np.nan, sharpe=np.nan)
    yrs = (eq.index[-1] - eq.index[0]).days / 365.25
    r = eq.pct_change().dropna()
    return dict(cagr_pct=round(((eq.iloc[-1] / eq.iloc[0]) ** (1 / max(yrs, 1e-9)) - 1) * 100, 2),
                max_dd_pct=round((eq / eq.cummax() - 1).min() * 100, 2),
                sharpe=round(r.mean() / r.std() * np.sqrt(252), 2) if r.std() > 0 else np.nan)

"""Oliver Velez-style study (separate preregistered study, backtest-only).

2-minute bars. Setup: 20/200 SMA "narrow state" in the last few bars, price on the trend side of both
SMAs with the 20 SMA sloping that way, and an entry bar: Elephant Bar (EB), Tail Bar (TB) or 180.
Entry: stop order 1 cent beyond the entry bar's high (long) / low (short), valid for 3 bars.
Stop: 1 cent beyond the entry bar's tail. Exit rule (fixed up front): 2R target, breakeven at +1R,
90-minute time stop, flat by 15:55.
"""
import numpy as np
import pandas as pd

import config
from common import Signal
from experts import Expert, _cents, _nan
from indicators import prepare_intraday

VZ_NARROW_PCT = 0.0020      # 20/200 SMA gap below 0.20% of price = narrow state
VZ_NARROW_LOOKBACK = 10     # ... at some point in the last 10 bars (20 minutes)
VZ_EB_MULT = 2.0            # elephant bar: body >= 2x the average body of the prior 20 bars
VZ_TRIGGER_BARS = 3         # stop-entry order stays live for 3 bars
VZ_MAX_POSITIONS = 10       # one expert here, so allow 10 concurrent positions (0.5% each)
VZ_TF = 2                   # bar size in minutes (--tf); rules are identical at every timeframe


def warmup_days():
    """Calendar days of 1-min history needed before the start so the 200-bar SMA is ready."""
    return max(6, int(200 * VZ_TF / 390 * 1.6) + 4)


def to_bars(df1, tf):
    """Session-anchored tf-minute bars (09:30 + k*tf) from 1-minute bars."""
    x = df1[["open", "high", "low", "close", "volume"]].sort_index()
    x = x[~x.index.duplicated(keep="last")]
    ny = x.index.tz_convert("America/New_York")
    mod = np.asarray(ny.hour * 60 + ny.minute - 570)
    x = x[(mod >= 0) & (mod < 390)]
    ny = ny[(mod >= 0) & (mod < 390)]
    mod = mod[(mod >= 0) & (mod < 390)]
    start = x.index - pd.to_timedelta(mod % tf, unit="min")      # bar start, anchored to 09:30
    g = x.groupby(start)
    out = pd.DataFrame({"open": g["open"].first(), "high": g["high"].max(), "low": g["low"].min(),
                        "close": g["close"].last(), "volume": g["volume"].sum()})
    out.index.name = x.index.name
    return out.dropna(subset=["open", "close"])


def to_2min(df1):
    return to_bars(df1, 2)


def add_velez(df):
    c, o, h, l = df["close"], df["open"], df["high"], df["low"]
    s20, s200 = c.rolling(20).mean(), c.rolling(200).mean()
    df["vz_s20"], df["vz_s200"] = s20, s200
    gap = (s20 - s200).abs() / c
    df["vz_narrow"] = gap.rolling(VZ_NARROW_LOOKBACK).min() < VZ_NARROW_PCT
    df["vz_slope"] = s20 - s20.shift(5)
    df["vz_body_avg"] = (c - o).abs().rolling(20).mean().shift(1)
    return df


def prepare_2min(df1):
    """Name kept for compatibility: builds VZ_TF-minute bars (2 by default)."""
    return add_velez(prepare_intraday(to_bars(df1, VZ_TF)))


def pattern(a, i):
    """('EB'|'TB'|'180', side) for bar i, or (None, None)."""
    o, h, l, c = a["open"][i], a["high"][i], a["low"][i], a["close"][i]
    po, pc = a["open"][i - 1], a["close"][i - 1]
    rng, body, avg = h - l, abs(c - o), a["vz_body_avg"][i]
    if rng <= 0 or _nan(avg) or avg <= 0:
        return None, None
    if body >= VZ_EB_MULT * avg and body >= 0.7 * rng:
        return "EB", (1 if c > o else -1)
    lower, upper = min(o, c) - l, h - max(o, c)
    if lower >= 0.5 * rng and c >= l + 0.66 * rng:
        return "TB", 1
    if upper >= 0.5 * rng and c <= l + 0.34 * rng:
        return "TB", -1
    if pc < po and c > o and c >= po and o <= pc and body >= 0.6 * rng:
        return "180", 1
    if pc > po and c < o and c <= po and o >= pc and body >= 0.6 * rng:
        return "180", -1
    return None, None


class VelezExpert(Expert):
    id, name, style = "VELEZ", "Velez 20/200 narrow-state breakout", "breakout"
    DEFAULTS = dict(start_mod=10, last_mod=config.LAST_ENTRY_MOD, rr=2.0, max_hold=90)

    def entry(self, sym, a, i, dirs):
        if i < 2 or not self.in_window(a, i) or not a["vz_narrow"][i]:
            return None
        kind, side = pattern(a, i)
        if side is None or side not in dirs:
            return None
        c, s20, s200, slope = a["close"][i], a["vz_s20"][i], a["vz_s200"][i], a["vz_slope"][i]
        if _nan(s20, s200, slope):
            return None
        if side > 0 and not (c > s20 and c > s200 and slope > 0):
            return None
        if side < 0 and not (c < s20 and c < s200 and slope < 0):
            return None
        h, l = float(a["high"][i]), float(a["low"][i])
        trigger = round(h + 0.01, 2) if side > 0 else round(l - 0.01, 2)
        stop = round(l - 0.01, 2) if side > 0 else round(h + 0.01, 2)
        risk = abs(trigger - stop)
        target = _cents(trigger + side * self.p["rr"] * risk, down=side < 0)
        return Signal(self.id, sym, side, trigger, stop, target, risk,
                      f"{kind} narrow-state @{trigger}", trigger=trigger)


def activate():
    """Study-specific engine settings."""
    config.MAX_POSITIONS_PER_EXPERT = VZ_MAX_POSITIONS
    config.PENDING_EXPIRY_S = VZ_TRIGGER_BARS * VZ_TF * 60
    print(f"STUDY: Velez {VZ_TF}-min (narrow < {VZ_NARROW_PCT:.2%}, EB/TB/180, stop-entry, 2R target, "
          f"max {VZ_MAX_POSITIONS} positions)")
    return [VelezExpert()]


def cost_check(arrays, sym_sector, start, end, slip_bps):

    e = VelezExpert()
    lo, hi = int(start.strftime("%Y%m%d")), int(end.strftime("%Y%m%d"))
    rows = []
    for s in sym_sector:
        a = arrays.get(s)
        if a is None:
            continue
        for i in np.flatnonzero((a["session"] >= lo) & (a["session"] <= hi)):
            if i < 200:
                continue
            sig = e.entry(s, a, i, {1, -1})
            if sig:

    print("\n================ VELEZ COST CHECK (no trades simulated) ================")
    if not rows:
        print("No setups found in this period.")
        return

    q = [0.1, 0.25, 0.5, 0.75, 0.9]

STOP_GRID_BPS = [5, 10, 15, 20, 25, 30, 40, 50, 60, 80, 100, 150, 200]


def _trades_for_scan(arrays, sym_sector, frames, start, end, slip, horizon_min):
    """Every Velez setup that actually triggered, with its full price path to the horizon.
    One trade per symbol at a time (a new setup is ignored until the previous path ends), so every
    stop in the grid is tested on exactly the same trades."""
    from common import allowed_dirs
    from regime import build_heatmap
    from datetime import date as _date
    e = VelezExpert()
    lo, hi = int(start.strftime("%Y%m%d")), int(end.strftime("%Y%m%d"))
    regime_cache, out = {}, []
    for sym, sector in sym_sector.items():
        a = arrays.get(sym)
        if a is None:
            continue
        mod, sess = a["mod"], a["session"]
        o, h, l, c = a["open"], a["high"], a["low"], a["close"]
        busy_until = -1
        for i in np.flatnonzero((sess >= lo) & (sess <= hi)):
            if i < 200 or i <= busy_until:
                continue
            d = int(sess[i])
            if d not in regime_cache:
                heat = build_heatmap(frames, _date(d // 10000, d // 100 % 100, d % 100))
                regime_cache[d] = heat["regime"].to_dict()
            dirs = allowed_dirs(e.style, regime_cache[d][sector])
            sig = e.entry(sym, a, i, dirs) if dirs else None
            if not sig:
                continue
            side, trig = sig.side, sig.trigger
            j = None
            for k in range(i + 1, min(i + 1 + VZ_TRIGGER_BARS, len(c))):
                if sess[k] != d:
                    break
                if (side > 0 and h[k] >= trig) or (side < 0 and l[k] <= trig):
                    j = k
                    break
            if j is None:
                continue
            fill = (max(o[j], trig) if side > 0 else min(o[j], trig)) * (1 + side * slip)
            last_mod = min(mod[j] + horizon_min, config.FLATTEN_MOD)
            k = j
            while k + 1 < len(c) and sess[k + 1] == d and mod[k + 1] <= last_mod:
                k += 1
            busy_until = k
            tail_bps = abs(trig - sig.stop) / trig * 1e4
            out.append(dict(day=d, symbol=sym, side=side, fill=fill, tail_bps=tail_bps,
                            o=o[j:k + 1].copy(), h=h[j:k + 1].copy(), l=l[j:k + 1].copy(), exit_close=c[k]))
    return out


def _result_bps(t, stop_bps, slip):
    """Net return in bps for one trade with a stop stop_bps away (None = no stop)."""
    side, fill = t["side"], t["fill"]
    if stop_bps is not None:
        stop = fill * (1 - side * stop_bps / 1e4)
        hit = (t["l"] <= stop) if side > 0 else (t["h"] >= stop)
        if hit.any():
            m = int(np.argmax(hit))
            px = stop if m == 0 else (min(t["o"][m], stop) if side > 0 else max(t["o"][m], stop))
            return side * (px * (1 - side * slip) - fill) / fill * 1e4, True
    px = t["exit_close"] * (1 - side * slip)
    return side * (px - fill) / fill * 1e4, False


def _grid(trades, slip, stops):
    rows = []
    for s in stops:
        pairs = [_result_bps(t, s if s != "tail" else t["tail_bps"], slip) for t in trades]
        res = np.array([p[0] for p in pairs])
        hit = np.mean([p[1] for p in pairs])
        loss = -res[res < 0].sum()
        rows.append(dict(stop=("none" if s is None else "bar tail (v1 rule)" if s == "tail" else f"{s} bps"),
                         stopped_pct=round(hit * 100, 1), avg_net_bps=round(res.mean(), 2),
                         win_pct=round((res > 0).mean() * 100, 1),
                         profit_factor=round(res[res > 0].sum() / loss, 3) if loss > 0 else float("inf"),
                         usd_per_trade=round(res.mean() / 1e4 * 500, 3)))
    return pd.DataFrame(rows)


def stop_scan(arrays, sym_sector, frames, start, end, slip_bps, horizon_min=90):
    """Run every triggered setup with NO stop to the horizon, then replay each stop width on the same
    trades. The stop is PICKED on the first 2/3 of days and CHECKED on the last 1/3."""
    slip = slip_bps / 1e4
    trades = _trades_for_scan(arrays, sym_sector, frames, start, end, slip, horizon_min)
    print(f"\n================ VELEZ STOP SCAN ({VZ_TF}-min bars, exit after {horizon_min} min "
          f"or 15:55, {slip_bps:g} bps/side) ================")
    if len(trades) < 30:
        print(f"Only {len(trades)} triggered setups - too few to judge.")
        return
    days = sorted({t["day"] for t in trades})
    cut = days[int(len(days) * 2 / 3)]
    pick = [t for t in trades if t["day"] < cut]
    check = [t for t in trades if t["day"] >= cut]
    print(f"triggered trades: {len(trades)} over {len(days)} days "
          f"(pick set {len(pick)} trades before {cut}, check set {len(check)} from {cut})")

    mae = np.array([(t["fill"] - t["l"].min()) / t["fill"] * 1e4 if t["side"] > 0
                    else (t["h"].max() - t["fill"]) / t["fill"] * 1e4 for t in trades])
    mfe = np.array([(t["h"].max() - t["fill"]) / t["fill"] * 1e4 if t["side"] > 0
                    else (t["fill"] - t["l"].min()) / t["fill"] * 1e4 for t in trades])
    q = [0.25, 0.5, 0.75, 0.9]
    print("worst move against (MAE), bps: " + "  ".join(f"p{int(x * 100)}={np.quantile(mae, x):.0f}" for x in q))
    print("best move in favour (MFE), bps: " + "  ".join(f"p{int(x * 100)}={np.quantile(mfe, x):.0f}" for x in q))

    stops = [None, "tail"] + STOP_GRID_BPS
    g_all = _grid(trades, slip, stops)
    print("\n--- all trades ---")
    print(g_all.to_string(index=False))
    g_pick = _grid(pick, slip, stops)
    best = g_pick.loc[g_pick["avg_net_bps"].idxmax()]
    g_check = _grid(check, slip, stops)
    chk = g_check.loc[g_check["stop"] == best["stop"]].iloc[0]
    none_chk = g_check.loc[g_check["stop"] == "none"].iloc[0]
    print(f"\nBEST STOP on pick set: {best['stop']}  ({best['avg_net_bps']:+.2f} bps/trade, "
          f"PF {best['profit_factor']})")
    print(f"same stop on CHECK set: {chk['avg_net_bps']:+.2f} bps/trade, PF {chk['profit_factor']}  "
          f"(no stop on check set: {none_chk['avg_net_bps']:+.2f})")
    gross = np.mean([_result_bps(t, None, 0.0)[0] for t in trades])
    print(f"\nno-stop result BEFORE costs: {gross:+.2f} bps/trade  |  round-trip cost: {2 * slip_bps:.1f} bps")
    ok = chk["avg_net_bps"] > 0 and best["avg_net_bps"] > 0
    print("VERDICT: " + ("PROMISING - best stop is profitable on both halves; next step is a full backtest "
                         "with this stop, then 2025 held-out data" if ok else
                         "NO EDGE - no stop is profitable on data it was not chosen on"))

"""Velez SWING method (Pristine Buy Setup, daily) and a Weinstein-hybrid version. Research only.

ARM A  Velez swing (Pristine Buy Setup, per his swing course):
  trend   rising 20-day SMA, rising 40-day SMA, 20 above 40   ("rising" = higher than 3 sessions ago)
  setup   3 consecutive lower highs AND lower lows ending on the signal day (a pullback in the uptrend)
  entry   buy stop 5 cents above the signal day's high, valid the next session only
  stop    5 cents below the signal day's low
  target  the swing high the pullback started from (sell-limit); none if price is already above it
  time    sell at the close of the 5th session held (Velez's 2-5 day swing)
ARM B  Weinstein hybrid: Arm A + the last completed week in Weinstein Stage 2
        (weekly close > rising 30-week SMA, Mansfield RS > 0)

import argparse
import os
from datetime import date, timedelta

import numpy as np
import pandas as pd

import weinstein as ws

TICK = 0.05
HOLD_DAYS = 5
SLOPE_DAYS = 3
RISK_PCT, MAX_POS_PCT, MAX_POSITIONS = 0.01, 0.10, 20
FILL_RULE = "pessimistic"   # same-bar ambiguity: "pessimistic" (stop assumed first) or "candle" (OHLC path rule)
DISCOVERY_END = date(2021, 12, 31)


def pbs_frame(d):
    """Signal-day features, all from the signal day and earlier."""
    c, h, l = d["close"], d["high"], d["low"]
    out = pd.DataFrame(index=d.index)
    s20, s40 = c.rolling(20).mean(), c.rolling(40).mean()
    out["trend"] = (s20 > s20.shift(SLOPE_DAYS)) & (s40 > s40.shift(SLOPE_DAYS)) & (s20 > s40)
    lh = (h < h.shift(1)) & (h.shift(1) < h.shift(2)) & (h.shift(2) < h.shift(3))
    ll = (l < l.shift(1)) & (l.shift(1) < l.shift(2)) & (l.shift(2) < l.shift(3))
    out["pbs"] = lh & ll
    out["setup"] = out["trend"] & out["pbs"]
    # swing high the pullback started from: highest high of the last 10 sessions before the signal day
    out["swing_high"] = h.shift(1).rolling(10).max()
    out["trigger"] = (h + TICK).round(2)
    out["stop"] = (l - TICK).round(2)
    return out


def stage2_daily(d, spy_w):
    """Weinstein Stage 2 of the last COMPLETED week, mapped onto each day (causal)."""
    f = ws.stage_frame(ws.weekly(d), spy_w)
    st2 = ((f["close"] > f["ma30"]) & f["slope_ok"] & (f["mansfield"] > 0)).fillna(False)
    prev = d.index.to_period("W-FRI") - 1
    return pd.Series(st2.reindex(prev).fillna(False).to_numpy(bool), index=d.index)


def run(panel, spy_daily, start, end, hybrid=False, cost_mult=1.0, equity0=100_000.0, market_filter=False):
    import swing
    spy_w = ws.weekly(pd.DataFrame({"open": spy_daily, "high": spy_daily, "low": spy_daily,
                                    "close": spy_daily, "volume": 0.0}))["close"]
    mkt = ws.spy_stage2(spy_daily)
    syms = sorted(s for s, d in panel.items() if len(d) >= 60)
    O, H, L, C = (pd.DataFrame({s: panel[s][k] for s in syms}).sort_index() for k in ("open", "high", "low", "close"))
    adv, close = ws._liquid({s: panel[s] for s in syms})
    member = swing.membership(adv, close, large_rank=0, min_adv=ws.MIN_ADV, min_price=ws.MIN_PRICE)
    dates = C.index
    per = dates.to_period("W-FRI")
    o, h, l, c = O.to_numpy(), H.to_numpy(), L.to_numpy(), C.to_numpy()
    a = adv.reindex(index=dates, columns=syms).to_numpy()
    mem = member.reindex(index=dates, columns=syms).fillna(False).to_numpy(bool)
    F = {s: pbs_frame(panel[s]).reindex(dates) for s in syms}
    setup = np.column_stack([F[s]["setup"].fillna(False).to_numpy(bool) for s in syms])
    trig = np.column_stack([F[s]["trigger"].to_numpy() for s in syms])
    stp = np.column_stack([F[s]["stop"].to_numpy() for s in syms])
    swh = np.column_stack([F[s]["swing_high"].to_numpy() for s in syms])
    if hybrid:
        st2 = np.column_stack([stage2_daily(panel[s], spy_w).reindex(dates).fillna(False).to_numpy(bool) for s in syms])
        setup &= st2
    di0 = int(np.searchsorted(dates, pd.Timestamp(start)))
    di1 = int(np.searchsorted(dates, pd.Timestamp(end), side="right"))
    cash, pos, pend, trades, curve, last = equity0, {}, {}, [], [], {}

    def close_pos(j, px, di, why):
        nonlocal cash
        p = pos.pop(j)
        cash += p["qty"] * px
        trades.append(dict(symbol=syms[j], signal_date=dates[p["sig"]].date(), entry_date=dates[p["di"]].date(),
                           exit_date=dates[di].date(), trigger=p["trigger"], stop=p["stop"],
                           target=(p["target"] if np.isfinite(p["target"]) else None),
                           entry=round(p["px"], 4), exit=round(px, 4), ret_pct=round((px / p["px"] - 1) * 100, 3),
                           r=round((px - p["px"]) / p["risk"], 3), days=di - p["di"] + 1, reason=why,
                           cost_bps=p["bps"], pnl=round(p["qty"] * (px - p["px"]), 2)))

    for di in range(max(di0, 1), di1):
        eq = curve[-1][1] if curve else equity0
        # 1) yesterday's setups: buy stop above the signal day's high, today only
        for j, sg in list(pend.items()):
            pend.pop(j)
            if j in pos or len(pos) >= MAX_POSITIONS or not (np.isfinite(h[di, j]) and h[di, j] >= sg["trigger"]):
                continue
            bps = ws.slip_bps(a[di, j], cost_mult)
            px = max(o[di, j], sg["trigger"]) * (1 + bps / 1e4)
            if px <= sg["stop"]:
                continue
            risk = px - sg["stop"]
            qty = int(min(RISK_PCT * eq / risk, MAX_POS_PCT * eq / px))
            if qty < 1 or qty * px > cash:
                continue
            cash -= qty * px
            tgt = round(float(sg["target"]), 2) if np.isfinite(sg["target"]) and sg["target"] > px else np.nan   # limit in cents
            pos[j] = dict(px=px, qty=qty, di=di, sig=sg["sig"], stop=sg["stop"], trigger=sg["trigger"],
                          target=tgt, risk=risk, bps=bps)
        # 2) intraday, using the candle to infer the order of the day's moves (standard OHLC path rule):
        #    up day (close > open): open -> low -> high -> close;  down day: open -> high -> low -> close.
        for j in list(pos):
            p = pos[j]
            if not np.isfinite(l[di, j]):
                continue
            up_day = (c[di, j] > o[di, j]) if FILL_RULE == "candle" else False   # pessimistic: treat as stop-first
            hit_stop = l[di, j] <= p["stop"]
            hit_tgt = np.isfinite(p["target"]) and h[di, j] >= p["target"]
            if di == p["di"]:
                # entry day: we bought on the way UP through the trigger. On an up day the low came first,
                # so the stop only counts if price fell back below it after the high - which an up-day
                # close above the open rules out for a stop below the trigger. On a down day the high
                # (our entry) came first, so a low below the stop does count.
                if hit_stop and not up_day:
                    close_pos(j, p["stop"] * (1 - p["bps"] / 1e4), di, "stop")
                elif hit_tgt:
                    close_pos(j, p["target"], di, "target")
                continue
            if hit_stop and hit_tgt:
                if up_day or FILL_RULE == "pessimistic":        # low first (or unknown) -> stopped
                    close_pos(j, min(o[di, j], p["stop"]) * (1 - p["bps"] / 1e4), di, "stop")
                else:             # high first -> target
                    close_pos(j, max(o[di, j], p["target"]), di, "target")
            elif hit_stop:
                close_pos(j, min(o[di, j], p["stop"]) * (1 - p["bps"] / 1e4), di, "stop")
            elif hit_tgt:
                close_pos(j, max(o[di, j], p["target"]), di, "target")
        # 3) time exit at the close of the 5th session held
        for j in list(pos):
            if np.isfinite(c[di, j]):
                last[j] = c[di, j]
                if di - pos[j]["di"] + 1 >= HOLD_DAYS:
                    close_pos(j, c[di, j] * (1 - pos[j]["bps"] / 1e4), di, "day5_close")
        # 4) today's setups -> tomorrow's buy stops
        if not (market_filter and not bool(mkt.get(per[di] - 1, False))):
            cand = [j for j in np.flatnonzero(setup[di] & mem[di]) if j not in pos]
            room = MAX_POSITIONS - len(pos)
            for j in cand[:max(room, 0)]:
                pend[j] = dict(trigger=trig[di, j], stop=stp[di, j], target=swh[di, j], sig=di)
        curve.append((dates[di], cash + sum(p["qty"] * last.get(j, p["px"]) for j, p in pos.items())))
    for j in list(pos):
        close_pos(j, last.get(j, pos[j]["px"]) * (1 - pos[j]["bps"] / 1e4), di1 - 1, "period_end")
    return pd.DataFrame(trades), pd.Series(dict(curve), dtype=float)


def _row(name, tr, eq):
    import swing
    if tr.empty:
        return dict(arm=name, trades=0)
    r, R = tr["ret_pct"], tr["r"]
    y = pd.to_datetime(tr["entry_date"]).dt.year
    yrs = sorted(y.unique())
    cut = yrs[int(len(yrs) * 2 / 3)] if len(yrs) >= 3 else yrs[-1]
    p = swing.perf(eq)
    return dict(arm=name, trades=len(tr), win_pct=round((r > 0).mean() * 100, 1), avg_pct=round(r.mean(), 3),
                avg_R=round(R.mean(), 3), t_stat=round(r.mean() / (r.std() / np.sqrt(len(r))), 2),
                pick_avg_R=round(R[y < cut].mean(), 3), check_avg_R=round(R[y >= cut].mean(), 3),
                cagr_pct=p["cagr_pct"], max_dd_pct=p["max_dd_pct"], sharpe=p["sharpe"])


def report(results, bench, start, out="velez_swing_results"):
    import swing
    os.makedirs(out, exist_ok=True)
    print("\n================ VELEZ SWING (Pristine Buy Setup) vs WEINSTEIN HYBRID ================")
    print("NOTE: universe = stocks listed TODAY (delisted names missing) -> survivorship bias flatters results.")
    rows = []
    for name, (tr, eq) in results.items():
        rows.append(_row(name, tr, eq))
        if not tr.empty:
            tr.to_csv(os.path.join(out, f"trades_{name.split()[0]}.csv"), index=False)
    for b, s in bench.items():
        p = swing.perf(s.loc[pd.Timestamp(start):])
        rows.append(dict(arm=f"buy & hold {b}", cagr_pct=p["cagr_pct"], max_dd_pct=p["max_dd_pct"], sharpe=p["sharpe"]))
    t = pd.DataFrame(rows).set_index("arm")
    print(t.to_string())
    t.to_csv(os.path.join(out, "arms.csv"))
    for name, (tr, _) in results.items():
        if not tr.empty:
            print(f"\n--- {name}: by exit reason ---")
            print(tr.groupby("reason")[["ret_pct", "r"]].agg(["count", "mean"]).round(3).to_string())
    for name, (tr, _) in results.items():
        if tr.empty:
            continue
        r = tr["ret_pct"]
        y = pd.to_datetime(tr["entry_date"]).dt.year
        yrs = sorted(y.unique())
        cut = yrs[int(len(yrs) * 2 / 3)] if len(yrs) >= 3 else yrs[-1]
        tstat = r.mean() / (r.std() / np.sqrt(len(r)))
        ok = tstat >= 2.5 and tr.loc[y < cut, "r"].mean() > 0 and tr.loc[y >= cut, "r"].mean() > 0
        print(f"\nVERDICT {name}: {'PASS' if ok else 'FAIL'} (needs t >= 2.5 and both halves positive; t = {tstat:.2f})")
    print(f"\nFiles written to {out}/")


def scan(panel, spy_daily, top=30):
    spy_w = ws.weekly(pd.DataFrame({"open": spy_daily, "high": spy_daily, "low": spy_daily,
                                    "close": spy_daily, "volume": 0.0}))["close"]
    rows = []
    for s, d in panel.items():
        if len(d) < 260:
            continue
        f = pbs_frame(d).iloc[-1]
        adv = float((d["close"] * d["volume"]).rolling(20).mean().iloc[-1])
        if not (bool(f["setup"]) and adv >= ws.MIN_ADV and d["close"].iloc[-1] >= ws.MIN_PRICE):
            continue
        st2 = bool(stage2_daily(d, spy_w).iloc[-1])
        rows.append(dict(symbol=s, close=round(d["close"].iloc[-1], 2), buy_above=f["trigger"], stop=f["stop"],
                         target=round(f["swing_high"], 2), risk_pct=round((f["trigger"] - f["stop"]) / f["trigger"] * 100, 1),
                         weinstein_stage2=st2))
    asof = max(d.index[-1] for d in panel.values())
    print(f"\n=== VELEZ SWING SETUPS as of {asof.date()} - buy stop valid next session only (research output) ===")
    if not rows:
        print("no setups")
        return
    print(pd.DataFrame(rows).sort_values(["weinstein_stage2", "risk_pct"], ascending=[False, True]).head(top)
          .to_string(index=False))

"""Weinstein Stage 2 breakout module: backtest + scanner + paper swing bot.

    python weinstein.py --start 2017-01-01 --end 2021-12-31      # discovery backtest
    python weinstein.py --bot                                      # plan only (dry run, no orders)
    python weinstein.py --bot --submit                             # paper orders (paper account only)

RULES (weekly bars, decided at the week's last close, acted on at the next open)
  BUY a Stage 2 breakout when ALL hold:
    - weekly close above the 30-week SMA, and the SMA is not falling (>= its value 4 weeks ago)
    - weekly close above the highest high of the prior 10 weeks (breakout from a base)
    - weekly volume >= 2x the average of the prior 10 weeks
    - Mansfield relative strength vs SPY > 0 (stock/SPY ratio above its 52-week average)
    - price >= $5, 20-day average dollar volume >= $5M
  INITIAL STOP 1% below the 10-week base low (skip the trade if that is more than 20% away).
  SELL when a weekly close is below the 30-week SMA (next open), or the stop is hit (checked daily).
  SIZE: risk 1% of equity per trade (entry - stop), capped at 10% of equity per position, max 20 positions.
"""
import argparse
import json
import os
import time
import uuid
from datetime import date, timedelta

import numpy as np
import pandas as pd

MA_WEEKS, SLOPE_WEEKS, BASE_WEEKS, VOL_MULT, RS_WEEKS = 30, 4, 10, 2.0, 52
STOP_BUFFER, MAX_STOP_PCT = 0.01, 0.20
RISK_PCT, MAX_POS_PCT, MAX_POSITIONS = 0.01, 0.10, 20
MIN_PRICE, MIN_ADV = 5.0, 5e6
DISCOVERY_END = date(2021, 12, 31)
COST_TIERS = [(50e6, 5.0), (20e6, 10.0), (0.0, 20.0)]
STATE_FILE = "weinstein_state.json"
STOP_RENEW_DAYS = 80
PREFIX = "WS"


# ---------------- weekly bars and signals (causal: a week is used only after it closes) ----------------
def weekly(d):
    """Daily OHLCV (date index) -> weekly bars indexed by week period (W-FRI)."""
    p = d.index.to_period("W-FRI")
    g = d.groupby(p)
    return pd.DataFrame({"open": g["open"].first(), "high": g["high"].max(), "low": g["low"].min(),
                         "close": g["close"].last(), "volume": g["volume"].sum()})


def stage_frame(w, bench_close_w):
    c, h, l, v = w["close"], w["high"], w["low"], w["volume"]
    out = pd.DataFrame(index=w.index)
    ma = c.rolling(MA_WEEKS).mean()
    out["ma30"] = ma
    out["slope_ok"] = ma >= ma.shift(SLOPE_WEEKS)
    out["base_high"] = h.shift(1).rolling(BASE_WEEKS).max()
    out["base_low"] = l.shift(1).rolling(BASE_WEEKS).min()
    out["vol_ratio"] = v / v.shift(1).rolling(BASE_WEEKS).mean()
    rs = c / bench_close_w.reindex(w.index)
    out["mansfield"] = (rs / rs.rolling(RS_WEEKS).mean() - 1) * 100
    out["stop"] = out["base_low"] * (1 - STOP_BUFFER)
    out["stop_pct"] = 1 - out["stop"] / c
    out["breakout"] = ((c > ma) & out["slope_ok"] & (c > out["base_high"]) & (out["vol_ratio"] >= VOL_MULT)
                       & (out["mansfield"] > 0) & (out["stop_pct"] > 0) & (out["stop_pct"] <= MAX_STOP_PCT))
    out["exit"] = c < ma
    out["close"] = c
    return out


def spy_stage2(bench_daily):
    """Weinstein's market check: is SPY itself in Stage 2 (above a rising 30-week SMA)?"""
    bw = weekly(pd.DataFrame({"open": bench_daily, "high": bench_daily, "low": bench_daily,
                              "close": bench_daily, "volume": 0.0}))["close"]
    ma = bw.rolling(MA_WEEKS).mean()
    return ((bw > ma) & (ma >= ma.shift(SLOPE_WEEKS))).fillna(False)


def slip_bps(adv, mult=1.0):
    for floor, bps in COST_TIERS:
        if adv >= floor:
            return bps * mult
    return COST_TIERS[-1][1] * mult


def _liquid(panel):
    adv = pd.DataFrame({s: (d["close"] * d["volume"]).rolling(20).mean() for s, d in panel.items()}).sort_index()
    close = pd.DataFrame({s: d["close"] for s, d in panel.items()}).sort_index()
    return adv, close


def build_signals(panel, bench_daily):
    bw = weekly(pd.DataFrame({"open": bench_daily, "high": bench_daily, "low": bench_daily,
                              "close": bench_daily, "volume": 0.0}))["close"]
    frames = {s: stage_frame(weekly(d), bw) for s, d in panel.items() if len(d) >= 60}
    wide = {k: pd.DataFrame({s: f[k] for s, f in frames.items()}).sort_index()
            for k in ("breakout", "exit", "stop", "mansfield", "close", "ma30", "vol_ratio")}
    return frames, wide

    _, wide = build_signals(panel, bench_daily)

                    if ex[j]:
                        pend_out[j] = "weekly_close_below_ma30"
                br = W["breakout"].loc[wk].to_numpy()
                st, ms = W["stop"].loc[wk].to_numpy(), W["mansfield"].loc[wk].to_numpy()
                cand = [j for j in range(len(syms)) if br[j] and mem[di, j] and j not in pos]
                if market_filter and not bool(mkt.get(wk, False)):
                    cand = []                                  # market not in Stage 2: no new longs
                cand.sort(key=lambda j: -ms[j])
                slots = MAX_POSITIONS - len(pos) + len(pend_out)
                pend_in = [(j, st[j], wk, a[di, j]) for j in cand[:max(slots, 0)]]
        curve.append((dates[di], cash + sum(p["qty"] * last.get(j, p["px"]) for j, p in pos.items())))
    for j in list(pos):
        close_pos(j, last.get(j, pos[j]["px"]) * (1 - pos[j]["bps"] / 1e4), di1 - 1, "period_end")
    return pd.DataFrame(trades), pd.Series(dict(curve), dtype=float)


def report(trades, eq, bench, out="weinstein_results"):
    import swing
    os.makedirs(out, exist_ok=True)
    print("\n================ WEINSTEIN STAGE 2 BREAKOUTS ================")
    print("NOTE: universe = stocks listed TODAY (delisted names missing) -> survivorship bias flatters results.")
    if trades.empty:
        print("No trades.")

    eq.to_csv(os.path.join(out, "equity.csv"))
    r, R = trades["ret_pct"], trades["r"]
    pf = r[r > 0].sum() / -r[r < 0].sum() if (r < 0).any() else float("inf")

          f"| profit factor {pf:.2f} | avg hold {trades['days'].mean():.0f} days | t-stat "
          f"{r.mean() / (r.std() / np.sqrt(len(r))):.2f} | best {R.max():+.1f}R")
    rows = {"strategy": swing.perf(eq)}
    for name, s in bench.items():
        s = s.reindex(eq.index).ffill()

    print("\n--- by exit reason ---")
    print(trades.groupby("reason")[["ret_pct", "r"]].agg(["count", "mean"]).round(3).to_string())
    print("\n--- by year ---")

def _last_completed_week(bench_daily, today):
    """Latest week whose final trading day is before today -> (period, last_day)."""
    d = bench_daily.index[bench_daily.index < pd.Timestamp(today)]
    per = d.to_period("W-FRI")
    cur = pd.Timestamp(today).to_period("W-FRI")
    done = per[per < cur]
    if not len(done):
        return None, None
    wk = done[-1]
    return wk, d[per == wk].max()

def plan(panel, bench_daily, today, equity, held, market_filter=False):
    """Decisions from the last COMPLETED week. Returns (week, fresh, exits, entries)."""
    frames, wide = build_signals(panel, bench_daily)
    wk, last_day = _last_completed_week(bench_daily, today)
    if wk is None:
        return None, False, [], []
    days_after = bench_daily.index[(bench_daily.index > last_day) & (bench_daily.index < pd.Timestamp(today))]
    fresh = len(days_after) == 0                     # today is the first session after that week
    adv, close = _liquid(panel)
    ex = [s for s in held if s in frames and wk in frames[s].index and frames[s].at[wk, "exit"] is not pd.NA
          and bool(frames[s].at[wk, "exit"])]
    rows = []
    for s, f in frames.items():
        if s in held or wk not in f.index or not (f.at[wk, "breakout"] is True or f.at[wk, "breakout"] == 1):
            continue
        px, a = float(f.at[wk, "close"]), float(adv[s].get(last_day, np.nan))
        if px < MIN_PRICE or not a >= MIN_ADV:
            continue
        stop = float(f.at[wk, "stop"])
        qty = int(min(RISK_PCT * equity / (px - stop), MAX_POS_PCT * equity / px))
        if qty >= 1:
            rows.append(dict(symbol=s, close=round(px, 2), stop=round(stop, 2),
                             stop_pct=round((1 - stop / px) * 100, 1), vol_x=round(float(f.at[wk, "vol_ratio"]), 1),
                             mansfield=round(float(f.at[wk, "mansfield"]), 1), qty=qty))
    rows.sort(key=lambda r: -r["mansfield"])
    if market_filter and not bool(spy_stage2(bench_daily).get(wk, False)):
        print("market filter: SPY is not in Stage 2 - no new entries this week")
        rows = []
    return wk, fresh, ex, rows[:max(MAX_POSITIONS - len(held) + len(ex), 0)]


def run_bot(tc, panel, bench_daily, submit=False, today=None, wait_fill_s=60, market_filter=False):
    import config
    from broker import call
    from alpaca.trading.enums import OrderSide, TimeInForce, QueryOrderStatus
    from alpaca.trading.requests import MarketOrderRequest, StopOrderRequest, GetOrdersRequest
    if submit and not config.PAPER:
        raise SystemExit("weinstein bot is paper-only: APCA_PAPER must be true")
    today = today or date.today()
    st = _load_state()
    clock = call(tc.get_clock, what="clock")
    if submit and not clock.is_open:
        print("market closed - nothing submitted (run during market hours, e.g. 09:35 ET)")
        submit = False
    acct = call(tc.get_account, what="account")
    equity = float(acct.equity)
    broker_pos = {p.symbol: abs(float(p.qty)) for p in call(tc.get_all_positions, what="positions")}
    held = {s for s in st["positions"] if s in broker_pos}
    for s in list(st["positions"]):
        if s not in broker_pos:
            print(f"{s}: no longer held (stopped out or closed) - removed from state")
            st["positions"].pop(s)
    open_orders = call(tc.get_orders, filter=GetOrdersRequest(status=QueryOrderStatus.OPEN), what="orders")
    open_by_sym, mine = {}, {}
    for od in open_orders:
        open_by_sym.setdefault(od.symbol, []).append(od)
        if (od.client_order_id or "").startswith(PREFIX + "_"):
            mine.setdefault(od.symbol, []).append(od)          # only this bot's orders are ever cancelled

    wk, fresh, exits, entries = plan(panel, bench_daily, today, equity, held, market_filter)
    print(f"\n=== WEINSTEIN BOT {'(SUBMITTING - PAPER)' if submit else '(DRY RUN - no orders)'} ===")
    print(f"equity ${equity:,.0f} | held {sorted(held)} | last completed week {wk} | fresh signals today: {fresh}")

    # 1) exits: weekly close below the 30-week MA
    for s in exits:
        qty = min(broker_pos[s], float(st["positions"][s].get("qty", broker_pos[s])))
        print(f"EXIT  {s}: weekly close below 30-week MA -> sell {qty:g}")
        if submit:
            for od in mine.get(s, []):
                call(tc.cancel_order_by_id, od.id, what=f"cancel {s}")
            time.sleep(1)
            call(tc.submit_order, order_data=MarketOrderRequest(
                symbol=s, qty=qty, side=OrderSide.SELL, time_in_force=TimeInForce.DAY,
                client_order_id=f"{PREFIX}_X_{s}_{uuid.uuid4().hex[:6]}"), what=f"exit {s}")
            st["positions"].pop(s, None)

    # 2) protective stops: every held position must have a live GTC stop.
    #    Alpaca cancels GTC orders 90 days after creation, so stops older than 80 days are renewed.
    now = pd.Timestamp.now(tz="UTC")
    for s in held - set(exits):
        for od in list(mine.get(s, [])):
            made = getattr(od, "created_at", None)
            if getattr(od, "stop_price", None) and made is not None and \
                    (now - pd.Timestamp(made).tz_convert("UTC")).days >= STOP_RENEW_DAYS:
                print(f"RENEW {s}: stop is {(now - pd.Timestamp(made).tz_convert('UTC')).days} days old "
                      f"(Alpaca cancels GTC at 90) -> re-placing")
                if submit:
                    call(tc.cancel_order_by_id, od.id, what=f"cancel old stop {s}")
                mine[s].remove(od)
        has_stop = any(getattr(od, "stop_price", None) for od in mine.get(s, []))
        if not has_stop:
            stp = st["positions"][s]["stop"]
            qty = min(broker_pos[s], float(st["positions"][s].get("qty", broker_pos[s])))
            print(f"STOP  {s}: no live stop found -> place GTC stop {stp}")
            if submit:
                call(tc.submit_order, order_data=StopOrderRequest(
                    symbol=s, qty=qty, side=OrderSide.SELL, stop_price=stp,
                    time_in_force=TimeInForce.GTC, client_order_id=f"{PREFIX}_S_{s}_{uuid.uuid4().hex[:6]}"),
                    what=f"stop {s}")

    # 3) entries: only on the first session after the signal week, once per symbol per week
    tried = set(st["attempted"].get(str(wk), []))
    if not fresh:
        print("no new entries today: this week's signals are no longer fresh (run on the first session of the week)")
        entries = []
    for e in entries:
        s = e["symbol"]
        if s in tried or s in broker_pos or s in open_by_sym:
            continue
        print(f"BUY   {s}: {e['qty']} sh ~${e['close']} stop {e['stop']} ({e['stop_pct']}% risk) "
              f"vol {e['vol_x']}x RS {e['mansfield']}")
        if not submit:
            continue
        st["attempted"].setdefault(str(wk), []).append(s)
        od = call(tc.submit_order, order_data=MarketOrderRequest(
            symbol=s, qty=e["qty"], side=OrderSide.BUY, time_in_force=TimeInForce.DAY,
            client_order_id=f"{PREFIX}_B_{s}_{uuid.uuid4().hex[:6]}"), what=f"buy {s}")
        filled, t0 = 0.0, time.time()
        while time.time() - t0 < wait_fill_s:
            o2 = call(tc.get_order_by_id, od.id, what=f"order {s}")
            filled = float(o2.filled_qty or 0)
            if str(getattr(o2.status, "value", o2.status)).lower() == "filled":
                break
            time.sleep(2)
        if filled > 0:
            call(tc.submit_order, order_data=StopOrderRequest(
                symbol=s, qty=filled, side=OrderSide.SELL, stop_price=e["stop"], time_in_force=TimeInForce.GTC,
                client_order_id=f"{PREFIX}_S_{s}_{uuid.uuid4().hex[:6]}"), what=f"stop {s}")
        st["positions"][s] = {"stop": e["stop"], "week": str(wk), "entry_ref": e["close"], "qty": filled or e["qty"]}
        print(f"      filled {filled:g} - GTC stop placed" if filled else "      not filled yet - stop added on next run")
    if submit:
        _save_state(st)
    return wk, fresh, exits, entries

===== ./zone_filter.py
"""365-day zone filter for the 5 intraday experts (research).

For each stock and each session, using ONLY the 365 daily candles before that session:
  nodes     = volume-at-price levels (zones.profile_nodes: 50 bins, 3-bin smoothing, >= 3% of volume)
  BUY zone  = nearest node BELOW the prior close  +/- ZONE_ATR x daily ATR(14)
  SELL zone = nearest node ABOVE the prior close  +/- ZONE_ATR x daily ATR(14)
Experts may go long only while the 1-min close is inside the buy zone, short only inside the sell zone.
"""
import numpy as np
import pandas as pd

import zones

LOOKBACK = 365
ZONE_ATR = 0.5
MIN_HISTORY = 200


def _session_ints(index):
    idx = pd.DatetimeIndex(index)
    if idx.tz is not None:
        idx = idx.tz_convert("America/New_York")
    return np.asarray(idx.year * 10000 + idx.month * 100 + idx.day)


def zone_table(daily):
    """daily OHLCV -> DataFrame indexed by session int: buy_lo, buy_hi, sell_lo, sell_hi (from PRIOR days only)."""
    d = daily.sort_index()
    d = d[~d.index.duplicated(keep="last")]
    sess = _session_ints(d.index)
    h, l, c, v = (d[k].to_numpy(float) for k in ("high", "low", "close", "volume"))
    atr = zones._atr(d).to_numpy()
    rows = {}
    for k in range(MIN_HISTORY, len(d)):
        w = slice(max(0, k - LOOKBACK), k)                   # strictly before session k
        nodes = zones.profile_nodes(h[w], l[w], c[w], v[w])
        pc, a = c[k - 1], atr[k - 1]
        if not np.isfinite(a) or not len(nodes):
            continue
        below, above = nodes[nodes < pc], nodes[nodes > pc]
        b = below.max() if len(below) else np.nan
        s = above.min() if len(above) else np.nan
        rows[int(sess[k])] = (b - ZONE_ATR * a, b + ZONE_ATR * a, s - ZONE_ATR * a, s + ZONE_ATR * a)
    return pd.DataFrame.from_dict(rows, orient="index", columns=["buy_lo", "buy_hi", "sell_lo", "sell_hi"])


def attach(arrays, tables):
    """Add zbuy_lo/zbuy_hi/zsell_lo/zsell_hi arrays (per 1-min bar) to each symbol's arrays."""
    for s, a in arrays.items():
        t = tables.get(s)
        n = a["n"]
        if t is None or t.empty:
            for k in ("zbuy_lo", "zbuy_hi", "zsell_lo", "zsell_hi"):
                a[k] = np.full(n, np.nan)
            continue
        r = t.reindex(a["session"].astype(int))
        a["zbuy_lo"], a["zbuy_hi"] = r["buy_lo"].to_numpy(), r["buy_hi"].to_numpy()
        a["zsell_lo"], a["zsell_hi"] = r["sell_lo"].to_numpy(), r["sell_hi"].to_numpy()


def build_tables(symbols, start, end, feed):
    from datetime import timedelta
    from alpaca.data.timeframe import TimeFrame
    from data import cached_bars, make_data_client
    from indicators import NY

    def utc(x):
        return pd.Timestamp(x, tz=NY).tz_convert("UTC").to_pydatetime()
    daily = cached_bars(make_data_client(), sorted(symbols), TimeFrame.Day, utc(start - timedelta(days=620)),
                        utc(end + timedelta(days=1)), feed, "all")
    return {s: zone_table(d) for s, d in daily.items()}

===== ./zones.py
"""Weinstein / Velez 365-day zone study (research + scanner).

    python zones.py --start 2017-01-01 --end 2021-12-31      # discovery backtest
    python zones.py --scan                                    # today's Stage 2 stocks in their buy zones
    python zones.py --zones AAPL NVDA CELH                    # print the zones for specific symbols

WHETHER (Weinstein, last COMPLETED week): Stage 2 = weekly close > 30-week SMA, SMA not falling, Mansfield RS > 0.
WHERE (365 daily candles): volume-at-price profile of the last 365 sessions (typical price x volume, 50 bins,
  3-bin smoothing). Nodes = local peaks holding >= 3% of the volume.
  BUY ZONE  = highest node below price that is >= 97% of the 30-week SMA (else the SMA itself), +/- 0.5 ATR(14).
  TP ZONE   = lowest node above 1.02 x price (overhead supply). None -> "open air": target = 2R.
WHEN (Velez, daily): a bullish tail bar, 180 or elephant bar whose low touches the buy zone and closes at or
  above it -> stop-entry 1 cent above that bar's high, live 3 sessions.
STOP  = 1 cent below min(bar low, zone low); skip if more than 12% away. Skip if TP1 < 1.5R away.
EXITS = half at TP1 (limit), then stop to breakeven; rest on a weekly close below the 30-week SMA. Stops daily.
SIZE  = 1% of equity at risk, capped at 10% of equity, max 20 positions. Long only.
"""
import argparse
import os
from datetime import date, timedelta

import numpy as np
import pandas as pd

import weinstein as ws

LOOKBACK, BINS, NODE_SHARE = 365, 50, 0.03
ZONE_ATR, MA_TOL, TP_GAP = 0.5, 0.97, 1.02
MAX_STOP_PCT, MIN_RR, OPEN_AIR_R = 0.12, 1.5, 2.0
TRIGGER_DAYS, TP1_FRACTION = 3, 0.5
RISK_PCT, MAX_POS_PCT, MAX_POSITIONS = 0.01, 0.10, 20
DISCOVERY_END = date(2021, 12, 31)


# ---------------- zones from 365 daily candles ----------------
def profile_nodes(high, low, close, volume, bins=BINS, share=NODE_SHARE):
    """Volume-at-price nodes (price levels) from a window of daily candles."""
    lo, hi = np.nanmin(low), np.nanmax(high)
    if not np.isfinite(lo) or hi <= lo:
        return np.array([])
    tp = (high + low + close) / 3
    ok = np.isfinite(tp) & np.isfinite(volume)
    hist, edges = np.histogram(tp[ok], bins=bins, range=(lo, hi), weights=volume[ok])
    if hist.sum() <= 0:
        return np.array([])
    sm = np.convolve(hist, np.ones(3) / 3, mode="same")
    frac = sm / sm.sum()
    centers = (edges[:-1] + edges[1:]) / 2
    peak = (frac >= np.roll(frac, 1)) & (frac >= np.roll(frac, -1)) & (frac >= share)
    peak[0] = peak[0] and frac[0] >= frac[1]
    peak[-1] = peak[-1] and frac[-1] >= frac[-2]
    return centers[peak]


def zones_at(d, i, ma30, atr):
    """Zones using the 365 candles ending at row i (inclusive). Returns dict or None."""
    j0 = max(0, i - LOOKBACK + 1)
    h, l, c, v = (d[k].to_numpy()[j0:i + 1] for k in ("high", "low", "close", "volume"))
    if len(c) < 200:
        return None
    px = c[-1]
    nodes = profile_nodes(h, l, c, v)
    below = nodes[(nodes < px) & (nodes >= MA_TOL * ma30)]
    support = below.max() if len(below) else ma30
    above = nodes[nodes > TP_GAP * px]
    return dict(support=support, zone_lo=support - ZONE_ATR * atr, zone_hi=support + ZONE_ATR * atr,
                tp1=(above.min() if len(above) else np.nan), n_nodes=len(nodes),
                p10=np.percentile(c, 10), p50=np.percentile(c, 50), p90=np.percentile(c, 90),
                hi365=h.max(), lo365=l.min())


def _atr(d, n=14):
    pc = d["close"].shift(1)
    tr = pd.concat([d["high"] - d["low"], (d["high"] - pc).abs(), (d["low"] - pc).abs()], axis=1).max(axis=1)
    return tr.ewm(alpha=1 / n, adjust=False, min_periods=n).mean()


def _pattern(d, i, body_avg):
    """Velez entry bar on daily candle i: ('EB'|'TB'|'180', side) or (None, None)."""
    import velez
    a = {"open": d["open"].to_numpy(), "high": d["high"].to_numpy(), "low": d["low"].to_numpy(),
         "close": d["close"].to_numpy(), "vz_body_avg": body_avg}
    return velez.pattern(a, i)


def weekly_state(d, spy_w):
    """Weinstein stage per completed week + zones computed at each Stage-2 week's last session."""
    w = ws.weekly(d)
    f = ws.stage_frame(w, spy_w)
    f["stage2"] = (f["close"] > f["ma30"]) & f["slope_ok"] & (f["mansfield"] > 0)
    per = d.index.to_period("W-FRI")
    last_row = pd.Series(np.arange(len(d)), index=per).groupby(level=0).max()
    atr = _atr(d).to_numpy()
    z = {}
    for wk in f.index[f["stage2"].fillna(False).astype(bool)]:
        i = int(last_row.get(wk, -1))
        if i >= 0 and np.isfinite(atr[i]):
            zz = zones_at(d, i, float(f.at[wk, "ma30"]), float(atr[i]))
            if zz:
                z[wk] = zz
    return f, z

def run(panel, spy_daily, start, end, cost_mult=1.0, equity0=100_000.0, universe="all", market_filter=False):
    import swing
    spy_w = ws.weekly(pd.DataFrame({"open": spy_daily, "high": spy_daily, "low": spy_daily,
                                    "close": spy_daily, "volume": 0.0}))["close"]
    mkt = ws.spy_stage2(spy_daily)
    syms = sorted(s for s, d in panel.items() if len(d) >= 260)
    O, H, L, C = (pd.DataFrame({s: panel[s][k] for s in syms}).sort_index() for k in ("open", "high", "low", "close"))
    adv, close = ws._liquid({s: panel[s] for s in syms})
    member = swing.membership(adv, close, large_rank=(swing.LARGE_CAP_RANK if universe == "midsmall" else 0),
                              min_adv=ws.MIN_ADV, min_price=ws.MIN_PRICE)
    dates = C.index
    per = dates.to_period("W-FRI")
    o, h, l, c = O.to_numpy(), H.to_numpy(), L.to_numpy(), C.to_numpy()
    a = adv.reindex(index=dates, columns=syms).to_numpy()
    mem = member.reindex(index=dates, columns=syms).fillna(False).to_numpy(bool)
    di0 = int(np.searchsorted(dates, pd.Timestamp(start)))
    di1 = int(np.searchsorted(dates, pd.Timestamp(end), side="right"))
    # per-symbol state, aligned to the global calendar
    import velez
    st, zn, pat_arr = {}, {}, {}
    for j, s in enumerate(syms):
        f, z = weekly_state(panel[s], spy_w)
        st[j], zn[j] = f, z
        pat_arr[j] = {"open": o[:, j], "high": h[:, j], "low": l[:, j], "close": c[:, j],
                      "vz_body_avg": pd.Series(np.abs(c[:, j] - o[:, j])).rolling(20).mean().shift(1).to_numpy()}

    cash, pos, pending, trades, curve, last = equity0, {}, {}, [], [], {}

    def sell(j, qty, px, di, why):
        nonlocal cash
        p = pos[j]
        cash += qty * px
        p["qty"] -= qty
        p["legs"].append((qty, px))
        if p["qty"] <= 0:
            pos.pop(j)
            tot = sum(q for q, _ in p["legs"])
            avg = sum(q * x for q, x in p["legs"]) / tot
            trades.append(dict(symbol=syms[j], entry_date=dates[p["di"]].date(), exit_date=dates[di].date(),
                               trigger=p["trigger"], stop0=p["stop0"], tp1=round(p["tp1"], 4),
                               entry=round(p["px"], 4), avg_exit=round(avg, 4), ret_pct=round((avg / p["px"] - 1) * 100, 3),
                               r=round((avg - p["px"]) / p["risk"], 3), days=di - p["di"], tp1_hit=p["tp1_done"],
                               open_air=p["open_air"], pattern=p["pat"], reason=why, cost_bps=p["bps"],
                               pnl=round(tot * (avg - p["px"]), 2)))

    for di in range(max(di0, 1), di1):
        prev_wk = per[di] - 1
        # 1) weekly-exit orders queued last session -> today's open
        for j in [k for k, p in pos.items() if p.get("exit_next")]:
            if np.isfinite(o[di, j]):
                sell(j, pos[j]["qty"], o[di, j] * (1 - pos[j]["bps"] / 1e4), di, "weekly_close_below_ma30")
        # 2) stop-entry orders
        eq = curve[-1][1] if curve else equity0
        for j in list(pending):
            sig = pending[j]
            if j in pos or di > sig["expire"] or len(pos) >= MAX_POSITIONS:
                pending.pop(j)
                continue
            if not (np.isfinite(h[di, j]) and h[di, j] >= sig["trigger"]):
                continue
            pending.pop(j)
            bps = ws.slip_bps(a[di, j], cost_mult)
            px = max(o[di, j], sig["trigger"]) * (1 + bps / 1e4)
            if px <= sig["stop"]:
                continue
            risk = px - sig["stop"]
            qty = int(min(RISK_PCT * eq / risk, MAX_POS_PCT * eq / px))
            if qty < 2 or qty * px > cash:
                continue
            tp1 = sig["tp1"] if np.isfinite(sig["tp1"]) else px + OPEN_AIR_R * risk
            cash -= qty * px
            pos[j] = dict(px=px, qty=qty, di=di, stop=sig["stop"], risk=risk, tp1=tp1, tp1_done=False,
                          open_air=not np.isfinite(sig["tp1"]), pat=sig["pat"], bps=bps, legs=[],
                          trigger=sig["trigger"], stop0=sig["stop"])
        # 3) intraday: stop first (pessimistic), then TP1
        for j in list(pos):
            p = pos[j]
            if not np.isfinite(l[di, j]):
                continue
            if l[di, j] <= p["stop"]:
                fill = p["stop"] if di == p["di"] else min(o[di, j], p["stop"])
                sell(j, p["qty"], fill * (1 - p["bps"] / 1e4), di, "breakeven_stop" if p["tp1_done"] else "stop")
                continue
            if not p["tp1_done"] and h[di, j] >= p["tp1"]:
                q = max(1, int(p["qty"] * TP1_FRACTION))
                fill = max(o[di, j], p["tp1"]) if di > p["di"] else p["tp1"]
                p["tp1_done"], p["stop"] = True, max(p["stop"], p["px"])
                sell(j, min(q, p["qty"]), fill, di, "tp1_then_rest")
        for j in pos:
            if np.isfinite(c[di, j]):
                last[j] = c[di, j]
        # 4) end of week: weekly close below the 30-week SMA -> sell the rest next open
        is_week_end = di + 1 >= len(dates) or per[di + 1] != per[di]
        if is_week_end:
            wk = per[di]
            for j, p in pos.items():
                f = st[j]
                if wk in f.index and bool(f.at[wk, "exit"]):
                    p["exit_next"] = True
        # 5) new Velez triggers inside the buy zone (Stage 2 from the last completed week)
        if len(pos) + len(pending) < MAX_POSITIONS and not (market_filter and not bool(mkt.get(prev_wk, False))):
            for j in range(len(syms)):
                if j in pos or j in pending or not mem[di, j]:
                    continue
                z = zn[j].get(prev_wk)
                if z is None or not np.isfinite(c[di, j]):
                    continue
                if not (l[di, j] <= z["zone_hi"] and c[di, j] >= z["zone_lo"]):
                    continue
                kind, side = velez.pattern(pat_arr[j], di)
                if side != 1:
                    continue
                trig = round(h[di, j] + 0.01, 2)
                stop = round(min(l[di, j], z["zone_lo"]) - 0.01, 2)
                if stop <= 0 or (trig - stop) / trig > MAX_STOP_PCT:
                    continue
                if np.isfinite(z["tp1"]) and (z["tp1"] - trig) < MIN_RR * (trig - stop):
                    continue
                pending[j] = dict(trigger=trig, stop=stop, tp1=z["tp1"], expire=di + TRIGGER_DAYS, pat=kind)
                if len(pos) + len(pending) >= MAX_POSITIONS:
                    break
        curve.append((dates[di], cash + sum(float(p["qty"]) * float(last.get(j, p["px"])) for j, p in pos.items())))
    for j in list(pos):
        sell(j, pos[j]["qty"], last.get(j, pos[j]["px"]) * (1 - pos[j]["bps"] / 1e4), di1 - 1, "period_end")
    return pd.DataFrame(trades), pd.Series(dict(curve), dtype=float)


def report(trades, eq, bench, out="zones_results"):
    import swing
    os.makedirs(out, exist_ok=True)
    print("\n================ WEINSTEIN / VELEZ 365-DAY ZONES ================")
    print("NOTE: universe = stocks listed TODAY (delisted names missing) -> survivorship bias flatters results.")
    if trades.empty:
        print("No trades.")
        return
    trades.to_csv(os.path.join(out, "trades.csv"), index=False)
    eq.to_csv(os.path.join(out, "equity.csv"))
    r, R = trades["ret_pct"], trades["r"]
    pf = r[r > 0].sum() / -r[r < 0].sum() if (r < 0).any() else float("inf")
    print(f"trades {len(trades)} | win {(r > 0).mean() * 100:.1f}% | avg {r.mean():+.2f}% | avg {R.mean():+.2f}R | "
          f"PF {pf:.2f} | TP1 hit {trades['tp1_hit'].mean() * 100:.0f}% | hold {trades['days'].mean():.0f}d | "
          f"t-stat {r.mean() / (r.std() / np.sqrt(len(r))):.2f}")
    rows = {"strategy": swing.perf(eq)}
    for name, s in bench.items():
        s = s.reindex(eq.index).ffill()
        rows[f"buy & hold {name}"] = swing.perf(s / s.iloc[0] * eq.iloc[0])
    print(pd.DataFrame(rows).T.to_string())
    y = pd.to_datetime(trades["entry_date"]).dt.year
    yrs = sorted(y.unique())
    cut = yrs[int(len(yrs) * 2 / 3)] if len(yrs) >= 3 else yrs[-1]
    for label, m in (("first 2/3 (pick)", y < cut), (f"from {cut} (check)", y >= cut)):
        if m.any():
            print(f"  {label:<18} trades {m.sum():>5}  avg {r[m].mean():+.2f}%  avg {R[m].mean():+.2f}R")
    for col in ("reason", "pattern", "open_air"):
        print(f"\n--- by {col} ---")
        print(trades.groupby(col)[["ret_pct", "r"]].agg(["count", "mean"]).round(3).to_string())
    print(f"\nFiles written to {out}/")


def show_zones(panel, spy_daily, symbols=None, only_setups=False, top=30):
    spy_w = ws.weekly(pd.DataFrame({"open": spy_daily, "high": spy_daily, "low": spy_daily,
                                    "close": spy_daily, "volume": 0.0}))["close"]
    rows = []
    for s in (symbols or sorted(panel)):
        d = panel.get(s)
        if d is None or len(d) < 260:
            continue
        w = ws.weekly(d)
        f = ws.stage_frame(w, spy_w)
        today_wk = d.index[-1].to_period("W-FRI")
        done = f.index[f.index < today_wk]
        if not len(done):
            continue
        wk = done[-1]
        stage2 = bool((f.at[wk, "close"] > f.at[wk, "ma30"]) and f.at[wk, "slope_ok"] and f.at[wk, "mansfield"] > 0)
        i_wk = int(np.flatnonzero(d.index.to_period("W-FRI") == wk)[-1])     # zones as of the completed week
        atr = float(_atr(d).iloc[i_wk])
        z = zones_at(d, i_wk, float(f.at[wk, "ma30"]), atr)
        if z is None:
            continue
        body = (d["close"] - d["open"]).abs().rolling(20).mean().shift(1).to_numpy()
        kind, side = _pattern(d, len(d) - 1, body)
        px = float(d["close"].iloc[-1])
        in_zone = d["low"].iloc[-1] <= z["zone_hi"] and px >= z["zone_lo"]
        setup = stage2 and in_zone and side == 1
        if only_setups and not setup:
            continue
        rows.append(dict(symbol=s, close=round(px, 2), stage2=stage2, ma30w=round(float(f.at[wk, "ma30"]), 2),
                         buy_zone=f"{z['zone_lo']:.2f}-{z['zone_hi']:.2f}",
                         tp1=("open air (2R)" if not np.isfinite(z["tp1"]) else round(z["tp1"], 2)),
                         p10_p50_p90=f"{z['p10']:.2f}/{z['p50']:.2f}/{z['p90']:.2f}",
                         hi365=round(z["hi365"], 2), in_zone=in_zone, bar=(kind if side == 1 else "-"),
                         SETUP="YES" if setup else ""))
    print(f"\n=== 365-DAY ZONES as of {max(d.index[-1] for d in panel.values()).date()} (research output, "
          f"not a recommendation) ===")
    if not rows:
        print("nothing to show" + (" - no setups today" if only_setups else ""))
        return
    print(pd.DataFrame(rows).head(top).to_string(index=False))


def main():
    import swing
    ap = argparse.ArgumentParser()
    ap.add_argument("--start")
    ap.add_argument("--end")
    ap.add_argument("--scan", action="store_true")
    ap.add_argument("--zones", nargs="+")
    ap.add_argument("--universe", choices=["all", "midsmall"], default="all")
    ap.add_argument("--universe-file")
    ap.add_argument("--cost-mult", type=float, default=1.0)
    ap.add_argument("--use-holdout", action="store_true")
    ap.add_argument("--market-filter", action="store_true", help="no new longs unless SPY is in Stage 2")
    a = ap.parse_args()
    today = date.today()
    if a.zones or a.scan:
        syms = a.zones or swing._symbols(a.universe_file)
        panel = swing.load_panel(sorted(syms), today - timedelta(days=800), today)
        spy = swing.load_panel(["SPY"], today - timedelta(days=800), today)["SPY"]["close"]
        show_zones(panel, spy, symbols=a.zones, only_setups=a.scan)
        return
    if not (a.start and a.end):
        raise SystemExit("give --start and --end, --scan, or --zones SYMBOLS")
    if pd.Timestamp(a.end).date() > DISCOVERY_END and not a.use_holdout:
        raise SystemExit(f"--end is after {DISCOVERY_END}: that data is HELD OUT.")
    start, end = pd.Timestamp(a.start).date(), pd.Timestamp(a.end).date()
    syms = swing._symbols(a.universe_file)
    panel = swing.load_panel(syms, start - timedelta(days=420), end + timedelta(days=1))

    for b in ("SPY", "IWM", "IJH"):
        p = swing.load_panel([b], start - timedelta(days=420), end + timedelta(days=1))

            bench[b] = p[b]["close"]
    trades, eq = run(panel, bench["SPY"], start, end, a.cost_mult, universe=a.universe, market_filter=a.market_filter)
    if a.market_filter:
        print("VARIANT: market filter ON (new longs only while SPY is in Stage 2)")
    report(trades, eq, {k: v.loc[pd.Timestamp(start):] for k, v in bench.items()},
           out="zones_results_mf" if a.market_filter else "zones_results")


if __name__ == "__main__":

"""Paper bot for the Weinstein / Velez 365-day zone strategy (zones.py rules).

    python zones_bot.py                 # plan: setups, triggers, management (no orders)
    python zones_bot.py --submit        # PAPER orders (paper account only)
    add --market-filter to require SPY in Stage 2 for new entries

Run it every ~15 minutes during market hours (cron). Each run:
  1. once per day: find setups on the last completed daily candle (Stage 2 + Velez bar inside the 365-day
     buy zone) -> pending triggers, valid for 3 sessions
  2. a pending stock trading at/above its trigger -> two GTC orders, protected from the moment they fill:
       lot A (half): bracket  = market entry + TP1 limit + stop
       lot B (rest): OTO      = market entry + stop
  3. TP1 filled -> lot B stop moved to breakeven (cancel leg, new stop; OTO legs cannot be modified)
  4. first session after a week closes below the 30-week SMA -> cancel this bot's orders, sell its shares
  5. any stop older than 80 days is renewed (Alpaca cancels GTC orders after 90 days)
Only orders tagged ZN_ are ever touched. Live fills happen at the market price when the trigger is seen,
so they can be a little worse than the backtest's fill at max(open, trigger).
"""
import argparse
import json
import os
import uuid
from datetime import date, timedelta

import numpy as np
import pandas as pd

import weinstein as ws
import zones

PREFIX = "ZN"
STATE_FILE = "zones_state.json"
STOP_RENEW_DAYS = 80
OPEN = {"new", "accepted", "held", "partially_filled", "pending_new", "pending_replace", "accepted_for_bidding"}


def _val(x):
    return str(getattr(x, "value", x)).lower()


def _cid(kind, sym):
    return f"{PREFIX}_{kind}_{sym}_{uuid.uuid4().hex[:6]}"


def load_state():
    if os.path.exists(STATE_FILE):
        try:
            return json.load(open(STATE_FILE))
        except Exception:   # noqa: BLE001
            os.replace(STATE_FILE, STATE_FILE + ".corrupt")
    return {"setup_day": None, "pending": {}, "positions": {}}


def save_state(st):
    tmp = STATE_FILE + ".tmp"
    json.dump(st, open(tmp, "w"), indent=2, default=str)
    os.replace(tmp, STATE_FILE)


def find_setups(panel, spy_daily, signal_day, equity, market_filter=False):
    """Same rules as zones.run: Stage 2 + zones from the week BEFORE the signal day's week,
    Velez bullish bar on the signal day's candle touching the buy zone."""
    import velez
    sd = pd.Timestamp(signal_day)
    prev_wk = sd.to_period("W-FRI") - 1
    if market_filter and not bool(ws.spy_stage2(spy_daily).get(prev_wk, False)):
        print("market filter: SPY not in Stage 2 - no new setups")
        return []
    spy_w = ws.weekly(pd.DataFrame({"open": spy_daily, "high": spy_daily, "low": spy_daily,
                                    "close": spy_daily, "volume": 0.0}))["close"]
    out = []
    for s, d in panel.items():
        d = d[d.index <= sd]
        if len(d) < 260 or d.index[-1] != sd:
            continue
        adv = float((d["close"] * d["volume"]).rolling(20).mean().iloc[-2])
        if not (adv >= ws.MIN_ADV and d["close"].iloc[-2] >= ws.MIN_PRICE):
            continue
        f = ws.stage_frame(ws.weekly(d), spy_w)
        if prev_wk not in f.index:
            continue
        r = f.loc[prev_wk]
        if not (r["close"] > r["ma30"] and r["slope_ok"] and r["mansfield"] > 0):
            continue
        per = d.index.to_period("W-FRI")
        i_wk = int(np.flatnonzero(per == prev_wk)[-1])
        z = zones.zones_at(d, i_wk, float(r["ma30"]), float(zones._atr(d).iloc[i_wk]))
        if z is None:
            continue
        lo, hi, cl = float(d["low"].iloc[-1]), float(d["high"].iloc[-1]), float(d["close"].iloc[-1])
        if not (lo <= z["zone_hi"] and cl >= z["zone_lo"]):
            continue
        a = {k: d[k].to_numpy() for k in ("open", "high", "low", "close")}
        a["vz_body_avg"] = (d["close"] - d["open"]).abs().rolling(20).mean().shift(1).to_numpy()
        kind, side = velez.pattern(a, len(d) - 1)
        if side != 1:
            continue
        trig, stop = round(hi + 0.01, 2), round(min(lo, z["zone_lo"]) - 0.01, 2)
        if stop <= 0 or (trig - stop) / trig > zones.MAX_STOP_PCT:
            continue
        open_air = not np.isfinite(z["tp1"])
        if not open_air and (z["tp1"] - trig) < zones.MIN_RR * (trig - stop):
            continue
        tp1 = round(trig + zones.OPEN_AIR_R * (trig - stop), 2) if open_air else round(float(z["tp1"]), 2)
        qty = int(min(zones.RISK_PCT * equity / (trig - stop), zones.MAX_POS_PCT * equity / trig))
        if qty >= 2:
            out.append(dict(symbol=s, trigger=trig, stop=stop, tp1=tp1, qty=qty, pattern=kind, open_air=open_air,
                            signal_day=str(sd.date()), zone=f"{z['zone_lo']:.2f}-{z['zone_hi']:.2f}"))
    return out


def run_bot(tc, panel, spy_daily, price_fn, submit=False, today=None, market_filter=False):
    import config
    from broker import call
    from alpaca.trading.enums import OrderClass, OrderSide, QueryOrderStatus, TimeInForce
    from alpaca.trading.requests import (GetOrderByIdRequest, GetOrdersRequest, MarketOrderRequest,
                                         StopLossRequest, StopOrderRequest, TakeProfitRequest)
    if submit and not config.PAPER:
        raise SystemExit("zones bot is paper-only: APCA_PAPER must be true")
    today = pd.Timestamp(today or date.today()).normalize()
    st = load_state()
    clock = call(tc.get_clock, what="clock")
    if submit and not clock.is_open:
        print("market closed - planning only, nothing submitted")
        submit = False
    equity = float(call(tc.get_account, what="account").equity)
    bpos = {p.symbol: abs(float(p.qty)) for p in call(tc.get_all_positions, what="positions")}
    orders = call(tc.get_orders, filter=GetOrdersRequest(status=QueryOrderStatus.OPEN), what="orders")
    mine, parents = {}, {}
    for o in orders:                                   # standalone orders this bot placed (tagged ZN_)
        if (o.client_order_id or "").startswith(PREFIX + "_") and _val(o.status) in OPEN:
            mine.setdefault(o.symbol, []).append(o)
    for s, p in st["positions"].items():               # bracket / OTO legs: found through OUR parent orders
        for key in ("a_id", "b_id"):
            if p.get(key):
                try:
                    par = call(tc.get_order_by_id, p[key], filter=GetOrderByIdRequest(nested=True), what=f"order {s}")
                except Exception as e:   # noqa: BLE001
                    print(f"WARN  {s}: could not load order {p[key]} ({e}) - management of it skipped this run")
                    continue
                parents[p[key]] = par
                for leg in (par.legs or []):
                    if _val(leg.status) in OPEN and all(leg.id != x.id for x in mine.get(s, [])):
                        mine.setdefault(s, []).append(leg)
    sessions = spy_daily.index[spy_daily.index < today]
    last_session = sessions[-1] if len(sessions) else None
    print(f"\n=== ZONES BOT {'(SUBMITTING - PAPER)' if submit else '(PLAN - no orders)'} === equity ${equity:,.0f}")

    # 0) reconcile: positions gone at the broker (stopped / sold) leave the state
    for s in list(st["positions"]):
        if s not in bpos and st["positions"][s].get("filled"):
            print(f"{s}: position closed at broker - removed")
            st["positions"].pop(s)

    # 1) setups once per completed session
    if last_session is not None and st["setup_day"] != str(last_session.date()):
        new = find_setups(panel, spy_daily, last_session, equity, market_filter)
        for s in list(st["pending"]):                 # a new session started: count down old triggers
            st["pending"][s]["sessions_left"] -= 1
            if st["pending"][s]["sessions_left"] <= 0:
                print(f"{s}: trigger expired unfilled")
                st["pending"].pop(s)
        for s in new:
            if s["symbol"] in st["positions"] or s["symbol"] in bpos:
                continue
            s["sessions_left"] = zones.TRIGGER_DAYS
            st["pending"][s["symbol"]] = s
        st["setup_day"] = str(last_session.date())
        print(f"setups from {last_session.date()}: {len(new)} | pending triggers: {len(st['pending'])}")

    # 2) triggers -> two protected lots
    slots = zones.MAX_POSITIONS - len(st["positions"])
    for s, p in list(st["pending"].items()):
        if slots <= 0:
            break
        if s in bpos or s in mine:
            continue
        px = price_fn(s)
        if px is None or px < p["trigger"]:
            continue
        if px <= p["stop"]:
            st["pending"].pop(s)
            continue
        qa = p["qty"] // 2
        qb = p["qty"] - qa
        print(f"BUY   {s}: {p['qty']} sh @~{px:.2f} (trigger {p['trigger']}) stop {p['stop']} TP1 {p['tp1']} "
              f"[{p['pattern']}, zone {p['zone']}{', open air' if p['open_air'] else ''}]")
        slots -= 1
        if not submit:
            continue
        a = call(tc.submit_order, order_data=MarketOrderRequest(
            symbol=s, qty=qa, side=OrderSide.BUY, time_in_force=TimeInForce.GTC, order_class=OrderClass.BRACKET,
            take_profit=TakeProfitRequest(limit_price=p["tp1"]), stop_loss=StopLossRequest(stop_price=p["stop"]),
            client_order_id=_cid("A", s)), what=f"lot A {s}")
        b = call(tc.submit_order, order_data=MarketOrderRequest(
            symbol=s, qty=qb, side=OrderSide.BUY, time_in_force=TimeInForce.GTC, order_class=OrderClass.OTO,
            stop_loss=StopLossRequest(stop_price=p["stop"]), client_order_id=_cid("B", s)), what=f"lot B {s}")
        st["positions"][s] = dict(qa=qa, qb=qb, stop=p["stop"], tp1=p["tp1"], entry_ref=px, a_id=str(a.id),
                                  b_id=str(b.id), be_moved=False, filled=True, since=str(today.date()))
        st["pending"].pop(s)

    # 3) management of held positions
    new_week = last_session is not None and len(sessions) >= 2 and \
        sessions[-1].to_period("W-FRI") != today.to_period("W-FRI")
    for s, p in list(st["positions"].items()):
        if s not in bpos:
            continue
        ords = mine.get(s, [])
        # 3a) weekly exit (first session after the week closed below the 30-week SMA)
        if new_week and s not in panel:
            print(f"WARN  {s}: no daily data this run - weekly exit check skipped (broker stop still protects it)")
        elif new_week:
            f = ws.stage_frame(ws.weekly(panel[s][panel[s].index < today]),
                               ws.weekly(pd.DataFrame({"open": spy_daily, "high": spy_daily, "low": spy_daily,
                                                       "close": spy_daily, "volume": 0.0}))["close"])
            wk = today.to_period("W-FRI") - 1
            if wk in f.index and bool(f.at[wk, "exit"]):
                held = min(bpos[s], p["qa"] + p["qb"] if not p.get("tp1_done") else p["qb"])
                print(f"EXIT  {s}: weekly close below 30-week SMA -> cancel ZN orders, sell {held:g}")
                if submit:
                    for o in ords:
                        call(tc.cancel_order_by_id, o.id, what=f"cancel {s}")
                    call(tc.submit_order, order_data=MarketOrderRequest(
                        symbol=s, qty=held, side=OrderSide.SELL, time_in_force=TimeInForce.DAY,
                        client_order_id=_cid("X", s)), what=f"exit {s}")
                    st["positions"].pop(s)
                continue
        # 3b) TP1 done? lot A's take-profit filled -> lot B stop to breakeven
        if not p["be_moved"]:
            a = parents.get(p["a_id"])
            tp_filled = a is not None and any(_val(l.status) == "filled" and getattr(l, "limit_price", None) for l in (a.legs or []))
            if tp_filled:
                be = round(float(p["entry_ref"]), 2)
                print(f"TP1   {s}: target hit -> lot B stop to breakeven {be}")
                p["tp1_done"] = True
                if submit:
                    for o in ords:
                        if getattr(o, "stop_price", None) and _val(o.status) in OPEN:
                            call(tc.cancel_order_by_id, o.id, what=f"cancel lot B stop {s}")
                    call(tc.submit_order, order_data=StopOrderRequest(
                        symbol=s, qty=p["qb"], side=OrderSide.SELL, stop_price=be, time_in_force=TimeInForce.GTC,
                        client_order_id=_cid("S", s)), what=f"BE stop {s}")
                    p["be_moved"], p["stop"] = True, be
        # 3c) renew stops older than 80 days (Alpaca cancels GTC after 90)
        now = pd.Timestamp.now(tz="UTC")
        for o in ords:
            made = getattr(o, "created_at", None)
            if getattr(o, "stop_price", None) and made is not None and \
                    (now - pd.Timestamp(made).tz_convert("UTC")).days >= STOP_RENEW_DAYS:
                q = float(getattr(o, "qty", 0) or 0)
                print(f"RENEW {s}: stop {o.stop_price} is {(now - pd.Timestamp(made).tz_convert('UTC')).days} days old")
                if submit and q > 0:
                    call(tc.cancel_order_by_id, o.id, what=f"cancel old stop {s}")
                    call(tc.submit_order, order_data=StopOrderRequest(
                        symbol=s, qty=q, side=OrderSide.SELL, stop_price=float(o.stop_price),
                        time_in_force=TimeInForce.GTC, client_order_id=_cid("S", s)), what=f"renew stop {s}")
    if st["pending"]:
        print("pending triggers: " + ", ".join(f"{k}@{v['trigger']}" for k, v in st["pending"].items()))
    save_state(st) if submit else None
    return st


def main():

            return None

            submit=a.submit, market_filter=a.market_filter)
