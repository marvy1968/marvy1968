import numpy as np

def calculate_efficiency_ratio(prices, period=14):
    if len(prices) < period + 1:
        return 0.0
    net_change = abs(prices[-1] - prices[-period-1])
    sum_changes = sum(abs(prices[i] - prices[i-1]) for i in range(-period, 0))
    return net_change / sum_changes if sum_changes > 0 else 0.0

def calculate_hurst_exponent(prices):
    if len(prices) < 30:
        return 0.5
    lags = range(2, 20)
    tau = [np.sqrt(np.std(np.subtract(prices[lag:], prices[:-lag]))) for lag in lags]
    poly = np.polyfit(np.log(lags), np.log(tau), 1)
    return poly[0] * 2.0

def classify_regime(prices):
    er = calculate_efficiency_ratio(prices)
    h = calculate_hurst_exponent(prices)
    
    if er >= 0.55 and h > 0.52:
        return "TREND", er, h
    elif er <= 0.30 or h < 0.48:
        return "CHOP", er, h
    else:
        return "SQUEEZE", er, h

    @staticmethod
    def analyze_holistic_market(df_m5, df_m15, df_h1, df_h4):
        bias_score = 0
        
        if not df_h4.empty and len(df_h4) >= 10:
            df_h4['sma20'] = df_h4['close'].rolling(10).mean()
            if df_h4['close'].iloc[-1] > df_h4['sma20'].iloc[-1]: bias_score += 3
            else: bias_score -= 3

        if not df_h1.empty and len(df_h1) >= 15:
            df_h1['sma20'] = df_h1['close'].rolling(15).mean()
            if df_h1['close'].iloc[-1] > df_h1['sma20'].iloc[-1]: bias_score += 2
            else: bias_score -= 2

        if not df_m15.empty and len(df_m15) >= 15:
            df_m15['sma20'] = df_m15['close'].rolling(15).mean()
            df_m15['vol_sma'] = df_m15['volume'].rolling(10).mean()
            latest_m15 = df_m15.iloc[-1]
            vol_multiplier = latest_m15['volume'] / max(latest_m15['vol_sma'], 1.0)
            vol_weight = 1.5 if vol_multiplier > 1.2 else 1.0

            if latest_m15['close'] > latest_m15['sma20']: bias_score += int(2 * vol_weight)
            else: bias_score -= int(2 * vol_weight)

        if not df_m5.empty and len(df_m5) >= 20:
            df_m5['sma20'] = df_m5['close'].rolling(20).mean()
            if df_m5['close'].iloc[-1] > df_m5['sma20'].iloc[-1]: bias_score += 1
            else: bias_score -= 1

        if bias_score >= 3: consensus = "STRONG_BULLISH"
        elif bias_score >= 1: consensus = "WEAK_BULLISH"
        elif bias_score <= -3: consensus = "STRONG_BEARISH"
        elif bias_score <= -1: consensus = "WEAK_BEARISH"
        else: consensus = "NEUTRAL"

        return consensus, bias_score

    @staticmethod
    def synthesize_dynamic_sl_tp(close, rec_sl, rec_tp, is_buy, df_m5, consensus):
        df_m5['atr'] = (df_m5['high'] - df_m5['low']).rolling(14).mean()
        atr = max(df_m5['atr'].iloc[-1], 0.060)

        tp_mult = 2.5 if "STRONG" in consensus else 2.0
        sl_mult = 1.2

        if is_buy:
            ai_sl = min(rec_sl, close - (sl_mult * atr)) if rec_sl > 0 else close - (sl_mult * atr)
            ai_tp = max(rec_tp, close + (tp_mult * atr)) if rec_tp > 0 else close + (tp_mult * atr)
            if (ai_tp - close) < MIN_TP_PIPS: ai_tp = close + MIN_TP_PIPS
            if (close - ai_sl) < MIN_SL_PIPS: ai_sl = close - MIN_SL_PIPS
        else:
            ai_sl = max(rec_sl, close + (sl_mult * atr)) if rec_sl > 0 else close + (sl_mult * atr)
            ai_tp = min(rec_tp, close - (tp_mult * atr)) if rec_tp > 0 else close - (tp_mult * atr)
            if (close - ai_tp) < MIN_TP_PIPS: ai_tp = close - MIN_TP_PIPS
            if (ai_sl - close) < MIN_SL_PIPS: ai_sl = close + MIN_SL_PIPS

        return round(ai_sl, 3), round(ai_tp, 3)

    @staticmethod
    def calculate_dynamic_risk_units(balance, entry_price, sl_price, confidence_score, num_active):
        """
        DYNAMIC SIZING ENGINE:
        Scales equity risk up to 1.0% per trade depending on trend conviction score.
        Score >= 5 (Strong Alignment) -> 1.00% Risk per trade
        Score >= 3 (Moderate Alignment) -> 0.75% Risk per trade
        Score <= 2 (Neutral / Reversion) -> 0.35%-0.50% Risk per trade
        """
        abs_score = abs(confidence_score)
        # AGGRESSIVE COMPOUNDING ENGINE: Reinvests NAV floating gains directly into unit sizing
        if abs_score >= 5:
            risk_pct = 0.0250  # 2.50% Equity Risk on High-Conviction Alignment
        elif abs_score >= 3:
            risk_pct = 0.0175  # 1.75% Equity Risk on Moderate Alignment
        else:
            risk_pct = 0.0100  # 1.00% Baseline Equity Risk

        risk_amount = balance * risk_pct
        price_risk = abs(entry_price - sl_price)
        if price_risk <= 0.005: price_risk = 0.040

        units = risk_amount / price_risk
        
        # Add offset per slot to eliminate FIFO same-size order rejection errors
        slot_offset = num_active * 11
        final_units = max(2000, min(units, 150000)) + slot_offset
        return final_units

        else:
            return "RANGING"

            "VOLATILITY_EXPANSION": ["BOT-1: STRUCTURAL BREAKOUT", "BOT-4: VOLATILITY EXPANSION"]
        }

#!/usr/bin/env python3
import os, sys, json, logging
from datetime import datetime, timezone
import numpy as np
import pandas as pd

class RangePressurePredictor:
    def __init__(self, symbol, df_5m, df_1h=None, df_daily=None):
        self.symbol = symbol
        self.df_5m = df_5m.copy()
        self.df_1h = df_1h.copy() if df_1h is not None else None
        self.df_daily = df_daily.copy() if df_daily is not None else None
        
    @staticmethod
    def calculate_atr(df, period=14):
        high = df['high'].values
        low = df['low'].values
        close = df['close'].values
        tr = np.maximum(high[1:] - low[1:], np.maximum(abs(high[1:] - close[:-1]), abs(low[1:] - close[:-1])))
        tr = np.insert(tr, 0, high[0] - low[0])
        return pd.Series(tr, index=df.index).rolling(period).mean().values

    def calculate_range_quality(self, lookback=24):
        df = self.df_5m.tail(lookback)
        if len(df) < lookback:
            return 0.0, 0.0, 0.0, 0.0

        close = df['close'].values
        high = df['high'].values
        low = df['low'].values

        range_high = float(np.max(high))
        range_low = float(np.min(low))
        range_span = range_high - range_low

        if range_span <= 0 or close[-1] <= 0:
            return 0.0, 0.0, 0.0, 0.0

        atr_5m = self.calculate_atr(self.df_5m, 14)
        curr_atr = atr_5m[-1]
        hist_atr_mean = float(np.mean(atr_5m[-100:])) if len(atr_5m) >= 100 else curr_atr
        atr_compression = max(0.0, min(1.0, 1.0 - (curr_atr / (hist_atr_mean + 1e-8))))

        sma20 = float(pd.Series(close).rolling(20).mean().values[-1])
        std20 = float(pd.Series(close).rolling(20).std().values[-1])
        bb_width = (4.0 * std20) / (sma20 + 1e-8)
        bb_compression = max(0.0, min(1.0, 1.0 - (bb_width * 12.0)))

        upper_touches = int(np.sum(high >= (range_high - 0.12 * range_span)))
        lower_touches = int(np.sum(low <= (range_low + 0.12 * range_span)))
        touch_score = min(1.0, (upper_touches + lower_touches) / 6.0)

        x = np.arange(len(close))
        r_sq = float(np.corrcoef(x, close)[0, 1] ** 2) if np.std(close) > 0 else 0.0
        drift_score = max(0.0, 1.0 - r_sq)

        quality_score = float(round((0.30 * touch_score + 0.30 * drift_score + 0.20 * atr_compression + 0.20 * bb_compression) * 100.0, 1))
        compression_score = float(round((0.50 * atr_compression + 0.50 * bb_compression) * 100.0, 1))

        return quality_score, compression_score, range_low, range_high

    def score_directional_pressure(self, range_low, range_high):
        df = self.df_5m.tail(30)
        close = df['close'].values
        high = df['high'].values
        low = df['low'].values
        vol = df['volume'].values if 'volume' in df.columns else np.ones(len(df))

        htf_vector = 0.0
        if self.df_1h is not None and len(self.df_1h) >= 50:
            c_1h = self.df_1h['close'].values
            ema20 = float(pd.Series(c_1h).ewm(span=20).mean().values[-1])
            ema50 = float(pd.Series(c_1h).ewm(span=50).mean().values[-1])
            if c_1h[-1] > ema20 > ema50: htf_vector = 1.0
            elif c_1h[-1] < ema20 < ema50: htf_vector = -1.0

        clv = float(np.mean(((close - low) - (high - close)) / (high - low + 1e-8)))

        price_diff = np.diff(close, prepend=close[0])
        up_vol = float(np.sum(vol[price_diff > 0]))
        down_vol = float(np.sum(vol[price_diff < 0]))
        vol_delta = (up_vol - down_vol) / (up_vol + down_vol + 1e-8)

        low_slope = float(np.polyfit(np.arange(10), low[-10:], 1)[0])
        high_slope = float(np.polyfit(np.arange(10), high[-10:], 1)[0])
        geom_vector = 0.0
        if low_slope > 0 and abs(high_slope) < low_slope: geom_vector = 0.80
        elif high_slope < 0 and abs(low_slope) < abs(high_slope): geom_vector = -0.80

        range_span = range_high - range_low + 1e-8
        bins = np.linspace(range_low, range_high, 10)
        hist, bin_edges = np.histogram(close, bins=bins, weights=vol)
        poc_bin_idx = int(np.argmax(hist))
        poc_price = (bin_edges[poc_bin_idx] + bin_edges[poc_bin_idx + 1]) / 2.0
        poc_rel_pos = (poc_price - range_low) / range_span
        poc_vector = (poc_rel_pos - 0.5) * 2.0

        sma20_seq = pd.Series(close).rolling(10).mean().values
        dev_seq = close - sma20_seq
        dev_clean = dev_seq[~np.isnan(dev_seq)]
        mom_slope = float(np.polyfit(np.arange(len(dev_clean)), dev_clean, 1)[0]) if len(dev_clean) >= 5 else 0.0
        mom_vector = max(-1.0, min(1.0, mom_slope * 5.0))

        rs_vector = 0.0
        if self.df_daily is not None and len(self.df_daily) >= 5:
            d_close = self.df_daily['close'].values
            ret_5d = (d_close[-1] - d_close[-5]) / d_close[-5]
            rs_vector = max(-1.0, min(1.0, ret_5d * 20.0))

            (0.25 * htf_vector) + (0.20 * vol_delta) + (0.15 * clv) +
            (0.15 * geom_vector) + (0.10 * poc_vector) + (0.10 * mom_vector) + (0.05 * rs_vector)

        p_up = round(expansion_prob * (up_weight / total_w), 2)
        p_down = round(expansion_prob * (down_weight / total_w), 2)

        return p_up, p_down, p_hold

def fetch_alpaca_summary():
    if not ALPACA_KEY or not ALPACA_SECRET:
        return None
    try:
        api = tradeapi.REST(key_id=ALPACA_KEY, secret_key=ALPACA_SECRET, base_url='https://paper-api.alpaca.markets', api_version='v2')
        acc = api.get_account()
        positions = api.list_positions()
        
        equity = float(acc.equity)
        buying_power = float(acc.buying_power)
        cash = float(acc.cash)
        
        stock_tickers = ['AAPL', 'NVDA', 'AMD', 'TSLA', 'MSFT', 'COIN', 'RBLX', 'AMZN', 'META', 'GOOGL', 'PLTR', 'BABA', 'NFLX', 'SBUX', 'BA', 'INTC', 'PYPL', 'DIS', 'MARA', 'RIOT']

        return {
            'equity': equity,
            'cash': cash,
            'buying_power': buying_power,
            'unrealized_pnl': unrealized_pnl,

        return None

def fetch_oanda_summary(days=1):
    if not OANDA_API_KEY or not OANDA_ACCOUNT_ID:
        return None
    headers = {"Authorization": f"Bearer {OANDA_API_KEY}"}
    summary_url = f"{OANDA_BASE_URL}/accounts/{OANDA_ACCOUNT_ID}/summary"
    try:
        req = urllib.request.Request(summary_url, headers=headers)
        with urllib.request.urlopen(req) as resp:
            acc = json.loads(resp.read().decode())['account']
            balance = float(acc['balance'])
            nav = float(acc['NAV'])
            unrealized_pnl = float(acc['unrealizedPL'])
            open_count = int(acc['openTradeCount'])
            
        # Transactions PnL
        cutoff_time = datetime.now(timezone.utc) - timedelta(days=days)
        realized_pnl = 0.0
        closed_trades = 0
        wins = 0
        
        tx_url = f"{OANDA_BASE_URL}/accounts/{OANDA_ACCOUNT_ID}/transactions/sinceid?id=1"
        req_tx = urllib.request.Request(tx_url, headers=headers)
        with urllib.request.urlopen(req_tx) as resp:
            txs = json.loads(resp.read().decode()).get('transactions', [])
            for tx in txs:
                if tx.get('type') == 'ORDER_FILL':
                    tx_time_str = tx.get('time', '')
                    if tx_time_str:
                        tx_dt = datetime.fromisoformat(tx_time_str.replace('Z', '+00:00'))
                        if tx_dt >= cutoff_time:
                            pl = float(tx.get('realizedPL', 0.0))
                            if pl != 0.0:
                                realized_pnl += pl
                                closed_trades += 1
                                if pl > 0: wins += 1
                                
        win_rate = (wins / closed_trades * 100) if closed_trades > 0 else 0.0
        return {
            'nav': nav,
            'balance': balance,
            'unrealized_pnl': unrealized_pnl,
            'open_count': open_count,
            'realized_pnl': realized_pnl,
            'closed_trades': closed_trades,
            'win_rate': win_rate
        }
    except Exception as e:
        logging.error(f"Error fetching OANDA summary: {e}")
        return None
