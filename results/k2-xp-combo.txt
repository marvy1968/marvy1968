"""Replay broker: fills at the engine's reference prices (+ slippage_bps adverse), no network, no orders."""
from __future__ import annotations
import itertools

class SimBroker:
    is_sim = True

    def __init__(self, ledger, start_equity: float, slippage_bps: float = 2.0, close_px=None):
        self.ledger, self.start, self.slip = ledger, start_equity, slippage_bps / 1e4
        self.close_px = close_px            # callable(sym) -> official close of the replay day
        self.n = itertools.count(1)

    def equity(self) -> float:
        r = self.ledger.query('SELECT COALESCE(SUM(pnl),0) FROM exits')[0][0]
        return self.start + float(r)

    def buying_power(self) -> float:
        return 4 * self.equity()

    def busy_symbols(self, xp_qty) -> set:
        return set()

    def can_short(self, sym) -> bool:
        return True

    def _px(self, px, side_buy):
        return round(px * (1 + self.slip) if side_buy else px * (1 - self.slip), 4)

    def enter(self, sg, qty, px, now):
        return dict(qty=qty, price=self._px(px, sg.side == 'long'), coid=f'{sg.system}-sim-{next(self.n)}')

    def exit(self, p, qty, reason, px, now):
        return dict(qty=qty, price=self._px(px, p.side == 'short'), coid=f'{p.system}-sim-{next(self.n)}')

    def place_stop(self, p): pass
    def sync_stop(self, p): pass
    def cancel_stop(self, p): pass
    def poll_stops(self, positions): return []

    def submit_moc(self, system, sym, side, qty, purpose, now, pos=None):
        return dict(coid=f'{system}-sim-moc-{next(self.n)}', sym=sym, side=side, qty=qty)

    def moc_fill(self, m, now):
        sym = m['sig'].symbol if m['kind'] == 'entry' else m['pos'].symbol
        px = self.close_px(sym) if self.close_px else None
        if px is None:
            return None
        buy = (m['kind'] == 'entry' and m['sig'].side == 'long') or (m['kind'] == 'exit' and m['pos'].side == 'short')
        return dict(qty=m['qty'], price=self._px(px, buy))

from __future__ import annotations

def load(path: str) -> dict:
    cfg = yaml.safe_load(open(path))
    base = os.path.dirname(os.path.dirname(os.path.abspath(path)))
    cfg['_base'] = base
    for k in ('db_path', 'reports_dir', 'log_file', 'control_dir'):

            cfg[k] = os.path.join(base, cfg[k])
    if cfg.get('env_file') and os.path.exists(cfg['env_file']):
        load_dotenv(cfg['env_file'], override=False)
    return cfg

    core_syms = [m for sec in core['sectors'].values() for m in sec['members']]
    return dict(core=core_syms, index=list(u['index']), sp500=list(sp['sp500']),
                sp1500=list(dict.fromkeys(sp['sp500'] + sp['sp400'] + sp['sp600'])),
                sector_of={m: k for k, sec in core['sectors'].items() for m in sec['members']})

"""Operator control.
  python -m xp.ctl list                         systems + state

from __future__ import annotations
import argparse, datetime as dt, glob, logging, os, sys
import pandas as pd
from .config import load, build_strategies
from .ledger import Ledger


def main(argv=None):
    ap = argparse.ArgumentParser()
    ap.add_argument('cmd', choices=['list', 'pause', 'stop', 'resume', 'flatten'])
    ap.add_argument('system', nargs='?')
    ap.add_argument('--config', default=os.path.join(os.path.dirname(os.path.dirname(os.path.abspath(__file__))), 'config', 'config.yaml'))
    ap.add_argument('--system', dest='sys2')
    ap.add_argument('--intraday-only', action='store_true')
    a = ap.parse_args(argv)
    cfg = load(a.config)
    cd = cfg['control_dir']; os.makedirs(cd, exist_ok=True)
    sysname = a.system or a.sys2
    if a.cmd == 'list':
        led = Ledger(cfg['db_path'])
        files = {os.path.basename(f) for f in glob.glob(os.path.join(cd, '*'))}
        for s in build_strategies(cfg):
            st = 'STOPPED' if f'{s.sid}.stop' in files or 'ALL.stop' in files else ('PAUSED' if f'{s.sid}.pause' in files or 'ALL.pause' in files else 'on')
            n = led.query("SELECT COUNT(*) FROM positions WHERE system=? AND status='open'", (s.sid,))[0][0]
            print(f'{s.sid:<15} {s.prefix:<20} {st:<8} open={n}  {s.expert}: {s.title}')
        return
    if a.cmd in ('pause', 'stop'):
        assert sysname, 'system name (or ALL) required'
        open(os.path.join(cd, f'{sysname}.{a.cmd}'), 'w').write(str(dt.datetime.now()))
        print(f'{a.cmd} flag set for {sysname}')
        return
    if a.cmd == 'resume':
        for ext in ('stop', 'pause'):
            f = os.path.join(cd, f'{sysname}.{ext}')
            if os.path.exists(f):
                os.remove(f)
        print(f'resumed {sysname}')
        return
    # direct flatten
    logging.basicConfig(level=logging.INFO, format='%(asctime)s %(levelname)s %(message)s')
    from .broker_live import AlpacaPaperBroker
    from .engine import Engine
    from .marketdata import MarketData
    from .strategies.base import Context
    from .data import Data, ET
    led = Ledger(cfg['db_path'])
    strats = build_strategies(cfg)
    dry = bool(cfg.get('dry_run', True))
    br = AlpacaPaperBroker(led, {s.sid: s.prefix for s in strats}, dry_run=dry)
    md = MarketData()
    eng = Engine(cfg, strats, br, led, md)
    now = pd.Timestamp.now(tz=ET)
    eng.ctx = Context(now=now, day=now.date(), session_open=now.normalize() + pd.Timedelta(hours=9, minutes=30),
                      session_close=now.normalize() + pd.Timedelta(hours=16), daily={}, md=md, universes={})
    md.session_open = eng.ctx.session_open
    targets = [p for p in eng.positions if (not sysname or p.system == sysname) and (not a.intraday_only or p.plan_obj().eod_flat)]
    if not targets:
        print('nothing to flatten'); return
    try:
        md.lastpx.update(Data().latest_trades([p.symbol for p in targets]))
    except Exception:  # noqa: BLE001
        pass

    eng.flatten(now, 'ctl flatten' + (' (intraday safety net)' if a.intraday_only else ''), system=sysname,
                intraday_only=a.intraday_only)
    print('flattened:', [(p.system, p.symbol) for p in targets])


if __name__ == '__main__':

"""Market data access. Real-time = IEX (free plan); history = SIP (free when older than 15 minutes).
Scanner = Alpaca screener (movers) + snapshots; catalysts = Alpaca News; earnings dates = Nasdaq calendar."""
from __future__ import annotations
import datetime as dt, json, logging, time, urllib.request
from zoneinfo import ZoneInfo
import pandas as pd
import requests
from .safety import credentials

ET = ZoneInfo('America/New_York')
log = logging.getLogger('xp.data')
DATA = 'https://data.alpaca.markets'
UA = {'User-Agent': 'Mozilla/5.0 (X11; Linux x86_64) Chrome/124', 'Accept': 'application/json'}


def _ts(x) -> str:
    if isinstance(x, str) and len(x) == 10:
        return x
    t = pd.Timestamp(x)
    if t.tzinfo is None:
        t = t.tz_localize(ET)
    return t.tz_convert('UTC').strftime('%Y-%m-%dT%H:%M:%SZ')


class Data:
    def __init__(self, rate_per_min: int = 150):
        k, s = credentials()
        self.h = {'APCA-API-KEY-ID': k, 'APCA-API-SECRET-KEY': s}
        self.s = requests.Session()
        self.calls: list[float] = []
        self.rate = rate_per_min

    def _get(self, path, params=None, base=DATA):
        for i in range(5):
            now = time.time()
            self.calls = [t for t in self.calls if now - t < 60]
            if len(self.calls) >= self.rate:
                time.sleep(60 - (now - self.calls[0]) + 0.1)
            self.calls.append(time.time())
            try:
                r = self.s.get(base + path, params=params, headers=self.h, timeout=30)
                if r.status_code == 429:
                    time.sleep(3 * (i + 1)); continue
                if 400 <= r.status_code < 500:
                    raise ValueError(f'{r.status_code} {path}: {r.text[:200]}')
                r.raise_for_status()
                return r.json()
            except requests.RequestException as e:
                if i == 4:
                    raise
                log.warning('data retry %s: %s', path, type(e).__name__); time.sleep(1.5 * (i + 1))

    def bars(self, symbols, start, end, timeframe='1Min', feed='iex', adjustment='split') -> dict:
        """Multi-symbol bars (paged). Returns sym -> DataFrame indexed by ET timestamps."""
        out: dict[str, list] = {}
        syms = list(dict.fromkeys(symbols))
        for i in range(0, len(syms), 100):
            chunk = syms[i:i + 100]
            params = dict(symbols=','.join(chunk), timeframe=timeframe, start=_ts(start), end=_ts(end), feed=feed, adjustment=adjustment, limit=10000)
            tok = None
            while True:
                if tok:
                    params['page_token'] = tok
                j = self._get('/v2/stocks/bars', params)
                for s, rows in (j.get('bars') or {}).items():
                    out.setdefault(s, []).extend(rows)
                tok = j.get('next_page_token')
                if not tok:
                    break
        res = {}
        for s, rows in out.items():
            df = pd.DataFrame(rows)
            df.index = pd.DatetimeIndex(pd.to_datetime(df['t'], utc=True)).tz_convert(ET)
            df = df.rename(columns={'o': 'open', 'h': 'high', 'l': 'low', 'c': 'close', 'v': 'volume'})[['open', 'high', 'low', 'close', 'volume']]
            res[s] = df[~df.index.duplicated()].sort_index()
        return res

    def minute_rth(self, symbols, start, end, feed='iex') -> dict:
        res = self.bars(symbols, start, end, '1Min', feed)
        return {s: d.between_time('09:30', '15:59') for s, d in res.items()}

    def daily(self, symbols, start, end) -> dict:
        res = self.bars(symbols, pd.Timestamp(start).date().isoformat(), pd.Timestamp(end).date().isoformat(), '1Day', 'sip')
        for s, d in res.items():
            d.index = d.index.normalize()
        return res

    def latest_trades(self, symbols, feed='iex') -> dict:
        out = {}
        syms = list(dict.fromkeys(symbols))
        for i in range(0, len(syms), 200):
            j = self._get('/v2/stocks/trades/latest', dict(symbols=','.join(syms[i:i + 200]), feed=feed))
            for s, t in (j.get('trades') or {}).items():
                out[s] = float(t['p'])
        return out

    def snapshots(self, symbols, feed='iex') -> dict:
        out = {}
        syms = list(dict.fromkeys(symbols))
        for i in range(0, len(syms), 200):
            j = self._get('/v2/stocks/snapshots', dict(symbols=','.join(syms[i:i + 200]), feed=feed))
            out.update(j or {})
        return out

    def movers(self, top=50) -> list:
        j = self._get('/v1beta1/screener/stocks/movers', dict(top=top))
        return j.get('gainers') or []

    def news_symbols(self, symbols, start, end) -> set:
        hit = set()
        syms = list(dict.fromkeys(symbols))
        for i in range(0, len(syms), 50):
            j = self._get('/v1beta1/news', dict(symbols=','.join(syms[i:i + 50]), start=_ts(start), end=_ts(end), limit=50))
            for n in j.get('news') or []:
                hit |= set(n.get('symbols') or [])
        return hit & set(syms)


def nasdaq_earnings(day: dt.date, timeout=15) -> dict:
    """sym -> 'bmo'|'amc'|'' for companies reporting on `day` (Nasdaq public calendar JSON)."""
    url = f'https://api.nasdaq.com/api/calendar/earnings?date={day.isoformat()}'
    with urllib.request.urlopen(urllib.request.Request(url, headers=UA), timeout=timeout) as r:
        d = json.load(r)
    out = {}
    for row in ((d.get('data') or {}).get('rows') or []):
        t = (row.get('time') or '').lower()
        out[(row.get('symbol') or '').replace('/', '.')] = 'bmo' if 'pre' in t else ('amc' if 'after' in t else '')
    return out

===== ./xp/engine.py
"""Execution engine shared by the replay (SimBroker) and the live service (AlpacaPaperBroker).

Responsibilities: run each system's rules on its own timeframe, arm/expire signals, enter when price trades through
the trigger, size at RISK_PCT of total account equity, manage positions per the system's Plan (stops, trails,
partials, targets, time/day exits, close-based exits, earnings exits), flatten intraday systems, enforce the
per-system and account-wide daily loss guards, and write everything to the ledger."""
from __future__ import annotations
import logging, math
import datetime as dt
import pandas as pd
from .model import Plan, Position, Signal
from .strategies.base import Context, TICK

log = logging.getLogger('xp.engine')


class Engine:
    def __init__(self, cfg: dict, strategies: list, broker, ledger, md, notifier=None):
        self.cfg, self.broker, self.ledger, self.md, self.notifier = cfg, broker, ledger, md, notifier
        self.strats = {s.sid: s for s in strategies}
        self.pending: dict[tuple, Signal] = {}
        self.positions: list[Position] = ledger.open_positions()
        self.halted: dict[str, str] = {}           # sid -> reason (for today)
        self.account_halt: str | None = None
        self.ctx: Context | None = None
        self.equity0 = None                        # equity at the start of the day (loss-cap basis)
        self.pending_moc: list[dict] = []          # MOC entries/exits awaiting the closing auction
        self.disabled: set = set()                 # manual per-system stop (control files / config)
        r = cfg.get('risk', {})
        self.risk_pct = float(r.get('risk_pct', 0.0025))
        self.max_pos_pct = float(r.get('max_position_pct', 0.10))
        self.gross_cap = float(r.get('xp_gross_cap', 1.0))
        self.overnight_cap = float(r.get('xp_overnight_cap', 0.6))
        self.bp_reserve = float(r.get('bp_reserve_pct', 0.25))
        self.sys_loss = float(r.get('system_daily_loss_pct', 0.01))
        self.acct_loss = float(r.get('account_daily_loss_pct', 0.02))
        self.max_open = int(r.get('max_open_per_system', 5))
        self.max_entries = int(r.get('max_entries_per_system_day', 10))
        self.chase_r = float(r.get('max_chase_r', 0.5))
        self.max_adv = float(r.get('max_pct_adv', 0.01))
        self.flat_time = cfg.get('schedule', {}).get('intraday_flatten', '15:55')
        self.no_entry_after = cfg.get('schedule', {}).get('no_new_entries_after', '15:30')
        # xp-combo HARD exits (Marvin 2026-10-09): every trade gets a stop no looser than hard_stop_pct from entry and a
        # take-profit no farther than hard_tp_pct from entry (the book's own stop/target is kept when it is tighter).
        # Both are placed at the broker AND enforced in software on every price tick (enforce_hard), so the position is
        # exited at market even if the broker stop/OCO is missing, rejected or cancelled.
        h = cfg.get('hard_exit') or {}
        self.hard_on = bool(h.get('enabled', False))
        self.hard_stop_pct = float(h.get('stop_pct', 1.0))
        self.hard_tp_pct = float(h.get('tp_pct', 1.0))
        self.hard_retry = float(h.get('retry_s', 15.0))

    # ------------------------------------------------------------------ day lifecycle
    def start_day(self, ctx: Context):
        self.ctx = ctx
        self.md.session_open = ctx.session_open
        self.pending.clear(); self.halted.clear(); self.account_halt = None; self.pending_moc = []
        self.equity0 = self.broker.equity()
        day = str(ctx.day)
        for p in self.positions:
            if p.entry_day < day and p.meta.get('last_day') != day:
                p.days_held += 1
                p.meta['last_day'] = day
                d = ctx.daily.get(p.symbol)
                if d is not None and not d.empty:
                    p.mark_open = float(d.close.iloc[-1])
                plan = p.plan_obj()
                if plan.trail == 'prevday' and p.days_held >= plan.trail_start_day and d is not None and not d.empty:
                    lvl = float(d.low.iloc[-1]) - TICK if p.side == 'long' else float(d.high.iloc[-1]) + TICK
                    self._move_stop(p, lvl, 'prior-day trail')
                strat = self.strats.get(p.system)
                if strat:
                    strat.on_new_day(p, ctx)
                if p.meta.pop('stop_changed', False):
                    self.broker.sync_stop(p)
                self.ledger.save_position(p)
        for s in self.strats.values():
            if s.sid in self.disabled:
                continue
            try:
                s.prep(ctx)
            except Exception as e:  # noqa: BLE001
                log.exception('prep %s failed', s.sid); self.ledger.event('error', s.sid, f'prep failed: {e!r}')
        self.ledger.event('info', '', f'start_day {day} equity={self.equity0:.2f} open_positions={len(self.positions)} '
                          + ' '.join(f'{s.sid}:{len(s.watch)}' for s in self.strats.values() if s.watch))

    def active_symbols(self) -> set:
        out = set()
        for s in self.strats.values():
            if s.sid in self.disabled:
                continue
            out |= set(s.symbols(self.ctx))
        out |= {p.symbol for p in self.positions}
        return out

    # ------------------------------------------------------------------ bars -> signals
    def on_minute(self, now: pd.Timestamp):
        ctx = self.ctx
        ctx.now = now; ctx.cache.clear()
        mins = int((now - ctx.session_open).total_seconds() // 60)
        if mins <= 0:
            return
        for s in self.strats.values():
            if s.sid in self.disabled or s.sid in self.halted or self.account_halt:
                continue
            if mins % s.tf:
                continue
            if now > ctx.at(self.no_entry_after):
                continue
            for sym in s.symbols(ctx):
                bars = self.md.tf(sym, s.tf, now)
                if bars is None or bars.empty or bars.index[-1] < ctx.session_open:
                    continue
                try:
                    sigs = s.on_bar(sym, bars, ctx)
                except Exception as e:  # noqa: BLE001
                    log.exception('on_bar %s %s', s.sid, sym); self.ledger.event('error', s.sid, f'on_bar {sym}: {e!r}', now)
                    continue
                for sg in sigs or []:
                    self._arm(sg, now)
        self._manage_bars(now)
        self._expire(now)
        t = now.strftime('%H:%M')
        for p in list(self.positions):
            pl = p.plan_obj()
            if pl.time_exit and t >= pl.time_exit and p.entry_day == str(ctx.day):
                self._exit(p, p.qty, f'time exit {pl.time_exit}', now)

    def _arm(self, sg: Signal, now):
        if sg.kind == 'moc':
            return self._enter_moc(sg, now)
        key = (sg.system, sg.symbol, sg.side)
        if any(p.symbol == sg.symbol and p.system == sg.system for p in self.positions):
            return
        self.pending[key] = sg
        sg.meta['sid_row'] = self.ledger.signal(now, self.ctx.day, sg, 'armed', f'expires {sg.expires}')

    def _expire(self, now):
        for k, sg in list(self.pending.items()):
            if sg.expires is not None and now >= sg.expires:
                self.pending.pop(k)

    # ------------------------------------------------------------------ prices -> entries / exits
    def on_prices(self, now, quotes: dict):
        """quotes: sym -> dict(open, high, low, last). Sim passes the minute bar; live passes the latest trade
        (and the latest bar's high/low)."""
        ctx = self.ctx
        for sym, q in quotes.items():
            if q.get('last') is not None:
                self.md.lastpx[sym] = q['last']
        sim = self.broker.is_sim
        self.enforce_hard(now, quotes)
        # exits first (stops in sim; targets/partials both)
        for p in list(self.positions):
            q = quotes.get(p.symbol)
            if not q or p.pending_exit:
                continue
            self._check_position(p, q, now, sim)
        for key, sg in list(self.pending.items()):
            q = quotes.get(sg.symbol)
            if not q or key not in self.pending:
                continue
            hit = q['high'] >= sg.trigger if sg.side == 'long' else q['low'] <= sg.trigger
            if not hit:
                continue
            self.pending.pop(key, None)
            if sg.meta.get('oco'):
                for k2 in [k for k in self.pending if k[0] == sg.system and k[1] == sg.symbol]:
                    self.pending.pop(k2, None)
            ref = q.get('open') if sim else q.get('last')
            px = (max(sg.trigger, ref) if sg.side == 'long' else min(sg.trigger, ref)) if ref is not None else sg.trigger
            self._try_enter(sg, px, now)
        self.guard(now)

    def _check_position(self, p: Position, q: dict, now, sim: bool):
        pl = p.plan_obj()
        o, h, l = q.get('open', q.get('last')), q['high'], q['low']
        fav = h if p.side == 'long' else l
        adv = l if p.side == 'long' else h
        p.meta['mfe_r'] = max(p.meta.get('mfe_r', 0.0), p.r_of(fav))
        if sim:
            stop_hit = adv <= p.stop if p.side == 'long' else adv >= p.stop
            if pl.broker_stop and stop_hit:
                fill = min(p.stop, o) if p.side == 'long' else max(p.stop, o)
                return self._record_exit(p, p.qty, fill, p.stop, 'stop', now)
        # partials (price based)
        for k, pt in enumerate(pl.partials):
            if k in p.partials_done or 'at_day' in pt:
                continue
            if self._tp_at_broker(p, k):                 # resting OCO target at the broker handles it (poll records)
                continue
            lvl = self._partial_level(p, pt)
            if lvl is None:
                continue
            if (fav >= lvl) if p.side == 'long' else (fav <= lvl):
                qty = max(1, int(math.floor(p.qty0 * pt.get('frac', 0.5))))
                qty = min(qty, p.qty)
                p.partials_done.append(k)
                fillref = (max(lvl, o) if p.side == 'long' else min(lvl, o)) if sim else q.get('last', lvl)
                self._exit(p, qty, f'partial {pt}', now, ref_px=lvl, sim_px=fillref)
                if p.qty > 0 and pt.get('be'):
                    self._move_stop(p, p.entry_px, 'breakeven after partial')
                if p.qty <= 0:
                    return
        tgt = self._target(p, pl)
        if tgt is not None and self._tp_at_broker(p, 'target'):
            tgt = None
        if tgt is not None and ((fav >= tgt) if p.side == 'long' else (fav <= tgt)):
            fillref = (max(tgt, o) if p.side == 'long' else min(tgt, o)) if sim else q.get('last', tgt)
            return self._exit(p, p.qty, 'target', now, ref_px=tgt, sim_px=fillref)
        if pl.be_at_r is not None and p.meta['mfe_r'] >= pl.be_at_r:
            self._move_stop(p, p.entry_px, f'breakeven at {pl.be_at_r}R')
        if p.meta.pop('stop_changed', False):
            self._sync_stop(p, now)

    # ------------------------------------------------------------------ broker-resting fixed targets
    def _live_targets_ok(self, p) -> bool:
        return (not getattr(self.broker, 'is_sim', False) and not getattr(self.broker, 'dry_run', True)
                and hasattr(self.broker, 'place_targets') and p.plan_obj().broker_stop)

    def _tp_at_broker(self, p, k) -> bool:
        return self._live_targets_ok(p) and self.broker.tranche_live(p, k)

    def target_tranches(self, p) -> list:
        """Fixed-price exits from the expert's plan -> broker tranches [{'k', 'qty', 'px'}]: each price partial
        (at_r / at_px / at_pct, sized like the software partial) and a full target (target_px / target_r / target_abs)
        for the rest. Time / close / trailing exits stay in software."""
        pl = p.plan_obj(); out = []; alloc = 0
        for k, pt in enumerate(pl.partials):
            if k in p.partials_done or 'at_day' in pt:
                continue
            lvl = self._partial_level(p, pt)
            if lvl is None:
                continue
            q = min(max(1, int(math.floor(p.qty0 * pt.get('frac', 0.5)))), p.qty - alloc)
            if q <= 0:
                continue
            out.append(dict(k=k, qty=int(q), px=round(float(lvl), 2), id=None, leg=None)); alloc += q
        tgt = self._target(p, pl)
        if tgt is not None and p.qty - alloc > 0:
            out.append(dict(k='target', qty=int(p.qty - alloc), px=round(float(tgt), 2), id=None, leg=None))
        return out

    def _init_targets(self, p):
        if self._live_targets_ok(p) and 'tp' not in p.meta:
            p.meta['tp'] = self.target_tranches(p)

    def _record_target_fill(self, p, t, qty, px, now):
        pl = p.plan_obj()
        if t['k'] == 'target':
            reason = 'target (broker limit)'
        else:
            pt = pl.partials[t['k']] if t['k'] < len(pl.partials) else {}
            if t['k'] not in p.partials_done:
                p.partials_done.append(t['k'])
            reason = f'partial {pt} (broker limit)'
        self._record_exit(p, min(int(qty), p.qty), float(px), t['px'], reason, now)
        if p.qty > 0 and p in self.positions and t['k'] != 'target' and pl.partials[t['k']].get('be'):
            self._move_stop(p, p.entry_px, 'breakeven after partial')

    @staticmethod
    def _partial_level(p, pt):
        if 'at_r' in pt:
            return p.entry_px + p.sign * pt['at_r'] * p.r_ps
        if 'at_px' in pt:
            return pt['at_px']
        if 'at_pct' in pt:
            return p.entry_px * (1 + p.sign * pt['at_pct'] / 100)
        return None

    @staticmethod
    def _target(p, pl: Plan):
        if pl.target_px is not None:
            return pl.target_px
        if pl.target_r is not None:
            return p.entry_px + p.sign * pl.target_r * p.r_ps
        if pl.target_abs is not None:
            return p.entry_px + p.sign * pl.target_abs
        return None

    def _move_stop(self, p: Position, lvl: float, why: str):
        lvl = round(lvl, 2)
        better = lvl > p.stop if p.side == 'long' else lvl < p.stop
        if better:
            p.stop = lvl
            p.meta['stop_changed'] = True
            p.meta.setdefault('stop_log', []).append([why, lvl])

    def _manage_bars(self, now):
        ctx = self.ctx
        mins = int((now - ctx.session_open).total_seconds() // 60)
        for p in list(self.positions):
            pl = p.plan_obj()
            if pl.trail != 'bar':
                continue
            strat = self.strats.get(p.system)
            tf = pl.trail_tf or (strat.tf if strat else 1)
            if mins % tf:
                continue
            bars = self.md.tf(p.symbol, tf, now)
            if bars is None or bars.empty:
                continue
            b = bars.iloc[-1]
            if bars.index[-1] < pd.Timestamp(p.entry_ts) - pd.Timedelta(minutes=0):
                continue
            if (b.close > p.entry_px) if p.side == 'long' else (b.close < p.entry_px):
                p.favorable_closes += 1
            if p.favorable_closes < pl.trail_after_bars:
                continue
            if pl.trail_after_r is not None and p.meta.get('mfe_r', 0) < pl.trail_after_r:
                continue
            lvl = b.low - TICK if p.side == 'long' else b.high + TICK
            self._move_stop(p, lvl, f'bar trail {tf}m')
            if p.meta.pop('stop_changed', False):
                # a stop at/through the market is an immediate exit
                last = self.md.last(p.symbol)
                if last is not None and ((last <= p.stop) if p.side == 'long' else (last >= p.stop)):
                    self._exit(p, p.qty, 'bar trail (stop through market)', now, ref_px=p.stop)
                    continue
                self._sync_stop(p, now)

    # ------------------------------------------------------------------ entries
    def _blocked(self, sg: Signal, now) -> str | None:
        ctx = self.ctx
        if self.account_halt:
            return f'account halt: {self.account_halt}'
        if sg.system in self.halted:
            return f'system halt: {self.halted[sg.system]}'
        if sg.system in self.disabled:
            return 'system disabled'
        if now > ctx.at(self.no_entry_after) and sg.kind != 'moc':
            return 'after last-entry time'
        if any(p.symbol == sg.symbol for p in self.positions):
            return 'symbol held by another xp system'
        if sum(1 for p in self.positions if p.system == sg.system) >= self.max_open:
            return 'max open positions for system'
        if self.ledger.entries_today(ctx.day, sg.system) >= self.max_entries:
            return 'max entries today for system'
        if sg.symbol in self.broker.busy_symbols({p.symbol: p.qty * p.sign for p in self.positions}):
            return 'symbol busy with another bot'
        if sg.side == 'short' and not self.broker.can_short(sg.symbol):
            return 'not shortable / not easy-to-borrow'
        strat = self.strats.get(sg.system)
        if strat and strat.style == 'swing' and sg.symbol in ctx.earnings_soon and sg.plan.exit_before_earnings:
            return 'earnings before next session'
        return None

    def size(self, sg: Signal, px: float, stop: float) -> tuple[int, str]:
        eq = self.broker.equity()
        r_ps = abs(px - stop)
        if r_ps <= 0:
            return 0, 'zero risk distance'
        qty = int(math.floor(self.risk_pct * eq / r_ps))
        cap = int(math.floor(self.max_pos_pct * eq / px))
        why = 'risk'
        if cap < qty:
            qty, why = cap, 'notional cap'
        d = self.ctx.daily.get(sg.symbol)
        if d is not None and len(d) >= 20 and self.max_adv > 0:
            adv = float(d.volume.iloc[-20:].mean())
            if adv > 0 and qty > self.max_adv * adv:
                qty, why = int(self.max_adv * adv), 'ADV cap'
        gross = sum(abs(p.qty) * (self.md.last(p.symbol) or p.entry_px) for p in self.positions)
        room = self.gross_cap * eq - gross
        strat = self.strats.get(sg.system)
        if strat and strat.style == 'swing':
            on = sum(abs(p.qty) * (self.md.last(p.symbol) or p.entry_px) for p in self.positions
                     if not p.plan_obj().eod_flat)
            room = min(room, self.overnight_cap * eq - on)
        bp = self.broker.buying_power()
        room = min(room, bp * (1 - self.bp_reserve))
        if qty * px > room:
            qty, why = int(max(0, room) // px), 'buying power / gross cap'
        return max(qty, 0), why

    def _resolve_stop(self, sg: Signal) -> float:
        buf = sg.meta.get('buf', TICK)
        if sg.stop_rule == 'lod':
            lvl = self.md.day_low(sg.symbol) if sg.side == 'long' else self.md.day_high(sg.symbol)
            if lvl is not None:
                return round(lvl - buf if sg.side == 'long' else lvl + buf, 2)
        if sg.stop_rule == 'lod_prev':
            lvl = self.md.day_low(sg.symbol) if sg.side == 'long' else self.md.day_high(sg.symbol)
            prev = sg.meta.get('prev')
            if lvl is not None and prev is not None:
                return round(min(lvl, prev) - buf if sg.side == 'long' else max(lvl, prev) + buf, 2)
        return sg.stop

    def _try_enter(self, sg: Signal, px: float, now):
        ctx = self.ctx
        why = self._blocked(sg, now)
        stop = self._resolve_stop(sg)
        if not why:
            if (sg.side == 'long' and stop >= px) or (sg.side == 'short' and stop <= px):
                why = f'stop {stop} on wrong side of price {px}'
        r_ps = abs(sg.trigger - stop)
        if not why and r_ps > 0 and abs(px - sg.trigger) > self.chase_r * r_ps:
            why = f'missed: price {px:.2f} ran > {self.chase_r}R past trigger {sg.trigger}'
        if not why and sg.max_stop_dist is not None and abs(px - stop) > sg.max_stop_dist:
            why = f'stop distance {abs(px - stop):.2f} > max {sg.max_stop_dist:.2f}'
        qty = 0
        if not why and self.hard_on:
            stop = self._clamp_stop(sg.side, px, stop)          # size on the effective (hard-clamped) stop
        if not why:
            qty, szwhy = self.size(sg, px, stop)
            if qty < 1:
                why = f'size 0 ({szwhy})'
        if why:
            self.ledger.signal(now, ctx.day, sg, 'skipped', why)
            return None
        res = self.broker.enter(sg, qty, px, now)
        if not res or res.get('qty', 0) <= 0:
            self.ledger.signal(now, ctx.day, sg, 'rejected', str(res.get('error') if res else 'no fill'))
            return None
        fq, fpx = int(res['qty']), float(res['price'])
        if sg.stop_rule:                       # re-anchor LOD stops on the actual fill
            stop = self._resolve_stop(sg)
        if self.hard_on:
            stop = self._clamp_stop(sg.side, fpx, stop)
        r_ps = abs(fpx - stop) if sg.plan.broker_stop else abs(sg.trigger - sg.stop)
        p = Position(system=sg.system, symbol=sg.symbol, side=sg.side, qty=fq, qty0=fq, entry_px=fpx, signal_px=sg.trigger,
                     entry_ts=str(now), entry_day=str(ctx.day), stop=round(stop, 2), init_stop=round(stop, 2),
                     r_ps=max(r_ps, 0.01), plan=sg.plan.__dict__.copy(), reason=sg.reason, meta={'last_day': str(ctx.day)})
        self._harden(p)
        self.ledger.save_position(p)
        self.ledger.fill(now, ctx.day, p.system, p.symbol, 'buy' if p.side == 'long' else 'sell', fq, fpx, sg.trigger,
                         'entry', res.get('coid'), p.pid)
        self.ledger.signal(now, ctx.day, sg, 'filled', f'qty={fq} px={fpx:.2f} stop={stop:.2f}')
        self.positions.append(p)
        log.info('ENTRY %s %s %s %d @ %.2f stop %.2f (%s)', p.system, p.side, p.symbol, fq, fpx, p.stop, p.reason)
        if p.plan_obj().broker_stop:
            self._init_targets(p)
            self._sync_stop(p, now, initial=True)
        if p in self.positions:
            self.ledger.save_position(p)
        return p

    def _enter_moc(self, sg: Signal, now):
        ctx = self.ctx
        why = self._blocked(sg, now)
        qty = 0
        if not why:
            qty, szwhy = self.size(sg, sg.trigger, sg.stop)
            if qty < 1:
                why = f'size 0 ({szwhy})'
        if why:
            self.ledger.signal(now, ctx.day, sg, 'skipped', why)
            return
        res = self.broker.submit_moc(sg.system, sg.symbol, 'buy' if sg.side == 'long' else 'sell', qty, 'entry', now)
        if not res or res.get('error'):
            self.ledger.signal(now, ctx.day, sg, 'rejected', str(res.get('error') if res else 'no order'))
            return
        self.ledger.signal(now, ctx.day, sg, 'moc_submitted', f'qty={qty}')
        self.pending_moc.append(dict(kind='entry', sig=sg, qty=qty, coid=res.get('coid'), id=res.get('id')))
        # reserve the symbol so no other xp system takes it before the close
        self.positions.append(Position(system=sg.system, symbol=sg.symbol, side=sg.side, qty=0, qty0=0, entry_px=sg.trigger,
                                       signal_px=sg.trigger, entry_ts=str(now), entry_day=str(ctx.day), stop=sg.stop,
                                       init_stop=sg.stop, r_ps=abs(sg.trigger - sg.stop), plan=sg.plan.__dict__.copy(),
                                       reason=sg.reason, pending_exit='moc_entry'))

    # ------------------------------------------------------------------ exits
    def _exit(self, p: Position, qty: int, reason: str, now, ref_px=None, sim_px=None, moc=False, _again=False):
        qty = min(int(qty), p.qty)
        if qty <= 0:
            return
        if moc:
            res = self.broker.submit_moc(p.system, p.symbol, 'sell' if p.side == 'long' else 'buy', qty, 'exit', now, pos=p)
            if res and not res.get('error'):
                self.pending_moc.append(dict(kind='exit', pos=p, qty=qty, coid=res.get('coid'), id=res.get('id'), reason=reason,
                                             ref=self.md.last(p.symbol)))
                if qty >= p.qty:
                    p.pending_exit = 'moc'
                self.ledger.save_position(p)
            else:
                self.ledger.event('error', p.system, f'MOC exit {p.symbol} failed: {res}', now)
            return
        ref = ref_px if ref_px is not None else self.md.last(p.symbol)
        res = self.broker.exit(p, qty, reason, sim_px if sim_px is not None else ref, now)
        if res and (res.get('stop_fill') or res.get('tp_fills')):
            for k, tq, tpx in res.get('tp_fills') or []:
                t = next((x for x in (p.meta.get('tp') or []) if x['k'] == k), dict(k=k, px=tpx))
                if p in self.positions and p.qty > 0:
                    self._record_target_fill(p, t, int(tq), tpx, now)
            if res.get('stop_fill') and p in self.positions and p.qty > 0:
                sq, spx = res['stop_fill']
                self._record_exit(p, min(sq, p.qty), spx, p.stop, 'stop', now)
            if p in self.positions and p.qty > 0 and not res.get('stop_fill') and not _again:
                return self._exit(p, min(qty, p.qty), reason, now, ref_px=ref_px, sim_px=sim_px, _again=True)
            return
        if not res or res.get('qty', 0) <= 0:
            self.ledger.event('error', p.system, f'exit {p.symbol} {qty} failed: {res}', now)
            return
        self._record_exit(p, int(res['qty']), float(res['price']), ref, reason, now, coid=res.get('coid'), placed=True)

    def _record_exit(self, p: Position, qty: int, px: float, ref, reason: str, now, coid=None, placed=False):
        ctx = self.ctx
        final = qty >= p.qty
        pnl = self.ledger.exit(now, ctx.day, p, qty, px, ref, reason, final)
        self.ledger.fill(now, ctx.day, p.system, p.symbol, 'sell' if p.side == 'long' else 'buy', qty, px, ref,
                         'exit:' + reason, coid, p.pid)
        p.qty -= qty; p.realized += pnl
        if final or p.qty <= 0:
            p.qty = 0
            if p in self.positions:
                self.positions.remove(p)
            if not reason.startswith('stop'):
                self.broker.cancel_stop(p)
            self.ledger.save_position(p, status='closed', closed=str(now))
        else:
            self.ledger.save_position(p)
            if placed and p.plan_obj().broker_stop:
                self._sync_stop(p, now, initial=True)   # re-place the protective stop for the remaining shares
        log.info('EXIT %s %s %d @ %.2f pnl %.2f (%s)', p.system, p.symbol, qty, px, pnl, reason)

    # ------------------------------------------------------------------ protective stops (live)
    def _stop_through(self, p: Position) -> bool:
        last = self.md.last(p.symbol)
        return last is not None and ((last <= p.stop) if p.side == 'long' else (last >= p.stop))

    def _sync_stop(self, p: Position, now, initial=False):
        """Place (initial) or move the broker stop to p.stop. If the level is already past the market (our last price,
        or the broker rejects it as such), exit at market through the normal exit path (recorded in the ledger) instead
        of leaving the position unprotected. The broker never cancels the old stop before a replacement is accepted."""
        if p.qty <= 0 or p.pending_exit or not p.plan_obj().broker_stop:
            return None
        live = not getattr(self.broker, 'is_sim', False)
        if live and self._stop_through(p):
            self._exit(p, p.qty, 'stop through market', now, ref_px=p.stop)
            return 'exited'
        res = self.broker.place_stop(p) if initial else self.broker.sync_stop(p)
        if res == 'through_market':
            self._exit(p, p.qty, 'stop through market (broker rejected stop)', now, ref_px=p.stop)
            return 'exited'
        if p in self.positions:
            self.ledger.save_position(p)
        return res

    def ensure_stops(self, now, retry_s: float = 60.0):
        """Live safety net (called every few seconds in session): any broker-stop position without a working stop gets
        one, or is exited at market if its stop is already past the market. Failed placements retry after retry_s."""
        if getattr(self.broker, 'is_sim', False):
            return
        ts = pd.Timestamp(now).timestamp()
        for p in list(self.positions):
            if (p.qty > 0 and not p.pending_exit and self._live_targets_ok(p) and 'tp' not in p.meta
                    and (p.stop_order_id or p.meta.get('stop_qty') == 0)):
                p.meta['tp'] = self.target_tranches(p)              # carried position: add its book targets at the broker
                if p.meta['tp']:
                    res = self.broker.add_targets(p)
                    self.ledger.event('info', p.system, f"{p.symbol} broker targets added: "
                                      f"{[(t['k'], t['qty'], t['px'], bool(t.get('id'))) for t in p.meta['tp']]} ({res})", now)
                    if res == 'through_market':
                        self._exit(p, p.qty, 'stop through market', now, ref_px=p.stop)
                        continue
                self.ledger.save_position(p)
            if p.qty <= 0 or p.pending_exit or p.stop_order_id or not p.plan_obj().broker_stop:
                continue
            if getattr(self.broker, 'fully_covered', None) and self.broker.fully_covered(p):
                continue
            if p.meta.get('stop_retry_at', 0) > ts:
                continue
            p.meta['stop_retry_at'] = ts + retry_s
            self.ledger.event('warn', p.system, f'{p.symbol} has no working broker stop (stop {p.stop}); re-protecting', now)
            res = self._sync_stop(p, now, initial=True)
            if res == 'ok':
                p.meta.pop('stop_retry_at', None)
                if p in self.positions:
                    self.ledger.save_position(p)

    def sync_broker_stops(self, now):
        """Live: record protective stops the broker filled."""
        for p, qty, px in self.broker.poll_stops(list(self.positions)):
            if p in self.positions and p.qty > 0:
                self._record_exit(p, min(qty, p.qty), px, p.stop, 'stop', now)
        for p, t, qty, px in (self.broker.pop_target_fills() if hasattr(self.broker, 'pop_target_fills') else []):
            if p in self.positions and p.qty > 0:
                self._record_target_fill(p, t, qty, px, now)
        for p in list(self.positions):
            if p.meta.get('tp') and p.qty > 0 and not p.pending_exit and self._live_targets_ok(p) \
                    and (p.meta.get('stop_changed') or self.broker.needs_sync(p)):
                p.meta.pop('stop_changed', None)
                self._sync_stop(p, now)

    # ------------------------------------------------------------------ close routines
    def close_routine(self, now):
        """~15:44 ET: decisions that the experts make 'on the close' are sent as market-on-close orders."""
        ctx = self.ctx
        ctx.now = now
        if self.account_halt:
            return
        for p in list(self.positions):
            if p.pending_exit:
                continue
            pl = p.plan_obj()
            if pl.eod_flat:
                continue
            strat = self.strats.get(p.system)
            reason = None
            if pl.max_days and p.days_held >= pl.max_days:
                reason = f'time exit day {pl.max_days}'
            elif pl.exit_before_earnings and p.symbol in ctx.earnings_soon:
                reason = 'earnings before next session'
            if not reason and strat:
                d = ctx.daily.get(p.symbol)
                last = self.md.last_prices([p.symbol]).get(p.symbol)
                if d is not None and last is not None:
                    dfd = pd.concat([d, pd.DataFrame({'open': [last], 'high': [last], 'low': [last], 'close': [last],
                                                      'volume': [0.0]}, index=[pd.Timestamp(ctx.day, tz=d.index.tz)])])
                    try:
                        reason = strat.close_exit(p, dfd, ctx)
                    except Exception as e:  # noqa: BLE001
                        self.ledger.event('error', p.system, f'close_exit {p.symbol}: {e!r}', now)
            if reason:
                self._exit(p, p.qty, reason, now, moc=True)
                continue
            for k, pt in enumerate(pl.partials):
                if k in p.partials_done or 'at_day' not in pt or p.days_held < pt['at_day']:
                    continue
                p.partials_done.append(k)
                qty = min(p.qty, max(1, int(p.qty0 * pt.get('frac', 0.5))))
                self._exit(p, qty, f'partial day {pt["at_day"]}', now, moc=True)
                if pt.get('be') and p.qty - qty > 0:
                    p.meta['be_after_moc'] = True
        for s in self.strats.values():
            if s.sid in self.disabled or s.sid in self.halted:
                continue
            try:
                for sg in s.on_close(ctx) or []:
                    self._arm(sg, now)
            except Exception as e:  # noqa: BLE001
                log.exception('on_close %s', s.sid); self.ledger.event('error', s.sid, f'on_close: {e!r}', now)

    def finalize_moc(self, now):
        """After the close: record closing-auction fills."""
        ctx = self.ctx
        for m in self.pending_moc:
            res = self.broker.moc_fill(m, now)
            if m['kind'] == 'entry':
                sg = m['sig']
                ph = next((p for p in self.positions if p.pending_exit == 'moc_entry' and p.system == sg.system
                           and p.symbol == sg.symbol), None)
                if ph:
                    self.positions.remove(ph)
                if not res or res.get('qty', 0) <= 0:
                    self.ledger.signal(now, ctx.day, sg, 'moc_unfilled', str(res))
                    continue
                fq, fpx = int(res['qty']), float(res['price'])
                nominal = abs(sg.trigger - sg.stop)
                stop = fpx - nominal if sg.side == 'long' else fpx + nominal
                p = Position(system=sg.system, symbol=sg.symbol, side=sg.side, qty=fq, qty0=fq, entry_px=fpx, signal_px=sg.trigger,
                             entry_ts=str(now), entry_day=str(ctx.day), stop=round(stop, 2), init_stop=round(stop, 2),
                             r_ps=max(nominal, 0.01), plan=sg.plan.__dict__.copy(), reason=sg.reason,
                             meta={'last_day': str(ctx.day)}, mark_open=None)
                self._harden(p)
                self.ledger.save_position(p)
                self.ledger.fill(now, ctx.day, p.system, p.symbol, 'buy' if p.side == 'long' else 'sell', fq, fpx, sg.trigger,
                                 'entry', m.get('coid'), p.pid)
                self.ledger.signal(now, ctx.day, sg, 'filled', f'MOC qty={fq} px={fpx:.2f}')
                self.positions.append(p)
                if p.plan_obj().broker_stop:
                    self.broker.place_stop(p)
            else:
                p = m['pos']
                if not res or res.get('qty', 0) <= 0:
                    p.pending_exit = None
                    self.ledger.event('error', p.system, f'MOC exit {p.symbol} unfilled: {res}', now)
                    if p.plan_obj().broker_stop and p.qty > 0:
                        self.broker.place_stop(p)
                    self.ledger.save_position(p)
                    continue
                p.pending_exit = None
                self._record_exit(p, int(res['qty']), float(res['price']), m.get('ref'), m['reason'], now, coid=m.get('coid'))
                if p.qty > 0 and p.meta.pop('be_after_moc', False):
                    self._move_stop(p, p.entry_px, 'breakeven after partial')
                    p.meta.pop('stop_changed', None)
                    self.broker.sync_stop(p)
                    self.ledger.save_position(p)
        self.pending_moc = []

    def flatten(self, now, reason: str, system: str | None = None, intraday_only=False):
        for p in list(self.positions):
            if system and p.system != system:
                continue
            if p.pending_exit:
                continue
            if intraday_only and not p.plan_obj().eod_flat:
                continue
            self._exit(p, p.qty, reason, now)
        for k in [k for k in self.pending if system is None or k[0] == system]:
            self.pending.pop(k, None)

    def eod_marks(self, now):
        ctx = self.ctx
        eq = self.broker.equity()
        for sid in self.strats:
            unreal = sum(self._unreal_today(p) for p in self.positions if p.system == sid)
            self.ledger.daily_row(ctx.day, sid, self.ledger.realized_today(ctx.day, sid), unreal, eq)
        for p in self.positions:
            last = self.md.last(p.symbol)
            if last is not None:
                p.meta['eod_mark'] = last
            self.ledger.save_position(p)

    # ------------------------------------------------------------------ xp-combo hard 1% stop / take-profit
    def _hard_levels(self, side: str, px: float):
        sg = 1 if side == 'long' else -1
        hs = px * (1 - sg * self.hard_stop_pct / 100.0)
        ht = px * (1 + sg * self.hard_tp_pct / 100.0)
        # round to the cent on the TIGHT side (never looser than the percentage)
        hs = math.ceil(hs * 100 - 1e-9) / 100 if sg > 0 else math.floor(hs * 100 + 1e-9) / 100
        ht = math.floor(ht * 100 + 1e-9) / 100 if sg > 0 else math.ceil(ht * 100 - 1e-9) / 100
        return hs, ht

    def _clamp_stop(self, side: str, px: float, stop: float) -> float:
        hs, _ = self._hard_levels(side, px)
        return max(stop, hs) if side == 'long' else min(stop, hs)

    def _harden(self, p: Position):
        """Fix the hard stop / take-profit for a new position (entry = actual fill). Stop = tighter of the book stop and
        hard_stop_pct; target = nearer of the book target and hard_tp_pct (book partials beyond it are dropped)."""
        if not self.hard_on:
            return
        hs, ht = self._hard_levels(p.side, p.entry_px)
        if (p.stop < hs) if p.side == 'long' else (p.stop > hs):
            p.stop = p.init_stop = hs
            p.r_ps = max(abs(p.entry_px - p.stop), 0.01)
        p.meta['hard_stop'] = round(p.stop, 2)
        pl = p.plan_obj()
        bt = self._target(p, pl)
        if bt is not None and p.sign * (bt - p.entry_px) > 0 and p.sign * (bt - ht) <= 0:
            tgt, src = round(bt, 2), 'book'
        else:
            tgt, src = ht, f'hard {self.hard_tp_pct:g}%'
        p.plan['target_px'] = tgt; p.plan['target_r'] = None; p.plan['target_abs'] = None
        keep = []
        for pt in p.plan.get('partials') or []:
            lvl = self._partial_level(p, pt)
            if lvl is None or p.sign * (lvl - tgt) < 0:
                keep.append(pt)
        p.plan['partials'] = keep
        p.meta['hard_tp'] = tgt
        p.meta['hard_src'] = dict(stop='hard' if p.stop == hs else 'book', tp=src)
        risk = abs(p.entry_px - p.stop); rew = abs(tgt - p.entry_px)
        p.meta['hard_rr'] = round(rew / risk, 2) if risk > 0 else None
        self.ledger.event('info', p.system, f'{p.symbol} {p.side} entry {p.entry_px:.2f}: HARD stop {p.stop:.2f} '
                          f'({p.meta["hard_src"]["stop"]}), TP {tgt:.2f} ({src}), R:R {p.meta["hard_rr"]}')

    def enforce_hard(self, now, quotes: dict):
        """Software ('virtual') stop/TP: on every quote, exit at market (cancelling the broker stop/OCOs first) when the
        last price is through the stop (hard stop, or the trailed stop if tighter) or through the take-profit.
        Works whether or not the broker-side orders exist; failed exits are retried every retry_s."""
        if not self.hard_on:
            return
        ts = pd.Timestamp(now).timestamp()
        for p in list(self.positions):
            if p.qty <= 0 or p.pending_exit or 'hard_stop' not in p.meta or p not in self.positions:
                continue
            q = quotes.get(p.symbol)
            last = q.get('last') if q else None
            if last is None:
                continue
            long_ = p.side == 'long'
            vs = max(p.meta['hard_stop'], p.stop) if long_ else min(p.meta['hard_stop'], p.stop)
            vt = p.meta.get('hard_tp')
            hit_s = last <= vs if long_ else last >= vs
            hit_t = vt is not None and (last >= vt if long_ else last <= vt)
            if not (hit_s or hit_t) or p.meta.get('hard_retry_at', 0) > ts:
                continue
            n = p.meta['hard_tries'] = p.meta.get('hard_tries', 0) + 1
            p.meta['hard_retry_at'] = ts + (self.hard_retry if n < 5 else 60.0)
            lvl = vs if hit_s else vt
            reason = f'HARD virtual {"stop" if hit_s else "TP"} {lvl:.2f}'
            self.ledger.event('warn' if n < 5 else 'error', p.system,
                              f'{p.symbol} last {last} -> {reason} (try {n}): cancel broker orders, exit at market', now)
            self._exit(p, p.qty, reason, now, ref_px=lvl, sim_px=last)

    # ------------------------------------------------------------------ guards
    def _unreal_today(self, p: Position) -> float:
        last = self.md.last(p.symbol)
        if last is None or p.qty <= 0:
            return 0.0
        ref = p.entry_px if p.entry_day == str(self.ctx.day) or p.mark_open is None else p.mark_open
        return (last - ref) * p.sign * p.qty

    def day_pnl(self, sid: str) -> float:
        return self.ledger.realized_today(self.ctx.day, sid) + sum(self._unreal_today(p) for p in self.positions if p.system == sid)

    def guard(self, now):
        if self.account_halt or self.equity0 is None:
            return
        tot = 0.0
        for sid in self.strats:
            pnl = self.day_pnl(sid)
            tot += pnl
            if sid not in self.halted and pnl <= -self.sys_loss * self.equity0:
                self.halted[sid] = f'daily loss {pnl:.2f} <= -{self.sys_loss:.1%} of equity'
                self.ledger.event('warn', sid, 'HALT ' + self.halted[sid], now)
                self.flatten(now, 'system daily loss cap', system=sid)
        if tot <= -self.acct_loss * self.equity0:
            self.account_halt = f'combined xp day P&L {tot:.2f} <= -{self.acct_loss:.1%} of equity'
            self.ledger.event('warn', '', 'ACCOUNT HALT ' + self.account_halt, now)
            self.flatten(now, 'account-wide xp loss guard')

"""Indicator helpers (pandas). All functions take/return pandas objects aligned to the input index."""
from __future__ import annotations
import numpy as np
import pandas as pd

def sma(s: pd.Series, n: int) -> pd.Series:
    return s.rolling(n, min_periods=n).mean()

def ema(s: pd.Series, n: int) -> pd.Series:
    return s.ewm(span=n, adjust=False, min_periods=n).mean()

def true_range(df: pd.DataFrame) -> pd.Series:
    pc = df['close'].shift(1)
    return pd.concat([df['high'] - df['low'], (df['high'] - pc).abs(), (df['low'] - pc).abs()], axis=1).max(axis=1)

def atr(df: pd.DataFrame, n: int = 14) -> pd.Series:
    return true_range(df).ewm(alpha=1 / n, adjust=False, min_periods=n).mean()   # Wilder

def rsi(s: pd.Series, n: int = 2) -> pd.Series:
    d = s.diff()
    up = d.clip(lower=0).ewm(alpha=1 / n, adjust=False, min_periods=n).mean()
    dn = (-d.clip(upper=0)).ewm(alpha=1 / n, adjust=False, min_periods=n).mean()
    rs = up / dn.replace(0, np.nan)
    out = 100 - 100 / (1 + rs)
    return out.where(dn != 0, 100.0)

def adx(df: pd.DataFrame, n: int = 14) -> pd.Series:
    h, l = df['high'], df['low']
    up, dn = h.diff(), -l.diff()
    pdm = pd.Series(np.where((up > dn) & (up > 0), up, 0.0), index=df.index)
    ndm = pd.Series(np.where((dn > up) & (dn > 0), dn, 0.0), index=df.index)
    tr = true_range(df).ewm(alpha=1 / n, adjust=False, min_periods=n).mean()
    pdi = 100 * pdm.ewm(alpha=1 / n, adjust=False, min_periods=n).mean() / tr
    ndi = 100 * ndm.ewm(alpha=1 / n, adjust=False, min_periods=n).mean() / tr
    dx = 100 * (pdi - ndi).abs() / (pdi + ndi).replace(0, np.nan)
    return dx.ewm(alpha=1 / n, adjust=False, min_periods=n).mean()

def resample_session(df1m: pd.DataFrame, tf: int) -> pd.DataFrame:
    """Aggregate RTH 1-min bars into tf-minute bars anchored at 09:30 ET each session (index = bucket start)."""
    if df1m is None or df1m.empty or tf == 1:
        return df1m
    idx = df1m.index
    mins = (idx.hour * 60 + idx.minute) - 570
    bucket = (mins // tf) * tf + 570
    start = idx.normalize() + pd.to_timedelta(bucket, unit='m')
    g = df1m.groupby(start)
    out = pd.DataFrame({'open': g['open'].first(), 'high': g['high'].max(), 'low': g['low'].min(),
                        'close': g['close'].last(), 'volume': g['volume'].sum()})
    out.index.name = None
    return out

def completed(df_tf: pd.DataFrame, tf: int, now: pd.Timestamp) -> pd.DataFrame:
    """Drop the still-forming last bar (bucket start + tf > now)."""
    if df_tf is None or df_tf.empty:
        return df_tf
    keep = df_tf.index + pd.Timedelta(minutes=tf) <= now
    return df_tf[keep]

def vwap_anchored(df: pd.DataFrame, anchor_pos: int) -> pd.Series:
    """Anchored VWAP from row position anchor_pos (typical price x volume)."""
    sub = df.iloc[anchor_pos:]
    tp = (sub['high'] + sub['low'] + sub['close']) / 3
    v = sub['volume'].replace(0, np.nan).fillna(0)
    return (tp * v).cumsum() / v.cumsum().replace(0, np.nan)

def daily_from_1m(df1m: pd.DataFrame) -> dict | None:
    if df1m is None or df1m.empty:
        return None
    return dict(open=float(df1m['open'].iloc[0]), high=float(df1m['high'].max()), low=float(df1m['low'].min()),
                close=float(df1m['close'].iloc[-1]), volume=float(df1m['volume'].sum()))

"""SQLite running tab. One database, every row tagged with its system (=client_order_id prefix), so each system has
its own ledger slice: signals, orders, fills, positions, exits (P&L, R multiple, slippage vs signal), daily rows."""
from __future__ import annotations
import json, sqlite3, threading, datetime as dt
from .model import Position

SCHEMA = """
CREATE TABLE IF NOT EXISTS signals(id INTEGER PRIMARY KEY, ts TEXT, day TEXT, system TEXT, symbol TEXT, side TEXT,
  kind TEXT, trigger REAL, stop REAL, reason TEXT, status TEXT, note TEXT);
CREATE TABLE IF NOT EXISTS orders(id INTEGER PRIMARY KEY, ts TEXT, day TEXT, system TEXT, symbol TEXT, side TEXT,
  qty INTEGER, otype TEXT, tif TEXT, price REAL, purpose TEXT, client_order_id TEXT, broker_id TEXT, status TEXT,
  pid INTEGER, dry_run INTEGER);
CREATE TABLE IF NOT EXISTS fills(id INTEGER PRIMARY KEY, ts TEXT, day TEXT, system TEXT, symbol TEXT, side TEXT,
  qty INTEGER, price REAL, ref_price REAL, purpose TEXT, client_order_id TEXT, pid INTEGER);
CREATE TABLE IF NOT EXISTS positions(pid INTEGER PRIMARY KEY, system TEXT, symbol TEXT, side TEXT, status TEXT,
  opened TEXT, closed TEXT, state TEXT);
CREATE TABLE IF NOT EXISTS exits(id INTEGER PRIMARY KEY, ts TEXT, day TEXT, system TEXT, symbol TEXT, side TEXT,
  pid INTEGER, qty INTEGER, entry_px REAL, exit_px REAL, ref_px REAL, pnl REAL, r_mult REAL, entry_slip REAL,
  exit_slip REAL, reason TEXT, final INTEGER);
CREATE TABLE IF NOT EXISTS events(id INTEGER PRIMARY KEY, ts TEXT, level TEXT, system TEXT, msg TEXT);
CREATE TABLE IF NOT EXISTS daily(day TEXT, system TEXT, realized REAL, unrealized REAL, equity REAL,
  PRIMARY KEY(day, system));
CREATE TABLE IF NOT EXISTS kv(k TEXT PRIMARY KEY, v TEXT);
CREATE INDEX IF NOT EXISTS ix_exits_sys ON exits(system, day);
CREATE INDEX IF NOT EXISTS ix_pos_status ON positions(status);
"""

class Ledger:
    def __init__(self, path: str):
        self.path = path
        self.db = sqlite3.connect(path, check_same_thread=False, isolation_level=None)
        self.db.execute('PRAGMA journal_mode=WAL')
        self.db.executescript(SCHEMA)
        self.lock = threading.Lock()

    def _x(self, sql, args=()):
        with self.lock:
            return self.db.execute(sql, args)

    # --- writes
    def signal(self, ts, day, s, status, note=''):
        return self._x('INSERT INTO signals(ts,day,system,symbol,side,kind,trigger,stop,reason,status,note) VALUES(?,?,?,?,?,?,?,?,?,?,?)',
                       (str(ts), str(day), s.system, s.symbol, s.side, s.kind, s.trigger, s.stop, s.reason, status, note)).lastrowid

    def order(self, ts, day, system, symbol, side, qty, otype, tif, price, purpose, coid, broker_id, status, pid, dry):
        return self._x('INSERT INTO orders(ts,day,system,symbol,side,qty,otype,tif,price,purpose,client_order_id,broker_id,status,pid,dry_run) '
                       'VALUES(?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)', (str(ts), str(day), system, symbol, side, int(qty), otype, tif,
                       price, purpose, coid, broker_id, status, pid, int(bool(dry)))).lastrowid

    def order_status(self, coid, status):
        self._x('UPDATE orders SET status=? WHERE client_order_id=?', (status, coid))

    def fill(self, ts, day, system, symbol, side, qty, price, ref, purpose, coid, pid):
        self._x('INSERT INTO fills(ts,day,system,symbol,side,qty,price,ref_price,purpose,client_order_id,pid) VALUES(?,?,?,?,?,?,?,?,?,?,?)',
                (str(ts), str(day), system, symbol, side, int(qty), price, ref, purpose, coid, pid))

    def save_position(self, p: Position, status='open', closed=None):
        st = json.dumps(p.to_dict(), default=str)
        if p.pid is None:
            p.pid = self._x('INSERT INTO positions(system,symbol,side,status,opened,closed,state) VALUES(?,?,?,?,?,?,?)',
                            (p.system, p.symbol, p.side, status, p.entry_ts, closed, st)).lastrowid
            self._x('UPDATE positions SET state=? WHERE pid=?', (json.dumps(p.to_dict(), default=str), p.pid))
        else:
            self._x('UPDATE positions SET status=?, closed=?, state=? WHERE pid=?', (status, closed, st, p.pid))
        return p.pid

    def exit(self, ts, day, p: Position, qty, exit_px, ref_px, reason, final):
        pnl = (exit_px - p.entry_px) * p.sign * qty
        r = pnl / (p.r_ps * p.qty0) if p.r_ps > 0 and p.qty0 > 0 else 0.0     # R of this slice relative to full initial risk
        es = (p.entry_px - p.signal_px) * p.sign                                 # >0 = paid up vs signal
        xs = ((ref_px - exit_px) * p.sign) if ref_px else 0.0                    # >0 = exit worse than reference
        self._x('INSERT INTO exits(ts,day,system,symbol,side,pid,qty,entry_px,exit_px,ref_px,pnl,r_mult,entry_slip,exit_slip,reason,final) '
                'VALUES(?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)', (str(ts), str(day), p.system, p.symbol, p.side, p.pid, int(qty),
                p.entry_px, exit_px, ref_px, pnl, r, es, xs, reason, int(final)))
        return pnl

    def event(self, level, system, msg, ts=None):
        self._x('INSERT INTO events(ts,level,system,msg) VALUES(?,?,?,?)', (str(ts or dt.datetime.now()), level, system, msg))

    def daily_row(self, day, system, realized, unrealized, equity):
        self._x('INSERT OR REPLACE INTO daily(day,system,realized,unrealized,equity) VALUES(?,?,?,?,?)',
                (str(day), system, realized, unrealized, equity))

    def kv_set(self, k, v):
        self._x('INSERT OR REPLACE INTO kv(k,v) VALUES(?,?)', (k, json.dumps(v, default=str)))

    def kv_get(self, k, default=None):
        r = self._x('SELECT v FROM kv WHERE k=?', (k,)).fetchone()
        return json.loads(r[0]) if r else default

    # --- reads
    def open_positions(self) -> list[Position]:
        out = []
        for (st,) in self._x("SELECT state FROM positions WHERE status='open' ORDER BY pid").fetchall():
            d = json.loads(st)
            out.append(Position(**d))
        return out

    def realized_today(self, day, system=None) -> float:
        q = 'SELECT COALESCE(SUM(pnl),0) FROM exits WHERE day=?' + (' AND system=?' if system else '')
        return float(self._x(q, (str(day), system) if system else (str(day),)).fetchone()[0])

    def entries_today(self, day, system) -> int:
        return int(self._x("SELECT COUNT(*) FROM fills WHERE day=? AND system=? AND purpose='entry'", (str(day), system)).fetchone()[0])

    def query(self, sql, args=()):
        return self._x(sql, args).fetchall()

        os.makedirs(cfg['control_dir'], exist_ok=True)
        self.ledger = Ledger(cfg['db_path'])
        self.U = universes(cfg)
        self.data = Data()
        self.strats = build_strategies(cfg)
        self.broker = AlpacaPaperBroker(self.ledger, {s.sid: s.prefix for s in self.strats}, dry_run=dry)
        self.md = MarketData(last_provider=lambda syms: self.data.latest_trades(syms, feed=cfg.get('feed', 'iex')))
        self.engine = Engine(cfg, self.strats, self.broker, self.ledger, self.md)
        self.notifier = Notifier(cfg)
        self.day = None; self.flags = set(); self.daily = {}; self.last_min = None; self.last_quote = 0.0
        self.session = None
        log.info(os.environ.get('XP_NAME', 'xp-test') + ' start: %d systems, dry_run=%s, telegram=%s', len(self.strats), dry, self.notifier.enabled)

    # --------------------------------------------------------------- calendar
    def session_today(self, now):
        from alpaca.trading.requests import GetCalendarRequest
        cal = self.broker.tc.get_calendar(GetCalendarRequest(start=now.date() - dt.timedelta(days=7), end=now.date() + dt.timedelta(days=7)))
        days = [c.date for c in cal]
        today = next((c for c in cal if c.date == now.date()), None)
        prev = max(d for d in days if d < now.date())
        nxt = min(d for d in days if d > now.date())
        if today is None:
            return None, prev, nxt
        def ts(x):
            if isinstance(x, dt.datetime):
                return pd.Timestamp(x).tz_localize(ET) if x.tzinfo is None else pd.Timestamp(x).tz_convert(ET)
            return pd.Timestamp(dt.datetime.combine(today.date, x), tz=ET)
        op, cl = ts(today.open), ts(today.close)
        return (op, cl), prev, nxt

    def t(self, key, default=None):
        return self.cfg['schedule'].get(key, default)

    # --------------------------------------------------------------- daily prep
    def prep(self, now):
        so, sc = self.session
        U = self.U
        syms = list(dict.fromkeys(U['core'] + U['index'] + U['sp1500']))
        t0 = time.time()
        self.daily = self.data.daily(syms, now.date() - dt.timedelta(days=560), self.prev_day)
        self.daily = {s: d[d.index.date < now.date()] for s, d in self.daily.items()}
        log.info('daily bars %d symbols in %.0fs', len(self.daily), time.time() - t0)
        soon, recent, ok = earnings_sets(now.date(), self.prev_day, self.next_day, nasdaq_earnings)
        if not ok:
            self.ledger.event('warn', '', 'earnings calendar fetch incomplete')
        ctx = Context(now=now, day=now.date(), session_open=so, session_close=sc, daily=self.daily, md=self.md,
                      universes=U, gappers={}, earnings_soon=soon, earnings_recent=recent)
        self.engine.start_day(ctx)
        self.warmup(self.engine.active_symbols())
        log.info('prep done: active symbols %d, watchlists %s', len(self.engine.active_symbols()),
                 {s.sid: sorted(s.watch)[:20] for s in self.strats if s.watch})

    def warmup(self, syms):
        syms = [s for s in syms if s not in self.md.bars]
        if not syms:
            return
        so = self.session[0]
        start = so - pd.Timedelta(days=7)
        end = min(pd.Timestamp.now(tz=ET) - pd.Timedelta(minutes=16), so)
        mb = self.data.minute_rth(syms, start, end, feed='sip')
        for s in syms:
            d = mb.get(s)
            if d is None:
                d = pd.DataFrame(columns=['open', 'high', 'low', 'close', 'volume'])
            days = sorted(set(d.index.date))[-3:]
            self.md.set_bars(s, d[[x in days for x in d.index.date]] if len(d) else d)
        log.info('warm-up minute bars for %d symbols', len(syms))

    # --------------------------------------------------------------- gappers
    def scan_gappers(self, now):
        """Gap = today's price (IEX open after 09:30, latest trade pre-market; must be stamped today) vs the SIP prior
        close. Candidates = Alpaca movers (whole market) + S&P 1500 + core."""
        cfg = self.cfg['scanner']
        movers = []
        try:
            movers = [g.get('symbol', '') for g in self.data.movers(cfg.get('movers_top', 50)) if clean_symbol(g.get('symbol', ''))]
        except Exception as e:  # noqa: BLE001
            log.warning('movers failed: %s', e)
        need = [s for s in movers if s not in self.daily]
        if need:
            try:
                extra = self.data.daily(need, now.date() - dt.timedelta(days=150), self.prev_day)
                for s, d in extra.items():
                    self.daily[s] = d[d.index.date < now.date()]
            except Exception as e:  # noqa: BLE001
                log.warning('gapper daily fetch failed: %s', e)
        cands = list(dict.fromkeys(movers + self.U['sp1500'] + self.U['core']))
        gp = {}
        try:
            snaps = self.data.snapshots(cands, feed=self.cfg.get('feed', 'iex'))
        except Exception as e:  # noqa: BLE001
            log.warning('snapshot gap scan failed: %s', e); snaps = {}
        after_open = now >= self.session[0]
        for s, sn in snaps.items():
            d = self.daily.get(s)
            if d is None or d.empty or not sn:
                continue
            pc = float(d.close.iloc[-1])
            db, lt = sn.get('dailyBar') or {}, sn.get('latestTrade') or {}
            px = None
            if after_open and db.get('t') and pd.Timestamp(db['t']).tz_convert(ET).date() == now.date():
                px = float(db['o'])
            elif lt.get('t') and pd.Timestamp(lt['t']).tz_convert(ET).date() == now.date():
                px = float(lt['p'])
            if px and pc > 0:
                gp[s] = dict(gap_pct=round((px / pc - 1) * 100, 2), prev_close=pc, price=px, open=px,
                             src='movers' if s in movers else 'snapshot')
        keep = {}
        for s, g in gp.items():
            d = self.daily.get(s)
            if g['gap_pct'] < cfg['gap_min_pct'] or g['price'] < 1.0 or d is None or len(d) < 20:
                continue
            if (d.volume.iloc[-20:] * d.close.iloc[-20:]).mean() < 2e6:
                continue
            keep[s] = g
        top = sorted(keep, key=lambda s: -keep[s]['gap_pct'])[:cfg.get('max_gappers', 40)]
        keep = {s: keep[s] for s in top}
        try:
            prev_close_ts = pd.Timestamp(dt.datetime.combine(self.prev_day, dt.time(16, 0)), tz=ET)
            news = self.data.news_symbols(list(keep), prev_close_ts, now) if keep else set()
        except Exception as e:  # noqa: BLE001
            log.warning('news failed: %s', e); news = set()
        for s in keep:
            keep[s]['news'] = s in news
        self.engine.ctx.gappers = keep
        self.engine.ctx.daily = self.daily
        log.info('gappers (%s): %s', hm(now), {s: (g['gap_pct'], g['news']) for s, g in list(keep.items())[:25]})
        self.warmup(set(keep))

    # --------------------------------------------------------------- control files
    def controls(self, now):
        cd = self.cfg['control_dir']
        for f in glob.glob(os.path.join(cd, '*.stop')) + glob.glob(os.path.join(cd, '*.pause')):
            sid, kind = os.path.basename(f).rsplit('.', 1)
            if sid == 'ALL':
                targets = list(self.engine.strats)
            else:
                targets = [sid] if sid in self.engine.strats else []
            for t in targets:
                if t not in self.engine.disabled:
                    self.engine.disabled.add(t)
                    self.ledger.event('warn', t, f'control file {kind}: no new entries', now)
                if kind == 'stop' and (t, 'flattened', str(now.date())) not in self.flags:
                    self.flags.add((t, 'flattened', str(now.date())))
                    self.engine.flatten(now, 'manual stop (control file)', system=t)
        active = {os.path.basename(f).rsplit('.', 1)[0] for f in glob.glob(os.path.join(cd, '*.stop')) + glob.glob(os.path.join(cd, '*.pause'))}
        for t in list(self.engine.disabled):
            if t not in active and 'ALL' not in active:
                self.engine.disabled.discard(t)
                self.ledger.event('info', t, 'control file removed: entries re-enabled', now)

    # --------------------------------------------------------------- intraday
    def minute(self, now):
        m = now.floor('min')
        syms = sorted(self.engine.active_symbols())
        if not syms:
            return
        start = m - pd.Timedelta(minutes=4)
        try:
            mb = self.data.minute_rth(syms, start, m, feed=self.cfg.get('feed', 'iex'))
        except Exception as e:  # noqa: BLE001
            log.warning('minute bars failed: %s', e); return
        quotes = {}
        for s, d in mb.items():
            d = d[d.index + pd.Timedelta(minutes=1) <= m]
            if d.empty:
                continue
            old = self.md.bars.get(s)
            new = d if old is None or old.empty else d[d.index > old.index[-1] - pd.Timedelta(minutes=1)]
            self.md.add_bars(s, d)
            if not new.empty:
                quotes[s] = dict(open=float(new.open.iloc[0]), high=float(new.high.max()), low=float(new.low.min()),
                                 last=float(new.close.iloc[-1]))
        live = self.live_quotes()
        for s, px in live.items():
            q = quotes.get(s, dict(open=px, high=px, low=px))
            q['last'] = px; q['high'] = max(q['high'], px); q['low'] = min(q['low'], px)
            quotes[s] = q
        self.engine.on_prices(now, quotes)
        self.engine.on_minute(m)

    def live_quotes(self) -> dict:
        syms = {k[1] for k in self.engine.pending} | {p.symbol for p in self.engine.positions if p.qty > 0}
        if not syms:
            return {}
        try:
            return self.data.latest_trades(sorted(syms), feed=self.cfg.get('feed', 'iex'))
        except Exception as e:  # noqa: BLE001
            log.warning('latest trades failed: %s', e); return {}

    def quick(self, now):
        q = self.live_quotes()
        if q:
            self.engine.on_prices(now, {s: dict(open=px, high=px, low=px, last=px) for s, px in q.items()})
        self.engine.sync_broker_stops(now)
        self.engine.ensure_stops(now)

    # --------------------------------------------------------------- after the close
    def after_close(self, now):
        self.engine.finalize_moc(now)
        last = self.data.latest_trades(sorted({p.symbol for p in self.engine.positions}), feed='iex') if self.engine.positions else {}
        self.md.lastpx.update(last)
        self.engine.eod_marks(now)

    def scorecard(self, now):
        eq = self.broker.equity()
        weekly = self.next_day.isocalendar()[1] != now.date().isocalendar()[1]
        monday = now.date() - dt.timedelta(days=now.weekday())
        rows = scorecard_rows(self.ledger, [s.sid for s in self.strats], now.date(), eq, week_from=monday)
        paths = export(self.ledger, rows, now.date(), self.cfg['reports_dir'], eq, weekly=weekly)
        log.info('reports: %s', paths)
        for msg in telegram_text(rows, now.date(), eq):
            self.notifier.send(msg)
        if weekly:
            for msg in telegram_text(rows, now.date(), eq, weekly=True):
                self.notifier.send(msg)
        halts = self.ledger.query("SELECT system, msg FROM events WHERE level='warn' AND ts LIKE ? AND msg LIKE '%HALT%'",
                                  (f'{now.date()}%',))
        if halts:
            self.notifier.send('halts today: ' + '; '.join(f'{a or "ALL"}: {b}' for a, b in halts)[:3000])

    # --------------------------------------------------------------- main loop
    def run(self):
        while True:
            try:
                self.tick()
            except KeyboardInterrupt:
                raise
            except Exception as e:  # noqa: BLE001
                log.exception('tick failed'); self.ledger.event('error', '', f'tick: {e!r}')
                time.sleep(10)
            time.sleep(2)

    def tick(self):
        now = pd.Timestamp.now(tz=ET)
        if self.day != now.date():
            self.day = now.date(); self.flags = set(); self.session = None
            sess, self.prev_day, self.next_day = self.session_today(now)
            self.session = sess
            if sess is None:
                log.info('%s is not a trading day', now.date())
        if self.session is None:
            time.sleep(60); return
        so, sc = self.session
        early = sc.hour < 16
        close_routine = (sc - pd.Timedelta(minutes=16)) if early else so.normalize() + pd.Timedelta(self.t('close_routine') + ':00')
        flat_t = (sc - pd.Timedelta(minutes=5)) if early else so.normalize() + pd.Timedelta(self.t('intraday_flatten') + ':00')
        fin_t = sc + pd.Timedelta(minutes=5)
        card_t = sc + pd.Timedelta(minutes=15)
        prep_t = so.normalize() + pd.Timedelta(self.t('prep') + ':00')
        if now < prep_t or now > card_t + pd.Timedelta(minutes=30):
            if 'prep' not in self.flags and now > card_t + pd.Timedelta(minutes=30):
                self.flags.add('prep')  # started after the day's work; nothing to do
            time.sleep(30); return
        if 'prep' not in self.flags:
            self.flags.add('prep'); self.prep(now)
            if now > so:
                self.scan_gappers(now); self.flags |= {f'scan{x}' for x in self.t('gapper_scan')}
        self.controls(now)
        for x in self.t('gapper_scan'):
            if f'scan{x}' not in self.flags and hm(now) >= x and (x < '09:30' or now >= so + pd.Timedelta(seconds=62)):
                self.flags.add(f'scan{x}'); self.scan_gappers(now)
        if so <= now <= sc + pd.Timedelta(seconds=30):
            m = now.floor('min')
            if self.last_min != m and now.second >= 3 and m > so:
                self.last_min = m; self.minute(now)
            elif time.time() - self.last_quote > 10:
                self.last_quote = time.time(); self.quick(now)
            if now >= close_routine and 'close' not in self.flags:
                self.flags.add('close'); self.engine.close_routine(now)
            if now >= flat_t and 'flat' not in self.flags:
                self.flags.add('flat'); self.engine.flatten(now, f'intraday flatten {hm(flat_t)}', intraday_only=True)
        if now >= fin_t and 'fin' not in self.flags:
            self.flags.add('fin'); self.after_close(now)
        if now >= card_t and 'card' not in self.flags:
            self.flags.add('card'); self.scorecard(now)

"""Intraday bar store shared by all systems: 1-min RTH bars (warm-up sessions + today) per symbol."""
from __future__ import annotations
import pandas as pd
from .indicators import resample_session, completed

COLS = ['open', 'high', 'low', 'close', 'volume']

class MarketData:
    def __init__(self, last_provider=None):
        self.bars: dict[str, pd.DataFrame] = {}
        self.lastpx: dict[str, float] = {}
        self.last_provider = last_provider          # callable(list[str]) -> {sym: px} for symbols without bars
        self._tf_cache: dict = {}
        self.session_open: pd.Timestamp | None = None

    def set_bars(self, sym: str, df: pd.DataFrame):
        df = df[COLS].astype(float) if not df.empty else pd.DataFrame(columns=COLS, dtype=float)
        self.bars[sym] = df.sort_index()
        if not df.empty:
            self.lastpx[sym] = float(df['close'].iloc[-1])

    def add_bars(self, sym: str, df: pd.DataFrame):
        if df is None or df.empty:
            return
        old = self.bars.get(sym)
        new = df[COLS].astype(float)
        if old is not None and not old.empty:
            new = pd.concat([old[~old.index.isin(new.index)], new]).sort_index()
        self.bars[sym] = new
        self.lastpx[sym] = float(new['close'].iloc[-1])

    def tf(self, sym: str, tf: int, now: pd.Timestamp) -> pd.DataFrame | None:
        df = self.bars.get(sym)
        if df is None or df.empty:
            return None
        if df.index[-1] + pd.Timedelta(minutes=1) > now:      # drop a still-forming minute
            df = df[df.index + pd.Timedelta(minutes=1) <= now]
            if df.empty:
                return None
        key = (sym, tf, df.index[-1], len(df))
        out = self._tf_cache.get(key)
        if out is None:
            out = completed(resample_session(df, tf), tf, now)
            if len(self._tf_cache) > 5000:
                self._tf_cache.clear()
            self._tf_cache[key] = out
        return out

    def today(self, sym: str) -> pd.DataFrame | None:
        df = self.bars.get(sym)
        if df is None or df.empty or self.session_open is None:
            return None
        return df[df.index >= self.session_open]

    def day_low(self, sym: str) -> float | None:
        t = self.today(sym)
        lo = float(t['low'].min()) if t is not None and not t.empty else None
        px = self.lastpx.get(sym)
        return min(x for x in (lo, px) if x is not None) if (lo is not None or px is not None) else None

    def day_high(self, sym: str) -> float | None:
        t = self.today(sym)
        hi = float(t['high'].max()) if t is not None and not t.empty else None
        px = self.lastpx.get(sym)
        return max(x for x in (hi, px) if x is not None) if (hi is not None or px is not None) else None

    def last(self, sym: str) -> float | None:
        return self.lastpx.get(sym)

    def last_prices(self, syms) -> dict:
        out = {s: self.lastpx[s] for s in syms if s in self.lastpx}
        missing = [s for s in syms if s not in out]
        if missing and self.last_provider:
            try:
                out.update(self.last_provider(missing))
            except Exception:  # noqa: BLE001
                pass
        return out

"""Plain data objects shared by the strategies, the engine, the sim broker and the live broker."""
from __future__ import annotations
from dataclasses import dataclass, field, asdict
from typing import Optional
import datetime as dt

@dataclass
class Plan:
    """How a position is managed after entry, as the expert prescribes. Every field is optional."""
    eod_flat: bool = True              # intraday system: flatten by the configured time (15:55 ET)
    trail: Optional[str] = None        # 'bar' (bar-by-bar on the strategy timeframe) | 'prevday' (prior day's low/high)
    trail_after_bars: int = 1          # 'bar' trail starts after this many favourable closes
    trail_tf: Optional[int] = None     # timeframe (min) for the 'bar' trail; default = the system's timeframe
    trail_after_r: Optional[float] = None   # trail only after price has reached this R multiple
    trail_start_day: int = 2           # 'prevday' trail starts on this holding day (entry day = 1)
    target_px: Optional[float] = None  # full exit at this price
    target_r: Optional[float] = None   # full exit at this R multiple
    target_abs: Optional[float] = None # full exit this many dollars beyond entry (Velez Guerrilla '$2 profit')
    partials: list = field(default_factory=list)   # [{'at_r':2,'frac':.5,'be':True}|{'at_px':..}|{'at_day':3,...}]
    be_at_r: Optional[float] = None    # move the stop to breakeven once price reaches this R
    time_exit: Optional[str] = None    # 'HH:MM' ET intraday time stop
    max_days: Optional[int] = None     # exit at the close of holding day N (entry day = 1)
    exit_before_earnings: bool = True  # swing: exit at the close before an earnings report
    broker_stop: bool = True           # False = the expert uses no stop (e.g. Connors RSI(2)); R uses a nominal risk

@dataclass
class Signal:
    system: str
    symbol: str
    side: str                          # 'long' | 'short'
    kind: str                          # 'stop' (enter when price trades through trigger) | 'market' | 'moc'
    trigger: float                     # signal price (stop level, or the reference price for market/moc)
    stop: float                        # protective stop (or nominal risk level when plan.broker_stop is False)
    plan: Plan
    reason: str = ''
    expires: Optional[dt.datetime] = None
    stop_rule: Optional[str] = None    # 'lod' = stop recomputed at fill as today's low/high -/+ 0.01
    max_stop_dist: Optional[float] = None   # skip if |entry-stop| exceeds this (e.g. Kullamaggie: stop <= ATR)
    meta: dict = field(default_factory=dict)
    created: Optional[dt.datetime] = None

    def risk_ps(self, entry: float | None = None) -> float:
        return abs((entry if entry is not None else self.trigger) - self.stop)

@dataclass
class Position:
    system: str
    symbol: str
    side: str
    qty: int
    qty0: int
    entry_px: float
    signal_px: float
    entry_ts: str
    entry_day: str
    stop: float
    init_stop: float
    r_ps: float
    plan: dict
    reason: str = ''
    pid: Optional[int] = None
    stop_order_id: Optional[str] = None
    days_held: int = 1
    partials_done: list = field(default_factory=list)
    favorable_closes: int = 0
    realized: float = 0.0
    mark_open: Optional[float] = None   # mark at the start of today (prior close) for day P&L of carried positions
    meta: dict = field(default_factory=dict)
    pending_exit: Optional[str] = None  # e.g. a submitted MOC exit awaiting its fill

    @property
    def sign(self) -> int:
        return 1 if self.side == 'long' else -1

    def plan_obj(self) -> Plan:
        return Plan(**self.plan)

    def r_of(self, px: float) -> float:
        return (px - self.entry_px) * self.sign / self.r_ps if self.r_ps > 0 else 0.0

    def to_dict(self) -> dict:
        return asdict(self)

"""Telegram send-only (sendMessage). Never polls for updates (master-trade-bot owns updates for this token).
Disabled unless config telegram.enabled AND env TELEGRAM_ENABLED=true AND not under test/replay. Never logs the token."""
import os, logging, requests
log = logging.getLogger('xp.notify')

class Notifier:
    def __init__(self, cfg: dict):
        t = cfg.get('telegram', {})
        self.token = os.environ.get('TELEGRAM_BOT_TOKEN')
        self.chat = str(t.get('chat_id') or os.environ.get('TELEGRAM_CHAT_ID') or '')
        self.tag = t.get('tag', 'xp PAPER')
        env_on = os.environ.get('TELEGRAM_ENABLED', 'false').lower() == 'true'
        testing = os.environ.get('XP_TESTING') == '1' or 'PYTEST_CURRENT_TEST' in os.environ
        self.enabled = bool(t.get('enabled') and env_on and self.token and self.chat and not testing)
        self.sent = []

    def send(self, text: str) -> bool:
        msg = f'[{self.tag}] {text}'
        self.sent.append(msg)
        if not self.enabled:
            log.info('telegram (disabled): %s', msg[:300]); return False
        try:
            r = requests.post(f'https://api.telegram.org/bot{self.token}/sendMessage', json={'chat_id': self.chat, 'text': msg},
                              timeout=15)
            if not r.ok:
                log.warning('telegram send failed: HTTP %s', r.status_code)
            return r.ok
        except Exception as e:  # noqa: BLE001

"""Scorecard: per-system trades, win rate, P&L, profit factor, avg R, max drawdown (today and cumulative);
CSV + markdown exports; Telegram text."""
from __future__ import annotations
import csv, datetime as dt, os
import math


def _stats(trades: list, equity: float) -> dict:
    n = len(trades)
    pnl = [t['pnl'] for t in trades]
    wins = [x for x in pnl if x > 0]; losses = [x for x in pnl if x <= 0]
    gp, gl = sum(wins), -sum(losses)
    pf = (gp / gl) if gl > 0 else (math.inf if gp > 0 else 0.0)
    eq, peak, mdd = 0.0, 0.0, 0.0
    for x in pnl:
        eq += x; peak = max(peak, eq); mdd = max(mdd, peak - eq)
    return dict(trades=n, win=(len(wins) / n * 100 if n else 0.0), pnl=sum(pnl), pf=pf,
                avg_r=(sum(t['r'] for t in trades) / n if n else 0.0), mdd=mdd, mdd_pct=(mdd / equity * 100 if equity else 0.0))


def closed_trades(ledger, system: str, day_from=None, day_to=None) -> list:
    """One row per closed position (partials summed). Day = day of the final exit."""
    q = ('SELECT pid, MAX(day), SUM(pnl), SUM(r_mult), MIN(ts) FROM exits WHERE system=? GROUP BY pid HAVING MAX(final)=1 '
         'ORDER BY MAX(ts)')
    out = []
    for pid, day, pnl, r, ts in ledger.query(q, (system,)):
        if day_from and day < str(day_from):
            continue
        if day_to and day > str(day_to):
            continue
        out.append(dict(pid=pid, day=day, pnl=pnl, r=r))
    return out


def scorecard_rows(ledger, systems: list, day, equity: float, week_from=None) -> list:
    rows = []
    for sid in systems:
        allt = closed_trades(ledger, sid)
        tod = [t for t in allt if t['day'] == str(day)]
        real_today = ledger.realized_today(day, sid)
        openn = ledger.query("SELECT COUNT(*) FROM positions WHERE system=? AND status='open'", (sid,))[0][0]
        r = dict(system=sid, today=_stats(tod, equity), cum=_stats(allt, equity), realized_today=real_today, open=openn)
        if week_from:
            r['week'] = _stats([t for t in allt if t['day'] >= str(week_from)], equity)
        rows.append(r)
    rows.sort(key=lambda r: -r['cum']['pnl'])
    return rows


def _pf(x):
    return 'inf' if x == math.inf else f'{x:.2f}'


def render_table(rows: list, title: str, key='cum') -> str:
    lines = [f'# {title}', '', f'| # | system | trades | win% | P&L $ | PF | avg R | maxDD $ | today trades | today P&L $ | open |',
             '|---|---|---|---|---|---|---|---|---|---|---|']
    for i, r in enumerate(rows, 1):
        c, t = r[key], r['today']
        lines.append(f"| {i} | {r['system']} | {c['trades']} | {c['win']:.0f} | {c['pnl']:.2f} | {_pf(c['pf'])} | {c['avg_r']:.2f} | "
                     f"{c['mdd']:.2f} | {t['trades']} | {r['realized_today']:.2f} | {r['open']} |")
    return '\n'.join(lines) + '\n'


def telegram_text(rows: list, day, equity: float, weekly=False) -> list:
    key = 'week' if weekly else 'cum'
    head = f"{'WEEKLY' if weekly else 'Daily'} scorecard {day}  equity ${equity:,.0f}\n"
    head += "rank system  | today n/P&L | " + ('week' if weekly else 'cum') + " n win% P&L PF avgR maxDD\n"
    body = []
    tot_t = sum(r['realized_today'] for r in rows); tot_c = sum(r[key]['pnl'] for r in rows)
    for i, r in enumerate(sorted(rows, key=lambda r: -r[key]['pnl']), 1):
        c, t = r[key], r['today']
        body.append(f"{i:>2} {r['system']:<14} {t['trades']}/{r['realized_today']:+.0f} | {c['trades']} {c['win']:.0f}% "
                    f"{c['pnl']:+.0f} {_pf(c['pf'])} {c['avg_r']:+.2f}R dd{c['mdd']:.0f}")
    foot = f"\nALL xp: today {tot_t:+.2f}  {'week' if weekly else 'cum'} {tot_c:+.2f}"
    msgs, cur = [], head
    for line in body:
        if len(cur) + len(line) > 3500:
            msgs.append(cur); cur = ''
        cur += line + '\n'
    msgs.append(cur + foot)
    return msgs


def export(ledger, rows: list, day, outdir: str, equity: float, weekly=False) -> list:
    os.makedirs(outdir, exist_ok=True)
    paths = []
    md = os.path.join(outdir, f'scorecard_{day}.md')
    open(md, 'w').write(render_table(rows, f'xp PAPER scorecard {day} (equity ${equity:,.0f})'))
    paths.append(md)
    cs = os.path.join(outdir, f'scorecard_{day}.csv')
    with open(cs, 'w', newline='') as f:
        w = csv.writer(f)
        w.writerow(['system', 'scope', 'trades', 'win_pct', 'pnl', 'profit_factor', 'avg_r', 'max_dd', 'max_dd_pct'])
        for r in rows:
            for scope in ('today', 'cum') + (('week',) if 'week' in r else ()):
                s = r[scope]
                w.writerow([r['system'], scope, s['trades'], round(s['win'], 1), round(s['pnl'], 2), _pf(s['pf']),
                            round(s['avg_r'], 3), round(s['mdd'], 2), round(s['mdd_pct'], 3)])
    paths.append(cs)
    tr = os.path.join(outdir, 'trades_all.csv')
    with open(tr, 'w', newline='') as f:
        w = csv.writer(f)
        cols = ['ts', 'day', 'system', 'symbol', 'side', 'pid', 'qty', 'entry_px', 'exit_px', 'ref_px', 'pnl', 'r_mult',
                'entry_slip', 'exit_slip', 'reason', 'final']
        w.writerow(cols)
        for row in ledger.query('SELECT ' + ','.join(cols) + ' FROM exits ORDER BY ts'):
            w.writerow(row)
    paths.append(tr)
    if weekly:
        wk = os.path.join(outdir, f'weekly_{day}.md')
        open(wk, 'w').write(render_table(rows, f'xp PAPER weekly summary, week ending {day}', key='week'))
        paths.append(wk)
    return paths

"""Gapper list + earnings sets (shared by live and replay)."""
from __future__ import annotations
import datetime as dt, re
import logging
log = logging.getLogger('xp.scan')
BAD = re.compile(r'(W|WS|U|R|RT)$')

def clean_symbol(s: str) -> bool:
    if not re.fullmatch(r'[A-Z]{1,5}(\.[AB])?', s or ''):
        return False
    return not (len(s) == 5 and BAD.search(s))

def earnings_sets(day: dt.date, prev_day: dt.date, next_day: dt.date, fetch) -> tuple[set, set, bool]:
    """(soon, recent, ok). soon = reports after today's close or before the next open (unknown time counts);
    recent = reported since the prior close (EP catalyst)."""
    soon, recent, ok = set(), set(), True
    try:
        today = fetch(day)
        soon |= {s for s, t in today.items() if t in ('amc', '')}
        recent |= {s for s, t in today.items() if t in ('bmo', '')}
    except Exception as e:  # noqa: BLE001
        ok = False; log.warning('earnings fetch %s failed: %s', day, type(e).__name__)
    try:
        nxt = fetch(next_day)
        soon |= {s for s, t in nxt.items() if t in ('bmo', '')}
    except Exception as e:  # noqa: BLE001
        ok = False; log.warning('earnings fetch %s failed: %s', next_day, type(e).__name__)
    try:
        prv = fetch(prev_day)
        recent |= {s for s, t in prv.items() if t in ('amc', '')}
    except Exception as e:  # noqa: BLE001
        ok = False; log.warning('earnings fetch %s failed: %s', prev_day, type(e).__name__)
    return soon, recent, ok

"""Strategy interface. A strategy only describes the expert's rules: it turns bars into Signals (with a Plan for
managing the position). Execution, sizing, ownership and safety live in the engine/broker, shared by every system."""
from __future__ import annotations
import datetime as dt
import os
from dataclasses import dataclass, field
from typing import Optional
import pandas as pd
from ..model import Plan, Signal, Position

TICK = 0.01
REGISTRY: dict[str, type] = {}

def register(cls):
    REGISTRY[cls.sid] = cls
    return cls

@dataclass
class Context:
    now: pd.Timestamp
    day: dt.date
    session_open: pd.Timestamp
    session_close: pd.Timestamp
    daily: dict                         # sym -> completed daily bars (through the prior session), SIP
    md: object                          # MarketData (intraday 1-min bars incl. warm-up)
    universes: dict                     # name -> list of symbols
    gappers: dict = field(default_factory=dict)        # sym -> {'gap_pct':..,'prev_close':..,'news':bool,...}
    earnings_soon: set = field(default_factory=set)    # report after today's close or before next open
    earnings_recent: set = field(default_factory=set)  # reported since the prior close (EP catalyst)
    news_syms: set = field(default_factory=set)        # Alpaca news since prior close
    cache: dict = field(default_factory=dict)

    def today(self, df: pd.DataFrame) -> pd.DataFrame:
        if df is None or df.empty:
            return df
        return df[df.index >= self.session_open]

    def at(self, hhmm: str) -> pd.Timestamp:
        h, m = map(int, hhmm.split(':'))
        return self.session_open.normalize() + pd.Timedelta(hours=h, minutes=m)

class Strategy:
    sid = 'base'                 # system id; client_order_id prefix = 'xp-' + sid + '-'
    expert = ''
    title = ''
    style = 'intraday'           # 'intraday' (flat by 15:55 ET) | 'swing' (holds overnight per its rules)
    tf = 1                       # bar timeframe (minutes) for on_bar
    universe = ('core',)         # names in ctx.universes, plus anything returned by watchlist()
    warmup_sessions = 0          # prior sessions of 1-min bars needed for indicators
    source = ''
    rules = ''
    approximations = ''
    defaults: dict = {}
    entry_window = ('09:31', '15:30')

    def __init__(self, params: Optional[dict] = None):
        self.p = {**self.defaults, **(params or {})}
        self.watch: dict = {}
        self.fired: set = set()      # (day, sym, tag) one-shot guards

    @property
    def prefix(self) -> str:
        # xp-combo: client_order_id root comes from XP_PREFIX (xp-combo service sets 'xc-'); default 'xp-' (xp-test)
        return f"{os.environ.get('XP_PREFIX', 'xp-')}{self.sid}-"

    # ---- hooks
    def prep(self, ctx: Context) -> None:
        """Pre-market: build today's watchlist from daily bars."""
        self.watch = {}

    def symbols(self, ctx: Context) -> list:
        out = []
        for u in self.universe:
            out += list(ctx.universes.get(u, []))
        out += list(self.watch.keys())
        return list(dict.fromkeys(out))

    def on_bar(self, sym: str, bars: pd.DataFrame, ctx: Context) -> list:
        return []

    def on_close(self, ctx: Context) -> list:
        """~15:44 ET: market-on-close entries (e.g. Connors buys on the close)."""
        return []

    def close_exit(self, pos: Position, dfd: pd.DataFrame, ctx: Context) -> Optional[str]:
        """~15:44 ET with today's provisional daily bar appended: return a reason to exit at the close."""
        return None

    def on_new_day(self, pos: Position, ctx: Context) -> None:
        """Pre-market, for positions carried overnight (e.g. Pristine 4-step stop raise)."""
        return None

    # ---- helpers
    def in_window(self, ctx: Context) -> bool:
        return ctx.at(self.entry_window[0]) <= ctx.now <= ctx.at(self.entry_window[1])

    def once(self, ctx: Context, sym: str, tag: str) -> bool:
        k = (ctx.day, sym, tag)
        if k in self.fired:
            return False
        self.fired.add(k)
        return True

    def sig(self, ctx, sym, side, trigger, stop, plan, reason, kind='stop', valid_min=None, **kw) -> Signal:
        exp = ctx.now + pd.Timedelta(minutes=valid_min) if valid_min else ctx.session_close
        return Signal(self.sid, sym, side, kind, round(float(trigger), 2), round(float(stop), 2), plan, reason,
                      expires=exp, created=ctx.now, **kw)

"""Non-Velez expert systems. Every rule cites its source in docs/SYSTEMS.md; numbers marked DEFAULT are where the
expert leaves a value open and a concrete choice was needed to code it."""
from __future__ import annotations

def gappers(ctx, min_gap, pmin, pmax, need_news=False, need_catalyst=False, top=5):
    out = []
    for s, g in ctx.gappers.items():
        px = g.get('price') or g.get('open')
        if px is None or not (pmin <= px <= pmax) or g.get('gap_pct', 0) < min_gap:
            continue
        if need_news and not g.get('news'):
            continue
        if need_catalyst and not (g.get('news') or s in ctx.earnings_recent):
            continue
        out.append((g['gap_pct'], s))
    return [s for _, s in sorted(out, reverse=True)[:top]]

@register
class AzizABCD(Strategy):
    sid = 'aziz-abcd'
    expert = 'Andrew Aziz'
    title = 'ABCD pattern on stocks in play'
    tf = 1
    universe = ()
    warmup_sessions = 1
    # BOOK (Aziz, Advanced Techniques ch.3 'Pre-market Gappers'): gap >= 2%, ADV > 500k, daily ATR >= $0.50,
    # >= 100k pre-market shares (NOT checkable on IEX), fundamental catalyst.
    defaults = dict(min_gap=2.0, pmin=5.0, pmax=200.0, top=8, min_ab_atr=3.0, min_rr=2.0, min_adv=500_000,
                    min_atr_d=0.50, need_catalyst=True)
    entry_window = ('09:35', '11:30')
    rules = ('Stocks in play: gappers >= 2% with a catalyst, $5-$200, 20-day avg volume >= 500k, daily ATR >= '
             "$0.50. 1-min: A = low before today's high B (A->B >= 3 ATR); price pulls back to C, which holds "
             'above A (retrace 20-62%). When a green bar makes a higher low above C, buy 1c over it; stop 1c under '
             'C. Sell half at B+1c and move the stop to breakeven; trail the rest under 5-min bars; skip unless D '
             '= C + (B - A) gives >= 2:1. Flat by the close.')

    def symbols(self, ctx):
        out = []
        for s in gappers(ctx, self.p['min_gap'], self.p['pmin'], self.p['pmax'], need_catalyst=self.p['need_catalyst'],
                         top=40):
            d = ctx.daily.get(s)
            if d is None or len(d) < 20:
                continue
            if d.volume.iloc[-20:].mean() < self.p['min_adv'] or atr(d, 14).iloc[-1] < self.p['min_atr_d']:
                continue
            out.append(s)
        return out[:self.p['top']]

    def on_bar(self, sym, bars, ctx):
        if bars is None or not self.in_window(ctx):
            return []
        tod = ctx.today(bars)
        if tod is None or len(tod) < 8:
            return []
        a14 = atr(bars, 14).iloc[-1]
        h, l, o, c = tod['high'].values, tod['low'].values, tod['open'].values, tod['close'].values
        i = len(tod) - 1
        b = int(np.argmax(h[:i]))
        if b < 2 or i - b < 2:
            return []
        a = int(np.argmin(l[:b + 1]))
        A, B = l[a], h[b]
        AB = B - A
        if not np.isfinite(a14) or AB < self.p['min_ab_atr'] * a14:
            return []
        C = l[b + 1:i + 1].min()
        if not (A + 0.38 * AB <= C <= B - 0.2 * AB):
            return []
        if not (c[i] > o[i] and l[i] > C and l[i] > l[i - 1] - 1e-9):
            return []
        entry, stop = h[i] + TICK, C - TICK
        D = C + AB
        if entry >= B or (D - entry) < self.p['min_rr'] * (entry - stop):
            return []
        if not self.once(ctx, sym, f'abcd{b}'):
            return []
        # BOOK step 5-6: sell half at D (the new high above B) and stop to break-even; sell the rest on a target or
        # when price makes a new low on the 5-min chart (coded: 5-min bar trail once B is reached). D = C + (B - A)
        # is kept only as the DEFAULT 2:1 screen; it is no longer a hard exit.
        b_r = max(float((B + TICK - entry) / (entry - stop)), 0.0)
        plan = Plan(eod_flat=True, partials=[{'at_px': round(float(B) + TICK, 2), 'frac': 0.5, 'be': True}],
                    trail='bar', trail_tf=5, trail_after_r=float(round(b_r, 2)), exit_before_earnings=False)
        return [self.sig(ctx, sym, 'long', entry, stop, plan, f'ABCD A={A:.2f} B={B:.2f} C={C:.2f}', valid_min=2)]

class CrabelORB(Strategy):
    sid = 'crabel-orb'
    expert = 'Toby Crabel'
    title = 'Opening range breakout with stretch (OCO)'
    tf = 1
    universe = ('core', 'index')
    # anchor='open': buy stop open + stretch / sell stop open - stretch (Crabel 1990 book, S&C V7:4 abstract, Crabel's
    # 2026 Substack). anchor='or5': legacy behaviour (first 5-min high/low +/- stretch, S&C V7:2 wording).
    # filt: None = trade every day (Crabel's unfiltered baseline); 'nr4' | 'nr7' | 'idnr4' | 'any' = only after that
    # prior-day contraction pattern (his book's main finding). The pattern is always written into the reason.
    defaults = dict(stretch_days=10, anchor='open', filt=None)
    entry_window = ('09:35', '15:30')
    rules = ('Stretch = 10-day average of |open - nearer extreme of the day|. On the first 1-min bar, OCO: buy '
             "stop at today's open + stretch, sell-short stop at open - stretch (anchor='open'; legacy "
             "anchor='or5' = first 5-min high/low +/- stretch); the first fill is the trade and cancels the other; "
             'the opposite level is the protective stop; exit on the close (15:55). Prior-day NR4/NR7/ID/ID-NR4 '
             'pattern logged (filter off).')

    @staticmethod
    def _pattern(d) -> str:
        if len(d) < 7:
            return ''
        r = (d.high - d.low).values
        nr4 = r[-1] < r[-4:-1].min()
        nr7 = r[-1] < r[-7:-1].min()
        ib = d.high.iloc[-1] < d.high.iloc[-2] and d.low.iloc[-1] > d.low.iloc[-2]
        tags = [t for t, ok in (('IDNR4', ib and nr4), ('NR7', nr7), ('NR4', nr4), ('ID', ib)) if ok]
        return '/'.join(tags)

    def prep(self, ctx):
        self.watch, self.stretch, self.pat = {}, {}, {}
        for sym in self.symbols(ctx):
            d = ctx.daily.get(sym)
            if d is None or len(d) < max(self.p['stretch_days'], 7):
                continue
            pat = self._pattern(d)
            f = self.p['filt']
            if f and not ((f == 'any' and pat) or (f == 'nr4' and 'NR4' in pat) or (f == 'nr7' and 'NR7' in pat)
                          or (f == 'idnr4' and 'IDNR4' in pat)):
                continue
            t = d.iloc[-self.p['stretch_days']:]
            self.stretch[sym] = float(np.minimum(t.high - t.open, t.open - t.low).mean())
            self.pat[sym] = pat or 'none'

    def on_bar(self, sym, bars, ctx):
        tod = ctx.today(bars)
        st = self.stretch.get(sym)
        if tod is None or st is None:
            return []
        if self.p['anchor'] == 'or5':
            if len(tod) != 5 or not self.once(ctx, sym, 'orb'):
                return []
            up, dn = tod['high'].max() + st, tod['low'].min() - st
        else:
            if len(tod) != 1 or not self.once(ctx, sym, 'orb'):
                return []
            o = float(tod['open'].iloc[0])
            up, dn = o + st, o - st
        plan = Plan(eod_flat=True, exit_before_earnings=False)
        tag = f"stretch {st:.2f}, prior day {self.pat.get(sym, '?')}"
        return [self.sig(ctx, sym, 'long', up, dn, plan, f'ORB long ({tag})', meta={'oco': True}),
                self.sig(ctx, sym, 'short', dn, up, plan, f'ORB short ({tag})', meta={'oco': True})]

                continue

            r2, s200 = rsi(cl, 2).iloc[-1], cl.iloc[-200:].mean()
            a = atr(d, 14).iloc[-1]
            if px > s200 and r2 < p['rsi_long']:

                cands.append((100 - r2, sym, 'short', px, px + p['nominal_atr'] * a))
        cands.sort()
        out = []
        for score, sym, side, px, nstop in cands[:p['max_new']]:
            plan = Plan(eod_flat=False, broker_stop=False, exit_before_earnings=True)

        if pos.side == 'long' and c > s5:
            return 'close > SMA5'
        if pos.side == 'short' and c < s5:
            return 'close < SMA5'
        return None

def zigzag(h: np.ndarray, l: np.ndarray, pct: float) -> list:
    """Alternating swing points [(idx, price, 'H'|'L')] with a pct reversal threshold."""
    pts, mode, ext_i, ext = [], None, 0, h[0]
    lo_i, lo = 0, l[0]
    for i in range(1, len(h)):
        if mode in (None, 'up'):
            if h[i] >= ext:
                ext, ext_i = h[i], i
            if l[i] <= ext * (1 - pct):
                pts.append((ext_i, ext, 'H')); mode = 'down'; lo, lo_i = l[i], i
        if mode == 'down':
            if l[i] <= lo:
                lo, lo_i = l[i], i
            if h[i] >= lo * (1 + pct):
                pts.append((lo_i, lo, 'L')); mode = 'up'; ext, ext_i = h[i], i
    if mode == 'down':
        pts.append((lo_i, lo, 'L'))
    return pts

def trend_template(d: pd.DataFrame) -> bool:
    c = d.close
    if len(c) < 260:
        return False
    s50, s150, s200 = sma(c, 50).iloc[-1], sma(c, 150).iloc[-1], sma(c, 200)
    px = c.iloc[-1]
    lo52, hi52 = d.low.iloc[-252:].min(), d.high.iloc[-252:].max()
    return (px > s150 and px > s200.iloc[-1] and s150 > s200.iloc[-1] and s200.iloc[-1] > s200.iloc[-22]
            and s50 > s150 and px > s50 and px >= 1.30 * lo52 and px >= 0.75 * hi52)

def rs_score(d: pd.DataFrame) -> float:
    c = d.close
    if len(c) < 253:
        return np.nan
    q = [c.iloc[-1] / c.iloc[-64] - 1, c.iloc[-64] / c.iloc[-127] - 1, c.iloc[-127] / c.iloc[-190] - 1, c.iloc[-190] / c.iloc[-253] - 1]
    return 0.4 * q[0] + 0.2 * (q[1] + q[2] + q[3])

def find_vcp(d: pd.DataFrame, lookback=65, zz=0.03, max_first=0.35, max_last=0.10):
    w = d.iloc[-lookback:]
    pts = zigzag(w.high.values, w.low.values, zz)
    pairs = []
    for k in range(len(pts) - 1):
        if pts[k][2] == 'H' and pts[k + 1][2] == 'L':
            pairs.append((pts[k], pts[k + 1]))
    if len(pairs) < 2:
        return None
    depths = [(H[1] - L[1]) / H[1] for H, L in pairs]
    tail = depths[-4:] if len(depths) >= 4 else depths
    n = 1
    for j in range(len(depths) - 1, 0, -1):
        if depths[j] < depths[j - 1]:
            n += 1
        else:
            break
    if n < 2:
        return None
    seq = depths[-n:]
    if seq[0] > max_first or seq[-1] > max_last:
        return None
    H, L = pairs[-1]
    px = w.close.iloc[-1]
    if not (L[1] <= px <= H[1]):
        return None
    vol_dry = w.volume.iloc[-10:].mean() < d.volume.iloc[-50:].mean()
    if not vol_dry:
        return None
    return dict(pivot=float(H[1]), low=float(L[1]), depths=[round(x * 100, 1) for x in seq])

# ------------------------------------------------------------------ shared helpers
def rs_scores(daily: dict, syms) -> pd.Series:
    """IBD-style weighted 12-month performance (40% last quarter + 20% each prior quarter) -> percentile 0-100."""
    out = {}
    for s in syms:
        d = daily.get(s)
        if d is None or len(d) < 253:
            continue
        c = d.close.values
        out[s] = 0.4 * (c[-1] / c[-64] - 1) + 0.2 * ((c[-64] / c[-127] - 1) + (c[-127] / c[-190] - 1) + (c[-190] / c[-253] - 1))
    ser = pd.Series(out, dtype=float).dropna()
    return ser.rank(pct=True) * 100 if len(ser) else ser


def weekly(d: pd.DataFrame, today) -> pd.DataFrame:
    """Completed weeks (Friday-labelled) from daily bars; drops the week that contains `today`."""
    w = d.resample('W-FRI').agg({'open': 'first', 'high': 'max', 'low': 'min', 'close': 'last', 'volume': 'sum'}).dropna()
    t = pd.Timestamp(today)
    if w.index.tz is not None and t.tzinfo is None:
        t = t.tz_localize(w.index.tz)
    return w[w.index < t.normalize()]


def ema_np(x: np.ndarray, n: int) -> np.ndarray:
    a = 2.0 / (n + 1); out = np.empty_like(x, dtype=float); out[0] = x[0]
    for i in range(1, len(x)):
        out[i] = a * x[i] + (1 - a) * out[i - 1]
    return out


def distribution_days(spy: pd.DataFrame, n: int = 25) -> int:
    """O'Neil distribution day: index closes down >= 0.2% on higher volume than the prior day (SPY as the proxy)."""
    if spy is None or len(spy) < n + 1:
        return 0
    s = spy.iloc[-(n + 1):]
    ch = s.close.pct_change(); vu = s.volume > s.volume.shift(1)
    return int(((ch <= -0.002) & vu).iloc[1:].sum())


class DailySwing(Strategy):

    tf = 1
    universe = ()
    scan_universe = 'sp500'
    max_watch = 25                                  # DEFAULT: watch at most 25 setups per system per day
    min_hist = 60

    def market(self, ctx) -> dict:
        return {}

    def scan(self, sym: str, d: pd.DataFrame, ext: dict):
        raise NotImplementedError

    def make_plan(self, w: dict) -> Plan:
        raise NotImplementedError

    def prep(self, ctx):
        self.watch = {}
        ext = self.market(ctx)
        ext['today'] = ctx.day
        if ext.get('skip'):
            return
        cands = []
        for s in ctx.universes.get(self.scan_universe, []):
            d = ctx.daily.get(s)
            if d is None or len(d) < self.min_hist:
                continue
            try:
                w = self.scan(s, d, ext)
            except Exception:  # noqa: BLE001  (a bad series must not kill the scan)
                w = None

        return True

        w = self.watch.get(sym)
        tod = ctx.today(bars)
        if not w or tod is None or tod.empty or not self.once(ctx, sym, self.sid):
            return []
        if not self.entry_ok(w, tod):
            return []
        return [self.sig(ctx, sym, w['side'], w['trig'], w['stop'], self.make_plan(w), w['why'])]


# ------------------------------------------------------------------ William J. O'Neil
def find_cup_handle(d: pd.DataFrame, p: dict):
    h, l, c, v = d.high.values, d.low.values, d.close.values, d.volume.values
    n = len(c)
    if n < 200:
        return None
    hw = p['handle_max'] + 1
    r = n - hw + int(np.argmax(h[-hw:]))                       # right lip = top of the handle
    if n - 1 - r < p['handle_min'] - 1:                         # handle >= 1 week (5 sessions incl. the lip bar)
        return None
    lo_a, hi_a = max(0, r - p['cup_max']), r - p['right_min']
    if hi_a - lo_a < 5:
        return None
    a = lo_a + int(np.argmax(h[lo_a:hi_a]))                     # left lip
    if n - 1 - a < p['cup_min']:                                # cup + handle >= 7 weeks
        return None
    if h[a + 1:r].max(initial=0) > max(h[a], h[r]) or h[r] > h[a] * 1.05:
        return None
    m = a + int(np.argmin(l[a:r]))
    depth = (h[a] - l[m]) / h[a]
    if not (p['depth_min'] <= depth <= p['depth_max']):
        return None
    pre = l[max(0, a - p['prior_lb']):a + 1]
    if len(pre) < 20 or h[a] / pre.min() - 1 < p['prior_up']:
        return None
    hl = l[r:].min()
    if (h[r] - hl) / h[r] > p['handle_depth']:                  # handle decline <= 12%
        return None
    if hl <= l[m] + 0.5 * (h[a] - l[m]):                        # handle in the upper half of the cup
        return None
    s50 = c[-50:].mean()
    if hl < s50:                                                # handle above the 10-week (50-day) line
        return None
    if v[r + 1:].size and v[r + 1:].mean() >= v[-50:].mean():   # volume dries up in the handle
        return None
    pivot = h[r] + p['pivot_add']
    if c[-1] >= pivot:
        return None
    return dict(pivot=float(pivot), depth=round(depth * 100, 1), cup_len=int(r - a), handle_len=int(n - 1 - r))

def weinstein_setup(d: pd.DataFrame, spyw: pd.DataFrame, today, p: dict, w: pd.DataFrame = None):
    w = weekly(d, today) if w is None else w
    if len(w) < 60 or spyw is None or len(spyw) < 60:
        return None
    wc, wh, wl = w.close.values, w.high.values, w.low.values
    ma = pd.Series(wc).rolling(30).mean().values
    sw = spyw.reindex(w.index).close
    if sw.isna().iloc[-1]:
        return None
    ratio = (w.close / sw).values
    rs = ratio[-1] / np.nanmean(ratio[-52:]) - 1                     # Mansfield RS (52-week basis)
    smac = spyw.close.rolling(30).mean().values
    mkt_up = spyw.close.values[-1] > smac[-1] and smac[-1] >= smac[-2]
    mkt_dn = spyw.close.values[-1] < smac[-1] and smac[-1] <= smac[-2]
    last = float(d.close.iloc[-1])
    B, N = p['base_weeks'], p['minor_weeks']
    res_i = len(wh) - B + int(np.argmax(wh[-B:]))
    sup_i = len(wl) - B + int(np.argmin(wl[-B:]))
    res, sup = wh[-B:].max(), wl[-B:].min()
    if (mkt_up and ma[-1] >= ma[-2] and res > ma[-1] and rs > 0 and len(wh) - 1 - res_i >= p['base_age']
            and wc[-1] < res and last < res and last >= res * (1 - p['near'])):
        trig = res + TICK
        stop = min(wl[-N:].min(), ma[-1]) - TICK
        return dict(side='long', trig=round(float(trig), 2), stop=round(float(stop), 2), rank=float(rs) * 100,
                    level=float(res), why=f'Stage 2 breakout over {res:.2f} (30w MA {ma[-1]:.2f}, Mansfield RS {rs:+.2f})')
    if (p['shorts'] and mkt_dn and ma[-1] <= ma[-2] and sup < ma[-1] and rs < 0 and len(wl) - 1 - sup_i >= p['base_age']
            and wc[-1] > sup and last > sup and last <= sup * (1 + p['near'])):
        trig = sup - TICK
        stop = max(wh[-N:].max(), ma[-1]) + TICK
        return dict(side='short', trig=round(float(trig), 2), stop=round(float(stop), 2), rank=float(-rs) * 100,
                    level=float(sup), why=f'Stage 4 breakdown under {sup:.2f} (30w MA {ma[-1]:.2f}, Mansfield RS {rs:+.2f})')
    return None


def weinstein_manage(pos, d: pd.DataFrame, today, p: dict):
    """New completed week -> (a) breakout-week volume check, (b) trail the stop. Returns new stop or None."""
    w = weekly(d, today)
    if len(w) < 35:
        return None
    wk = str(w.index[-1].date())
    if pos.meta.get('wk') == wk:
        return None
    pos.meta['wk'] = wk
    entry_wk = pd.Timestamp(pos.entry_day)
    if w.index.tz is not None:
        entry_wk = entry_wk.tz_localize(w.index.tz)
    if pos.side == 'long' and 'vol_ok' not in pos.meta:
        bw = w[w.index >= entry_wk]
        if len(bw):
            j = w.index.get_loc(bw.index[0])
            if j >= 4:
                ok = w.volume.iloc[j] >= p['vol_mult'] * w.volume.iloc[j - 4:j].mean()
                pos.meta['vol_ok'] = bool(ok)
    ma = w.close.rolling(30).mean().values
    N = p['minor_weeks']
    if pos.side == 'long':
        minor = w.low.values[-N:].min()
        lvl = (min(minor, ma[-1]) if ma[-1] > ma[-2] else minor) - TICK      # MA flat/down -> tighten to minor low
        return round(float(lvl), 2) if lvl > pos.stop else None
    minor = w.high.values[-N:].max()
    lvl = (max(minor, ma[-1]) if ma[-1] < ma[-2] else minor) + TICK
    return round(float(lvl), 2) if lvl < pos.stop else None

# ------------------------------------------------------------------ Nicolas Darvas
def darvas_box(h: np.ndarray, l: np.ndarray, t: int, k: int = 3):
    """Box after a top at index t: top confirmed when the next k highs stay below it; bottom = a low not undercut on the
    next k days (may form on the top day). Returns (top, bottom, confirmed_idx) or None."""
    n = len(h)
    if t + k >= n or (h[t + 1:t + k + 1] >= h[t]).any():
        return None
    run = np.inf
    for j in range(t, n - k):
        run = min(run, l[j])
        if l[j] <= run and (l[j + 1:j + k + 1] > l[j]).all():
            return float(h[t]), float(l[j]), max(j + k, t + k)
    return None


def darvas_setup(d: pd.DataFrame, p: dict):
    h, l, c = d.high.values, d.low.values, d.close.values
    n = len(h)
    if n < p['hi_days'] + 5:
        return None
    hh = pd.Series(h).rolling(p['hi_days']).max().values
    idx = np.where(h[p['hi_days']:] >= hh[p['hi_days']:])[0]
    if not len(idx):
        return None
    t = int(idx[-1]) + p['hi_days']                                   # most recent 52-week high
    box = darvas_box(h, l, t, p['k'])
    if not box:
        return None
    top, bot, ci = box
    if (l[ci + 1:] <= bot).any() or (h[t + 1:] > top).any():          # box broken / already out
        return None
    if c[-1] >= top:
        return None
    return dict(top=top, bottom=bot)


def darvas_trail(d: pd.DataFrame, entry_day: str, cur_stop: float, p: dict):
    """A new, higher box completed since the entry -> stop 1/8 under its bottom."""
    h, l = d.high.values, d.low.values
    t0 = pd.Timestamp(entry_day)
    if d.index.tz is not None:
        t0 = t0.tz_localize(d.index.tz)
    pos0 = int(d.index.searchsorted(t0))
    if pos0 >= len(h) - p['k'] - 1:
        return None
    seg = h[pos0:]
    t = pos0 + len(seg) - 1 - int(np.argmax(seg[::-1]))                # last occurrence of the highest high since entry
    box = darvas_box(h, l, t, p['k'])
    if not box:
        return None
    lvl = round(box[1] - p['eighth'], 2)
    return lvl if lvl > cur_stop else None

def landry_setup(d: pd.DataFrame, p: dict):
    h, l, c, v = d.high.values, d.low.values, d.close.values, d.volume.values
    n = len(c)
    if n < 60 or c[-1] < p['min_px'] or v[-50:].mean() < p['min_vol']:
        return None
    lr = np.diff(np.log(c[-51:]))
    hv = lr.std(ddof=1) * np.sqrt(252) * 100
    if not (p['hv_min'] <= hv <= p['hv_max']):
        return None
    x = c[-120:]
    s10 = x[-10:].mean()
    e20, e30 = ema_np(x, 20), ema_np(x, 30)
    if s10 > e20[-1] > e30[-1] and e20[-1] > e20[-6]:
        side = 'long'
    elif p['shorts'] and s10 < e20[-1] < e30[-1] and e20[-1] < e20[-6]:
        side = 'short'
    else:
        return None
    if side == 'long':
        hh = pd.Series(h).rolling(20).max().values
        recent = any(h[k] >= hh[k] for k in range(n - 1 - p['max_pb'], n - 1))
        if not recent or not (l[-1] < l[-2] and l[-1] < l[-3]):
            return None
        return dict(side='long', trig=round(h[-1] + p['wiggle'], 2), stop=round(l[-1] - TICK, 2), rank=float(hv),
                    why=f'Trend Knockout long (HV50 {hv:.0f})')
    ll = pd.Series(l).rolling(20).min().values
    recent = any(l[k] <= ll[k] for k in range(n - 1 - p['max_pb'], n - 1))
    if not recent or not (h[-1] > h[-2] and h[-1] > h[-3]):
        return None
    return dict(side='short', trig=round(l[-1] - p['wiggle'], 2), stop=round(h[-1] + TICK, 2), rank=float(hv),
                why=f'Trend Knockout short (HV50 {hv:.0f})')

                    rank=float(rng[-1] / rng[-10:-1].mean()), why=f'Expansion Breakdown (60-day low, widest range of 10)')
    return None

"""Oliver Velez systems (each is its own system / client_order_id prefix). Sources are in docs/SYSTEMS.md.

Shared intraday definitions (Velez, "The Powerful Elephant Bar", "The 90% Pattern", "From $50K to $500K" videos):
  SMA20 / SMA200 (simple) on the system's timeframe; picture of power = 20 rising, price & 20 above the 200.
  Elephant bar (EB): range >= 2x mean range of prior 20 bars, body >= 70% of range ("very solid"), not "pennies"
  (range >= 0.25% of price, DEFAULT), and it clears the prior 5 bars (Velez: "all you need is to clear five bars").
  Management (Velez method summary): initial stop beyond the signal bar; after one bar moves in favour, trail the
  stop bar-by-bar under each prior bar's low (over each prior bar's high for shorts); flat by the close.

Book-review corrections (VELEZ_BOOK.md, 2026-10-02):
  * Velez's position rules (RBI talk "From $50K to $500K"): trade WITH the 20's slope, buy above / short below the
    200, enter NEAR the 20, act only in the picture of power; stop playing in a WIDE state ("three-finger spread").
  * EB entry: in the last seconds of the EB or 1c over its high on the NEXT bar only ("The Powerful Elephant Bar"),
    and it must be the FIRST elephant ("nothing strong to the left of it").
  * Bull/Bear 180 ("The Bull 180" talk): fat red bar, the very next bar is green and takes out the red high -> buy
    1c over the RED bar's high while the green bar is still forming; stop 1c under the green bar's low.
  * Pristine action step 4: trail under each prior bar only after two bars in your direction are complete.
  * Tools & Tactics (2000) ch.7 Secret #9: "11:15 to 2:15 EST Is the Worst Possible Time to Trade" (optional block).
"""
from __future__ import annotations

def frame(bars: pd.DataFrame, ctx, sym: str, tf: int) -> dict | None:
    if bars is None or len(bars) < 210:
        return None
    key = ('velez', sym, tf, bars.index[-1])
    f = ctx.cache.get(key)
    if f is None:
        rng = bars['high'] - bars['low']
        f = dict(df=bars, s20=sma(bars['close'], 20), s200=sma(bars['close'], 200), atr=atr(bars, 14), rng=rng,
                 mrng=rng.shift(1).rolling(20).mean())
        ctx.cache[key] = f
    return f

def slope(s: pd.Series, i: int, lag: int = 5) -> float:
    a, b = s.iloc[i], s.iloc[i - lag]
    return (a - b) / b * 100 if b else 0.0

def is_eb(f, i, p) -> str | None:
    df = f['df']; o, h, l, c = (df[k].iloc[i] for k in ('open', 'high', 'low', 'close'))
    r = h - l
    if r <= 0 or not np.isfinite(f['mrng'].iloc[i]) or r < p['eb_range_mult'] * f['mrng'].iloc[i]:
        return None
    if abs(c - o) < p['eb_body_pct'] * r or r < p['eb_min_pct'] / 100 * c:
        return None
    if c > o and c > df['high'].iloc[i - 5:i].max():
        return 'long'
    if c < o and c < df['low'].iloc[i - 5:i].min():
        return 'short'
    return None

def flat_narrow(f, i, p) -> bool:
    c = f['df']['close'].iloc[i]
    gap = abs(f['s20'].iloc[i] - f['s200'].iloc[i]) / c * 100
    flat20 = abs(slope(f['s20'], i)) < p['flat_slope_pct']
    flat200 = abs(slope(f['s200'], i)) < p['flat_slope_pct']
    return gap <= p['fn_max_gap_pct'] and (flat20 or flat200)

def bar_plan(trail_after_bars: int = 1, **kw) -> Plan:
    return Plan(eod_flat=True, trail='bar', trail_after_bars=trail_after_bars, exit_before_earnings=False, **kw)

COMMON = dict(eb_range_mult=2.0, eb_body_pct=0.70, eb_min_pct=0.25, flat_slope_pct=0.05, fn_max_gap_pct=0.30,
              near_atr=0.25,
              valid_bars=1,            # was 3 (DEFAULT). Velez: "the next bar must clear the high of the elephant bar"
              wide_max_gap_pct=1.0,    # DEFAULT: |SMA20-SMA200| > 1.0% of price = wide state -> no new entries
              first_lookback=10,       # DEFAULT window for "nothing strong to the left" (first elephant only)
              midday_block=None)       # optional ('11:15', '14:15') ET = Tools & Tactics Secret #9; off by default

def wide(f, i, p) -> bool:
    """Velez 'wide state': the 20 has separated far from the 200 (trend mature) -> stop entering with the trend."""
    c = f['df']['close'].iloc[i]
    return abs(f['s20'].iloc[i] - f['s200'].iloc[i]) / c * 100 > p['wide_max_gap_pct']

def first_eb(f, i, p, side) -> bool:
    """Velez: buy the FIRST elephant - no same-direction elephant-sized bar in the prior first_lookback bars."""
    df = f['df']
    for j in range(max(1, i - p['first_lookback']), i):
        r, m = df['high'].iloc[j] - df['low'].iloc[j], f['mrng'].iloc[j]
        up = df['close'].iloc[j] > df['open'].iloc[j]
        if np.isfinite(m) and r >= p['eb_range_mult'] * m and up == (side == 'long'):
            return False
    return True

class _VelezIntraday(Strategy):

    defaults = dict(COMMON)

            return []
        side = is_eb(f, i, self.p)
        if not side:
            return []
        c, hi_ma, lo_ma = df['close'].iloc[i], max(f['s20'].iloc[i], f['s200'].iloc[i]), min(f['s20'].iloc[i], f['s200'].iloc[i])
        if (side == 'long' and c <= hi_ma) or (side == 'short' and c >= lo_ma) or not self.once(ctx, sym, 'fn'):
            return []
        stop = df['low'].iloc[i] - TICK if side == 'long' else df['high'].iloc[i] + TICK
        return [self.sig(ctx, sym, side, self.entry_px(df, i, side), stop, bar_plan(time_exit=self.p['time_exit']),
                         f'{side} F&N 20/200 open + EB', valid_min=self.tf)]

    rules = ('2-min. Picture of power (SMA20 rising, price & 20 above the 200), not in a wide state. The 1st or 2nd '
             'small red bar (range <= median of prior 10) right after green, pulling back near the 20 (low <= SMA20 + '
             '0.5 ATR) -> buy stop 1c over the red bar; stop 1c under it; trail bar-by-bar. GBI is the short mirror in '
             'a falling-20 regime.')

    def on_bar(self, sym, bars, ctx):
        f = frame(bars, ctx, sym, self.tf)
        if f is None or not self.in_window(ctx):
            return []
        df, i, p = f['df'], len(f['df']) - 1, self.p
        o, h, l, c = (df[k].iloc[i] for k in ('open', 'high', 'low', 'close'))
        s20, s200, a = f['s20'].iloc[i], f['s200'].iloc[i], f['atr'].iloc[i]
        sl = slope(f['s20'], i)
        small = (h - l) <= f['rng'].iloc[i - 10:i].median()
        if not small or wide(f, i, p):          # Velez: "I would stop the game at wide"
            return []
        col = np.sign(df['close'].values - df['open'].values)
        if c < o and sl > p['flat_slope_pct'] and c > s200 and s20 > s200 and l <= s20 + p['near_atr_rbi'] * a:
            k = 0
            while k < 5 and col[i - k] < 0:
                k += 1
            if 1 <= k <= p['max_counter'] and col[i - k] > 0:
                return [self.sig(ctx, sym, 'long', h + TICK, l - TICK, bar_plan(), f'RBI #{k}', valid_min=2 * self.tf)]
        if c > o and sl < -p['flat_slope_pct'] and c < s200 and s20 < s200 and h >= s20 - p['near_atr_rbi'] * a:
            k = 0
            while k < 5 and col[i - k] > 0:
                k += 1
            if 1 <= k <= p['max_counter'] and col[i - k] < 0:
                return [self.sig(ctx, sym, 'short', l - TICK, h + TICK, bar_plan(), f'GBI #{k}', valid_min=2 * self.tf)]
        return []

class Velez180(_VelezIntraday):
    sid = 'velez-180'
    title = 'Bullish / bearish 180 (two-bar U-turn)'
    tf = 2
    defaults = dict(COMMON, range_mult=1.0, require_pop=False)
    rules = ('2-min. Bar1 = fat red bar (body >= 70% of range, range >= avg range) at/near the SMA20 or SMA200. Buy stop '
             '1c over bar1 high valid for the NEXT bar only (the green bar must immediately wipe out the red, entry while '
             'it is still forming); stop 1c under the green bar low (proxy: 1c under bar1 low); trail bar-by-bar. Bear '
             '180 mirror.')
    approximations = ('Stop proxy: Velez stops 1c under the GREEN (entry) bar low, unknown when the order is armed; the '
                      'red bar low is used (slightly wider). require_pop (picture-of-power filter) is OFF: the 180 talk '
                      'shows 180s both near and away from the 20 without stating a 200-position rule (UNCERTAIN).')

    def on_bar(self, sym, bars, ctx):
        f = frame(bars, ctx, sym, self.tf)
        if f is None or not self.in_window(ctx):
            return []
        df, i, p = f['df'], len(f['df']) - 1, self.p
        b = df.iloc[i]
        r = b.high - b.low
        if r <= 0 or r < p['range_mult'] * f['mrng'].iloc[i] or abs(b.close - b.open) < p['eb_body_pct'] * r:
            return []
        a, s20, s200 = f['atr'].iloc[i], f['s20'].iloc[i], f['s200'].iloc[i]
        tol = p['near_atr'] * a
        sl = slope(f['s20'], i)
        def near(px):
            return abs(px - s20) <= tol or abs(px - s200) <= tol
        pb = df.iloc[i - 1]
        if (b.close > b.open and pb.close < pb.open and b.high > pb.high) or \
                (b.close < b.open and pb.close > pb.open and b.low < pb.low):
            return []                       # DEFAULT: this bar just completed a 180 - don't arm the opposite one
        if b.close < b.open and (near(b.low) or near(b.high)):          # fat red -> bull 180 armed
            if p['require_pop'] and not (sl > 0 and s20 > s200):
                return []
            return [self.sig(ctx, sym, 'long', b.high + TICK, b.low - TICK, bar_plan(), 'bullish 180 (fat red armed)',
                             valid_min=self.tf)]
        if b.close > b.open and (near(b.high) or near(b.low)):          # fat green -> bear 180 armed
            if p['require_pop'] and not (sl < 0 and s20 < s200):
                return []
            return [self.sig(ctx, sym, 'short', b.low - TICK, b.high + TICK, bar_plan(), 'bearish 180 (fat green armed)',
                             valid_min=self.tf)]
        return []

class VelezTail(_VelezIntraday):
    sid = 'velez-tail'
    title = 'Bottoming / topping tail after 3-5 bars (Pristine micro)'
    tf = 5
    warmup_sessions = 1
    defaults = dict(COMMON, min_bars=3, max_bars=5, tail_pct=0.5)
    rules = ('5-min. A 3-5 bar drop (3-5 consecutive red bars OR 3-5 consecutive lower highs), then a bar that makes a '
             'lower low but leaves a lower tail >= 50% of its range -> buy stop 1c over the tail bar; stop 1c under its '
             'low; trail bar-by-bar after two favourable bars. Topping tail after a 3-5 bar rally = short mirror.')

    def on_bar(self, sym, bars, ctx):
        if bars is None or len(bars) < 25 or not self.in_window(ctx):
            return []
        df, i, p = bars, len(bars) - 1, self.p
        b = df.iloc[i]; r = b.high - b.low
        if r <= 0:
            return []
        lower = min(b.open, b.close) - b.low
        upper = b.high - max(b.open, b.close)
        o, c, h, l = df['open'].values, df['close'].values, df['high'].values, df['low'].values
        def run(test):                       # consecutive bars before the tail bar satisfying test(j)
            k = 0
            while k < p['max_bars'] + 1 and i - 1 - k >= 1 and test(i - 1 - k):
                k += 1
            return k
        dn = max(run(lambda j: c[j] < o[j]), run(lambda j: h[j] < h[j - 1]))
        up = max(run(lambda j: c[j] > o[j]), run(lambda j: l[j] > l[j - 1]))
        if p['min_bars'] <= dn <= p['max_bars'] and b.low < l[i - 1] and lower >= p['tail_pct'] * r:
            return [self.sig(ctx, sym, 'long', b.high + TICK, b.low - TICK, bar_plan(trail_after_bars=2),
                             f'bottoming tail after {dn}-bar drop', valid_min=2 * self.tf)]
        if p['min_bars'] <= up <= p['max_bars'] and b.high > h[i - 1] and upper >= p['tail_pct'] * r:
            return [self.sig(ctx, sym, 'short', b.low - TICK, b.high + TICK, bar_plan(trail_after_bars=2),
                             f'topping tail after {up}-bar rally', valid_min=2 * self.tf)]
        return []

class VelezMR(_VelezIntraday):
    sid = 'velez-mr'
    title = 'Momentum reversal: Green Bar Reversal / Red Bar Reversal after a 3-5 bar run'
    tf = 5
    warmup_sessions = 1
    defaults = dict(COMMON, min_bars=3, max_bars=5)
    rules = ('5-min. After a steady 3-5 bar decline (consecutive red bars) a green bar closes above the prior red '
             "bar's open (it 'corrects' the red bar) -> buy when the next bar exceeds its high; stop 1c under the lower "
             'of the reversal bar / prior bar; trail bar-by-bar. Red Bar Reversal after 3-5 green bars = short.')

    def on_bar(self, sym, bars, ctx):
        if bars is None or len(bars) < 25 or not self.in_window(ctx):
            return []
        df, i, p = bars, len(bars) - 1, self.p
        col = np.sign(df['close'].values - df['open'].values)
        b, pb = df.iloc[i], df.iloc[i - 1]
        def run(sign):
            k = 0
            while k < p['max_bars'] + 1 and col[i - 1 - k] == sign:
                k += 1
            return k
        if col[i] > 0:
            k = run(-1)
            if p['min_bars'] <= k <= p['max_bars'] and b.close > pb.open:
                return [self.sig(ctx, sym, 'long', b.high + TICK, min(b.low, pb.low) - TICK, bar_plan(trail_after_bars=2),
                                 f'green bar reversal after {k} red', valid_min=self.tf)]
        if col[i] < 0:
            k = run(1)
            if p['min_bars'] <= k <= p['max_bars'] and b.close < pb.open:
                return [self.sig(ctx, sym, 'short', b.low - TICK, max(b.high, pb.high) + TICK, bar_plan(trail_after_bars=2),
                                 f'red bar reversal after {k} green', valid_min=self.tf)]
        return []

class VelezGuerrilla(Strategy):
    sid = 'velez-guer'
    expert = 'Oliver Velez'
    title = 'Guerrilla tactics: Gap-n-Snap / Gap-n-Crap / 20-20 plays (1-3 day)'
    style = 'swing'
    tf = 1
    universe = ()
    defaults = dict(min_price=20.0, min_range=1.0, gap=0.50, near=0.50, buf=0.07, target=2.0, max_days=3)
    rules = ('Daily setup: down 2 days in a row, wide-range day (>= $1.00) that opened in the top 20% and closed in the '
             'bottom 20% of its range. Gap-n-Snap: next day gaps down >= $0.50 under that low -> buy $0.07 over the '
             "prior low; stop $0.07 under today's low. Bullish 20/20: opens within $0.50 of the prior low -> wait 30 "
             "min, buy $0.07 over the first-30-min high; stop $0.07 under the lower of today's/yesterday's low. Exit at "
             '+$2.00 or the close of day 3. Gap-n-Crap / Bearish 20/20 are the short mirrors after 2 up days.')

    def prep(self, ctx):
        self.watch = {}
        p = self.p
        for sym in ctx.universes.get('core', []):
            d = ctx.daily.get(sym)
            if d is None or len(d) < 4:
                continue
            y, y1, y2 = d.iloc[-1], d.iloc[-2], d.iloc[-3]
            r = y.high - y.low
            if y.close < p['min_price'] or r < p['min_range']:
                continue
            if y.close < y1.close < y2.close and y.open >= y.low + 0.8 * r and y.close <= y.low + 0.2 * r:
                self.watch[sym] = dict(side='long', y=y)
            elif y.close > y1.close > y2.close and y.open <= y.low + 0.2 * r and y.close >= y.low + 0.8 * r:
                self.watch[sym] = dict(side='short', y=y)

    def on_bar(self, sym, bars, ctx):
        w = self.watch.get(sym)
        if not w:
            return []
        tod = ctx.today(bars)
        if tod is None or tod.empty:
            return []
        p, y, o = self.p, w['y'], float(tod['open'].iloc[0])
        plan = Plan(eod_flat=False, target_abs=p['target'], max_days=p['max_days'], exit_before_earnings=True)
        if w['side'] == 'long':
            if o <= y.low - p['gap'] and self.once(ctx, sym, 'gns'):
                return [self.sig(ctx, sym, 'long', y.low + p['buf'], tod['low'].min() - p['buf'], plan, 'Gap-n-Snap',
                                 stop_rule='lod', meta={'buf': p['buf']})]
            if abs(o - y.low) <= p['near'] and ctx.now >= ctx.at('10:00') and self.once(ctx, sym, '2020'):
                first = tod[tod.index < ctx.at('10:00')]
                return [self.sig(ctx, sym, 'long', first['high'].max() + p['buf'], min(tod['low'].min(), y.low) - p['buf'],
                                 plan, 'Bullish 20/20 play', stop_rule='lod_prev', meta={'buf': p['buf'], 'prev': float(y.low)})]
        else:
            if o >= y.high + p['gap'] and self.once(ctx, sym, 'gnc'):
                return [self.sig(ctx, sym, 'short', y.high - p['buf'], tod['high'].max() + p['buf'], plan, 'Gap-n-Crap',
                                 stop_rule='lod', meta={'buf': p['buf']})]
            if abs(o - y.high) <= p['near'] and ctx.now >= ctx.at('10:00') and self.once(ctx, sym, '2020'):
                first = tod[tod.index < ctx.at('10:00')]
                return [self.sig(ctx, sym, 'short', first['low'].min() - p['buf'], max(tod['high'].max(), y.high) + p['buf'],
                                 plan, 'Bearish 20/20 play', stop_rule='lod_prev', meta={'buf': p['buf'], 'prev': float(y.high)})]
        return []

class VelezPBS(Strategy):
    sid = 'velez-pbs'
    expert = 'Oliver Velez'
    title = 'Pristine Buy/Sell Setup swing (Key Buy / PBS action steps)'
    style = 'swing'
    tf = 1
    universe = ()
    defaults = dict(min_bars=3, max_bars=5, near_atr=0.25, max_days=5, new_high_within=8, new_high_lookback=20,
                    gap_wait='10:00', gap_buf=0.06)
    rules = ('Daily. 20-day SMA rising and above the 40-day; a recent new high (highest high of the prior 20 days, made '
             '<= 8 bars ago) followed by 3-5 consecutive lower-high days pulling back into the 20 (low <= SMA20 + 0.25 '
             "ATR). Buy stop 1c over the last pullback day's high; if the stock opens above that level, wait 30 min and "
             "buy $0.06 over the day's high (30-minute rule). Stop 1c under the lower of the entry day's low and the "
             "prior day's low. Minimum target = the prior pivot (new) high. From day 3 trail under each prior day's low; "
             'time exit day 5. PSS = short mirror.')

    def prep(self, ctx):
        self.watch = {}
        p = self.p
        for sym in ctx.universes.get('core', []):
            d = ctx.daily.get(sym)
            if d is None or len(d) < 60:
                continue
            s20, s40, a = sma(d.close, 20), sma(d.close, 40), atr(d, 14)
            i = len(d) - 1
            hs, ls = d.high.values, d.low.values
            k_dn = 0
            while k_dn < p['max_bars'] + 1 and hs[i - k_dn] < hs[i - k_dn - 1]:
                k_dn += 1
            k_up = 0
            while k_up < p['max_bars'] + 1 and ls[i - k_up] > ls[i - k_up - 1]:
                k_up += 1
            y = d.iloc[-1]
            lb = p['new_high_lookback']
            if (s20.iloc[i] > s20.iloc[i - 5] and s20.iloc[i] > s40.iloc[i] and p['min_bars'] <= k_dn <= p['max_bars']
                    and y.low <= s20.iloc[i] + p['near_atr'] * a.iloc[i]):
                piv = i - k_dn                                     # the bar the lower-high run started from
                if k_dn <= p['new_high_within'] and hs[piv] >= hs[piv - lb:piv].max():     # Key Buy criterion 1
                    self.watch[sym] = dict(side='long', trig=y.high + TICK, prev=float(y.low), tgt=float(hs[piv]), k=k_dn)
            elif (s20.iloc[i] < s20.iloc[i - 5] and s20.iloc[i] < s40.iloc[i] and p['min_bars'] <= k_up <= p['max_bars']
                  and y.high >= s20.iloc[i] - p['near_atr'] * a.iloc[i]):
                piv = i - k_up
                if k_up <= p['new_high_within'] and ls[piv] <= ls[piv - lb:piv].min():
                    self.watch[sym] = dict(side='short', trig=y.low - TICK, prev=float(y.high), tgt=float(ls[piv]), k=k_up)

    def on_bar(self, sym, bars, ctx):
        w = self.watch.get(sym)
        if not w:
            return []
        tod = ctx.today(bars)
        if tod is None or tod.empty:
            return []
        p, long_ = self.p, w['side'] == 'long'
        o = float(tod['open'].iloc[0])
        gapped = (o > w['trig']) if long_ else (o < w['trig'])
        plan = Plan(eod_flat=False, target_px=w['tgt'], trail='prevday', trail_start_day=3, max_days=p['max_days'],
                    exit_before_earnings=True)
        tag = f"{'PBS' if long_ else 'PSS'} after {w['k']}-bar pullback"
        if not gapped:
            if not self.once(ctx, sym, 'pbs'):
                return []
            if (w['tgt'] - w['trig']) * (1 if long_ else -1) <= 0:          # no room to the minimum target
                return []
            return [self.sig(ctx, sym, w['side'], w['trig'], w['prev'] - TICK if long_ else w['prev'] + TICK, plan, tag,
                             stop_rule='lod_prev', meta={'buf': TICK, 'prev': w['prev']})]
        # Tools & Tactics ch.14: gap over the trigger -> 30-minute rule (new day high after 30 min, stop under the LOD)
        if ctx.now < ctx.at(p['gap_wait']) or not self.once(ctx, sym, 'pbs30'):
            return []
        first = tod[tod.index < ctx.at(p['gap_wait'])]
        if first.empty:
            return []
        if long_:
            trig, stop = float(first['high'].max()) + p['gap_buf'], float(tod['low'].min()) - p['gap_buf']
        else:
            trig, stop = float(first['low'].min()) - p['gap_buf'], float(tod['high'].max()) + p['gap_buf']
        if (w['tgt'] - trig) * (1 if long_ else -1) <= 0:
            plan.target_px = None                                       # already past the pivot: trail/time only
        return [self.sig(ctx, sym, w['side'], trig, stop, plan, tag + ' (30-min gap rule)', stop_rule='lod',
                         meta={'buf': p['gap_buf']})]
