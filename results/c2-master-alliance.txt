    def record_result(
        self,
        won: bool,
        expert: str,
        trading_date: str,
    ):
        scopes = [
            "account",
            f"expert:{expert}",
        ]

        for scope in scopes:
            row = self._get(scope)

            current_streak = (
                int(row["loss_streak"])
                if row
                and row["trading_date"]
                == trading_date
                else 0
            )

            if won:
                self._save(
                    scope,
                    0,
                    None,
                    trading_date,
                )
                continue

            new_streak = (
                current_streak + 1
            )

            blocked_until = None

            if (
                new_streak
                >= self.loss_limit
            ):
                blocked_until = (
                    self._now()
                    + timedelta(
                        minutes=(
                            self.cooldown_minutes
                        )
                    )
                ).isoformat()

            self._save(
                scope,
                new_streak,
                blocked_until,
                trading_date,
            )

    def blocked(
        self,
        expert: str,
        trading_date: str,
    ) -> tuple[bool, str]:
        self.reset_if_new_day(
            trading_date
        )

        now = self._now()

        for scope in (
            "account",
            f"expert:{expert}",
        ):
            row = self._get(scope)

            if not row:
                continue

            blocked_until = row[
                "blocked_until"
            ]

            if not blocked_until:
                continue

            expiry = datetime.fromisoformat(
                blocked_until
            )

            if now < expiry:
                return (
                    True,
                    (
                        f"{scope}_cooldown_until_"
                        f"{expiry.isoformat()}"
                    ),
                )

        return False, "clear"

def safe_divide(
    numerator: float,
    denominator: float,
) -> float:
    if abs(denominator) < 1e-12:
        return 0.0

    return numerator / denominator


def maximum_drawdown(
    results: pd.Series,
) -> float:
    if results.empty:
        return 0.0

    equity = np.concatenate(
        (
            np.array([0.0]),
            results.cumsum().to_numpy(
                dtype=float
            ),
        )
    )

    peaks = np.maximum.accumulate(equity)

    return float(
        np.max(peaks - equity)
    )


def longest_loss_streak(
    results: pd.Series,
) -> int:
    longest = 0
    current = 0

    for value in results:
        if value < 0:
            current += 1
            longest = max(
                longest,
                current,
            )
        else:
            current = 0

    return longest


def statistics(
    results: pd.Series,
) -> dict[str, Any]:
    values = pd.to_numeric(
        results,
        errors="coerce",
    ).dropna()

    if values.empty:
        return {
            "trades": 0,
            "win_rate": 0.0,
            "profit_factor": 0.0,
            "expectancy_r": 0.0,
            "net_r": 0.0,
            "maximum_drawdown_r": 0.0,
            "longest_loss_streak": 0,
        }

    winning = values.loc[
        values > 0
    ]

    losing = values.loc[
        values < 0
    ]

    gross_profit = float(
        winning.sum()
    )

    gross_loss = float(
        -losing.sum()
    )

    profit_factor = (
        gross_profit / gross_loss
        if gross_loss > 0
        else (
            float("inf")
            if gross_profit > 0
            else 0.0
        )
    )

    return {
        "trades": int(len(values)),
        "win_rate": float(
            (values > 0).mean()
        ),
        "profit_factor": float(
            profit_factor
        ),
        "expectancy_r": float(
            values.mean()
        ),
        "net_r": float(
            values.sum()
        ),
        "maximum_drawdown_r": (
            maximum_drawdown(values)
        ),
        "longest_loss_streak": (
            longest_loss_streak(values)
        ),
    }


def load_candidates(
    path: Path,
) -> pd.DataFrame:
    if not path.exists():
        raise FileNotFoundError(
            f"Missing candidate file: {path}"
        )

    frame = pd.read_csv(
        path,
        low_memory=False,
    )

    required = {
        "expert",
        "symbol",
        "direction",
        "stock_regime",
        "time_bucket",
        "entry_time",
        "result_r",
        "mae_r",
        "mfe_r",
    }

    missing = sorted(
        required - set(frame.columns)
    )

    if missing:
        raise ValueError(
            "Missing columns: "
            + ", ".join(missing)
        )

    frame["entry_time"] = pd.to_datetime(
        frame["entry_time"],
        errors="coerce",
        utc=True,
    )

    frame = frame.loc[
        frame["entry_time"].notna()
    ].copy()

    frame["trading_date"] = (
        frame["entry_time"]
        .dt.tz_convert("America/New_York")
        .dt.normalize()
        .dt.tz_localize(None)
    )

    for column in (
        "expert",
        "direction",
        "stock_regime",
        "time_bucket",
    ):
        frame[column] = (
            frame[column]
            .fillna("unknown")
            .astype(str)
            .str.strip()
            .str.lower()
        )

    frame["symbol"] = (
        frame["symbol"]
        .fillna("unknown")
        .astype(str)
        .str.strip()
        .str.upper()
    )

    for column in (
        "result_r",
        "mae_r",
        "mfe_r",
    ):
        frame[column] = pd.to_numeric(
            frame[column],
            errors="coerce",
        )

    frame = frame.loc[
        frame["result_r"].notna()
    ].copy()

    frame["adverse_r"] = (
        frame["mae_r"]
        .fillna(0.0)
        .abs()
    )

    frame["favorable_r"] = (
        frame["mfe_r"]
        .fillna(0.0)
        .clip(lower=0.0)
    )

    return frame.sort_values(
        "entry_time"
    ).reset_index(
        drop=True
    )


def select_segment(
    frame: pd.DataFrame,
    definition: dict[str, Any],
) -> pd.DataFrame:
    mask = pd.Series(
        True,
        index=frame.index,
    )

    if "expert" in definition:
        mask &= (
            frame["expert"]
            == definition["expert"].lower()
        )

    if "direction" in definition:
        mask &= (
            frame["direction"]
            == definition["direction"].lower()
        )

    for column in (
        "stock_regime",
        "time_bucket",
    ):
        accepted = definition.get(column)

        if accepted:
            mask &= frame[column].isin(
                {
                    str(value).lower()
                    for value in accepted
                }
            )

    symbols = definition.get("symbol")

    if symbols:
        mask &= frame["symbol"].isin(
            {
                str(value).upper()
                for value in symbols
            }
        )

    excluded = definition.get(
        "exclude_time_bucket"
    )

    if excluded:
        mask &= ~frame[
            "time_bucket"
        ].isin(
            {
                str(value).lower()
                for value in excluded
            }
        )

    return frame.loc[
        mask
    ].copy()


def simulate_results(
    frame: pd.DataFrame,
    stop_r: float,
    target_r: float,
    extra_cost_r: float,
) -> pd.Series:
    adverse = frame[
        "adverse_r"
    ].to_numpy(
        dtype=float
    )

    favorable = frame[
        "favorable_r"
    ].to_numpy(
        dtype=float
    )

    original = frame[
        "result_r"
    ].to_numpy(
        dtype=float
    )

    stop_hit = adverse >= stop_r
    target_hit = favorable >= target_r

    # Conservative rule: if both were reachable,
    # count the stop as occurring first.
    optimized = np.where(
        stop_hit,
        -stop_r,
        np.where(
            target_hit,
            target_r,
            original,
        ),
    )

    optimized -= extra_cost_r

    return pd.Series(
        optimized,
        index=frame.index,
        dtype=float,
    )


def build_grid(
    settings: dict[str, Any],
) -> list[tuple[float, float]]:
    stops = settings.get(
        "stop_r_grid",
        [
            0.60,
            0.75,
            0.90,
            1.00,
            1.10,
            1.25,
            1.50,
        ],
    )

    targets = settings.get(
        "target_r_grid",
        [
            0.75,
            1.00,
            1.25,
            1.50,
            1.75,
            2.00,
            2.50,
            3.00,
        ],
    )

    minimum_rr = float(
        settings.get(
            "minimum_reward_risk",
            1.0,
        )
    )

    return [
        (
            float(stop_r),
            float(target_r),
        )
        for stop_r in stops
        for target_r in targets
        if (
            float(target_r)
            / float(stop_r)
            >= minimum_rr
        )
    ]


def score_result(
    stats: dict[str, Any],
    minimum_trades: int,
    drawdown_penalty: float,
) -> float:
    if stats["trades"] < minimum_trades:
        return -1_000_000.0

    pf = min(
        stats["profit_factor"],
        3.0,
    )

    expectancy = (
        stats["expectancy_r"]
        * 4.0
    )

    drawdown = (
        stats["maximum_drawdown_r"]
        * drawdown_penalty
    )

    return float(
        pf
        + expectancy
        - drawdown
    )


def optimize(
    training: pd.DataFrame,
    grid: list[tuple[float, float]],
    settings: dict[str, Any],
) -> tuple[
    tuple[float, float] | None,
    dict[str, Any],
    pd.DataFrame,
]:
    rows: list[dict[str, Any]] = []

    best_parameters = None
    best_statistics: dict[str, Any] = {}
    best_score = -float("inf")

    minimum_trades = int(
        settings.get(
            "optimization_minimum_training_trades",
            30,
        )
    )

    drawdown_penalty = float(
        settings.get(
            "drawdown_penalty",
            0.015,
        )
    )

    extra_cost = float(
        settings.get(
            "additional_cost_r",
            0.0,
        )
    )

    for stop_r, target_r in grid:
        results = simulate_results(
            training,
            stop_r,
            target_r,
            extra_cost,
        )

        stats = statistics(results)

        score = score_result(
            stats,
            minimum_trades,
            drawdown_penalty,
        )

        rows.append(
            {
                "stop_r": stop_r,
                "target_r": target_r,
                "reward_risk": (
                    target_r / stop_r
                ),
                "score": score,
                **stats,
            }
        )

        if score > best_score:
            best_score = score
            best_parameters = (
                stop_r,
                target_r,
            )
            best_statistics = stats

    grid_frame = pd.DataFrame(rows)

    if not grid_frame.empty:
        grid_frame = grid_frame.sort_values(
            [
                "score",
                "expectancy_r",
                "profit_factor",
            ],
            ascending=[
                False,
                False,
                False,
            ],
        ).reset_index(
            drop=True
        )

    return (
        best_parameters,
        best_statistics,
        grid_frame,
    )


def chronological_split(
    frame: pd.DataFrame,
    fraction: float,
) -> tuple[
    pd.DataFrame,
    pd.DataFrame,
]:
    ordered = frame.sort_values(
        "entry_time"
    ).reset_index(
        drop=True
    )

    cut = int(
        len(ordered) * fraction
    )

    cut = max(
        1,
        min(
            cut,
            len(ordered) - 1,
        ),
    )

    return (
        ordered.iloc[:cut].copy(),
        ordered.iloc[cut:].copy(),
    )


def walk_forward(
    frame: pd.DataFrame,
    grid: list[tuple[float, float]],
    settings: dict[str, Any],
) -> pd.DataFrame:
    training_days = int(
        settings.get(
            "walk_forward_training_days",
            45,
        )
    )

    testing_days = int(
        settings.get(
            "walk_forward_testing_days",
            15,
        )
    )

    minimum_training = int(
        settings.get(
            "walk_forward_minimum_training_trades",
            20,
        )
    )

    minimum_testing = int(
        settings.get(
            "walk_forward_minimum_testing_trades",
            5,
        )
    )

    extra_cost = float(
        settings.get(
            "additional_cost_r",
            0.0,
        )
    )

    rows: list[dict[str, Any]] = []

    start = frame[
        "trading_date"
    ].min()

    final = frame[
        "trading_date"
    ].max()

    window = 0

    while True:
        training_end = (
            start
            + pd.Timedelta(
                days=training_days
            )
        )

        testing_end = (
            training_end
            + pd.Timedelta(
                days=testing_days
            )
        )

        if training_end > final:
            break

        training = frame.loc[
            (
                frame["trading_date"] >= start
            )
            & (
                frame["trading_date"]
                < training_end
            )
        ]

        testing = frame.loc[
            (
                frame["trading_date"]
                >= training_end
            )
            & (
                frame["trading_date"]
                < testing_end
            )
        ]

        if (
            len(training) >= minimum_training
            and len(testing) >= minimum_testing
        ):
            (
                parameters,
                training_stats,
                _,
            ) = optimize(
                training,
                grid,
                {
                    **settings,
                    "optimization_minimum_training_trades": (
                        minimum_training
                    ),
                },
            )

            if parameters is not None:
                stop_r, target_r = parameters

                testing_results = simulate_results(
                    testing,
                    stop_r,
                    target_r,
                    extra_cost,
                )

                testing_stats = statistics(
                    testing_results
                )

                window += 1

                rows.append(
                    {
                        "window": window,
                        "training_start": (
                            start.date()
                        ),
                        "testing_start": (
                            training_end.date()
                        ),
                        "testing_end": (
                            testing_end.date()
                        ),
                        "selected_stop_r": stop_r,
                        "selected_target_r": (
                            target_r
                        ),
                        "training_trades": (
                            training_stats["trades"]
                        ),
                        "training_profit_factor": (
                            training_stats[
                                "profit_factor"
                            ]
                        ),
                        "training_expectancy_r": (
                            training_stats[
                                "expectancy_r"
                            ]
                        ),
                        "testing_trades": (
                            testing_stats["trades"]
                        ),
                        "testing_win_rate": (
                            testing_stats["win_rate"]
                        ),
                        "testing_profit_factor": (
                            testing_stats[
                                "profit_factor"
                            ]
                        ),
                        "testing_expectancy_r": (
                            testing_stats[
                                "expectancy_r"
                            ]
                        ),
                        "testing_net_r": (
                            testing_stats["net_r"]
                        ),
                        "testing_drawdown_r": (
                            testing_stats[
                                "maximum_drawdown_r"
                            ]
                        ),
                        "testing_profitable": (
                            testing_stats["net_r"]
                            > 0.0
                        ),
                    }
                )

        start += pd.Timedelta(
            days=testing_days
        )

        if start > final:
            break

    return pd.DataFrame(rows)


def monte_carlo(
    results: pd.Series,
    simulations: int,
    seed: int,
) -> dict[str, float]:
    values = pd.to_numeric(
        results,
        errors="coerce",
    ).dropna().to_numpy(
        dtype=float
    )

    if values.size == 0:
        return {
            "profitable_probability": 0.0,
            "median_ending_r": 0.0,
            "drawdown_95_r": 0.0,
        }

    rng = np.random.default_rng(seed)

    endings = np.empty(simulations)
    drawdowns = np.empty(simulations)

    for index in range(simulations):
        sample = rng.choice(
            values,
            size=values.size,
            replace=True,
        )

        equity = np.concatenate(
            (
                np.array([0.0]),
                np.cumsum(sample),
            )
        )

        peaks = np.maximum.accumulate(
            equity
        )

        endings[index] = equity[-1]

        drawdowns[index] = np.max(
            peaks - equity
        )

    return {
        "profitable_probability": float(
            np.mean(endings > 0)
        ),
        "median_ending_r": float(
            np.median(endings)
        ),
        "drawdown_95_r": float(
            np.percentile(
                drawdowns,
                95,
            )
        ),
    }


def qualify(
    test_stats: dict[str, Any],
    walk: pd.DataFrame,
    simulation: dict[str, float],
    settings: dict[str, Any],
) -> tuple[str, str]:
    reasons: list[str] = []

    minimum_test_trades = int(
        settings.get(
            "qualification_minimum_test_trades",
            25,
        )
    )

    minimum_pf = float(
        settings.get(
            "qualification_minimum_profit_factor",
            1.20,
        )
    )

    minimum_expectancy = float(
        settings.get(
            "qualification_minimum_expectancy_r",
            0.08,
        )
    )

    maximum_drawdown = float(
        settings.get(
            "qualification_maximum_drawdown_r",
            12.0,
        )
    )

    minimum_windows = int(
        settings.get(
            "qualification_minimum_walk_forward_windows",
            2,
        )
    )

    minimum_window_rate = float(
        settings.get(
            "qualification_minimum_profitable_window_rate",
            0.60,
        )
    )

    minimum_mc = float(
        settings.get(
            "qualification_minimum_monte_carlo_probability",
            0.70,
        )
    )

    maximum_mc_drawdown = float(
        settings.get(
            "qualification_maximum_monte_carlo_95_drawdown_r",
            18.0,
        )
    )

    window_rate = (
        float(
            walk["testing_profitable"].mean()
        )
        if not walk.empty
        else 0.0
    )

    if (
        test_stats["trades"]
        < minimum_test_trades
    ):
        reasons.append(
            f"test trades {test_stats['trades']} "
            f"< {minimum_test_trades}"
        )

    if (
        test_stats["profit_factor"]
        < minimum_pf
    ):
        reasons.append(
            f"test PF {test_stats['profit_factor']:.2f} "
            f"< {minimum_pf:.2f}"
        )

    if (
        test_stats["expectancy_r"]
        < minimum_expectancy
    ):
        reasons.append(
            f"test expectancy "
            f"{test_stats['expectancy_r']:+.3f}R "
            f"< {minimum_expectancy:+.3f}R"
        )

    if (
        test_stats["maximum_drawdown_r"]
        > maximum_drawdown
    ):
        reasons.append(
            f"test drawdown "
            f"{test_stats['maximum_drawdown_r']:.2f}R "
            f"> {maximum_drawdown:.2f}R"
        )

    if len(walk) < minimum_windows:
        reasons.append(
            f"walk-forward windows {len(walk)} "
            f"< {minimum_windows}"
        )

    if window_rate < minimum_window_rate:
        reasons.append(
            f"profitable window rate "
            f"{window_rate:.1%} "
            f"< {minimum_window_rate:.1%}"
        )

    if (
        simulation[
            "profitable_probability"
        ]
        < minimum_mc
    ):
        reasons.append(
            f"Monte Carlo probability "
            f"{simulation['profitable_probability']:.1%} "
            f"< {minimum_mc:.1%}"
        )

    if (
        simulation["drawdown_95_r"]
        > maximum_mc_drawdown
    ):
        reasons.append(
            f"Monte Carlo 95% drawdown "
            f"{simulation['drawdown_95_r']:.2f}R "
            f"> {maximum_mc_drawdown:.2f}R"
        )

    if not reasons:
        return (
            "validated_candidate",
            "all validation standards passed",
        )

    if (
        test_stats["expectancy_r"] > 0
        and test_stats["profit_factor"] > 1
    ):
        return (
            "research",
            "; ".join(reasons),
        )

    return (
        "reject",
        "; ".join(reasons),
    )


def markdown_table(
    frame: pd.DataFrame,
) -> str:
    if frame.empty:
        return "_No results._"

    columns = [
        "segment",
        "expert",
        "status",
        "total_trades",
        "selected_stop_r",
        "selected_target_r",
        "test_trades",
        "test_win_rate",
        "test_profit_factor",
        "test_expectancy_r",
        "test_net_r",
        "test_drawdown_r",
        "walk_forward_windows",
        "profitable_window_rate",
        "monte_carlo_profitable_probability",
        "monte_carlo_95_drawdown_r",
        "reasons",
    ]

    columns = [
        column
        for column in columns
        if column in frame.columns
    ]

    headings = [
        column.replace(
            "_",
            " ",
        ).title()
        for column in columns
    ]

    lines = [
        "| "
        + " | ".join(headings)
        + " |",
        "|"
        + "|".join(
            "---"
            for _ in columns
        )
        + "|",
    ]

    percent_columns = {
        "test_win_rate",
        "profitable_window_rate",
        "monte_carlo_profitable_probability",
    }

    for _, row in frame.iterrows():
        values: list[str] = []

        for column in columns:
            item = row[column]

            if pd.isna(item):
                text = ""

            elif column in percent_columns:
                text = f"{float(item):.1%}"

            elif column in {
                "test_expectancy_r",
                "test_net_r",
                "test_drawdown_r",
                "monte_carlo_95_drawdown_r",
            }:
                text = f"{float(item):.3f}"

            elif column in {
                "selected_stop_r",
                "selected_target_r",
                "test_profit_factor",
            }:
                text = f"{float(item):.3f}"

            else:
                text = str(item)

            values.append(
                text.replace(
                    "|",
                    "\\|",
                )
            )

        lines.append(
            "| "
            + " | ".join(values)
            + " |"
        )

    return "\n".join(lines)

def load_candidates(
    path: Path,
) -> pd.DataFrame:
    if not path.exists():
        raise FileNotFoundError(
            f"Missing Part 2G candidate file: {path}"
        )

    frame = pd.read_csv(
        path,
        low_memory=False,
    )

    required = {
        "expert",
        "symbol",
        "direction",
        "stock_regime",
        "time_bucket",
        "entry_time",
        "result_r",
        "mae_r",
        "mfe_r",
    }

    missing = sorted(
        required - set(frame.columns)
    )

    if missing:
        raise ValueError(
            "Candidate file missing columns: "
            + ", ".join(missing)
        )

    frame["entry_time"] = pd.to_datetime(
        frame["entry_time"],
        errors="coerce",
        utc=True,
    )

    frame = frame.loc[
        frame["entry_time"].notna()
    ].copy()

    if "exit_time" in frame.columns:
        frame["exit_time"] = pd.to_datetime(
            frame["exit_time"],
            errors="coerce",
            utc=True,
        )
    else:
        frame["exit_time"] = pd.NaT

    frame["trading_date"] = (
        frame["entry_time"]
        .dt.tz_convert(
            "America/New_York"
        )
        .dt.normalize()
        .dt.tz_localize(None)
    )

    for column in (
        "expert",
        "direction",
        "stock_regime",
        "time_bucket",
        "sector",
    ):
        if column not in frame.columns:
            frame[column] = "unknown"

        frame[column] = (
            frame[column]
            .fillna("unknown")
            .astype(str)
            .str.strip()
            .str.lower()
        )

    frame["symbol"] = (
        frame["symbol"]
        .fillna("unknown")
        .astype(str)
        .str.strip()
        .str.upper()
    )

    for column in (
        "result_r",
        "mae_r",
        "mfe_r",
        "confidence",
    ):
        if column not in frame.columns:
            frame[column] = np.nan

        frame[column] = pd.to_numeric(
            frame[column],
            errors="coerce",
        )

    frame = frame.loc[
        frame["result_r"].notna()
    ].copy()

    frame["adverse_r"] = (
        frame["mae_r"]
        .fillna(0.0)
        .abs()
    )

    frame["favorable_r"] = (
        frame["mfe_r"]
        .fillna(0.0)
        .clip(lower=0.0)
    )

    return frame.sort_values(
        [
            "entry_time",
            "symbol",
            "expert",
        ]
    ).reset_index(
        drop=True
    )


def segment_mask(
    frame: pd.DataFrame,
    definition: dict[str, Any],
) -> pd.Series:
    mask = frame["expert"].eq(
        definition["expert"]
    )

    symbols = definition.get(
        "symbols"
    )

    if symbols:
        mask &= frame["symbol"].isin(
            symbols
        )

    direction = definition.get(
        "direction"
    )

    if direction:
        mask &= frame["direction"].eq(
            direction
        )

    regimes = definition.get(
        "stock_regimes"
    )

    if regimes:
        mask &= frame[
            "stock_regime"
        ].isin(
            regimes
        )

    buckets = definition.get(
        "time_buckets"
    )

    if buckets:
        mask &= frame[
            "time_bucket"
        ].isin(
            {
                value.lower()
                for value in buckets
            }
        )

    return mask


def simulate_segment_results(
    frame: pd.DataFrame,
    stop_r: float,
    target_r: float,
    extra_cost_r: float,
) -> pd.Series:
    stop_hit = (
        frame["adverse_r"]
        .to_numpy(dtype=float)
        >= stop_r
    )

    target_hit = (
        frame["favorable_r"]
        .to_numpy(dtype=float)
        >= target_r
    )

    original = frame[
        "result_r"
    ].to_numpy(
        dtype=float
    )

    results = np.where(
        stop_hit,
        -stop_r,
        np.where(
            target_hit,
            target_r,
            original,
        ),
    )

    return pd.Series(
        results - extra_cost_r,
        index=frame.index,
        dtype=float,
    )


def build_segment_trades(
    candidates: pd.DataFrame,
    extra_cost_r: float,
) -> pd.DataFrame:
    outputs: list[pd.DataFrame] = []

    for (
        segment_name,
        definition,
    ) in VALIDATED_SEGMENTS.items():
        subset = candidates.loc[
            segment_mask(
                candidates,
                definition,
            )
        ].copy()

        if subset.empty:
            continue

        subset["segment"] = segment_name
        subset["priority"] = int(
            definition["priority"]
        )

        subset["portfolio_stop_r"] = float(
            definition["stop_r"]
        )

        subset["portfolio_target_r"] = float(
            definition["target_r"]
        )

        subset["portfolio_result_r"] = (
            simulate_segment_results(
                subset,
                float(
                    definition["stop_r"]
                ),
                float(
                    definition["target_r"]
                ),
                extra_cost_r,
            )
        )

        outputs.append(
            subset
        )

    if not outputs:
        return pd.DataFrame()

    combined = pd.concat(
        outputs,
        ignore_index=True,
    )

    return combined.sort_values(
        [
            "entry_time",
            "symbol",
            "priority",
        ],
        ascending=[
            True,
            True,
            True,
        ],
    ).reset_index(
        drop=True
    )


def remove_duplicate_membership(
    frame: pd.DataFrame,
) -> pd.DataFrame:
    if frame.empty:
        return frame

    return (
        frame.sort_values(
            [
                "priority",
                "segment",
            ]
        )
        .drop_duplicates(
            subset=[
                "entry_time",
                "symbol",
                "expert",
                "direction",
            ],
            keep="first",
        )
        .sort_values(
            [
                "entry_time",
                "priority",
                "symbol",
            ]
        )
        .reset_index(
            drop=True
        )
    )


def resolve_conflicts(
    frame: pd.DataFrame,
    policy: str,
) -> tuple[
    pd.DataFrame,
    pd.DataFrame,
]:
    if frame.empty:
        return (
            frame.copy(),
            pd.DataFrame(),
        )

    working = frame.copy()

    working = working.sort_values(
        [
            "entry_time",
            "symbol",
            "priority",
            "segment",
        ],
        ascending=[
            True,
            True,
            True,
            True,
        ],
    ).reset_index(
        drop=True
    )

    group_columns = [
        "entry_time",
        "symbol",
    ]

    working[
        "_direction_count"
    ] = (
        working.groupby(
            group_columns,
            observed=True,
            sort=False,
        )[
            "direction"
        ]
        .transform(
            "nunique"
        )
    )

    working[
        "_group_rank"
    ] = (
        working.groupby(
            group_columns,
            observed=True,
            sort=False,
        )
        .cumcount()
    )

    same_direction = (
        working[
            "_direction_count"
        ]
        == 1
    )

    conflicting = (
        working[
            "_direction_count"
        ]
        > 1
    )

    if policy == "skip":
        accepted_mask = (
            same_direction
            & (
                working[
                    "_group_rank"
                ]
                == 0
            )
        )

        rejected_reason = np.where(
            conflicting,
            "direction_conflict",
            "duplicate_same_direction",
        )

    else:
        accepted_mask = (
            working[
                "_group_rank"
            ]
            == 0
        )

        rejected_reason = np.where(
            conflicting,
            "lost_priority_conflict",
            "duplicate_same_direction",
        )

    accepted = (
        working.loc[
            accepted_mask
        ]
        .drop(
            columns=[
                "_direction_count",
                "_group_rank",
            ]
        )
        .sort_values(
            [
                "entry_time",
                "priority",
                "symbol",
            ]
        )
        .reset_index(
            drop=True
        )
    )

    rejected = working.loc[
        ~accepted_mask
    ].copy()

    if not rejected.empty:
        rejected[
            "portfolio_rejection_reason"
        ] = rejected_reason[
            ~accepted_mask.to_numpy()
        ]

        rejected = (
            rejected.drop(
                columns=[
                    "_direction_count",
                    "_group_rank",
                ]
            )
            .reset_index(
                drop=True
            )
        )

    else:
        rejected = pd.DataFrame(
            columns=list(
                frame.columns
            )
            + [
                "portfolio_rejection_reason"
            ]
        )

    return accepted, rejected



def apply_constraints(
    frame: pd.DataFrame,
    *,
    maximum_positions: int,
    maximum_sector_positions: int,
    maximum_daily_trades: int,
    maximum_daily_loss_r: float,
) -> tuple[
    pd.DataFrame,
    pd.DataFrame,
]:
    if frame.empty:
        return frame, pd.DataFrame()

    accepted = []
    rejected = []

    open_positions: list[
        dict[str, Any]
    ] = []

    daily_trade_count: dict[
        str,
        int,
    ] = {}

    daily_net_r: dict[
        str,
        float,
    ] = {}

    for _, row in frame.sort_values(
        [
            "entry_time",
            "priority",
        ]
    ).iterrows():
        timestamp = row[
            "entry_time"
        ]

        trading_date = str(
            row[
                "trading_date"
            ].date()
        )

        open_positions = [
            position
            for position in open_positions
            if (
                pd.isna(
                    position[
                        "exit_time"
                    ]
                )
                or position[
                    "exit_time"
                ] > timestamp
            )
        ]

        count = daily_trade_count.get(
            trading_date,
            0,
        )

        net_r = daily_net_r.get(
            trading_date,
            0.0,
        )

        reason = None

        if count >= maximum_daily_trades:
            reason = (
                "maximum_daily_trades"
            )

        elif (
            net_r
            <= -abs(
                maximum_daily_loss_r
            )
        ):
            reason = "daily_loss_limit"

        elif (
            len(open_positions)
            >= maximum_positions
        ):
            reason = (
                "maximum_simultaneous_positions"
            )

        else:
            sector = row.get(
                "sector",
                "unknown",
            )

            sector_count = sum(
                position["sector"]
                == sector
                for position in open_positions
            )

            if (
                sector_count
                >= maximum_sector_positions
            ):
                reason = (
                    "maximum_sector_positions"
                )

        if reason is not None:
            rejected_row = row.copy()

            rejected_row[
                "portfolio_rejection_reason"
            ] = reason

            rejected.append(
                rejected_row
            )

            continue

        accepted.append(
            row
        )

        daily_trade_count[
            trading_date
        ] = count + 1

        daily_net_r[
            trading_date
        ] = (
            net_r
            + float(
                row[
                    "portfolio_result_r"
                ]
            )
        )

        open_positions.append(
            {
                "exit_time": row.get(
                    "exit_time"
                ),
                "sector": row.get(
                    "sector",
                    "unknown",
                ),
            }
        )

    return (
        pd.DataFrame(accepted),
        pd.DataFrame(rejected),
    )


def statistics(
    results: pd.Series,
) -> dict[str, Any]:
    values = pd.to_numeric(
        results,
        errors="coerce",
    ).dropna()

    if values.empty:
        return {
            "trades": 0,
            "wins": 0,
            "losses": 0,
            "win_rate": 0.0,
            "profit_factor": 0.0,
            "expectancy_r": 0.0,
            "net_r": 0.0,
            "maximum_drawdown_r": 0.0,
            "longest_loss_streak": 0,
        }

    wins = values.loc[
        values > 0
    ]

    losses = values.loc[
        values < 0
    ]

    gross_profit = float(
        wins.sum()
    )

    gross_loss = float(
        -losses.sum()
    )

    equity = np.concatenate(
        (
            [0.0],
            values.cumsum().to_numpy(
                dtype=float
            ),
        )
    )

    peaks = np.maximum.accumulate(
        equity
    )

    drawdown = float(
        np.max(
            peaks - equity
        )
    )

    longest = 0
    current = 0

    for value in values:
        if value < 0:
            current += 1
            longest = max(
                longest,
                current,
            )
        else:
            current = 0

    return {
        "trades": int(
            len(values)
        ),
        "wins": int(
            len(wins)
        ),
        "losses": int(
            len(losses)
        ),
        "win_rate": float(
            (values > 0).mean()
        ),
        "profit_factor": (
            gross_profit / gross_loss
            if gross_loss > 0
            else (
                float("inf")
                if gross_profit > 0
                else 0.0
            )
        ),
        "expectancy_r": float(
            values.mean()
        ),
        "net_r": float(
            values.sum()
        ),
        "maximum_drawdown_r": (
            drawdown
        ),
        "longest_loss_streak": (
            longest
        ),
    }


def grouped_statistics(
    frame: pd.DataFrame,
    column: str,
) -> pd.DataFrame:
    rows = []

    if (
        frame.empty
        or column not in frame.columns
    ):
        return pd.DataFrame()

    for value, group in frame.groupby(
        column,
        observed=True,
        dropna=False,
    ):
        rows.append(
            {
                column: value,
                **statistics(
                    group[
                        "portfolio_result_r"
                    ]
                ),
            }
        )

    return (
        pd.DataFrame(rows)
        .sort_values(
            [
                "expectancy_r",
                "profit_factor",
            ],
            ascending=[
                False,
                False,
            ],
        )
        .reset_index(
            drop=True
        )
    )


def daily_statistics(
    frame: pd.DataFrame,
) -> pd.DataFrame:
    if frame.empty:
        return pd.DataFrame()

    daily = (
        frame.groupby(
            "trading_date",
            as_index=False,
        )
        .agg(
            trades=(
                "portfolio_result_r",
                "size",
            ),
            net_r=(
                "portfolio_result_r",
                "sum",
            ),
        )
        .sort_values(
            "trading_date"
        )
    )

    daily["profitable"] = (
        daily["net_r"] > 0
    )

    return daily


def chronological_holdout(
    frame: pd.DataFrame,
    fraction: float,
) -> tuple[
    pd.DataFrame,
    pd.DataFrame,
]:
    ordered = frame.sort_values(
        "entry_time"
    ).reset_index(
        drop=True
    )

    cut = int(
        len(ordered)
        * fraction
    )

    cut = max(
        1,
        min(
            cut,
            len(ordered) - 1,
        ),
    )

    return (
        ordered.iloc[
            :cut
        ].copy(),
        ordered.iloc[
            cut:
        ].copy(),
    )


def monte_carlo(
    results: pd.Series,
    simulations: int,
    seed: int,
) -> dict[str, Any]:
    values = pd.to_numeric(
        results,
        errors="coerce",
    ).dropna().to_numpy(
        dtype=float
    )

    if values.size == 0:
        return {
            "simulations": simulations,
            "profitable_probability": 0.0,
            "loss_probability": 1.0,
            "median_ending_r": 0.0,
            "ending_r_5th_percentile": 0.0,
            "ending_r_95th_percentile": 0.0,
            "median_drawdown_r": 0.0,
            "drawdown_95th_percentile_r": 0.0,
        }

    rng = np.random.default_rng(
        seed
    )

    endings = np.empty(
        simulations
    )

    drawdowns = np.empty(
        simulations
    )

    for index in range(
        simulations
    ):
        sample = rng.choice(
            values,
            size=values.size,
            replace=True,
        )

        equity = np.concatenate(
            (
                [0.0],
                np.cumsum(sample),
            )
        )

        peaks = np.maximum.accumulate(
            equity
        )

        endings[index] = equity[-1]

        drawdowns[index] = np.max(
            peaks - equity
        )

    return {
        "simulations": simulations,
        "profitable_probability": float(
            np.mean(
                endings > 0
            )
        ),
        "loss_probability": float(
            np.mean(
                endings < 0
            )
        ),
        "median_ending_r": float(
            np.median(
                endings
            )
        ),
        "ending_r_5th_percentile": float(
            np.percentile(
                endings,
                5,
            )
        ),
        "ending_r_95th_percentile": float(
            np.percentile(
                endings,
                95,
            )
        ),
        "median_drawdown_r": float(
            np.median(
                drawdowns
            )
        ),
        "drawdown_95th_percentile_r": float(
            np.percentile(
                drawdowns,
                95,
            )
        ),
    }


def build_portfolio(
    candidates: pd.DataFrame,
    settings: dict[str, Any],
    cost_r: float,
    conflict_policy: str,
) -> tuple[
    pd.DataFrame,
    pd.DataFrame,
]:
    segment_trades = build_segment_trades(
        candidates,
        cost_r,
    )

    segment_trades = (
        remove_duplicate_membership(
            segment_trades
        )
    )

    conflict_accepted, conflict_rejected = (
        resolve_conflicts(
            segment_trades,
            conflict_policy,
        )
    )

    portfolio, constraint_rejected = (
        apply_constraints(
            conflict_accepted,
            maximum_positions=int(
                settings.get(
                    "maximum_simultaneous_positions",
                    4,
                )
            ),
            maximum_sector_positions=int(
                settings.get(
                    "maximum_sector_positions",
                    2,
                )
            ),
            maximum_daily_trades=int(
                settings.get(
                    "maximum_daily_trades",
                    8,
                )
            ),
            maximum_daily_loss_r=float(
                settings.get(
                    "maximum_daily_loss_r",
                    3.0,
                )
            ),
        )
    )

    rejected_frames = [
        frame
        for frame in (
            conflict_rejected,
            constraint_rejected,
        )
        if not frame.empty
    ]

    rejected = (
        pd.concat(
            rejected_frames,
            ignore_index=True,
        )
        if rejected_frames
        else pd.DataFrame()
    )

    return portfolio, rejected


def stress_tests(
    candidates: pd.DataFrame,
    settings: dict[str, Any],
) -> pd.DataFrame:
    rows = []

    costs = settings.get(
        "stress_cost_r_values",
        [
            0.0,
            0.02,
            0.05,
            0.10,
        ],
    )

    policies = settings.get(
        "stress_conflict_policies",
        [
            "skip",
            "priority",
        ],
    )

    for cost_r in costs:
        for policy in policies:
            portfolio, _ = (
                build_portfolio(
                    candidates,
                    settings,
                    float(cost_r),
                    policy,
                )
            )

            stats = statistics(
                portfolio[
                    "portfolio_result_r"
                ]
            )

            rows.append(
                {
                    "additional_cost_r": (
                        float(cost_r)
                    ),
                    "conflict_policy": (
                        policy
                    ),
                    **stats,
                }
            )

    return pd.DataFrame(
        rows
    )


def qualify(
    holdout: dict[str, Any],
    simulation: dict[str, Any],
    stress: pd.DataFrame,
    settings: dict[str, Any],
) -> tuple[
    str,
    list[str],
]:
    reasons = []

    if (
        holdout["trades"]
        < int(
            settings.get(
                "qualification_minimum_holdout_trades",
                100,
            )
        )
    ):
        reasons.append(
            "insufficient holdout trades"
        )

    if (
        holdout["profit_factor"]
        < float(
            settings.get(
                "qualification_minimum_profit_factor",
                1.25,
            )
        )
    ):
        reasons.append(
            "holdout profit factor below standard"
        )

    if (
        holdout["expectancy_r"]
        < float(
            settings.get(
                "qualification_minimum_expectancy_r",
                0.10,
            )
        )
    ):
        reasons.append(
            "holdout expectancy below standard"
        )

    if (
        holdout["maximum_drawdown_r"]
        > float(
            settings.get(
                "qualification_maximum_drawdown_r",
                15.0,
            )
        )
    ):
        reasons.append(
            "holdout drawdown above standard"
        )

    if (
        simulation[
            "profitable_probability"
        ]
        < float(
            settings.get(
                "qualification_minimum_monte_carlo_probability",
                0.75,
            )
        )
    ):
        reasons.append(
            "Monte Carlo probability below standard"
        )

    if (
        simulation[
            "drawdown_95th_percentile_r"
        ]
        > float(
            settings.get(
                "qualification_maximum_monte_carlo_95_drawdown_r",
                25.0,
            )
        )
    ):
        reasons.append(
            "Monte Carlo drawdown above standard"
        )

    stressed = stress.loc[
        (
            stress[
                "additional_cost_r"
            ]
            >= 0.05
        )
        & (
            stress[
                "conflict_policy"
            ]
            == "skip"
        )
    ]

    if not stressed.empty:
        row = stressed.iloc[0]

        if (
            row["profit_factor"] < 1.10
            or row["expectancy_r"] <= 0
        ):
            reasons.append(
                "failed +0.05R cost stress"
            )

    return (
        (
            "validated_portfolio"
            if not reasons
            else "research_only"
        ),
        reasons,
    )


def markdown_table(
    frame: pd.DataFrame,
    columns: list[str],
) -> str:
    available = [
        column
        for column in columns
        if column in frame.columns
    ]

    if frame.empty or not available:
        return "_No records._"

    lines = [
        "| "
        + " | ".join(
            column.replace(
                "_",
                " ",
            ).title()
            for column in available
        )
        + " |",
        "|"
        + "|".join(
            "---"
            for _ in available
        )
        + "|",
    ]

    percentage_columns = {
        "win_rate",
        "profitable_probability",
        "loss_probability",
    }

    for _, row in frame[
        available
    ].iterrows():
        values = []

        for column in available:
            item = row[column]

            if pd.isna(item):
                text = ""

            elif (
                column
                in percentage_columns
            ):
                text = (
                    f"{float(item):.1%}"
                )

            elif isinstance(
                item,
                (
                    float,
                    np.floating,
                ),
            ):
                text = (
                    f"{float(item):.3f}"
                )

            else:
                text = str(item)

            values.append(
                text.replace(
                    "|",
                    "\\|",
                )
            )

        lines.append(
            "| "
            + " | ".join(values)
            + " |"
        )

    return "\n".join(lines)

    def manage_virtual_stops(
        self,
    ) -> None:
        positions = read_json(
            POSITIONS_FILE,
            {},
        )

        changed = False

        for symbol, position in list(
            positions.items()
        ):
            regime = self.regimes.get(
                symbol
            )

            if regime is None:
                continue

            try:
                current_price = (
                    latest_price(
                        self.client,
                        symbol,
                    )
                )

            except Exception as exc:
                append_jsonl(
                    EVENT_LOG,
                    {
                        "timestamp": (
                            utc_now()
                            .isoformat()
                        ),
                        "event": (
                            "price_error"
                        ),
                        "symbol": symbol,
                        "error": (
                            f"{type(exc).__name__}:"
                            f"{exc}"
                        ),
                    },
                )
                continue

            long_trade = (
                position[
                    "side"
                ]
                in {
                    "buy",
                    "long",
                }
            )

            best_price = float(
                position.get(
                    "best_price",
                    position[
                        "entry_price"
                    ],
                )
            )

            best_price = (
                max(
                    best_price,
                    current_price,
                )
                if long_trade
                else min(
                    best_price,
                    current_price,
                )
            )

            five_minute = next(
                (
                    result
                    for result
                    in regime.timeframe_results
                    if result[
                        "timeframe"
                    ]
                    == "5Min"
                ),
                None,
            )

            if five_minute is None:
                continue

            old_stop = float(
                position[
                    "virtual_stop"
                ]
            )

            (
                new_stop,
                current_r,
            ) = calculate_virtual_stop(
                side=(
                    position[
                        "side"
                    ]
                ),
                entry=float(
                    position[
                        "entry_price"
                    ]
                ),
                initial_stop=float(
                    position[
                        "initial_stop"
                    ]
                ),
                current_stop=(
                    old_stop
                ),
                best_price=(
                    best_price
                ),
                current_price=(
                    current_price
                ),
                atr_value=float(
                    five_minute[
                        "atr"
                    ]
                ),
                regime=(
                    regime.regime
                ),
                config=(
                    self.config[
                        "virtual_stop"
                    ]
                ),
            )

            position[
                "best_price"
            ] = best_price

            position[
                "virtual_stop"
            ] = new_stop

            changed = True

            if (
                abs(
                    new_stop
                    - old_stop
                )
                > 1e-9
            ):
                append_csv(
                    STOP_LOG,
                    {
                        "timestamp": (
                            utc_now()
                            .isoformat()
                        ),
                        "symbol": symbol,
                        "current_price": (
                            current_price
                        ),
                        "best_price": (
                            best_price
                        ),
                        "current_r": (
                            current_r
                        ),
                        "old_virtual_stop": (
                            old_stop
                        ),
                        "new_virtual_stop": (
                            new_stop
                        ),
                        "symbol_regime": (
                            regime.regime
                        ),
                        "paper_only": True,
                    },
                )

            stop_hit = (
                (
                    long_trade
                    and current_price
                    <= new_stop
                )
                or (
                    not long_trade
                    and current_price
                    >= new_stop
                )
            )

            if stop_hit:
                close_position(
                    self.client,
                    symbol,
                )

                self.record_exit(
                    position,
                    current_price,
                    "virtual_stop",
                )

                positions.pop(
                    symbol,
                    None,
                )

                changed = True

        if changed:
            atomic_json(
                POSITIONS_FILE,
                positions,
            )

    def cycle(self) -> None:
        current = utc_now()

        now_et = (
            current.astimezone(
                NEW_YORK

        )

        )

            self.last_refresh
            is None

                - self.last_refresh

                    "bar_refresh_seconds"

    def run(
        self,
        once: bool,
    ) -> None:

                return

            time.sleep(
                max(
                    5,
                    int(
                        self.config[
                            "poll_seconds"
                        ]
                    ),
                )
            )

from dataclasses import dataclass
from datetime import datetime, timedelta
from zoneinfo import ZoneInfo


@dataclass
class SessionStatus:
    market_open: bool
    entries_allowed: bool
    flatten_required: bool
    now_et: datetime
    entry_start_et: datetime
    flatten_time_et: datetime
    close_et: datetime
    reason: str


class SessionManager:
    def __init__(
        self,
        client,
        entry_start: str = "09:45",
        flatten_minutes_before_close: int = 10,
        timezone_name: str = (
            "America/New_York"
        ),
    ):
        self.client = client

        self.timezone = ZoneInfo(
            timezone_name
        )

        hour, minute = (
            int(value)
            for value in entry_start.split(":")
        )

        self.entry_hour = hour
        self.entry_minute = minute

        self.flatten_minutes = int(
            flatten_minutes_before_close
        )

    def status(self) -> SessionStatus:
        clock = self.client.clock()

        now_et = datetime.fromisoformat(
            clock["timestamp"]
        ).astimezone(
            self.timezone
        )

        close_et = datetime.fromisoformat(
            clock["next_close"]
        ).astimezone(
            self.timezone
        )

        market_open = bool(
            clock["is_open"]
        )

        session_date = (
            now_et.date()
            if market_open
            else datetime.fromisoformat(
                clock["next_open"]
            ).astimezone(
                self.timezone
            ).date()
        )

        entry_start_et = datetime(
            session_date.year,
            session_date.month,
            session_date.day,
            self.entry_hour,
            self.entry_minute,
            tzinfo=self.timezone,
        )

        flatten_time_et = (
            close_et
            - timedelta(
                minutes=self.flatten_minutes
            )
        )

        flatten_required = (
            market_open
            and now_et >= flatten_time_et
        )

        entries_allowed = (
            market_open
            and now_et >= entry_start_et
            and now_et < flatten_time_et
        )

        if not market_open:
            reason = "market_closed"

        elif now_et < entry_start_et:
            reason = "waiting_until_0945_et"

        elif flatten_required:
            reason = "flatten_window"

        else:
            reason = "entries_allowed"

        return SessionStatus(
            market_open=market_open,
            entries_allowed=entries_allowed,
            flatten_required=flatten_required,
            now_et=now_et,
            entry_start_et=entry_start_et,
            flatten_time_et=flatten_time_et,
            close_et=close_et,
            reason=reason,
        )

from __future__ import annotations

from io import StringIO
from pathlib import Path

import pandas as pd
import requests

from core.models import UniverseStock


SP500_URL = (
    "https://en.wikipedia.org/wiki/"
    "List_of_S%26P_500_companies"
)


class SP500Universe:
    def __init__(
        self,
        client,
        config: dict,
        sector_etfs: dict[str, str],
    ):
        self.client = client
        self.config = config
        self.sector_etfs = sector_etfs

    @staticmethod
    def normalize_symbol(
        symbol: str,
    ) -> str:
        return (
            str(symbol)
            .strip()
            .upper()
            .replace(".", "-")
        )

    def load_constituents(
        self,
    ) -> pd.DataFrame:
        cache_path = Path(
            self.config.get(
                "constituents_cache",
                "state/sp500_constituents.csv",
            )
        )

        cache_path.parent.mkdir(
            parents=True,
            exist_ok=True,
        )

        try:
            response = requests.get(
                SP500_URL,
                headers={
                    "User-Agent": (
                        "Mozilla/5.0 "
                        "(X11; Linux x86_64) "
                        "AppleWebKit/537.36 "
                        "Chrome/126 Safari/537.36"
                    ),
                    "Accept-Language": "en-US,en;q=0.9",
                },
                timeout=30,
            )

            response.raise_for_status()

            tables = pd.read_html(
                StringIO(response.text)
            )

            if not tables:
                raise RuntimeError(
                    "No S&P 500 table returned"
                )

            dataframe = tables[0][
                [
                    "Symbol",
                    "Security",
                    "GICS Sector",
                ]
            ].copy()

            dataframe.columns = [
                "symbol",
                "company",
                "sector",
            ]

            dataframe["symbol"] = (
                dataframe["symbol"]
                .map(self.normalize_symbol)
            )

            dataframe.to_csv(
                cache_path,
                index=False,
            )

            return dataframe

        except Exception as error:
            if not cache_path.exists():
                raise RuntimeError(
                    "Could not download S&P 500 "
                    "constituents and no cache exists: "
                    f"{error}"
                ) from error

            print(
                "Using cached S&P 500 constituents:",
                cache_path,
            )

            return pd.read_csv(
                cache_path
            )

    def build(
        self,
    ) -> list[UniverseStock]:
        constituents = (
            self.load_constituents()
        )

        assets = self.client.assets()

        asset_map = {
            asset["symbol"]: asset
            for asset in assets
            if (
                asset.get("tradable")
                and asset.get("status")
                == "active"
            )
        }

        constituents = constituents[
            constituents["symbol"].isin(
                asset_map.keys()
            )
        ].copy()

        snapshots = self.client.snapshots(
            constituents[
                "symbol"
            ].tolist()
        )

        minimum_price = float(
            self.config.get(
                "minimum_price",
                5.0,
            )
        )

        maximum_price = float(
            self.config.get(
                "maximum_price",
                500.0,
            )
        )

        minimum_dollar_volume = float(
            self.config.get(
                "minimum_previous_day_dollar_volume",
                25_000_000,
            )
        )

        maximum_spread_pct = float(
            self.config.get(
                "maximum_spread_pct",
                0.20,
            )
        )

        candidates = []

        for row in constituents.itertuples(
            index=False
        ):
            snapshot = snapshots.get(
                row.symbol,
                {},
            )

            latest_trade = (
                snapshot.get("latestTrade")
                or {}
            )

            latest_quote = (
                snapshot.get("latestQuote")
                or {}
            )

            daily_bar = (
                snapshot.get("dailyBar")
                or {}
            )

            previous_bar = (
                snapshot.get("prevDailyBar")
                or {}
            )

            price = float(
                latest_trade.get("p")
                or daily_bar.get("c")
                or previous_bar.get("c")
                or 0.0
            )

            bid = float(
                latest_quote.get("bp", 0.0)
                or 0.0
            )

            ask = float(
                latest_quote.get("ap", 0.0)
                or 0.0
            )

            previous_close = float(
                previous_bar.get(
                    "c",
                    price,
                )
                or price
            )

            previous_volume = float(
                previous_bar.get(
                    "v",
                    0.0,
                )
                or 0.0
            )

            dollar_volume = (
                previous_close
                * previous_volume
            )

            quote_available = (
                bid > 0
                and ask > bid
            )

            if quote_available:
                midpoint = (
                    bid + ask
                ) / 2.0

                spread_pct = (
                    ask - bid
                ) / midpoint * 100
            else:
                # Weekend and closed-market IEX snapshots
                # may omit a current bid/ask. Universe
                # selection may continue, but live execution
                # will require a fresh valid quote.
                spread_pct = 0.0

            if not (
                minimum_price
                <= price
                <= maximum_price
            ):
                continue

            if (
                dollar_volume
                < minimum_dollar_volume
            ):
                continue

            if (
                quote_available
                and spread_pct
                > maximum_spread_pct
            ):
                continue

            sector_etf = (
                self.sector_etfs.get(
                    row.sector
                )
            )

            if not sector_etf:
                continue

            asset = asset_map[
                row.symbol
            ]

            candidates.append(
                {
                    "symbol": row.symbol,
                    "company": row.company,
                    "sector": row.sector,
                    "sector_etf": sector_etf,
                    "price": price,
                    "previous_volume":
                        previous_volume,
                    "dollar_volume":
                        dollar_volume,
                    "spread_pct":
                        spread_pct,
                    "shortable": bool(
                        asset.get(
                            "shortable",
                            False,
                        )
                    ),
                    "fractionable": bool(
                        asset.get(
                            "fractionable",
                            False,
                        )
                    ),
                }
            )

        dataframe = pd.DataFrame(
            candidates
        )

        if dataframe.empty:
            return []

        requested_count = int(
            self.config.get(
                "stocks_per_sector",
                5,
            )
        )

        maximum_count = int(
            self.config.get(
                "maximum_stocks_per_sector",
                10,
            )
        )

        selected_count = max(
            1,
            min(
                requested_count,
                maximum_count,
            ),
        )

        dataframe = dataframe.sort_values(
            [
                "sector",
                "dollar_volume",
                "spread_pct",
            ],
            ascending=[
                True,
                False,
                True,
            ],
        )

        selected = (
            dataframe.groupby(
                "sector",
                group_keys=False,
            )
            .head(selected_count)
            .copy()
        )

        selected["rank_in_sector"] = (
            selected.groupby(
                "sector"
            )["dollar_volume"]
            .rank(
                method="first",
                ascending=False,
            )
            .astype(int)
        )

        selected_cache = Path(
            self.config.get(
                "selected_cache",
                "state/selected_universe.csv",
            )
        )

        selected_cache.parent.mkdir(
            parents=True,
            exist_ok=True,
        )

        selected.to_csv(
            selected_cache,
            index=False,
        )

        return [
            UniverseStock(
                symbol=row.symbol,
                company=row.company,
                sector=row.sector,
                sector_etf=row.sector_etf,
                price=float(row.price),
                previous_volume=float(
                    row.previous_volume
                ),
                dollar_volume=float(
                    row.dollar_volume
                ),
                spread_pct=float(
                    row.spread_pct
                ),
                rank_in_sector=int(
                    row.rank_in_sector
                ),
            )
            for row in selected.itertuples(
                index=False
            )
        ]
