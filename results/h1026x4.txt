class AlpacaStyleSimulation:

        # Emulating a standard retail trading account balance footprint
        self.cash = 100000.0
        self.portfolio_value = 100000.0
        # Alpaca API tracking format: { symbol: {"qty": float, "avg_entry": float, "side": str} }
        self.alpaca_positions = {} 

    def run_simulation(self, symbol, trend_bias="RANGE"):
        log.info(f"🔮 Initializing Alpaca-Style Order Simulation for {symbol} | Profile: {trend_bias}")
        
        np.random.seed(101)
        base_price = 150.00 if "USD" in symbol else 1.1000
        ticks = np.random.normal(0, 0.40, 100)
        
        if trend_bias == "BEARISH":
            ticks -= 0.15
        elif trend_bias == "BULLISH":
            ticks += 0.15
            
        prices = base_price + np.cumsum(ticks)
        df = pd.DataFrame({'high': prices + 0.20, 'low': prices - 0.20, 'close': prices})
        
        # Overlay structural logic mathematical indicators
        close = df['close']
        df['ema_fast'] = close.ewm(span=Config.FAST_EMA, adjust=False).mean()
        df['ema_slow'] = close.ewm(span=Config.SLOW_EMA, adjust=False).mean()
        df['bb_mid'] = close.rolling(window=20).mean()
        df['bb_std'] = close.rolling(window=20).std()
        df['bb_upper'] = df['bb_mid'] + (2.0 * df['bb_std'])
        df['bb_lower'] = df['bb_mid'] - (2.0 * df['bb_std'])
        df['tr'] = np.maximum(df['high'] - df['low'], np.maximum(abs(df['high'] - close.shift(1)), abs(df['low'] - close.shift(1))))
        df['atr'] = df['tr'].rolling(window=14).mean()
        df['adx'] = (abs(df['ema_fast'] - df['ema_slow']) / df['atr']) * 12.0
        df['adx'] = df['adx'].fillna(15.0).clip(5.0, 45.0)

        trades = 0
        pnl = 0.0

        for i in range(25, len(df)):
            row = df.iloc[i]
            price = row['close']
            adx = row['adx']
            has_position = symbol in self.alpaca_positions

            # --- ALPACA ORDER GATEWAY SUBMISSION SIMULATION ---
            if adx < Config.RANGE_ADX_MAX and not has_position:
                if price <= row['bb_lower']:
                    # Emulate an Alpaca SDK Market Order payload logic array
                    self.alpaca_positions[symbol] = {"qty": 100.0, "avg_entry": price, "side": "long"}
                    log.info(f"📦 [ALPACA SUBMIT] buy 100.0 shares {symbol} at market. Executed @ ${price:.2f} (ADX: {adx:.1f})")
                    trades += 1
                elif price >= row['bb_upper']:
                    self.alpaca_positions[symbol] = {"qty": 100.0, "avg_entry": price, "side": "short"}
                    log.info(f"📦 [ALPACA SUBMIT] sell_short 100.0 shares {symbol} at market. Executed @ ${price:.2f} (ADX: {adx:.1f})")
                    trades += 1

            # --- ALPACA TARGET UNWIND MATRIX ---
            if has_position:
                pos = self.alpaca_positions[symbol]
                entry = pos['avg_entry']
                side = pos['side']
                closed = False

                if side == "long":
                    if price >= entry + 1.25 or price >= row['bb_mid']:
                        log.info(f"💰 [ALPACA TP] Target hit. Closing long position for {symbol} @ ${price:.2f}")
                        pnl += (price - entry) * pos['qty']
                        closed = True
                elif side == "short":
                    if price <= entry - 1.25 or price <= row['bb_mid']:
                        log.info(f"💰 [ALPACA TP] Target hit. Closing short position for {symbol} @ ${price:.2f}")
                        pnl += (entry - price) * pos['qty']
                        closed = True

                if closed:
                    del self.alpaca_positions[symbol]

        print(f"\n=== 🇺🇸 ALPACA MATRIX COMPLIANCE REPORT ===")
        print(f"Asset Verified:           {symbol}")
        print(f"Alpaca Order Signals:     {trades}")
        print(f"Calculated Simulation PnL: ${pnl:.2f}")
        print(f"Active Live Footprint:    {self.alpaca_positions}")
        print("===========================================\n")

def calculate_atr(df: pd.DataFrame, period: int = 14) -> pd.Series:
    tr = pd.concat([
        (df["high"] - df["low"]),
        (df["high"] - df["close"].shift(1)).abs(),
        (df["low"] - df["close"].shift(1)).abs()
    ], axis=1).max(axis=1)
    return tr.rolling(period).mean()

def calculate_adx(df: pd.DataFrame, period: int = 14) -> pd.Series:
    plus_dm = df["high"].diff().where(lambda x: (x > 0) & (x > df["low"].diff().abs()), 0)
    minus_dm = df["low"].diff().abs().where(lambda x: (x > 0) & (x > df["high"].diff()), 0)
    tr = pd.concat([(df["high"] - df["low"]), (df["high"] - df["close"].shift(1)).abs(), (df["low"] - df["close"].shift(1)).abs()], axis=1).max(axis=1)
    atr_p = tr.rolling(period).mean()
    plus_di = 100 * (plus_dm.rolling(period).mean() / (atr_p + 1e-9))
    minus_di = 100 * (minus_dm.rolling(period).mean() / (atr_p + 1e-9))
    dx = 100 * (plus_di - minus_di).abs() / (plus_di + minus_di + 1e-9)
    return dx.rolling(period).mean()

import numpy as np

import logging

logging.basicConfig(level=logging.INFO, format='%(asctime)s [%(levelname)s] %(message)s')
log = logging.getLogger("UNIFIED_BEAST")

class Config:
    RANGE_ADX_MAX = 22.0      
    TREND_ADX_MIN = 22.0      
    BASE_RISK_PCT = 0.01      
    FAST_EMA = 9
    SLOW_EMA = 20

class OandaUnifiedBeast:
    def __init__(self):
        self.equity = 66158.39
        self.active_positions = {} # Structured as {symbol: []} to handle multiple concurrent positions safely
        
    def analyze_regime(self, df):
        close = df['close']
        high = df['high']
        low = df['low']
        
        df['ema_fast'] = close.ewm(span=Config.FAST_EMA, adjust=False).mean()
        df['ema_slow'] = close.ewm(span=Config.SLOW_EMA, adjust=False).mean()
        
        df['bb_mid'] = close.rolling(window=20).mean()
        df['bb_std'] = close.rolling(window=20).std()
        df['bb_upper'] = df['bb_mid'] + (2.0 * df['bb_std'])
        df['bb_lower'] = df['bb_mid'] - (2.0 * df['bb_std'])
        
        df['tr'] = np.maximum(high - low, np.maximum(abs(high - close.shift(1)), abs(low - close.shift(1))))
        df['atr'] = df['tr'].rolling(window=14).mean()
        df['adx'] = (abs(df['ema_fast'] - df['ema_slow']) / df['atr']) * 15.0
        df['adx'] = df['adx'].fillna(15.0).clip(5.0, 45.0)
        
        return df

    def run_internal_simulation(self, symbol, trend_bias="BULLISH"):
        log.info(f"🔮 Starting Internal Simulation Engine for {symbol} | Regime Focus: {trend_bias}")
        
        if symbol not in self.active_positions:
            self.active_positions[symbol] = []
            
        np.random.seed(42)
        base_price = 1.0850 if "USD" in symbol else 158.20
        ticks = np.random.normal(0, 0.0010, 100)
        
        if trend_bias == "BULLISH":
            ticks += 0.0004  
        elif trend_bias == "BEARISH":
            ticks -= 0.0004  
            
        prices = base_price + np.cumsum(ticks)
        
        df = pd.DataFrame({
            'high': prices + 0.0005,
            'low': prices - 0.0005,
            'close': prices
        })
        
        df = self.analyze_regime(df)
        trades_executed = 0
        pnl_accumulated = 0.0
        
        for i in range(25, len(df)):
            row = df.iloc[i]
            price = row['close']
            adx = row['adx']
            
            # Count active positions for this symbol
            symbol_pos_count = len(self.active_positions[symbol])
            
            # --- REGIME INTERLOCK BOARD ---
            if adx < Config.RANGE_ADX_MAX:
                if price <= row['bb_lower'] and symbol_pos_count == 0:
                    self.active_positions[symbol].append({'dir': 'BUY', 'entry': price, 'type': 'RANGE'})

                    trades_executed += 1
                elif price >= row['bb_upper'] and symbol_pos_count == 0:
                    self.active_positions[symbol].append({'dir': 'SELL', 'entry': price, 'type': 'RANGE'})
                    log.info(f"🔄 [RANGE MATCH] {symbol} overbought at {price:.5f} (ADX: {adx:.1f}). Opening Mean-Reversion Short.")
                    trades_executed += 1
                    
            elif adx >= Config.TREND_ADX_MIN:
                fast, slow = row['ema_fast'], row['ema_slow']
                
                if fast > slow and price <= fast and symbol_pos_count == 0:
                    self.active_positions[symbol].append({'dir': 'BUY', 'entry': price, 'type': 'BULL_SCALP'})

                        pnl_accumulated += (entry - price) * 100000
                        closed = True
                        
                if not closed:

def calculate_dynamic_trailing_stop(position_type, entry_price, current_price, atr):
    """
    Dynamically squeezes trailing stops as profits expand to completely stop give-backs.
    """
    if position_type == "long":
        profit_dist = current_price - entry_price
    else:
        profit_dist = entry_price - current_price

    # Calculate current progress in multiples of ATR
    atr_multiple = profit_dist / atr if atr > 0 else 0
    
    # Default initial fallback trail distance (1.5 ATR room to breathe)

    # --- THE AGGRESSIVE PROFIT RUNWAY LOGIC ---

        # Milestone 3: Extreme extension -> Suffocate the trade with a tight 0.25 ATR trail

    elif atr_multiple >= 1.5:
        # Milestone 2: High profit expansion -> Tighten down to 0.50 ATR trail
        trail_distance = 0.50 * atr

        trail_distance = 1.00 * atr

    if position_type == "long":

    return dynamic_stop_price
