def calculate_chop(df, length=14):
    # M-29 Logic: High CHOP (>61.8) = Stay out of the range
    tr1 = pd.DataFrame(df['high'] - df['low'])
    tr2 = pd.DataFrame(abs(df['high'] - df['close'].shift(1)))
    tr3 = pd.DataFrame(abs(df['low'] - df['close'].shift(1)))
    tr = pd.concat([tr1, tr2, tr3], axis=1).max(axis=1)
    atr_sum = tr.rolling(window=length).sum()
    max_high = df['high'].rolling(window=length).max()
    min_low = df['low'].rolling(window=length).min()
    # Fractal dimension formula
    chop = 100 * np.log10(atr_sum / (max_high - min_low)) / np.log10(length)
    return chop.iloc[-1]

SECTOR_MAP = {
    "NVDA": "XLK", "AAPL": "XLK", "AMD": "XLK",
    "TSLA": "XLY", "AMZN": "XLY",
    "BTC/USD": "CRYPTO", "ETH/USD": "CRYPTO"
}

def get_sector_bias(df_sector):
    """Returns 'BULL', 'BEAR', or 'CHOP' based on Weinstein Stages"""
    df_sector['EMA_30'] = ta.ema(df_sector['close'], length=30)
    last = df_sector.iloc[-1]
    prev = df_sector.iloc[-2]
    
    # Stage 2: Price above EMA and EMA rising
    if last['close'] > last['EMA_30'] and last['EMA_30'] > prev['EMA_30']:
        return "BULL"
    # Stage 4: Price below EMA and EMA falling
    elif last['close'] < last['EMA_30'] and last['EMA_30'] < prev['EMA_30']:
        return "BEAR"
    else:
        return "CHOP"

def atr_wilder(df: pd.DataFrame, n: int) -> pd.Series:
    high = df["high"]
    low = df["low"]
    close = df["close"]
    prev_close = close.shift(1)
    tr = pd.concat(
        [
            (high - low).abs(),
            (high - prev_close).abs(),
            (low - prev_close).abs(),
        ],
        axis=1,
    ).max(axis=1)
    return tr.ewm(alpha=1.0 / n, adjust=False).mean()


def vwap_daily(df: pd.DataFrame) -> pd.Series:
    df = df.copy()
    if df.index.tz is None:
        trade_dates = df.index.date
    else:
        trade_dates = df.index.tz_convert(ET).date
    df["trade_date"] = trade_dates

    def _calc(group: pd.DataFrame) -> pd.Series:
        pv = (group["close"] * group["volume"]).cumsum()
        vv = group["volume"].cumsum().replace(0, np.nan)
        return pv / vv

    return df.groupby("trade_date", group_keys=False).apply(_calc)


@dataclass
class Features:
    trend: bool
    above_20: bool
    above_200: bool
    elephant: bool
    rvol: bool
    vwap_ok: bool
    breakout: bool
    close_strong: bool
    atr_filter_ok: bool


@dataclass
class Signal:
    ok: bool
    side: str
    entry_ref: float
    stop: float
    tp: float
    score: float
    reason: str
    feat: Features


class VelezMasterSniper:

    def save_pos(self):
        jsave(POS_STATE, self.pos)

    def log_trade(self, sym: str, action: str, qty: int, price: float, stop: float, tp: float, reason: str):
        with TRADES_CSV.open("a") as f:
            f.write(
                f"{datetime.now(ET).isoformat()},{sym},{action},{qty},{price:.4f},{stop:.4f},{tp:.4f},{reason}\n"
            )

    def in_window(self) -> bool:
        now = datetime.now(ET).strftime("%H:%M")
        return self.win_start <= now <= self.win_end

    def get_buying_power(self) -> float:
        acct = self.trading.get_account()
        try:
            return float(acct.buying_power)
        except Exception:
            return float(acct.cash)

    def fetch_bars(self, sym: str) -> pd.DataFrame:
        end = datetime.now(ET) - timedelta(minutes=self.delay_min)
        start = end - timedelta(days=7)

        req = StockBarsRequest(
            symbol_or_symbols=[sym],
            timeframe=TimeFrame.Minute,
            start=start,
            end=end,
            feed=self.feed,
        )

        bars = self.data.get_stock_bars(req)
        raw = bars.data.get(sym, [])

        rows = []
        for b in raw:
            rows.append(
                {
                    "ts": pd.Timestamp(b.timestamp),
                    "open": float(b.open),
                    "high": float(b.high),
                    "low": float(b.low),
                    "close": float(b.close),
                    "volume": float(b.volume),
                }
            )

        df = pd.DataFrame(rows)
        if df.empty:
            return df

        df = df.tail(self.bar_limit).copy()
        df = df.set_index("ts").sort_index()

        if df.index.tz is None:
            df.index = df.index.tz_localize("UTC")
        else:
            df.index = df.index.tz_convert("UTC")

        return df

    def compute_qty(self, buying_power: float, entry: float, stop: float, risk_pct: float) -> int:
        risk_dollars = buying_power * risk_pct
        risk_per_share = abs(entry - stop)
        if risk_per_share <= 0:
            return 0

        qty_by_risk = int(risk_dollars / risk_per_share)
        qty_by_notional = int((buying_power * self.max_notional_pct) / max(entry, 0.01))
        qty_by_pos_cap = int((buying_power * self.max_pos_pct) / max(entry, 0.01))

        qty = min(qty_by_risk, qty_by_notional, qty_by_pos_cap, self.max_qty)
        return max(qty, 0)

    def build_signal(self, df: pd.DataFrame) -> Signal:
        if df is None or df.empty:
            return Signal(
                False, "", 0.0, 0.0, 0.0, 0.0, "no bars",
                Features(False, False, False, False, False, False, False, False, False),
            )

        min_bars = max(self.ema_slow + 5, self.breakout_lookback + 5, self.atr_len + 5, self.rvol_len + 5)
        if len(df) < min_bars:
            return Signal(
                False, "", 0.0, 0.0, 0.0, 0.0, f"not enough bars ({len(df)}/{min_bars})",
                Features(False, False, False, False, False, False, False, False, False),
            )

        d = df.copy()
        d["ema_fast"] = ema(d["close"], self.ema_fast)
        d["ema_mid"] = ema(d["close"], self.ema_mid)
        d["ema_slow"] = ema(d["close"], self.ema_slow)
        d["atr"] = atr_wilder(d, self.atr_len)
        d["vol_avg"] = d["volume"].rolling(self.rvol_len).mean()
        d["vwap"] = vwap_daily(d)
        
        # --- THE VELEZ SNIPER ENTRY LOGIC (FIXED POSITION) ---
        last = d.iloc[-1]
        
        # 1. LONG SIGNAL
        if (last["close"] > last["open"]) and (last["close"] > last["ema_fast"]):
            return Signal(True, "BUY", last["close"], last["close"] * 0.995, last["close"] * 1.015, 0.0, "BULLISH", Features(True, False, False, False, False, False, False, False, False))
        
        # 2. SHORT SIGNAL
        if (last["close"] < last["open"]) and (last["close"] < last["ema_fast"]):
            return Signal(True, "SELL", last["close"], last["close"] * 1.005, last["close"] * 0.985, 0.0, "BEARISH", Features(False, True, False, False, False, False, False, False, False))
        
        return Signal(False, "", 0.0, 0.0, 0.0, 0.0, "no trend", Features(False, False, False, False, False, False, False, False, False))

        last = d.iloc[-1]
        entry_ref = float(last["close"])
        atr_last = float(last["atr"]) if np.isfinite(last["atr"]) else 0.0

        if atr_last <= 0:
            return Signal(
                False, "", 0.0, 0.0, 0.0, 0.0, "bad atr",
                Features(False, False, False, False, False, False, False, False, False),
            )

        atr_pct = atr_last / max(entry_ref, 0.01)
        atr_filter_ok = self.atr_pct_min <= atr_pct <= self.atr_pct_max
        if not atr_filter_ok:
            return Signal(
                False, "", 0.0, 0.0, 0.0, 0.0, f"atr filter fail ({atr_pct:.4f})",
                Features(False, False, False, False, False, False, False, False, False),
            )

        trend = bool(last["ema_fast"] > last["ema_mid"] > last["ema_slow"])
        above_20 = bool(entry_ref > float(last["ema_fast"]))
        above_200 = bool(entry_ref > float(last["ema_slow"]))
        vwap_ok = bool(np.isfinite(last["vwap"]) and entry_ref > float(last["vwap"]))

        body = abs(float(last["close"]) - float(last["open"]))
        candle_range = float(last["high"]) - float(last["low"])
        elephant = bool(
            candle_range > 0
            and body > (0.8 * candle_range)
            and body > (1.2 * atr_last)
            and float(last["close"]) > float(last["open"])
        )

        breakout_high = d["high"].iloc[-self.breakout_lookback:-1].max()
        breakout = bool(entry_ref > float(breakout_high))

        close_strong = bool(
            entry_ref >= (float(last["low"]) + 0.7 * (float(last["high"]) - float(last["low"])))
        )

        rvol = bool(
            np.isfinite(last["vol_avg"]) and float(last["volume"]) >= float(last["vol_avg"]) * self.rvol_min
        )

        feat = Features(
            trend=trend,
            above_20=above_20,
            above_200=above_200,
            elephant=elephant,
            rvol=rvol,
            vwap_ok=vwap_ok,
            breakout=breakout,
            close_strong=close_strong,
            atr_filter_ok=atr_filter_ok,
        )

        score = 0.0
        score += 0.30 if elephant else 0.0
        score += 0.15 if above_20 else 0.0
        score += 0.15 if above_200 else 0.0
        score += 0.10 if vwap_ok else 0.0
        score += 0.15 if breakout else 0.0
        score += 0.10 if rvol else 0.0
        score += 0.05 if close_strong else 0.0

        if not trend:
            return Signal(False, "", 0.0, 0.0, 0.0, score, "trend not aligned", feat)

        if score < self.min_score:
            return Signal(False, "", 0.0, 0.0, 0.0, score, "score too low", feat)

        stop = entry_ref - (atr_last * self.atr_stop_mult)
        tp = entry_ref + (atr_last * self.atr_tp_mult)

        if stop <= 0 or tp <= entry_ref:
            return Signal(False, "", 0.0, 0.0, 0.0, score, "bad stop/tp", feat)

        return Signal(
            True,
            "long",
            entry_ref,
            stop,
            tp,
            score,
            f"velez elephant | breakout={int(breakout)} rvol={int(rvol)} vwap={int(vwap_ok)}",
            feat,
        )

    def exit_trade(self, sym: str, qty: int, reason: str, price: float, stop: float, tp: float):
        order = MarketOrderRequest(
            symbol=sym,
            qty=qty,
            side=OrderSide.SELL,
            time_in_force=TimeInForce.DAY,
        )
        self.trading.submit_order(order_data=order)
        self.log.info(f"EXIT {sym} | {reason} | px={price:.2f}")
        self.log_trade(sym, reason, qty, price, stop, tp, reason)
        self.pos[sym] = {"in_pos": False}
        self.save_pos()

    def manage_positions(self, sym: str, current_price: float):
        p = self.pos.get(sym)
        if not p or not p.get("in_pos"):
            return

        qty = int(p["qty"])
        entry = float(p["entry"])
        stop = float(p["stop"])
        tp = float(p["tp"])

        if current_price <= stop:
            self.exit_trade(sym, qty, "STOP", current_price, stop, tp)
            return

        if current_price >= tp:
            self.exit_trade(sym, qty, "TP", current_price, stop, tp)
            return

        if not p.get("be", False):
            if current_price >= (entry + (entry - stop)):
                p["stop"] = entry
                p["be"] = True
                self.pos[sym] = p
                self.save_pos()
                self.log.info(f"🟨 Move to BE for {sym}")

    def run(self):
        self.log.info("🚀 starting main loop")

        while True:
            try:
                if not self.in_window():
                    time.sleep(30)
                    continue

                buying_power = self.get_buying_power()

                for sym in self.symbols:
                    df = self.fetch_bars(sym)
                    if df.empty:
                        self.log.info(f"… {sym} no bars")
                        continue

                    current_price = float(df.iloc[-1]["close"])

                    if self.pos.get(sym, {}).get("in_pos"):
                        self.manage_positions(sym, current_price)
                        continue

                    sig = self.build_signal(df)
                    if not sig.ok:
                        self.log.info(f"… {sym} no entry: {sig.reason}")
                        continue

                    risk_pct = self.risk_min_pct + (self.risk_max_pct - self.risk_min_pct) * clamp(sig.score, 0.0, 1.0)
                    qty = self.compute_qty(buying_power, sig.entry_ref, sig.stop, risk_pct)

                    if qty <= 0:
                        self.log.info(f"⚠️ {sym} qty=0 (stop wide or caps). skip")
                        continue

                    order = MarketOrderRequest(
                        symbol=sym,
                        qty=qty,
                        side=OrderSide.BUY,
                        time_in_force=TimeInForce.DAY,
                    )

                    self.log.info(
                        f"🟩 ENTRY {sym} LONG qty={qty} px={sig.entry_ref:.2f} stop={sig.stop:.2f} tp={sig.tp:.2f} "
                        f"risk={risk_pct*100:.2f}% score={sig.score:.2f} | "
                        f"trend={int(sig.feat.trend)},above20={int(sig.feat.above_20)},above200={int(sig.feat.above_200)},"
                        f"elephant={int(sig.feat.elephant)},rvol={int(sig.feat.rvol)},vwap={int(sig.feat.vwap_ok)},"
                        f"breakout={int(sig.feat.breakout)},atr_ok={int(sig.feat.atr_filter_ok)}"
                    )

                    try:
                        self.trading.submit_order(order_data=order)
                    except Exception as e:
                        self.log.error(f"❌ submit_order error: {repr(e)}")
                        continue

                    self.pos[sym] = {
                        "in_pos": True,
                        "qty": qty,
                        "entry": sig.entry_ref,
                        "stop": sig.stop,
                        "tp": sig.tp,
                        "side": "long",
                        "be": False,
                        "ts": datetime.utcnow().isoformat(),
                        "score": sig.score,
                    }
                    self.save_pos()
                    self.log_trade(sym, "ENTRY", qty, sig.entry_ref, sig.stop, sig.tp, f"score={sig.score:.2f}")

                    time.sleep(60)

                time.sleep(self.sleep_s)

            except KeyboardInterrupt:
                self.log.info("👋 stopped by user")
                break
            except Exception as e:
                self.log.error(f"❌ Loop error: {repr(e)}")
                self.log.error(traceback.format_exc())
                time.sleep(10)


if __name__ == "__main__":
    VelezMasterSniper().run()
